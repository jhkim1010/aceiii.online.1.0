+        ),
+      };
+    case 'medio_pago':
+      return {
+        joins: [
+          `LEFT JOIN payment_methods pm ON pm.id = ${k} AND (pm.store_id = $1::int OR pm.store_id IS NULL)`,
+        ],
+        label: lookup('pm.title'),
+        extra: 'pm.slug AS medio_pago_slug',
+      };
+    case 'categoria':
+      return {
+        joins: [
+          `LEFT JOIN categories lc ON lc.id = ${k} AND lc.store_id = $1::int`,
+        ],
+        label: lookup('lc.name'),
+      };
+    case 'temporada':
+      return {
+        joins: [
+          `LEFT JOIN seasons ls ON ls.id = ${k} AND ls.store_id = $1::int`,
+        ],
+        label: lookup('ls.name'),
+      };
+    case 'color':
+      return {
+        joins: [
+          `LEFT JOIN colors lco ON lco.id = ${k} AND lco.store_id = $1::int`,
+        ],
+        label: lookup('lco.name'),
+      };
+    case 'talle':
+      return {
+        joins: [
+          `LEFT JOIN sizes lsz ON lsz.id = ${k} AND lsz.store_id = $1::int`,
+        ],
+        label: lookup('lsz.name'),
+      };
+    case 'producto':
+      return {
+        joins: [
+          `LEFT JOIN products lp ON lp.id = ${k} AND lp.store_id = $1::int`,
+        ],
+        label: lookup(
+          `NULLIF(TRIM(COALESCE(lp.sku, '') || ' ' || COALESCE(lp.name, '')), '')`,
+        ),
+      };
+    default:
+      throw new Error(`dimensión desconocida: ${String(dim)}`);
+  }

+/** Predicado de cada filtro (el valor viaja en `$n::int[]`). */
+const PREDICADO_FILTRO: Record<FiltroId, string> = {
+  sucursales: 'venta.branch_id',
+  vendedores: 'venta.seller_id',
+  mediosPago: 'spm.payment_method_id',
+  categorias: 'COALESCE(p.category_id, m.category_id)',
+  temporadas: 'COALESCE(p.season_id, m.season_id)',
+  colores: 'p.color_id',
+  talles: 'p.size_id',
+};

+function agregados(grano: Grano, necesitaIA: boolean): string[] {
+  if (grano === 'pago') {
+    return ['SUM(spm.amount) AS a_venta'];
+  }
+  if (grano === 'item') {
+    // Phase 117-13: el signo del monto manda sobre unidades, costo y bases de línea (Anulación con
+    // cantidad positiva). La cabecera usa `items_agg` firmado; acá se firma cada línea.
+    const f = (expr: string) => firmadoSql('venta.sgn', expr);
+
+    return [
+      `SUM(${f('si.subtotal')}) AS a_venta`,
+      'COUNT(DISTINCT venta.sale_id) FILTER (WHERE venta.sgn = 1) - COUNT(DISTINCT venta.sale_id) FILTER (WHERE venta.sgn = -1) AS a_sgn',
+      `SUM(${f('si.quantity')}) AS a_units`,
+      `SUM(${f('si.subtotal')}) AS a_items_total`,
+      `SUM(${f('si.quantity * si.unit_cost')}) FILTER (WHERE si.unit_cost IS NOT NULL) AS a_costo`,
+      `SUM(${f('si.subtotal')}) FILTER (WHERE si.unit_cost IS NOT NULL) AS a_vcc`,
+    ];
+  }
+  const base = [
+    'SUM(venta.total_amount) AS a_venta',
+    'SUM(venta.sgn) AS a_sgn',
+  ];
+  if (!necesitaIA) return base;
+
+  return [
+    ...base,
+    'SUM(ia.units) AS a_units',
+    'SUM(ia.items_total) AS a_items_total',
+    'SUM(ia.costo) AS a_costo',
+    'SUM(ia.venta_con_costo) AS a_vcc',
+  ];

+  alias: string;

+/** Expresiones finales (consulta externa) de cada métrica sobre las columnas `g.*`. */
+function expresionMetrica(
+  m: MetricaId,
+  grano: Grano,
+  conDev: boolean,
+  atribuible: boolean,
+): Salida[] {
+  const ventaNeta = atribuible
+    ? '(g.a_venta - COALESCE(g.d_monto, 0))'
+    : 'g.a_venta';
+  const noSoportada = (): never => {
+    throw new Error(`métrica ${m} no soportada en grano ${grano}`);
+  };
+  switch (m) {
+    case 'venta_neta':
+      return [{ sql: `${ventaNeta}::float8`, alias: 'venta_neta' }];
+    case 'unidades':
+      if (grano === 'pago') noSoportada();
+
+      return [{ sql: 'g.a_units::float8', alias: 'unidades' }];
+    case 'tickets':
+      if (grano === 'pago') noSoportada();
+
+      return [{ sql: 'COALESCE(g.a_sgn, 0)::int', alias: 'tickets' }];
+    case 'ticket_promedio':
+      if (grano !== 'cabecera') noSoportada();
+
+      return [
+        {
+          sql: `(${ventaNeta} / NULLIF(g.a_sgn, 0))::float8`,
+          alias: 'ticket_promedio',
+        },
+      ];
+    case 'aur':
+      if (grano === 'pago') noSoportada();
+
+      return [
+        {
+          sql: '(g.a_items_total / NULLIF(g.a_units, 0))::float8',
+          alias: 'aur',
+        },
+      ];
+    case 'devoluciones':
+      if (!conDev) noSoportada();
+
+      return [{ sql: 'COALESCE(g.d_monto, 0)::float8', alias: 'devoluciones' }];
+    case 'margen': {
+      if (grano === 'pago') noSoportada();
+      const margen = atribuible
+        ? '((g.a_vcc - g.a_costo) - COALESCE(g.d_monto, 0) + COALESCE(g.d_costo, 0))'
+        : '(g.a_vcc - g.a_costo)';
+
+      return [
+        { sql: `${margen}::float8`, alias: 'margen' },
+        {
+          sql: `(${margen} / NULLIF(g.a_vcc, 0))::float8`,
+          alias: 'margen_pct',
+        },
+        {
+          sql: '(g.a_vcc / NULLIF(g.a_items_total, 0))::float8',
+          alias: 'cobertura_costo',
+        },
+      ];
+    }
+    default:
+      return noSoportada();
+  }
+}

+export function compilarPedido(
+  p: PedidoNormalizado,
+  ctx: { storeId: number; tz: string; modo: 'pantalla' | 'csv' },
+): ReporteCompilado {
+  const grano = p.grano;
+  const dims = p.groupBy;
+  const metricas = p.metricasEfectivas;
+  const atribuible = grano === 'cabecera' && p.devolucionesAtribuibles;
+  const conDev =
+    grano === 'cabecera' && (metricas.includes('devoluciones') || atribuible);
+
+  if (conDev && !dims.every((d) => DIMS_DEVOLUCION.includes(d))) {
+    throw new Error('devoluciones sólo se combinan por sucursal y tiempo');
+  }
+
+  // ---- bind: $1..$4 fijos + un int[] por filtro presente (orden de FILTROS) ----
+  const bind: unknown[] = [ctx.storeId, ctx.tz, p.desde, p.hasta];
+  const idxFiltro = new Map<FiltroId, number>();
+  for (const f of Object.keys(FILTROS) as FiltroId[]) {
+    const ids = p.filters?.[f];
+    if (ids && ids.length) {
+      bind.push(ids);
+      idxFiltro.set(f, bind.length);
+    }
+  }
+  const predicados = (permitidos?: FiltroId[]): string[] =>
+    [...idxFiltro.entries()]
+      .filter(([f]) => !permitidos || permitidos.includes(f))
+      .map(([f, n]) => `${PREDICADO_FILTRO[f]} = ANY($${n}::int[])`);
+
+  const necesitaIA =
+    grano === 'cabecera' &&
+    metricas.some((m) => m === 'unidades' || m === 'aur' || m === 'margen');
+
+  // ---- CTE compartidas ----
+  const ctes: string[] = [ventaHeaderCte(PH)];
+  if (necesitaIA) ctes.push(itemsAggCte({ firmado: true }));
+  if (conDev) ctes.push(devolucionesLocalCte(PH));
+  // Grano pago (CODEX 117-11): un pago con monto NULL no puede aportar el total COMPLETO del ticket
+  // (con 2 pagos null se contaba dos veces). Se reparte lo que falta (total − pagos con monto)
+  // en partes iguales entre los pagos null; sin pagos queda una fila con el total («Sin medio de
+  // pago»). Las ventanas se calculan ANTES de filtrar por medio de pago.
+  if (grano === 'pago')
+    ctes.push(`pagos AS (
+        SELECT v.sale_id, spm.payment_method_id,
+               COALESCE(spm.amount,
+                        (v.total_amount - COALESCE(SUM(spm.amount) OVER (PARTITION BY v.sale_id), 0))
+                        / NULLIF(COUNT(*) FILTER (WHERE spm.amount IS NULL) OVER (PARTITION BY v.sale_id), 0)
+               ) AS amount
+          FROM venta v
+          LEFT JOIN sale_payment_methods spm ON spm.sale_id = v.sale_id
+      )`);
+
+  // ---- FROM interno según grano ----
+  const joinsDim: string[] = [];
+  let from = 'FROM venta';
+  if (grano === 'cabecera') {
+    if (necesitaIA)
+      from += ' LEFT JOIN items_agg ia ON ia.sale_id = venta.sale_id';
+  } else if (grano === 'pago') {
+    from += ' JOIN pagos spm ON spm.sale_id = venta.sale_id';
+  } else {
+    from +=
+      ' JOIN sale_items si ON si.sale_id = venta.sale_id' +
+      ' LEFT JOIN products p ON p.id = si.product_id AND p.store_id = $1::int' +
+      ' LEFT JOIN products m ON m.id = p.parent_id AND m.store_id = $1::int';
+  }
+
+  const claves = dims.map((d) => {
+    if (DIMS_VENTA.includes(d)) return { dim: d, key: claveVenta(d, 'venta') };
+    if (d === 'medio_pago') return { dim: d, key: 'spm.payment_method_id' };
+    const ci = claveItem(d);
+    ci.joins.forEach((j) => !joinsDim.includes(j) && joinsDim.push(j));
+
+    return { dim: d, key: ci.key };
+  });
+
+  const aggs = agregados(grano, necesitaIA);
+  const where = predicados();
+
+  const selectInterno = (conClaves: boolean) => {
+    const ks = conClaves ? claves.map((c) => `${c.key} AS ${c.dim}_key`) : [];
+    const sel = [...ks, ...aggs].join(', ');
+    const w = where.length ? ` WHERE ${where.join(' AND ')}` : '';
+    const gb =
+      conClaves && claves.length
+        ? ` GROUP BY ${claves.map((c) => c.key).join(', ')}`
+        : '';
+
+    return `SELECT ${sel} ${from}${joinsDim.length && conClaves ? ' ' + joinsDim.join(' ') : ''}${w}${gb}`;
+  };
+
+  /** CTE `g`: una fila por grupo (o una sola) con los agregados crudos. */
+  const cteG = (conClaves: boolean): string[] => {
+    if (!conDev) return [`g AS (${selectInterno(conClaves)})`];
+    const ks = conClaves ? claves.map((c) => c.dim + '_key') : [];
+    const cols = [
+      'a_venta',
+      'a_sgn',
+      'a_units',
+      'a_items_total',
+      'a_costo',
+      'a_vcc',
+    ].filter((c) => aggs.some((a) => a.endsWith(` AS ${c}`)));
+    const nulos = (lista: string[]) =>
+      lista.map((c) => `NULL::double precision AS ${c}`);
+    const wDev = predicados(['sucursales']);
+    const ksDev = conClaves
+      ? claves.map(
+          (c) => `${claveVenta(c.dim, 'devoluciones')} AS ${c.dim}_key`,
+        )
+      : [];
+    const gbDev =
+      conClaves && claves.length
+        ? ` GROUP BY ${claves.map((c) => claveVenta(c.dim, 'devoluciones')).join(', ')}`
+        : '';
+    const wDevSql = wDev.length
+      ? ` WHERE ${wDev.map((x) => x.replace('venta.branch_id', 'devoluciones.branch_id')).join(' AND ')}`
+      : '';
+    const gv = `gv AS (${selectInterno(conClaves)})`;
+    const gd = `gd AS (SELECT ${[...ksDev, 'SUM(devoluciones.monto) AS d_monto', 'SUM(devoluciones.costo_devuelto) AS d_costo'].join(', ')} FROM devoluciones${wDevSql}${gbDev})`;
+    const sumas = [...cols, 'd_monto', 'd_costo'].map(
+      (c) => `SUM(${c}) AS ${c}`,
+    );
+    const uni =
+      `SELECT ${[...ks, ...cols, ...nulos(['d_monto', 'd_costo'])].join(', ')} FROM gv` +
+      ` UNION ALL SELECT ${[...ks, ...nulos(cols), 'd_monto', 'd_costo'].join(', ')} FROM gd`;
+    const gbG = conClaves && ks.length ? ` GROUP BY ${ks.join(', ')}` : '';
+    const g = `g AS (SELECT ${[...ks, ...sumas].join(', ')} FROM (${uni}) u${gbG})`;
+
+    return [gv, gd, g];
+  };
+
+  const salidas = metricas.flatMap((m) =>
+    expresionMetrica(m, grano, conDev, atribuible),
+  );
+  const aliasesSalida = salidas.map((s) => s.alias);
+
+  // ---- límite ----
+  const limiteModo =
+    ctx.modo === 'csv' ? LIMITES.csvMaxFilas : LIMITES.pantallaMaxFilas;
+  const topN =
+    p.top && p.top > 0 ? Math.min(Math.trunc(p.top), limiteModo) : limiteModo;
+
+  // ---- ORDER BY (alias de catálogo + desempate por claves) ----
+  const ordenPor = (): string => {
+    const piezas: string[] = [];
+    const by = p.sort?.by;
+    const dir = p.sort?.dir === 'asc' ? 'ASC' : 'DESC';
+    if (by && aliasesSalida.includes(by)) {
+      piezas.push(`${by} ${dir} NULLS LAST`);
+    } else if (by && (dims as string[]).includes(by)) {
+      const tiempo = DIMENSIONES[by as DimensionId].grupo === 'tiempo';
+      piezas.push(`${by}_${tiempo ? 'key' : 'label'} ${dir} NULLS LAST`);
+    } else if (
+      !dims.some((d) => DIMENSIONES[d].grupo === 'tiempo') &&
+      aliasesSalida.length
+    ) {
+      piezas.push(`${aliasesSalida[0]} DESC NULLS LAST`);
+    }
+    for (const d of dims) piezas.push(`${d}_key ASC NULLS LAST`);
+
+    return piezas.length ? ` ORDER BY ${piezas.join(', ')}` : '';
+  };
+
+  // ---- consulta principal ----
+  const lookups = dims.map((d) => etiqueta(d));
+  const colsDim = dims.flatMap((d, i) => [
+    `g.${d}_key AS ${d}_key`,
+    `${lookups[i].label} AS ${d}_label`,
+    ...(lookups[i].extra ? [lookups[i].extra] : []),
+  ]);
+  const joinsExt = [...new Set(lookups.flatMap((l) => l.joins))];
+  const selectExt = [
+    ...colsDim,
+    ...salidas.map((s) => `${s.sql} AS ${s.alias}`),
+  ].join(', ');
+  /**
+   * `claveFiltro` = índice del bind jsonb con las claves del período ACTUAL (sólo comparación):
+   * así el Top N de la comparación mira las mismas filas que se muestran (no su propio Top N).
+   */
+  const armarPrincipal = (
+    claveFiltro: { n: number; dims: DimensionId[] } | null,
+  ): string =>
+    `WITH ${[...ctes, ...cteG(true)].join(',\n      ')}\n` +
+    `SELECT ${selectExt} FROM g${joinsExt.length ? ' ' + joinsExt.join(' ') : ''}` +
+    (claveFiltro
+      ? ` WHERE jsonb_build_array(${claveFiltro.dims.map((d) => `g.${d}_key::text`).join(', ')})` +
+        ` IN (SELECT jsonb_array_elements($${claveFiltro.n}::jsonb))`
+      : '') +
+    `${ordenPor()} LIMIT ${(claveFiltro ? limiteModo : topN) + 1}`;
+  const sqlPrincipal = armarPrincipal(null);
+
+  // ---- totales: mismo universo, sin claves ni LIMIT ----
+  const sqlTotales =
+    `WITH ${[...ctes, ...cteG(false)].join(',\n      ')}\n` +
+    `SELECT ${salidas.map((s) => `${s.sql} AS ${s.alias}`).join(', ')} FROM g`;
+
+  // ---- fuentes (D-28) ----
+  const sqlFuentes = fuentesSql(PH);
+  const bindFuentes: unknown[] = [ctx.storeId, ctx.tz, p.desde, p.hasta];
+
+  const principal: ConsultaCompilada = {
+    sql: sqlPrincipal,
+    bind,
+    etiqueta: 'principal',
+  };
+  const totales: ConsultaCompilada = {
+    sql: sqlTotales,
+    bind,
+    etiqueta: 'totales',
+  };
+  const fuentes: ConsultaCompilada = {
+    sql: sqlFuentes,
+    bind: bindFuentes,
+    etiqueta: 'fuentes',
+  };
+
+  // ---- comparación: mismo SQL, fechas distintas en $3/$4 (D-31a) ----
+  let comparacion: ReporteCompilado['comparacion'] = null;
+  if (p.comparacion) {
+    const bindComp = [...bind];
+    bindComp[2] = p.comparacion.desde;
+    bindComp[3] = p.comparacion.hasta;
+    // Las dimensiones de tiempo con bucket (día/semana/mes) se alinean por posición en el post-proceso:
+    // sus claves cambian entre períodos. Todas las demás se buscan por la MISMA clave que el actual.
+    const dimsClave = dims.filter(
+      (d) => d !== 'dia' && d !== 'semana' && d !== 'mes',
+    );
+    const principalComp: ConsultaCompilada = dimsClave.length
+      ? {
+          sql: armarPrincipal({ n: bindComp.length + 1, dims: dimsClave }),
+          bind: [...bindComp, '[]'],
+          etiqueta: 'comp_principal',
+          completarBind: (previas) => {
+            const vistas = new Set<string>();
+            const claves: Array<Array<string | null>> = [];
+
+            for (const r of (previas.principal ?? []).slice(0, topN)) {
+              const k = dimsClave.map((d) => {
+                const v = r[`${d}_key`];
+
+                return v === null || v === undefined
+                  ? null
+                  : String(v as string | number);
+              });
+              const id = JSON.stringify(k);
+
+              if (!vistas.has(id)) {
+                vistas.add(id);
+                claves.push(k);
+              }
+            }
+
+            return [...bindComp, JSON.stringify(claves)];
+          },
+        }
+      : { sql: sqlPrincipal, bind: bindComp, etiqueta: 'comp_principal' };
+
+    // D-31a: si el período incluye hoy, el TOTAL se compara además sobre el rango completo (hasta ayer).
+    const comp = p.comparacion;
+    const rc = rangoCompletoHastaAyer(p.desde, p.hasta, p.hoy);
+    let comparable: NonNullable<ReporteCompilado['comparacion']>['comparable'] =
+      null;
+
+    if (rc.parcial && rc.hastaComparable) {
+      const bindAct = [...bind];
+      const bindAnt = [...bindComp];
+
+      bindAct[3] = rc.hastaComparable;
+      // el rango de comparación se recorta los mismos días que el actual
+      const recorte = diasEnRango(rc.hastaComparable, p.hasta) - 1;
+
+      bindAnt[3] = restarDias(comp.hasta, recorte);
+      comparable = {
+        hastaComparable: rc.hastaComparable,
+        actual: {
+          sql: sqlTotales,
+          bind: bindAct,
+          etiqueta: 'total_comparable',
+        },
+        anterior: {
+          sql: sqlTotales,
+          bind: bindAnt,
+          etiqueta: 'comp_total_comparable',
+        },
+      };
+    }
+
+    comparacion = {
+      principal: principalComp,
+      totales: { sql: sqlTotales, bind: bindComp, etiqueta: 'comp_totales' },
+      comparable,
+    };
+  }
+
+  // ---- columnas ----
+  const columnas: ColumnaDef[] = dims.map((d) => ({
+    id: d,
+    labelEs: DIMENSIONES[d].labelEs,
+    tipo: 'dimension',
+    formato: 'texto',
+  }));
+  for (const m of metricas) {
+    const def = METRICAS[m];
+    columnas.push({
+      id: m,
+      labelEs: def.labelEs,
+      tipo: 'metrica',
+      formato: def.formato,
+      definicionEs:
+        m === 'venta_neta' && grano === 'item'
+          ? `${def.definicionEs} Por producto, color, talle o categoría no se restan devoluciones.`
+          : def.definicionEs,
+    });
+    if (m === 'margen') {
+      columnas.push(
+        {
+          id: 'margen_pct',
+          labelEs: 'Margen %',
+          tipo: 'metrica',
+          formato: 'porcentaje',
+        },
+        {
+          id: 'cobertura_costo',
+          labelEs: 'Cobertura de costo',
+          tipo: 'metrica',
+          formato: 'porcentaje',
+          definicionEs: 'Parte de la venta que tiene costo cargado.',
+        },
+      );
+    }
+  }
+
+  return {
+    grano,
+    principal,
+    totales,
+    fuentes,
+    comparacion,
+    columnas,
+    limite: limiteModo,
+  };
+}

+ * sin separador de miles ni «$», fin de línea CRLF. Están como constantes para cambiarlas
+ * en un solo lugar.
+ *
+ * T-117-28: un texto que empieza con `= + - @ TAB CR` se prefija con apóstrofe para que
+ * Excel no lo ejecute como fórmula. Los NÚMEROS negativos no se prefijan (no son texto).
+ */
+
+export const CSV_BOM = '﻿';
+export const CSV_SEPARADOR = ';';
+export const CSV_DECIMAL = ',';
+export const CSV_EOL = '\r\n';
+
+// CODEX 117-11: también fórmulas precedidas por espacios/controles (incl. LF) y LF/TAB/CR iniciales.
+const PELIGROSO = /^(?:\s*[=+\-@]|[\t\r\n])/;
+
+function campo(valor: unknown): string {
+  if (valor === null || valor === undefined) return '';
+  let s: string;
+  if (typeof valor === 'number') {
+    s = Number.isFinite(valor) ? String(valor).replace('.', CSV_DECIMAL) : '';
+  } else {
+    s =
+      valor instanceof Date
+        ? valor.toISOString()
+        : String(valor as string | boolean | bigint);
+    if (PELIGROSO.test(s)) s = `'${s}`;
+  }
+  if (/[;"\r\n]/.test(s)) s = `"${s.replace(/"/g, '""')}"`;
+
+  return s;
+}
+
+export function toCsv(
+  columnas: Array<{ id: string; labelEs: string; formato?: string }>,
+  filas: Array<Record<string, unknown>>,
+): string {
+  const lineas = [columnas.map((c) => campo(c.labelEs)).join(CSV_SEPARADOR)];
+  for (const f of filas) {
+    lineas.push(columnas.map((c) => campo(f[c.id])).join(CSV_SEPARADOR));
+  }
+
+  return CSV_BOM + lineas.join(CSV_EOL) + CSV_EOL;

+    if (x && typeof x === 'object') {

+// ---------------------------------------------------------------- utilidades de fecha (UTC)
+const MS_DIA = 86_400_000;
+
+const aMs = (s: string): number =>
+  Date.UTC(
+    Number(s.slice(0, 4)),
+    Number(s.slice(5, 7)) - 1,
+    Number(s.slice(8, 10)),
+  );
+
+function deMs(ms: number): string {
+  const d = new Date(ms);
+
+  return `${String(d.getUTCFullYear()).padStart(4, '0')}-${String(d.getUTCMonth() + 1).padStart(2, '0')}-${String(d.getUTCDate()).padStart(2, '0')}`;
+}
+
+const lunesDe = (s: string): string => {
+  const dow = new Date(aMs(s)).getUTCDay();
+
+  return deMs(aMs(s) - ((dow + 6) % 7) * MS_DIA);
+};
+
+const mesIndice = (s: string): number =>
+  Number(s.slice(0, 4)) * 12 + Number(s.slice(5, 7)) - 1;
+
+/** Posición del bucket respecto del inicio del rango (alinea fila i ↔ fila i, sin depender de huecos). */
+function indiceBucket(dim: DimBucket, key: string, desde: string): number {
+  if (dim === 'dia') return Math.round((aMs(key) - aMs(desde)) / MS_DIA);
+
+  if (dim === 'semana')
+    return Math.round((aMs(lunesDe(key)) - aMs(lunesDe(desde))) / (7 * MS_DIA));
+
+  return mesIndice(key) - mesIndice(desde);
+}
+
+/** Inicio del bucket que está `idx` posiciones después del inicio del rango. */
+function claveBucketEn(dim: DimBucket, desde: string, idx: number): string {
+  if (dim === 'dia') return deMs(aMs(desde) + idx * MS_DIA);
+
+  if (dim === 'semana') return deMs(aMs(lunesDe(desde)) + 7 * idx * MS_DIA);
+
+  const total = mesIndice(desde) + idx;
+
+  return `${String(Math.floor(total / 12)).padStart(4, '0')}-${String((total % 12) + 1).padStart(2, '0')}-01`;
+}
+
+const esBucket = (d: DimensionId): d is DimBucket =>
+  d === 'dia' || d === 'semana' || d === 'mes';
+
+// ---------------------------------------------------------------- números y etiquetas
+function aNumero(v: unknown): number | null {
+  if (v === null || v === undefined || v === '') return null;
+
+  const n = typeof v === 'number' ? v : Number(v);
+
+  return Number.isFinite(n) ? n : null;
+}
+
+const aTexto = (v: unknown): string | null =>
+  v === null || v === undefined ? null : String(v as string | number);
+
+/** Etiquetas canónicas de medios de pago (los slugs de la base varían). */
+export function normalizarMedioPago(
+  slug: string | null,
+  title: string | null,
+): string {
+  const s = (slug ?? '')
+    .toLowerCase()
+    .normalize('NFD')
+    .replace(/[\u0300-\u036f]/g, '')
+    .replace(/[_\s]+/g, '-')
+    .trim();
+
+  if (s === 'tarjeta-credito' || s === 'tarjeta-de-credito')
+    return 'Tarjeta de crédito';
+
+  if (s === 'tarjeta-debito' || s === 'tarjeta-de-debito')
+    return 'Tarjeta de débito';
+
+  if (s === 'efectivo') return 'Efectivo';
+
+  const t = (title ?? '').trim();
+
+  if (t) return t;
+
+  return 'Sin medio de pago';
+}
+
+function vacioDe(d: DimensionId): string {
+  return (
+    DIMENSIONES[d].vacioEs ?? `Sin ${DIMENSIONES[d].labelEs.toLowerCase()}`
+  );
+}
+
+/** Fila cruda → fila de salida: una columna por dimensión (etiqueta) + `<dim>_key` + métricas numéricas. */
+function normalizarFila(
+  raw: Fila,
+  dims: DimensionId[],
+  idsMetricas: string[],
+): Fila {
+  const out: Fila = {};
+
+  for (const d of dims) {
+    const key = raw[`${d}_key`] ?? null;
+    const label = aTexto(raw[`${d}_label`]);
+
+    out[`${d}_key`] = key;
+
+    if (key === null) out[d] = vacioDe(d);
+    else if (d === 'medio_pago')
+      out[d] = normalizarMedioPago(aTexto(raw.medio_pago_slug), label);
+    else out[d] = label ?? String(key as string | number);
+  }
+
+  for (const id of idsMetricas) out[id] = aNumero(raw[id]);
+
+  return out;
+}
+
+function varPct(actual: number | null, comp: number | null): number | null {
+  if (actual === null || comp === null || comp === 0) return null;
+
+  return (actual - comp) / comp;
+}
+
+// ---------------------------------------------------------------- días sin ventas
+const METRICAS_EN_CERO = new Set<string>([
+  'venta_neta',
+  'unidades',
+  'tickets',
+  'devoluciones',
+  'margen',
+]);
+
+function rellenarDias(
+  filas: Fila[],
+  dims: DimensionId[],
+  desde: string,
+  hasta: string,
+  idsMetricas: string[],
+  sort: PedidoNormalizado['sort'],
+  limite: number,
+): Fila[] {
+  if (!dims.includes('dia')) return filas;
+
+  if (!dims.every((d) => d === 'dia' || d === 'sucursal')) return filas;
+
+  if (sort?.by && sort.by !== 'dia') return filas;
+
+  const dias = diasEnRango(desde, hasta);
+
+  if (dias > LIMITES.diaMaxDias) return filas;
+
+  const conSucursal = dims.includes('sucursal');
+  const combos = new Map<string, Fila>();
+
+  if (conSucursal) {
+    for (const f of filas) {
+      const k = String((f.sucursal_key as string | number | null) ?? 'null');
+
+      if (!combos.has(k)) combos.set(k, f);
+    }
+  } else {
+    combos.set('', {});
+  }
+
+  if (combos.size === 0 || dias * combos.size > limite) return filas;
+
+  const existentes = new Map<string, Fila>();
+
+  for (const f of filas)
+    existentes.set(
+      `${String(f.dia_key as string)}|${conSucursal ? String((f.sucursal_key as string | number | null) ?? 'null') : ''}`,
+      f,
+    );
+
+  const listaDias: string[] = [];
+
+  for (let i = 0; i < dias; i++) listaDias.push(deMs(aMs(desde) + i * MS_DIA));
+
+  if (sort?.dir === 'desc') listaDias.reverse();
+
+  const out: Fila[] = [];
+
+  for (const dia of listaDias) {
+    for (const [k, molde] of combos) {
+      const hay = existentes.get(`${dia}|${k}`);
+
+      if (hay) {
+        out.push(hay);
+        continue;
+      }
+
+      const vacia: Fila = { dia, dia_key: dia };
+
+      if (conSucursal) {
+        vacia.sucursal = molde.sucursal;
+        vacia.sucursal_key = molde.sucursal_key;
+      }
+
+      for (const id of idsMetricas)
+        vacia[id] = METRICAS_EN_CERO.has(id) ? 0 : null;
+
+      out.push(vacia);
+    }
+  }
+
+  return out;
+}

+// ---------------------------------------------------------------- orden de talles
+function ordenarPorTalle(filas: Fila[], dims: DimensionId[]): Fila[] {
+  const cmp = (a: Fila, b: Fila): number => {
+    for (const d of dims) {
+      let r: number;
+
+      if (d === 'talle') {
+        r = compararTalles(
+          a.talle_key === null ? null : String(a.talle as string),
+          b.talle_key === null ? null : String(b.talle as string),
+        );
+      } else {
+        r = String(a[d] as string).localeCompare(
+          String(b[d] as string),
+          'es-AR',
+          {
+            numeric: true,
+          },
+        );
+      }
+
+      if (r !== 0) return r;
+    }
+
+    return 0;
+  };
+
+  return [...filas].sort(cmp);
+}

+// ---------------------------------------------------------------- principal
+export function postProcesar(input: {
+  pedido: PedidoNormalizado;
+  compilado: ReporteCompilado;
+  brutos: Record<string, Record<string, unknown>[]>;
+  resumenEs: string;
+}): ResultadoReporte {
+  const { pedido, compilado, brutos, resumenEs } = input;
+  const dims = pedido.groupBy;
+  const efectivas: MetricaId[] = pedido.metricasEfectivas;
+  const idsMetricas = compilado.columnas
+    .filter((c) => c.tipo === 'metrica')
+    .map((c) => c.id);
+
+  // ---- truncamiento: el SQL pide límite+1 para detectar el exceso ----
+  const top =
+    pedido.top && pedido.top > 0
+      ? Math.min(Math.trunc(pedido.top), compilado.limite)
+      : null;
+  const tope = top ?? compilado.limite;
+  const crudas = brutos.principal ?? [];
+  // Un Top N pedido es un corte deliberado: no es truncamiento.
+  const truncado = top === null && crudas.length > tope;
+
+  let filas = crudas
+    .slice(0, tope)
+    .map((r) => normalizarFila(r, dims, idsMetricas));
+
+  if (!truncado) {
+    filas = rellenarDias(
+      filas,
+      dims,
+      pedido.desde,
+      pedido.hasta,
+      idsMetricas,
+      pedido.sort,
+      compilado.limite,
+    );
+  }
+
+  if (dims.includes('talle') && !pedido.sort?.by)
+    filas = ordenarPorTalle(filas, dims);
+
+  const totales = normalizarFila(brutos.totales?.[0] ?? {}, [], idsMetricas);
+
+  // ---- columnas ----
+  const columnas: ColumnaDef[] = [...compilado.columnas];
+
+  // ---- participación: el denominador es el TOTAL, nunca la suma de las filas mostradas ----
+  if (pedido.participacion) {
+    for (const m of ['venta_neta', 'unidades'] as const) {
+      if (!efectivas.includes(m)) continue;
+
+      const total = aNumero(totales[m]);
+
+      for (const f of filas) {
+        const v = aNumero(f[m]);
+
+        f[`${m}_pct`] =
+          v === null || total === null || total === 0 ? null : v / total;
+      }
+
+      totales[`${m}_pct`] = total === null || total === 0 ? null : 1;
+      columnas.push({
+        id: `${m}_pct`,
+        labelEs: `${m === 'venta_neta' ? 'Venta neta' : 'Unidades'} %`,
+        tipo: 'metrica',
+        formato: 'porcentaje',
+        definicionEs: 'Parte del total del período (no del Top N mostrado).',
+      });
+    }
+  }
+
+  // ---- comparación ----
+  let comparacion: ResultadoReporte['comparacion'] = null;
+  let hayParcial = false;
+
+  if (pedido.comparacion && compilado.comparacion) {
+    const comp = pedido.comparacion;
+    const bucketDims = dims.filter(esBucket);
+    // La comparación ya viene filtrada por las claves del período actual (compilador): no se recorta
+    // por el Top N sino por el límite del modo.
+    const compFilasBase = (brutos.comp_principal ?? [])
+      .slice(0, compilado.limite)
+      .map((r) => normalizarFila(r, dims, idsMetricas));
+    const compFilas = rellenarDias(
+      compFilasBase,
+      dims,
+      comp.desde,
+      comp.hasta,
+      idsMetricas,
+      pedido.sort,
+      compilado.limite,
+    );
+    const compTotales = normalizarFila(
+      brutos.comp_totales?.[0] ?? {},
+      [],
+      idsMetricas,
+    );
+
+    const claveCompuesta = (f: Fila, desde: string): string =>
+      dims
+        .map((d) =>
+          esBucket(d)
+            ? `#${indiceBucket(d, String(f[`${d}_key`] as string), desde)}`
+            : String((f[`${d}_key`] as string | number | null) ?? 'null'),
+        )
+        .join('|');
+
+    const porClave = new Map<string, Fila>();
+
+    for (const f of compFilas) porClave.set(claveCompuesta(f, comp.desde), f);
+
+    for (const f of filas) {
+      const par = porClave.get(claveCompuesta(f, pedido.desde));
+      // D-31/D-31a: bucket incompleto (o con el día en curso) → sin variación.
+      const parcial = bucketDims.some((d) => {

+        if (bucketContieneHoy(d, key, pedido.hoy)) return true;

+          d,
+          comp.desde,
+          indiceBucket(d, key, pedido.desde),
+        );
+
+        return !bucketCompleto(d, contraparte, comp.desde, comp.hasta);
+      });
+
+      f.parcial = parcial;
+
+      if (parcial) hayParcial = true;
+
+      for (const m of efectivas) {
+        const anterior = par ? aNumero(par[m]) : null;
+
+        f[`${m}_comp`] = anterior;
+        f[`${m}_var_pct`] = parcial ? null : varPct(aNumero(f[m]), anterior);
+      }
+    }
+
+    for (const m of efectivas) {
+      const base = compilado.columnas.find((c) => c.id === m);
+
+      columnas.push(
+        {
+          id: `${m}_comp`,
+          labelEs: `${base?.labelEs ?? m} (comparación)`,
+          tipo: 'metrica',
+          formato: base?.formato,
+        },

+          labelEs: `${base?.labelEs ?? m} var. %`,
+          tipo: 'metrica',
+          formato: 'porcentaje',
+        },
+      );
+    }
+
+    const variacionTotal: NonNullable<
+      ResultadoReporte['comparacion']
+    >['variacionTotal'] = {};
+
+    const rc = rangoCompletoHastaAyer(pedido.desde, pedido.hasta, pedido.hoy);
+    const totActual = normalizarFila(
+      brutos.total_comparable?.[0] ?? {},
+      [],
+      idsMetricas,
+    );
+    const totAnterior = normalizarFila(
+      brutos.comp_total_comparable?.[0] ?? {},
+      [],
+      idsMetricas,
+    );
+
+    for (const m of ['venta_neta', 'unidades'] as const) {
+      if (!efectivas.includes(m)) continue;
+
+      if (!rc.parcial) {
+        const actual = aNumero(totales[m]) ?? 0;
+        const anterior = aNumero(compTotales[m]) ?? 0;
+
+        variacionTotal[m] = {
+          actual,
+          anterior,
+          varPct: varPct(actual, anterior),
+          parcial: false,
+          hastaComparable: pedido.hasta,
+        };
+      } else if (rc.hastaComparable) {
+        const actual = aNumero(totActual[m]) ?? 0;
+        const anterior = aNumero(totAnterior[m]) ?? 0;
+
+        variacionTotal[m] = {
+          actual,
+          anterior,
+          varPct: varPct(actual, anterior),
+          parcial: true,
+          hastaComparable: rc.hastaComparable,
+        };
+      } else {
+        variacionTotal[m] = {
+          actual: null,
+          anterior: null,
+          varPct: null,
+          parcial: true,
+          hastaComparable: null,
+        };
+      }
+    }
+
+    comparacion = {
+      desde: comp.desde,
+      hasta: comp.hasta,
+      filas: compFilas,
+      totales: compTotales,
+      variacionTotal,
+    };
+  }
+
+  // ---- fuentes (D-28) ----
+  const fuentes: Record<string, number> = { legacy: 0, pos: 0 };
+
+  for (const r of brutos.fuentes ?? []) {
+    const nombre = aTexto(r.source);
+
+    if (nombre) fuentes[nombre] = aNumero(r.ventas) ?? 0;
+  }
+
+  // ---- notas ----
+  const notasEs: string[] = [];
+
+  if (fuentes.legacy > 0)
+    notasEs.push(
+      `Incluye ${fuentes.legacy} ventas importadas del sistema anterior`,
+    );
+
+  if (!pedido.devolucionesAtribuibles && efectivas.includes('venta_neta'))
+    notasEs.push(
+      'Las devoluciones no se restan en esta vista (sólo por sucursal y tiempo)',
+    );
+
+  if (efectivas.includes('margen')) {
+    const cob = aNumero(totales.cobertura_costo);
+
+    if (cob !== null)
+      notasEs.push(
+        `Margen sobre líneas con costo cargado (${Math.round(cob * 100)} % de la venta)`,
+      );
+  }
+
+  if (hayParcial)
+    notasEs.push(
+      'Los períodos incompletos (semana o mes cortados, o el día en curso) no muestran variación.',
+    );
+
+  return {
+    columnas,
+    filas,
+    totales,
+    comparacion,
+    meta: {
+      grano: compilado.grano,
+      desde: pedido.desde,
+      hasta: pedido.hasta,
+      truncado,
+      filas: filas.length,
+      fuentes,
+      omitidas: pedido.omitidas,
+      resumenEs,
+      definicionVentaEs: VENTA_DEFINICION_ES,
+      catalogVersion: CATALOG_VERSION,
+      notasEs,
+    },
+  };
+}

+/**
+ * Orden de talles para el reporte a medida (D-23, PEDIDO 40 «Ordenar talles por número»).
+ *
+ * `sizes` no tiene columna de orden y los nombres reales mezclan números («38», «1.2MM»),
+ * letras («S», «XL», «2XL»), series «N1…N12» y texto libre («Talle Única», «CHICO»).
+ * Regla (grupos, en este orden) — la MISMA que la app (`ventago-app/src/utils/orden-talles.ts`,
+ * PEDIDO 40 ya desplegado): letras primero, después números. Un talle no puede salir en otro
+ * orden en el reporte que en la ficha de producto.
+ *   0. escalera de letras XXS, XS, S, M, L, XL, XXL (=2XL), XXXL (=3XL), 4XL
+ *   1. numéricos ascendentes (incluye «TALLE 1» → 1 y «1.2MM» → 1.2)
+ *   2. serie N por número (N1, N2, N12)
+ *   3. el resto alfabético sin mayúsculas ni acentos, números naturales
+ *   4. sin talle (null) al final
+ */
+
+const ESCALERA = ['XXS', 'XS', 'S', 'M', 'L', 'XL', 'XXL', 'XXXL', '4XL'];
+const ALIAS_LETRA: Record<string, string> = { '2XL': 'XXL', '3XL': 'XXXL' };
+
+function grupo(t: string): { g: number; n: number } {
+  const s = t.trim();
+  const up = s.toUpperCase();
+  const letra = ESCALERA.indexOf(ALIAS_LETRA[up] ?? up);
+  if (letra >= 0) return { g: 0, n: letra };
+  const num = /^(?:TALLE\s*)?(\d+(?:[.,]\d+)?)(?:\s*MM)?$/.exec(up);
+  if (num) return { g: 1, n: Number(num[1].replace(',', '.')) };
+  const serie = /^N\s*(\d+)$/.exec(up);
+  if (serie) return { g: 2, n: Number(serie[1]) };
+
+  return { g: 3, n: 0 };
+}
+
+export function compararTalles(a: string | null, b: string | null): number {
+  if (a == null && b == null) return 0;
+  if (a == null) return 1;
+  if (b == null) return -1;
+  const x = grupo(a);
+  const y = grupo(b);
+  if (x.g !== y.g) return x.g - y.g;
+  if (x.g !== 3 && x.n !== y.n) return x.n - y.n;
+
+  return a.localeCompare(b, 'es-AR', { numeric: true, sensitivity: 'base' });
+}

// Clases de lock (primer int de la forma de dos ints). Elegidas fuera del alcance de un store id.
export const LOCK_CLASE_TIENDA = 1170001;
export const LOCK_CLASE_USUARIO = 1170002;
export const LOCK_CLASE_GLOBAL = 1170003;

const RETRY_AFTER_MS = 2000;
const GLOBAL_MAX_DEFAULT = 4;
const GLOBAL_MAX_MIN = 1;
const GLOBAL_MAX_MAX = 6;

/** REPORTE_MEDIDA_GLOBAL_MAX: entero 1..6; inválido o ausente → 4; fuera de rango → se recorta. */
export function globalMax(
  raw: string | undefined = process.env.REPORTE_MEDIDA_GLOBAL_MAX,
): number {
  if (raw === undefined || !/^\d+$/.test(raw.trim())) return GLOBAL_MAX_DEFAULT;
  const n = Number(raw.trim());
  if (n === 0) return GLOBAL_MAX_DEFAULT;

  return Math.min(GLOBAL_MAX_MAX, Math.max(GLOBAL_MAX_MIN, n));
}

function ocupado(code: string, messageEs: string): HttpException {
  return new HttpException(
    { code, messageEs, retryAfterMs: RETRY_AFTER_MS },
    429,
  );
}

/** Traduce cualquier error de PG a HttpException sin exponer SQL ni binds. */
export function mapearErrorPg(e: unknown): HttpException {
  if (e instanceof HttpException) return e;
  const err = e as {
    original?: { code?: string };
    parent?: { code?: string };
    code?: string;
  };
  const pgCode = err?.original?.code ?? err?.parent?.code ?? err?.code;
  if (pgCode === '57014') {
    return new HttpException(
      {
        code: 'CONSULTA_DEMASIADO_LARGA',
        messageEs:
          'La consulta superó 5 s (20 s para CSV). Achicá el período, agregá un filtro o un Top N.',
      },
      408,
    );
  }

  return new InternalServerErrorException({
    code: 'ERROR_CONSULTA',
    messageEs: 'No se pudo ejecutar la consulta.',
  });
}

@Injectable()
export class ReporteMedidaExecutorService {
  private readonly logger = new Logger(ReporteMedidaExecutorService.name);

  constructor(@InjectConnection() private readonly sequelize: Sequelize) {}

  async ejecutar(
    ctx: ContextoEjecucion,
    consultas: ConsultaCompilada[],
  ): Promise<Record<string, Record<string, unknown>[]>> {
    const timeoutMs =
      ctx.modo === 'csv' ? LIMITES.timeoutCsvMs : LIMITES.timeoutPantallaMs;
    if (!Number.isInteger(timeoutMs) || timeoutMs <= 0) {
      throw new InternalServerErrorException({ code: 'ERROR_CONSULTA' });
    }
    const slots = globalMax();
    const t0 = Date.now();
    let filas = 0;

    try {
      const out = await this.sequelize.transaction(async (t) => {
        await this.sequelize.query(
          `SET TRANSACTION READ ONLY; SET LOCAL statement_timeout = '${timeoutMs}ms'; SET LOCAL idle_in_transaction_session_timeout = '${timeoutMs}ms'`,
          { transaction: t },
        );

        const tryLock = async (
          clase: number,
          clave: number,
        ): Promise<boolean> => {
          const rows = await this.sequelize.query<{ ok: boolean }>(
            'SELECT pg_try_advisory_xact_lock($1::int, $2::int) AS ok',
            { bind: [clase, clave], type: QueryTypes.SELECT, transaction: t },
          );

          return rows[0]?.ok === true;
        };

        if (!(await tryLock(LOCK_CLASE_TIENDA, ctx.storeId))) {
          throw ocupado(
            'TIENDA_OCUPADA',
            'Ya hay un reporte corriendo para esta tienda. Probá de nuevo en unos segundos.',
          );
        }
        if (!(await tryLock(LOCK_CLASE_USUARIO, ctx.userId))) {
          throw ocupado(
            'USUARIO_OCUPADO',
            'Ya tenés un reporte corriendo. Esperá a que termine.',
          );
        }
        let slot = false;
        for (let i = 0; i < slots && !slot; i++) {
          slot = await tryLock(LOCK_CLASE_GLOBAL, i);
        }
        if (!slot) {
          throw ocupado(
            'SERVIDOR_OCUPADO',
            'El servidor está procesando otros reportes. Probá de nuevo en unos segundos.',
          );
        }

        const res: Record<string, Record<string, unknown>[]> = {};
        for (const c of consultas) {
          // El tope es de la CORRIDA, no de cada consulta (hallazgo CODEX 117-05):
          // cada consulta recibe sólo el tiempo que queda.
          const restante = timeoutMs - (Date.now() - t0);
          if (restante <= 0) {
            throw mapearErrorPg({ code: '57014' });
          }
          await this.sequelize.query(
            `SET LOCAL statement_timeout = '${Math.max(1, Math.floor(restante))}ms'`,
            { transaction: t },
          );
          const rows = await this.sequelize.query<Record<string, unknown>>(
            c.sql,
            {
              bind: (c.completarBind ? c.completarBind(res) : c.bind) as never,
              type: QueryTypes.SELECT,
              transaction: t,
            },
          );
          res[c.etiqueta] = rows;
          filas += rows.length;
        }

        return res;
      });

      this.logger.log(
        JSON.stringify({
          storeId: ctx.storeId,
          userId: ctx.userId,
          pedidoHash: ctx.pedidoHash,
          grano: ctx.grano,
          modo: ctx.modo,
          ms: Date.now() - t0,
          filas,
        }),
      );

      return out;
    } catch (e) {
      throw mapearErrorPg(e);
    }
  }
}
/**
 * [Phase 117-01] «Venta» — la definición única del universo de ventas de «Reporte a medida».
 *
 * Todo número del reporte a medida sale de estas CTE. Phase 105 §1 ya había centralizado las
 * constantes (`sale-status.constants.ts`); aquí se componen en UN solo fragmento SQL para que
 * ningún query del reporte a medida repita la regla (D-01).
 *
 * Reglas (cada una con su origen):
 *  · `s.store_id = <store>` va PRIMERO en el WHERE — aislamiento de tenant (T-117-01).
 *  · `activity_type = 'sale'` (SOLO_VENTA_SQL) — movido / devuelto / fallado son filas de `sales`
 *    con status 'Pagado' y cantidad > 0; contarlas rompió los reportes (E3 de Phase 105).
 *  · status ∈ ACCOUNTING_SALE_STATUSES — «Anulado» (original, +) y «Anulación» (reverso, −) se
 *    cuentan CADA UNO en su día con su signo (`sgn`); así no se mueve un día ya cerrado.
 *  · `credit_payment_id IS NULL` (EXCLUDE_DEUDA_PAGO_SQL) — un cobro de deuda no es venta.
 *  · Devengado (D-26): la venta cuenta el día en que se hizo, aunque haya quedado en crédito.
 *  · Día local de la tienda: COALESCE(sale_day_local, sale_date AT TIME ZONE tz). Una venta de
 *    las 23:30 en Buenos Aires cae en el día de Buenos Aires, no en el de UTC.
 *  · SIN predicado sobre `source` (D-17/D-28): ventas importadas del sistema legacy y ventas POS
 *    del mismo día se cuentan JUNTAS (unión de fuentes, sin deduplicar). El desglose por origen
 *    (`fuentesSql`) y el monitor de posibles duplicados (`posiblesDuplicadosSql`) hacen visible
 *    cualquier doble conteo.
 *
 * ★ Los builders SOLO insertan los placeholders recibidos (`$1::int`, …). Ningún valor de usuario
 *   viaja en el texto SQL (T-117-02); los estados salen de constantes del servidor.
 */
import {
  ACCOUNTING_SALE_STATUSES,
  EXCLUDE_DEUDA_PAGO_SQL,
  saleBranchSql,
  SOLO_VENTA_SQL,
} from './sale-status.constants';

/** Placeholders de bind (nunca valores). Ej.: `{ store: '$1::int', tz: '$2::text', from: '$3::date', to: '$4::date' }`. */
export type VentaPlaceholders = {
  store: string;
  tz: string;
  from: string;
  to: string;

/** Constantes del servidor (no son entrada de usuario) como lista de literales SQL. */
const estadosSql = ACCOUNTING_SALE_STATUSES.map((x) => `'${x}'`).join(', ');

/** Día local de la tienda para el alias dado. */
const diaLocalSql = (alias: string, tz: string) =>
  `COALESCE(${alias}.sale_day_local, (${alias}.sale_date AT TIME ZONE ${tz})::date)`;

/**
 * `venta AS (...)` — una fila por venta contable del universo.
 * Columnas: sale_id, status, sgn, total_amount, seller_id, source, branch_id, dia, hora, sale_date.
 */
export const ventaHeaderCte = (p: VentaPlaceholders): string => `venta AS (
        SELECT s.id AS sale_id,
               s.status,
               CASE WHEN s.status = 'Anulación' THEN -1 ELSE 1 END AS sgn,
               s.total_amount,
               s.seller_id,
               s.source,
               ${saleBranchSql('s')} AS branch_id,
               ${diaLocalSql('s', p.tz)} AS dia,
               EXTRACT(HOUR FROM (s.sale_date AT TIME ZONE ${p.tz}))::int AS hora,
               s.sale_date
          FROM sales s
         WHERE s.store_id = ${p.store}
           AND ${SOLO_VENTA_SQL('s')}
           AND s.status IN (${estadosSql})
           AND ${EXCLUDE_DEUDA_PAGO_SQL('s')}
           -- rango grueso por índice (±1 día cubre cualquier huso); el exacto va abajo
           AND s.sale_date >= (${p.from} - 1)
           AND s.sale_date <  (${p.to} + 2)
           AND ${diaLocalSql('s', p.tz)} BETWEEN ${p.from} AND ${p.to}
      )`;

/**
 * Valor de una línea con el SIGNO DEL MONTO de su venta (Phase 117-13).
 *
 * En «Anulación» el monto (`total_amount`) es negativo pero la línea puede traer `quantity`
 * (y `subtotal`) POSITIVOS — medido en local: 128 de 11.624 líneas con cantidad positiva y 271 con
 * subtotal no negativo. Sumarlas tal cual HACE SUBIR unidades y costo en vez de compensar la
 * venta anulada. `-ABS(x)` fuerza el signo sin doblarlo cuando el dato ya viene negativo.
 * `sgn` es la columna `venta.sgn` (−1 para Anulación).
 */
export const firmadoSql = (sgn: string, expr: string): string =>
  `CASE WHEN ${sgn} = -1 THEN -ABS(${expr}) ELSE ${expr} END`;

/**
 * `items_agg AS (...)` — una fila por venta (guardia contra fan-out: se pre-agrega antes de unir).
 * Lee la CTE `venta`. Columnas: sale_id, units, items_total, costo, venta_con_costo.
 *
 * `firmado: true` (sólo Reporte a medida) aplica el signo del monto a unidades, costo y bases
 * de línea. El valor por defecto es el comportamiento histórico, que comparte Ganancia —
 * reconciliarlo es de Phase 105.
 */
export const itemsAggCte = (opts: { firmado?: boolean } = {}): string => {
  if (!opts.firmado) {
    return `items_agg AS (
        SELECT si.sale_id,
               SUM(si.quantity) AS units,
               SUM(si.subtotal) AS items_total,
               SUM(si.quantity * si.unit_cost) FILTER (WHERE si.unit_cost IS NOT NULL) AS costo,
               SUM(si.subtotal) FILTER (WHERE si.unit_cost IS NOT NULL) AS venta_con_costo
          FROM sale_items si
         WHERE si.sale_id IN (SELECT sale_id FROM venta)
         GROUP BY si.sale_id
      )`;
  }
  const qty = firmadoSql('v.sgn', 'si.quantity');
  const sub = firmadoSql('v.sgn', 'si.subtotal');

  return `items_agg AS (
        SELECT si.sale_id,
               SUM(${qty}) AS units,
               SUM(${sub}) AS items_total,
               SUM(${firmadoSql('v.sgn', 'si.quantity * si.unit_cost')}) FILTER (WHERE si.unit_cost IS NOT NULL) AS costo,
               SUM(${sub}) FILTER (WHERE si.unit_cost IS NOT NULL) AS venta_con_costo
          FROM sale_items si
          JOIN venta v ON v.sale_id = si.sale_id
         GROUP BY si.sale_id
      )`;
};

 * a ≤ 300 s de distancia y con EXACTAMENTE el mismo multiconjunto de ítems (producto × cantidad).
 * No borra ni corrige nada: sólo hace visible un doble conteo posible.
 * Columnas: legacy_id, pos_id, total, delta_seconds.
 */
export const posiblesDuplicadosSql = (storePlaceholder: string): string => `
        WITH cand AS (
          SELECT s.id, s.source, s.total_amount, s.sale_date,
                 (SELECT string_agg(x.product_id || 'x' || x.q, ',' ORDER BY x.product_id, x.q)
                    FROM (SELECT si.product_id, SUM(si.quantity) AS q
                            FROM sale_items si
                           WHERE si.sale_id = s.id
                           GROUP BY si.product_id) x) AS firma
            FROM sales s
           WHERE s.store_id = ${storePlaceholder}
             AND ${SOLO_VENTA_SQL('s')}
             AND s.status IN (${estadosSql})
             AND ${EXCLUDE_DEUDA_PAGO_SQL('s')}
             AND s.source IN ('legacy', 'pos')
        )
        SELECT l.id AS legacy_id,
               p.id AS pos_id,
               l.total_amount AS total,
               ABS(EXTRACT(EPOCH FROM (l.sale_date - p.sale_date)))::int AS delta_seconds
          FROM cand l
          JOIN cand p
            ON p.source = 'pos'
           AND l.total_amount = p.total_amount
           AND ABS(EXTRACT(EPOCH FROM (l.sale_date - p.sale_date))) <= 300
           AND l.firma IS NOT NULL
           AND l.firma = p.firma
         WHERE l.source = 'legacy'
         ORDER BY l.id, p.id`;

// Reporte a medida — resolución de períodos y períodos de comparación (D-14, D-21, D-31, D-31a).
// Funciones PURAS sobre cadenas 'YYYY-MM-DD'. Sólo se usan métodos getUTC* y Date.UTC:
// el resultado no depende de la zona horaria del proceso.
import type { DimensionId, PeriodoPedido } from './order.types';

const MS_DIA = 86_400_000;

export class ErrorPeriodo extends Error {
  readonly code = 'PERIODO_INVALIDO';

  constructor(mensaje: string) {
    super(mensaje);
    this.name = 'ErrorPeriodo';
  }
}

type Ymd = { y: number; m: number; d: number };

function diasDelMes(y: number, m: number): number {
  return new Date(Date.UTC(y, m, 0)).getUTCDate();
}

function parse(s: string): Ymd {
  const r = /^(\d{4})-(\d{2})-(\d{2})$/.exec(s ?? '');

  if (!r) throw new ErrorPeriodo(`Fecha inválida: ${s}`);

  const y = Number(r[1]);
  const m = Number(r[2]);
  const d = Number(r[3]);

  if (m < 1 || m > 12 || d < 1 || d > diasDelMes(y, m))
    throw new ErrorPeriodo(`Fecha inválida: ${s}`);

  return { y, m, d };
}

function fmt({ y, m, d }: Ymd): string {
  return `${String(y).padStart(4, '0')}-${String(m).padStart(2, '0')}-${String(d).padStart(2, '0')}`;
}

function toMs(s: string): number {
  const { y, m, d } = parse(s);

  return Date.UTC(y, m - 1, d);
}

function fromMs(ms: number): string {
  const dt = new Date(ms);

  return fmt({
    y: dt.getUTCFullYear(),
    m: dt.getUTCMonth() + 1,
    d: dt.getUTCDate(),
  });
}

function sumarDias(s: string, n: number): string {
  return fromMs(toMs(s) + n * MS_DIA);
}

/** Suma meses (puede ser negativo); el día se recorta al fin del mes de destino. */
function sumarMeses(s: string, n: number): string {
  const { y, m, d } = parse(s);
  const total = y * 12 + (m - 1) + n;
  const ny = Math.floor(total / 12);
  const nm = (total % 12) + 1;

  return fmt({ y: ny, m: nm, d: Math.min(d, diasDelMes(ny, nm)) });
}

function esUltimoDiaDelMes(s: string): boolean {
  const { y, m, d } = parse(s);

  return d === diasDelMes(y, m);
}

function finDeMes(s: string): string {
  const { y, m } = parse(s);

  return fmt({ y, m, d: diasDelMes(y, m) });
}

function inicioDeMes(s: string): string {
  const { y, m } = parse(s);

  return fmt({ y, m, d: 1 });
}

/** Lunes (ISO) de la semana que contiene el día. */
function lunesDe(s: string): string {
  const dow = new Date(toMs(s)).getUTCDay(); // 0 = domingo

  return sumarDias(s, -((dow + 6) % 7));
}

/** Días del rango, inclusivo. */
export function diasEnRango(desde: string, hasta: string): number {
  return Math.round((toMs(hasta) - toMs(desde)) / MS_DIA) + 1;
}

/**
 * Resuelve el período contra el día de ejecución (D-14).
 * `hoy` debe venir del reloj de la base en la zona de la tienda
 * (`(now() AT TIME ZONE tz)::date`), nunca de un Date de JS.
 */
export function resolvePeriodo(
  p: PeriodoPedido,
  hoy: string,
): { desde: string; hasta: string } {
  toMs(hoy);

  if ('from' in p) {
    const desde = fmt(parse(p.from));
    const hasta = fmt(parse(p.to));

    if (desde > hasta)
      throw new ErrorPeriodo('La fecha inicial es posterior a la final.');

    return { desde, hasta };
  }

  switch (p.preset) {
    case 'hoy':
      return { desde: hoy, hasta: hoy };
    case 'ayer': {
      const a = sumarDias(hoy, -1);

      return { desde: a, hasta: a };
    }
    case 'esta_semana':
      return { desde: lunesDe(hoy), hasta: hoy };
    case 'semana_pasada': {
      const l = lunesDe(hoy);

      return { desde: sumarDias(l, -7), hasta: sumarDias(l, -1) };
    }
    case 'este_mes':
      return { desde: inicioDeMes(hoy), hasta: hoy };
    case 'mes_pasado': {
      const ini = sumarMeses(inicioDeMes(hoy), -1);

      return { desde: ini, hasta: finDeMes(ini) };
    }
    case 'ultimos_7_dias':
      return { desde: sumarDias(hoy, -6), hasta: hoy };
    case 'ultimos_30_dias':
      return { desde: sumarDias(hoy, -29), hasta: hoy };
    case 'ultimos_90_dias':
      return { desde: sumarDias(hoy, -89), hasta: hoy };
    case 'ultimos_12_meses':
      return { desde: sumarDias(sumarMeses(hoy, -12), 1), hasta: hoy };
    default:
      throw new ErrorPeriodo('Período desconocido.');
  }
}

/** D-31: calendario = este_mes | mes_pasado; todo lo demás es rodante. */
export function clasePeriodo(p: PeriodoPedido): 'rodante' | 'calendario' {
  return 'preset' in p && (p.preset === 'este_mes' || p.preset === 'mes_pasado')
    ? 'calendario'
    : 'rodante';
}

/**
 * D-31/D-31a: el TIPO de período manda. Calendario → siempre calendario.
 * Rodante: grano mes → calendario; el resto (o sin dimensión de tiempo) → semanal.
 */
export function alineacionComparacion(
  p: PeriodoPedido,
  groupBy: DimensionId[],
): 'semanal' | 'calendario' {
  if (clasePeriodo(p) === 'calendario') return 'calendario';

  return groupBy.includes('mes') ? 'calendario' : 'semanal';
}

/** Período de comparación (D-31 semanal / D-31a calendario). */
export function periodoComparacion(
  desde: string,
  hasta: string,
  tipo: 'periodo_anterior' | 'anio_anterior',
  alineacion: 'semanal' | 'calendario',
): { desde: string; hasta: string } {
  if (alineacion === 'semanal') {
    const k =
      tipo === 'anio_anterior'
        ? 364
        : Math.ceil(diasEnRango(desde, hasta) / 7) * 7;

    return { desde: sumarDias(desde, -k), hasta: sumarDias(hasta, -k) };
  }

  let m = 12;

  if (tipo === 'periodo_anterior') {
    m = 1;

    while (sumarMeses(hasta, -m) >= desde) m++;
  }

  const nuevoHasta = esUltimoDiaDelMes(hasta)
    ? finDeMes(sumarMeses(hasta, -m))
    : sumarMeses(hasta, -m);

  return { desde: sumarMeses(desde, -m), hasta: nuevoHasta };
}

/** ¿La semana ISO / el mes completo cae dentro de [desde, hasta]? (D-31, buckets parciales). */
export function bucketCompleto(
  dim: 'semana' | 'mes',
  bucketInicio: string,
  desde: string,
  hasta: string,
): boolean {
  const fin =
    dim === 'semana' ? sumarDias(bucketInicio, 6) : finDeMes(bucketInicio);

  return bucketInicio >= desde && fin <= hasta;
}

/** D-31a: el día en curso vuelve parcial su bucket (día, semana o mes). */
export function bucketContieneHoy(
  dim: 'dia' | 'semana' | 'mes',
  bucketInicio: string,
  hoy: string,
): boolean {
  if (dim === 'dia') return bucketInicio === hoy;

  if (dim === 'semana')
    return hoy >= bucketInicio && hoy <= sumarDias(bucketInicio, 6);

  return hoy >= inicioDeMes(bucketInicio) && hoy <= finDeMes(bucketInicio);
}

/**
 * D-31a: ¿el período incluye hoy (o el futuro)? Entonces su último día es parcial y la variación
 * del TOTAL sólo es honesta sobre el rango completo (hasta ayer). `hastaComparable` es ese fin
 * (null si no queda ningún día completo: el período es sólo hoy).
 */
export function rangoCompletoHastaAyer(
  desde: string,
  hasta: string,
  hoy: string,
): { parcial: boolean; hastaComparable: string | null } {
  if (hasta < hoy) return { parcial: false, hastaComparable: hasta };

  const ayer = sumarDias(hoy, -1);
  const fin = ayer < hasta ? ayer : hasta;

  return { parcial: true, hastaComparable: fin >= desde ? fin : null };
}

/** Resta `n` días a una fecha (para recortar el último día parcial del rango de comparación). */
export function restarDias(s: string, n: number): string {
  return sumarDias(s, -n);
}

## Hallazgos

1. **MEDIUM — Los días futuros muestran comparaciones como si fueran períodos completos**  
   **Archivo:** `src/app/reporte-medida/order.validator.ts:506` y `src/app/reporte-medida/post-process.ts:442`  
   **Por qué es real:** `resolvePeriodo()` acepta un `to` posterior a `ctx.hoy`. En el post-proceso, `bucketContieneHoy()` sólo marca como parcial el bucket que contiene hoy. Para dimensión `dia`, cualquier fecha futura pasa por `if (d === 'dia') return false`, por lo que puede recibir `*_var_pct` contra un día histórico aunque todavía no haya ocurrido. `variacionTotal` sí se recorta hasta ayer, pero las variaciones por fila no.  
   **Corrección mínima:** rechazar períodos explícitos con `hasta > ctx.hoy` en el validador. Si se quieren admitir, marcar como parcial cualquier bucket diario `key >= hoy` y hacer equivalente el tratamiento para semanas/meses futuros.

2. **MEDIUM — La comparación de un Top N puede perder contrapartes por expansión de claves**  
   **Archivo:** `src/app/reporte-medida/compiler.ts:607-638`, especialmente `:619` y `:571`; `src/app/reporte-medida/post-process.ts:403-410`  
   **Por qué es real:** para `groupBy: ['dia', 'producto']`, `completarBind` extrae sólo `producto_key` de las N filas actuales. La consulta comparativa trae entonces todos los días de esos productos y aplica `LIMIT pantallaMaxFilas + 1`. Cien productos por 31 días pueden producir 3.100 grupos; sólo se ejecutan/usan aproximadamente 1.000, dejando filas actuales sin contraparte aunque ésta exista. El validador permite esta combinación hasta 31 días y además reduce la estimación completa a `top`.  
   **Corrección mínima:** pasar y filtrar las claves completas de las filas actuales, transformando los buckets temporales a su contraparte comparativa, en vez de filtrar únicamente las dimensiones no temporales. Alternativamente, usar una CTE de claves solicitadas y unirla con los grupos comparativos, sin un límite independiente que pueda cortar contrapartes.
