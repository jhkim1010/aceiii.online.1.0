// Una corrida = UNA transacción READ ONLY en UNA conexión, con statement_timeout fijado por
// constantes del servidor (nunca por el pedido) y límites de concurrencia por advisory locks
// de TRANSACCIÓN (pg_try_advisory_xact_lock): se sueltan solos al terminar la tx, también por
// timeout/error. Los locks de sesión están prohibidos (pgbouncer los filtraría).
// Dentro de la tx sólo van consultas: nada de HTTP, archivos ni caché.

  HttpException,
  Injectable,
  InternalServerErrorException,
  Logger,
} from '@nestjs/common';
import { InjectConnection } from '@nestjs/sequelize';
import { QueryTypes } from 'sequelize';
import { Sequelize } from 'sequelize-typescript';

import { LIMITES } from './catalog/catalog';
import { ConsultaCompilada, ContextoEjecucion } from './consulta.types';

// Clases de lock (primer int de la forma de dos ints). Elegidas fuera del alcance de un store id.
export const LOCK_CLASE_TIENDA = 1170001;
export const LOCK_CLASE_USUARIO = 1170002;
export const LOCK_CLASE_GLOBAL = 1170003;

const RETRY_AFTER_MS = 2000;
const GLOBAL_MAX_DEFAULT = 4;
const GLOBAL_MAX_MIN = 1;
const GLOBAL_MAX_MAX = 6;

/** REPORTE_MEDIDA_GLOBAL_MAX: entero 1..6; inválido o ausente → 4; fuera de rango → se recorta. */
export function globalMax(
  raw: string | undefined = process.env.REPORTE_MEDIDA_GLOBAL_MAX,
): number {
  if (raw === undefined || !/^\d+$/.test(raw.trim())) return GLOBAL_MAX_DEFAULT;
  const n = Number(raw.trim());
  if (n === 0) return GLOBAL_MAX_DEFAULT;

  return Math.min(GLOBAL_MAX_MAX, Math.max(GLOBAL_MAX_MIN, n));
}

function ocupado(code: string, messageEs: string): HttpException {
  return new HttpException(
    { code, messageEs, retryAfterMs: RETRY_AFTER_MS },
    429,
  );
}

/** Traduce cualquier error de PG a HttpException sin exponer SQL ni binds. */
export function mapearErrorPg(e: unknown): HttpException {
  if (e instanceof HttpException) return e;
  const err = e as {
    original?: { code?: string };
    parent?: { code?: string };
    code?: string;
  };
  const pgCode = err?.original?.code ?? err?.parent?.code ?? err?.code;
  if (pgCode === '57014') {
    return new HttpException(
      {
        code: 'CONSULTA_DEMASIADO_LARGA',
        messageEs:
          'La consulta superó 5 s (20 s para CSV). Achicá el período, agregá un filtro o un Top N.',
      },
      408,
    );
  }

  return new InternalServerErrorException({
    code: 'ERROR_CONSULTA',
    messageEs: 'No se pudo ejecutar la consulta.',
  });
}

@Injectable()
export class ReporteMedidaExecutorService {
  private readonly logger = new Logger(ReporteMedidaExecutorService.name);

  constructor(@InjectConnection() private readonly sequelize: Sequelize) {}

  async ejecutar(
    ctx: ContextoEjecucion,
    consultas: ConsultaCompilada[],
  ): Promise<Record<string, Record<string, unknown>[]>> {
    const timeoutMs =
      ctx.modo === 'csv' ? LIMITES.timeoutCsvMs : LIMITES.timeoutPantallaMs;
    if (!Number.isInteger(timeoutMs) || timeoutMs <= 0) {
      throw new InternalServerErrorException({ code: 'ERROR_CONSULTA' });
    }
    const slots = globalMax();
    const t0 = Date.now();
    let filas = 0;

    try {
      const out = await this.sequelize.transaction(async (t) => {
        await this.sequelize.query(
          `SET TRANSACTION READ ONLY; SET LOCAL statement_timeout = '${timeoutMs}ms'`,
          { transaction: t },
        );

        const tryLock = async (
          clase: number,
          clave: number,
        ): Promise<boolean> => {
          const rows = await this.sequelize.query<{ ok: boolean }>(
            'SELECT pg_try_advisory_xact_lock($1::int, $2::int) AS ok',
            { bind: [clase, clave], type: QueryTypes.SELECT, transaction: t },
          );

          return rows[0]?.ok === true;
        };

        if (!(await tryLock(LOCK_CLASE_TIENDA, ctx.storeId))) {
          throw ocupado(
            'TIENDA_OCUPADA',
            'Ya hay un reporte corriendo para esta tienda. Probá de nuevo en unos segundos.',
          );
        }
        if (!(await tryLock(LOCK_CLASE_USUARIO, ctx.userId))) {
          throw ocupado(
            'USUARIO_OCUPADO',
            'Ya tenés un reporte corriendo. Esperá a que termine.',
          );
        }
        let slot = false;
        for (let i = 0; i < slots && !slot; i++) {
          slot = await tryLock(LOCK_CLASE_GLOBAL, i);
        }
        if (!slot) {
          throw ocupado(
            'SERVIDOR_OCUPADO',
            'El servidor está procesando otros reportes. Probá de nuevo en unos segundos.',
          );
        }

        const res: Record<string, Record<string, unknown>[]> = {};
        for (const c of consultas) {
          const rows = await this.sequelize.query<Record<string, unknown>>(
            c.sql,
            {
              bind: c.bind as never,
              type: QueryTypes.SELECT,
              transaction: t,
            },
          );
          res[c.etiqueta] = rows;
          filas += rows.length;
        }

        return res;
      });

      this.logger.log(
        JSON.stringify({
          storeId: ctx.storeId,
          userId: ctx.userId,
          pedidoHash: ctx.pedidoHash,
          grano: ctx.grano,
          modo: ctx.modo,
          ms: Date.now() - t0,
          filas,
        }),
      );

      return out;
    } catch (e) {
      throw mapearErrorPg(e);
    }
  }
}

export const LOCK_CLASE_TIENDA = 1170001;
export const LOCK_CLASE_USUARIO = 1170002;
export const LOCK_CLASE_GLOBAL = 1170003;

const RETRY_AFTER_MS = 2000;
const GLOBAL_MAX_DEFAULT = 4;
const GLOBAL_MAX_MIN = 1;
const GLOBAL_MAX_MAX = 6;

/** REPORTE_MEDIDA_GLOBAL_MAX: entero 1..6; inválido o ausente → 4; fuera de rango → se recorta. */
export function globalMax(
  raw: string | undefined = process.env.REPORTE_MEDIDA_GLOBAL_MAX,
): number {
  if (raw === undefined || !/^\d+$/.test(raw.trim())) return GLOBAL_MAX_DEFAULT;
  const n = Number(raw.trim());
  if (n === 0) return GLOBAL_MAX_DEFAULT;

  return Math.min(GLOBAL_MAX_MAX, Math.max(GLOBAL_MAX_MIN, n));
}

function ocupado(code: string, messageEs: string): HttpException {
  return new HttpException(
    { code, messageEs, retryAfterMs: RETRY_AFTER_MS },
    429,
  );
}

/** Traduce cualquier error de PG a HttpException sin exponer SQL ni binds. */
export function mapearErrorPg(e: unknown): HttpException {
  if (e instanceof HttpException) return e;
  const err = e as {
    original?: { code?: string };
    parent?: { code?: string };
    code?: string;
  };
  const pgCode = err?.original?.code ?? err?.parent?.code ?? err?.code;
  if (pgCode === '57014') {
    return new HttpException(
      {
        code: 'CONSULTA_DEMASIADO_LARGA',
        messageEs:
          'La consulta superó 5 s (20 s para CSV). Achicá el período, agregá un filtro o un Top N.',
      },
      408,
    );
  }

  return new InternalServerErrorException({
    code: 'ERROR_CONSULTA',
    messageEs: 'No se pudo ejecutar la consulta.',
  });
}

  constructor(@InjectConnection() private readonly sequelize: Sequelize) {}

  async ejecutar(
    ctx: ContextoEjecucion,
    consultas: ConsultaCompilada[],
  ): Promise<Record<string, Record<string, unknown>[]>> {
    const timeoutMs =
      ctx.modo === 'csv' ? LIMITES.timeoutCsvMs : LIMITES.timeoutPantallaMs;
    if (!Number.isInteger(timeoutMs) || timeoutMs <= 0) {
      throw new InternalServerErrorException({ code: 'ERROR_CONSULTA' });
    }
    const slots = globalMax();
    const t0 = Date.now();
    let filas = 0;

    try {
      const out = await this.sequelize.transaction(async (t) => {
        await this.sequelize.query(
          `SET TRANSACTION READ ONLY; SET LOCAL statement_timeout = '${timeoutMs}ms'`,
          { transaction: t },
        );

        const tryLock = async (
          clase: number,
          clave: number,
        ): Promise<boolean> => {
          const rows = await this.sequelize.query<{ ok: boolean }>(
            'SELECT pg_try_advisory_xact_lock($1::int, $2::int) AS ok',
            { bind: [clase, clave], type: QueryTypes.SELECT, transaction: t },
          );

          return rows[0]?.ok === true;
        };

        if (!(await tryLock(LOCK_CLASE_TIENDA, ctx.storeId))) {
          throw ocupado(
            'TIENDA_OCUPADA',
            'Ya hay un reporte corriendo para esta tienda. Probá de nuevo en unos segundos.',
          );
        }
        if (!(await tryLock(LOCK_CLASE_USUARIO, ctx.userId))) {
          throw ocupado(
            'USUARIO_OCUPADO',
            'Ya tenés un reporte corriendo. Esperá a que termine.',
          );
        }
        let slot = false;
        for (let i = 0; i < slots && !slot; i++) {
          slot = await tryLock(LOCK_CLASE_GLOBAL, i);
        }
        if (!slot) {
          throw ocupado(
            'SERVIDOR_OCUPADO',
            'El servidor está procesando otros reportes. Probá de nuevo en unos segundos.',
          );
        }

        const res: Record<string, Record<string, unknown>[]> = {};
        for (const c of consultas) {
          const rows = await this.sequelize.query<Record<string, unknown>>(
            c.sql,
            {
              bind: c.bind as never,
              type: QueryTypes.SELECT,
              transaction: t,
            },
          );
          res[c.etiqueta] = rows;
          filas += rows.length;
        }

        return res;
      });

      this.logger.log(
        JSON.stringify({
          storeId: ctx.storeId,
          userId: ctx.userId,
          pedidoHash: ctx.pedidoHash,
          grano: ctx.grano,
          modo: ctx.modo,
          ms: Date.now() - t0,
          filas,
        }),
      );

      return out;
    } catch (e) {
      throw mapearErrorPg(e);
    }
  }
}
