+/** D-19: la disponibilidad se cachea 60 s por tienda (regla de datos de referencia). */
+export const DISPONIBILIDAD_TTL_MS = 60_000;
+
+/**
+ * Una sola consulta por tienda (T-117-30). D-30: NONE/General no cuentan como valor;
+ * categoría/temporada = COALESCE(hijo, madre). Todo predicado está atado a la tienda
+ * ($1) — madre, categorías, temporadas, colores y talles (T-117-29).
+ */
+const SQL_CONTEOS = `
+  SELECT
+    (SELECT COUNT(DISTINCT c.id)::int
+       FROM products p
+       LEFT JOIN products m ON m.id = p.parent_id AND m.store_id = $1
+       JOIN categories c ON c.id = COALESCE(p.category_id, m.category_id) AND c.store_id = $1
+      WHERE p.store_id = $1 AND p.status <> 'deactivated'
+        AND upper(trim(c.name)) NOT IN ('NONE', 'GENERAL')) AS categorias,
+    (SELECT COUNT(DISTINCT s.id)::int
+       FROM products p
+       LEFT JOIN products m ON m.id = p.parent_id AND m.store_id = $1
+       JOIN seasons s ON s.id = COALESCE(p.season_id, m.season_id) AND s.store_id = $1
+      WHERE p.store_id = $1 AND p.status <> 'deactivated'
+        AND upper(trim(s.name)) NOT IN ('NONE', 'GENERAL')) AS temporadas,
+    (SELECT COUNT(DISTINCT co.id)::int
+       FROM products p
+       JOIN colors co ON co.id = p.color_id AND co.store_id = $1
+      WHERE p.store_id = $1 AND p.status <> 'deactivated'
+        AND upper(trim(co.name)) NOT IN ('NONE', 'GENERAL')) AS colores,
+    (SELECT COUNT(DISTINCT sz.id)::int
+       FROM products p
+       JOIN sizes sz ON sz.id = p.size_id AND sz.store_id = $1
+      WHERE p.store_id = $1 AND p.status <> 'deactivated'
+        AND upper(trim(sz.name)) NOT IN ('NONE', 'GENERAL')) AS talles,
+    (SELECT COUNT(*)::int FROM branches b WHERE b.store_id = $1) AS sucursales
+`;
+
+export interface DisponibilidadTienda {
+  disponibilidad: Record<DimensionId, DisponibilidadDimension>;
+  sucursalesCount: number;
+}
+
+@Injectable()
+export class ReporteMedidaDisponibilidadService {
+  constructor(
+    @InjectConnection() private readonly sequelize: Sequelize,
+    private readonly cache: MemoryCacheService,
+  ) {}
+
+  async obtener(storeId: number): Promise<DisponibilidadTienda> {
+    return this.cache.getOrLoad(
+      storeKey('reporte-medida:disponibilidad', storeId),
+      DISPONIBILIDAD_TTL_MS,
+      async () => {
+        const [row] = await this.sequelize.query<ConteosCatalogo>(SQL_CONTEOS, {
+          bind: [storeId],
+          type: QueryTypes.SELECT,
+        });
+        const conteos: ConteosCatalogo = {
+          categorias: Number(row?.categorias ?? 0),
+          temporadas: Number(row?.temporadas ?? 0),
+          colores: Number(row?.colores ?? 0),
+          talles: Number(row?.talles ?? 0),
+          sucursales: Number(row?.sucursales ?? 0),
+        };
+
+        return {
+          disponibilidad: decidirDisponibilidad(conteos),
+          sucursalesCount: conteos.sucursales,
+        };
+      },
+    );
+  }

+import type { DimensionId } from './order.types';
+
+/** Conteos de valores REALES (sin NONE/General) por dimensión de catálogo de una tienda. */
+export interface ConteosCatalogo {
+  categorias: number;
+  temporadas: number;
+  colores: number;
+  talles: number;
+  sucursales: number;
+}
+
+export type DisponibilidadDimension = {
+  disponible: boolean;
+  motivoEs?: string;
+  cardinalidad?: number;
+};
+
+function porConteo(
+  n: number,
+  unico: string,
+  ninguno: string,
+): DisponibilidadDimension {
+  if (n >= 2) return { disponible: true, cardinalidad: n };
+  if (n === 1) {
+    return { disponible: false, cardinalidad: 1, motivoEs: unico };
+  }
+
+  return { disponible: false, cardinalidad: 0, motivoEs: ninguno };
+}
+
+/**
+ * D-19: una dimensión de catálogo con un único valor (o sólo NONE/General, D-30) no sirve
+ * para agrupar. Tiempo, producto, vendedor y medio de pago están siempre disponibles;
+ * sucursal requiere al menos 2 sucursales.
+ */
+export function decidirDisponibilidad(
+  c: ConteosCatalogo,
+): Record<DimensionId, DisponibilidadDimension> {
+  const siempre = (): DisponibilidadDimension => ({ disponible: true });
+
+  return {
+    sucursal:
+      c.sucursales >= 2
+        ? { disponible: true, cardinalidad: c.sucursales }
+        : {
+            disponible: false,
+            cardinalidad: c.sucursales,
+            motivoEs: 'Esta tienda tiene una sola sucursal',
+          },
+    dia: siempre(),
+    semana: siempre(),
+    mes: siempre(),
+    dia_semana: siempre(),
+    hora: siempre(),
+    vendedor: siempre(),
+    medio_pago: siempre(),
+    producto: siempre(),
+    categoria: porConteo(
+      c.categorias,
+      'Esta tienda tiene una sola categoría cargada',
+      'Esta tienda no tiene categorías cargadas',
+    ),
+    temporada: porConteo(
+      c.temporadas,
+      'Esta tienda tiene una sola temporada cargada',
+      'Esta tienda no tiene temporadas cargadas',
+    ),
+    color: porConteo(
+      c.colores,
+      'Esta tienda tiene un solo color cargado',
+      'Esta tienda no tiene colores cargados',
+    ),
+    talle: porConteo(
+      c.talles,
+      'Esta tienda tiene un solo talle cargado',
+      'Esta tienda no tiene talles cargados',
+    ),
+  };
+}

+    bind: bindFuentes,
+    etiqueta: 'fuentes',
+  };

+    bindComp[3] = p.comparacion.hasta;

+      principal: {
+        sql: sqlPrincipal,
+        bind: bindComp,
+        etiqueta: 'comp_principal',
+      },
+      totales: { sql: sqlTotales, bind: bindComp, etiqueta: 'comp_totales' },

+          formato: 'porcentaje',
+          definicionEs: 'Parte de la venta que tiene costo cargado.',
+        },
+      );
+    }
+  }

+  return {
+    grano,
+    principal,
+    totales,
+    fuentes,
+    comparacion,
+    columnas,
+    limite: limiteModo,
+  };

+/**
+ * CSV del reporte a medida (CODEX: ≤10.000 filas, armado en el servidor).
+ *
+ * Elecciones para Excel es-AR (supuesto A6 de la investigación): BOM UTF-8 para que Excel
+ * detecte la codificación, `;` como separador (la coma es el decimal), coma decimal,
+ * sin separador de miles ni «$», fin de línea CRLF. Están como constantes para cambiarlas
+ * en un solo lugar.
+ *
+ * T-117-28: un texto que empieza con `= + - @ TAB CR` se prefija con apóstrofe para que
+ * Excel no lo ejecute como fórmula. Los NÚMEROS negativos no se prefijan (no son texto).
+ */
+
+export const CSV_BOM = '﻿';
+export const CSV_SEPARADOR = ';';
+export const CSV_DECIMAL = ',';
+export const CSV_EOL = '\r\n';
+
+const PELIGROSOS = new Set(['=', '+', '-', '@', '\t', '\r']);
+
+function campo(valor: unknown): string {
+  if (valor === null || valor === undefined) return '';
+  let s: string;
+  if (typeof valor === 'number') {
+    s = Number.isFinite(valor) ? String(valor).replace('.', CSV_DECIMAL) : '';
+  } else {
+    s =
+      valor instanceof Date
+        ? valor.toISOString()
+        : String(valor as string | boolean | bigint);
+    if (s.length > 0 && PELIGROSOS.has(s[0])) s = `'${s}`;
+  }
+  if (/[;"\r\n]/.test(s)) s = `"${s.replace(/"/g, '""')}"`;
+
+  return s;
+}
+
+export function toCsv(
+  columnas: Array<{ id: string; labelEs: string; formato?: string }>,
+  filas: Array<Record<string, unknown>>,
+): string {
+  const lineas = [columnas.map((c) => campo(c.labelEs)).join(CSV_SEPARADOR)];
+  for (const f of filas) {
+    lineas.push(columnas.map((c) => campo(f[c.id])).join(CSV_SEPARADOR));
+  }
+
+  return CSV_BOM + lineas.join(CSV_EOL) + CSV_EOL;
+}

+//
+// Una corrida = UNA transacción READ ONLY en UNA conexión, con statement_timeout fijado por
+// constantes del servidor (nunca por el pedido) y límites de concurrencia por advisory locks
+// de TRANSACCIÓN (pg_try_advisory_xact_lock): se sueltan solos al terminar la tx, también por
+// timeout/error. Los locks de sesión están prohibidos (pgbouncer los filtraría).
+// Dentro de la tx sólo van consultas: nada de HTTP, archivos ni caché.

+  Injectable,

+export const LOCK_CLASE_TIENDA = 1170001;
+export const LOCK_CLASE_USUARIO = 1170002;
+export const LOCK_CLASE_GLOBAL = 1170003;

+const RETRY_AFTER_MS = 2000;
+const GLOBAL_MAX_DEFAULT = 4;
+const GLOBAL_MAX_MIN = 1;
+const GLOBAL_MAX_MAX = 6;
+
+/** REPORTE_MEDIDA_GLOBAL_MAX: entero 1..6; inválido o ausente → 4; fuera de rango → se recorta. */
+export function globalMax(
+  raw: string | undefined = process.env.REPORTE_MEDIDA_GLOBAL_MAX,
+): number {
+  if (raw === undefined || !/^\d+$/.test(raw.trim())) return GLOBAL_MAX_DEFAULT;
+  const n = Number(raw.trim());
+  if (n === 0) return GLOBAL_MAX_DEFAULT;
+
+  return Math.min(GLOBAL_MAX_MAX, Math.max(GLOBAL_MAX_MIN, n));
+}

+function ocupado(code: string, messageEs: string): HttpException {
+  return new HttpException(
+    { code, messageEs, retryAfterMs: RETRY_AFTER_MS },
+    429,

+/** Traduce cualquier error de PG a HttpException sin exponer SQL ni binds. */
+export function mapearErrorPg(e: unknown): HttpException {
+  if (e instanceof HttpException) return e;
+  const err = e as {
+    original?: { code?: string };
+    parent?: { code?: string };
+    code?: string;
+  };
+  const pgCode = err?.original?.code ?? err?.parent?.code ?? err?.code;

+        messageEs:
+          'La consulta superó 5 s (20 s para CSV). Achicá el período, agregá un filtro o un Top N.',
+      },
+      408,
+    );
+  }
+
+  return new InternalServerErrorException({
+    code: 'ERROR_CONSULTA',
+    messageEs: 'No se pudo ejecutar la consulta.',
+  });

+
+  constructor(@InjectConnection() private readonly sequelize: Sequelize) {}

+  async ejecutar(
+    ctx: ContextoEjecucion,
+    consultas: ConsultaCompilada[],
+  ): Promise<Record<string, Record<string, unknown>[]>> {
+    const timeoutMs =
+      ctx.modo === 'csv' ? LIMITES.timeoutCsvMs : LIMITES.timeoutPantallaMs;
+    if (!Number.isInteger(timeoutMs) || timeoutMs <= 0) {
+      throw new InternalServerErrorException({ code: 'ERROR_CONSULTA' });
+    }
+    const slots = globalMax();
+    const t0 = Date.now();
+    let filas = 0;
+
+    try {
+      const out = await this.sequelize.transaction(async (t) => {
+        await this.sequelize.query(
+          `SET TRANSACTION READ ONLY; SET LOCAL statement_timeout = '${timeoutMs}ms'; SET LOCAL idle_in_transaction_session_timeout = '${timeoutMs}ms'`,
+          { transaction: t },
+        );
+
+        const tryLock = async (
+          clase: number,
+          clave: number,
+        ): Promise<boolean> => {
+          const rows = await this.sequelize.query<{ ok: boolean }>(
+            'SELECT pg_try_advisory_xact_lock($1::int, $2::int) AS ok',
+            { bind: [clase, clave], type: QueryTypes.SELECT, transaction: t },
+          );
+
+          return rows[0]?.ok === true;
+        };
+
+        if (!(await tryLock(LOCK_CLASE_TIENDA, ctx.storeId))) {
+          throw ocupado(
+            'TIENDA_OCUPADA',
+            'Ya hay un reporte corriendo para esta tienda. Probá de nuevo en unos segundos.',
+          );
+        }
+        if (!(await tryLock(LOCK_CLASE_USUARIO, ctx.userId))) {
+          throw ocupado(
+            'USUARIO_OCUPADO',
+            'Ya tenés un reporte corriendo. Esperá a que termine.',
+          );
+        }
+        let slot = false;
+        for (let i = 0; i < slots && !slot; i++) {
+          slot = await tryLock(LOCK_CLASE_GLOBAL, i);
+        }
+        if (!slot) {
+          throw ocupado(
+            'SERVIDOR_OCUPADO',
+            'El servidor está procesando otros reportes. Probá de nuevo en unos segundos.',
+          );
+        }
+
+        const res: Record<string, Record<string, unknown>[]> = {};
+        for (const c of consultas) {
+          // El tope es de la CORRIDA, no de cada consulta (hallazgo CODEX 117-05):
+          // cada consulta recibe sólo el tiempo que queda.
+          const restante = timeoutMs - (Date.now() - t0);
+          if (restante <= 0) {
+            throw mapearErrorPg({ code: '57014' });
+          }
+          await this.sequelize.query(
+            `SET LOCAL statement_timeout = '${Math.max(1, Math.floor(restante))}ms'`,
+            { transaction: t },
+          );
+          const rows = await this.sequelize.query<Record<string, unknown>>(
+            c.sql,
+            {
+              bind: c.bind as never,
+              type: QueryTypes.SELECT,
+              transaction: t,
+            },
+          );
+          res[c.etiqueta] = rows;
+          filas += rows.length;
+        }
+
+        return res;
+      });
+
+      this.logger.log(
+        JSON.stringify({
+          storeId: ctx.storeId,
+          userId: ctx.userId,
+          pedidoHash: ctx.pedidoHash,
+          grano: ctx.grano,
+          modo: ctx.modo,
+          ms: Date.now() - t0,
+          filas,
+        }),
+      );
+
+      return out;
+    } catch (e) {
+      throw mapearErrorPg(e);
+    }
+  }
+}

+/** sha256 del JSON canónico (claves ordenadas) — sólo para logs. */
+export function hashPedido(p: PedidoReporte): string {
+  const canon = (x: unknown): unknown => {
+    if (Array.isArray(x)) return x.map(canon);
+
+    if (x && typeof x === 'object') {
+      const o = x as Record<string, unknown>;
+
+      return Object.keys(o)
+        .sort()
+        .reduce<Record<string, unknown>>((acc, k) => {
+          acc[k] = canon(o[k]);
+
+          return acc;
+        }, {});
+    }
+
+    return x;
+  };
+
+  return createHash('sha256')
+    .update(JSON.stringify(canon(p)))
+    .digest('hex');

+// Funciones PURAS sobre cadenas 'YYYY-MM-DD'. Sólo se usan métodos getUTC* y Date.UTC:

+import type { DimensionId, PeriodoPedido } from './order.types';
+
+const MS_DIA = 86_400_000;
+
+export class ErrorPeriodo extends Error {
+  readonly code = 'PERIODO_INVALIDO';
+
+  constructor(mensaje: string) {
+    super(mensaje);
+    this.name = 'ErrorPeriodo';
+  }
+}
+
+type Ymd = { y: number; m: number; d: number };
+
+function diasDelMes(y: number, m: number): number {
+  return new Date(Date.UTC(y, m, 0)).getUTCDate();
+}
+
+function parse(s: string): Ymd {
+  const r = /^(\d{4})-(\d{2})-(\d{2})$/.exec(s ?? '');
+
+  if (!r) throw new ErrorPeriodo(`Fecha inválida: ${s}`);
+
+  const y = Number(r[1]);
+  const m = Number(r[2]);
+  const d = Number(r[3]);
+
+  if (m < 1 || m > 12 || d < 1 || d > diasDelMes(y, m))
+    throw new ErrorPeriodo(`Fecha inválida: ${s}`);
+
+  return { y, m, d };
+}
+
+function fmt({ y, m, d }: Ymd): string {
+  return `${String(y).padStart(4, '0')}-${String(m).padStart(2, '0')}-${String(d).padStart(2, '0')}`;
+}
+
+function toMs(s: string): number {
+  const { y, m, d } = parse(s);
+
+  return Date.UTC(y, m - 1, d);
+}
+
+function fromMs(ms: number): string {
+  const dt = new Date(ms);
+
+  return fmt({
+    y: dt.getUTCFullYear(),
+    m: dt.getUTCMonth() + 1,
+    d: dt.getUTCDate(),
+  });
+}
+
+function sumarDias(s: string, n: number): string {
+  return fromMs(toMs(s) + n * MS_DIA);
+}
+
+/** Suma meses (puede ser negativo); el día se recorta al fin del mes de destino. */
+function sumarMeses(s: string, n: number): string {
+  const { y, m, d } = parse(s);
+  const total = y * 12 + (m - 1) + n;
+  const ny = Math.floor(total / 12);
+  const nm = (total % 12) + 1;
+
+  return fmt({ y: ny, m: nm, d: Math.min(d, diasDelMes(ny, nm)) });
+}
+
+function esUltimoDiaDelMes(s: string): boolean {
+  const { y, m, d } = parse(s);
+
+  return d === diasDelMes(y, m);
+}
+
+function finDeMes(s: string): string {
+  const { y, m } = parse(s);
+
+  return fmt({ y, m, d: diasDelMes(y, m) });
+}
+
+function inicioDeMes(s: string): string {
+  const { y, m } = parse(s);
+
+  return fmt({ y, m, d: 1 });
+}
+
+/** Lunes (ISO) de la semana que contiene el día. */
+function lunesDe(s: string): string {
+  const dow = new Date(toMs(s)).getUTCDay(); // 0 = domingo
+
+  return sumarDias(s, -((dow + 6) % 7));

+/** Días del rango, inclusivo. */
+export function diasEnRango(desde: string, hasta: string): number {
+  return Math.round((toMs(hasta) - toMs(desde)) / MS_DIA) + 1;

+/**
+ * Resuelve el período contra el día de ejecución (D-14).
+ * `hoy` debe venir del reloj de la base en la zona de la tienda
+ * (`(now() AT TIME ZONE tz)::date`), nunca de un Date de JS.
+ */
+export function resolvePeriodo(
+  p: PeriodoPedido,
+  hoy: string,
+): { desde: string; hasta: string } {
+  toMs(hoy);
+
+  if ('from' in p) {
+    const desde = fmt(parse(p.from));
+    const hasta = fmt(parse(p.to));
+
+    if (desde > hasta)
+      throw new ErrorPeriodo('La fecha inicial es posterior a la final.');
+
+    return { desde, hasta };
+  }
+
+  switch (p.preset) {
+    case 'hoy':
+      return { desde: hoy, hasta: hoy };
+    case 'ayer': {
+      const a = sumarDias(hoy, -1);
+
+      return { desde: a, hasta: a };
+    }
+    case 'esta_semana':
+      return { desde: lunesDe(hoy), hasta: hoy };
+    case 'semana_pasada': {
+      const l = lunesDe(hoy);

+      return { desde: sumarDias(l, -7), hasta: sumarDias(l, -1) };
+    }
+    case 'este_mes':

+    case 'mes_pasado': {
+      const ini = sumarMeses(inicioDeMes(hoy), -1);
+
+      return { desde: ini, hasta: finDeMes(ini) };
+    }
+    case 'ultimos_7_dias':
+      return { desde: sumarDias(hoy, -6), hasta: hoy };
+    case 'ultimos_30_dias':
+      return { desde: sumarDias(hoy, -29), hasta: hoy };
+    case 'ultimos_90_dias':
+      return { desde: sumarDias(hoy, -89), hasta: hoy };
+    case 'ultimos_12_meses':
+      return { desde: sumarDias(sumarMeses(hoy, -12), 1), hasta: hoy };
+    default:
+      throw new ErrorPeriodo('Período desconocido.');
+  }

+  return 'preset' in p && (p.preset === 'este_mes' || p.preset === 'mes_pasado')

+    : 'rodante';

+ * Rodante: grano mes → calendario; el resto (o sin dimensión de tiempo) → semanal.
+ */
+export function alineacionComparacion(
+  p: PeriodoPedido,
+  groupBy: DimensionId[],
+): 'semanal' | 'calendario' {
+  if (clasePeriodo(p) === 'calendario') return 'calendario';
+
+  return groupBy.includes('mes') ? 'calendario' : 'semanal';

+/** Período de comparación (D-31 semanal / D-31a calendario). */
+export function periodoComparacion(
+  desde: string,
+  hasta: string,
+  tipo: 'periodo_anterior' | 'anio_anterior',
+  alineacion: 'semanal' | 'calendario',
+): { desde: string; hasta: string } {
+  if (alineacion === 'semanal') {
+    const k =
+      tipo === 'anio_anterior'
+        ? 364
+        : Math.ceil(diasEnRango(desde, hasta) / 7) * 7;
+
+    return { desde: sumarDias(desde, -k), hasta: sumarDias(hasta, -k) };
+  }
+
+  let m = 12;
+
+  if (tipo === 'periodo_anterior') {
+    m = 1;
+
+    while (sumarMeses(hasta, -m) >= desde) m++;
+  }
+
+  const nuevoHasta = esUltimoDiaDelMes(hasta)
+    ? finDeMes(sumarMeses(hasta, -m))
+    : sumarMeses(hasta, -m);
+
+  return { desde: sumarMeses(desde, -m), hasta: nuevoHasta };
+}

+/** ¿La semana ISO / el mes completo cae dentro de [desde, hasta]? (D-31, buckets parciales). */
+export function bucketCompleto(
+  dim: 'semana' | 'mes',
+  bucketInicio: string,
+  desde: string,
+  hasta: string,
+): boolean {
+  const fin =
+    dim === 'semana' ? sumarDias(bucketInicio, 6) : finDeMes(bucketInicio);
+
+  return bucketInicio >= desde && fin <= hasta;
+}

+/** D-31a: el día en curso vuelve parcial su bucket (día, semana o mes). */
+export function bucketContieneHoy(
+  dim: 'dia' | 'semana' | 'mes',
+  bucketInicio: string,
+  hoy: string,
+): boolean {
+  if (dim === 'dia') return bucketInicio === hoy;
+
+  if (dim === 'semana')
+    return hoy >= bucketInicio && hoy <= sumarDias(bucketInicio, 6);
+
+  return hoy >= inicioDeMes(bucketInicio) && hoy <= finDeMes(bucketInicio);

+/**
+ * Orden de talles para el reporte a medida (D-23, PEDIDO 40 «Ordenar talles por número»).
+ *
+ * `sizes` no tiene columna de orden y los nombres reales mezclan números («38», «1.2MM»),
+ * letras («S», «XL», «2XL»), series «N1…N12» y texto libre («Talle Única», «CHICO»).
+ * Regla (grupos, en este orden) — la MISMA que la app (`ventago-app/src/utils/orden-talles.ts`,
+ * PEDIDO 40 ya desplegado): letras primero, después números. Un talle no puede salir en otro
+ * orden en el reporte que en la ficha de producto.
+ *   0. escalera de letras XXS, XS, S, M, L, XL, XXL (=2XL), XXXL (=3XL), 4XL
+ *   1. numéricos ascendentes (incluye «TALLE 1» → 1 y «1.2MM» → 1.2)
+ *   2. serie N por número (N1, N2, N12)
+ *   3. el resto alfabético sin mayúsculas ni acentos, números naturales
+ *   4. sin talle (null) al final
+ */
+
+const ESCALERA = ['XXS', 'XS', 'S', 'M', 'L', 'XL', 'XXL', 'XXXL', '4XL'];
+const ALIAS_LETRA: Record<string, string> = { '2XL': 'XXL', '3XL': 'XXXL' };
+
+function grupo(t: string): { g: number; n: number } {
+  const s = t.trim();
+  const up = s.toUpperCase();
+  const letra = ESCALERA.indexOf(ALIAS_LETRA[up] ?? up);
+  if (letra >= 0) return { g: 0, n: letra };
+  const num = /^(?:TALLE\s*)?(\d+(?:[.,]\d+)?)(?:\s*MM)?$/.exec(up);
+  if (num) return { g: 1, n: Number(num[1].replace(',', '.')) };
+  const serie = /^N\s*(\d+)$/.exec(up);

### HIGH — Las anulaciones no descuentan importes, unidades ni costos

**Archivos:** `src/app/reports/venta-universe.ts:60-68`, `src/app/reporte-medida/compiler.ts:257-279`

`ventaHeaderCte` calcula `sgn`, pero deja `total_amount` sin signo. Luego:

- Cabecera suma `SUM(venta.total_amount)`.
- Ítem suma `si.subtotal`, `si.quantity` y costos sin multiplicar por `venta.sgn`.
- Pago suma `spm.amount` sin signo.

Por tanto, una `Anulación` incrementa la venta, unidades y margen en lugar de revertirlos. Sólo `tickets` usa el signo.

**Arreglo:** producir importes firmados en el universo o multiplicar todos los agregados contables por `venta.sgn`, por ejemplo:

```sql
SUM(venta.sgn * venta.total_amount)
SUM(venta.sgn * si.subtotal)
SUM(venta.sgn * si.quantity)
SUM(venta.sgn * si.quantity * si.unit_cost)
SUM(venta.sgn * spm.amount)
```

Aplicar la misma regla en `items_agg`, incluyendo `items_total`, `costo` y `venta_con_costo`. Revisar también si las filas de anulación tienen ítems/pagos propios o deben referenciar la venta original.

### HIGH — JOINs tenant-scoped sin `store_id`

**Archivos:** `src/app/reporte-medida/compiler.ts:421-430`, `src/app/reports/venta-universe.ts:88-105`, `src/app/reports/venta-universe.ts:128-132`

Hay accesos y JOINs por `sale_id` solamente:

```sql
LEFT JOIN sale_payment_methods spm ON spm.sale_id = venta.sale_id
JOIN sale_items si ON si.sale_id = venta.sale_id
...
WHERE si.sale_id IN (SELECT sale_id FROM venta)
```

También las subconsultas de devoluciones leen `sale_items` sólo por `sale_id`.

Esto incumple la premisa “toda tabla tiene `store_id`” y deja el aislamiento dependiendo de que los IDs sean globalmente únicos y de que nunca existan datos inconsistentes. Además, el filtro `mediosPago` opera sobre esas filas no acotadas.

**Arreglo:** agregar el tenant a todos los accesos:

```sql
... ON spm.sale_id = venta.sale_id
   AND spm.store_id = $1::int

... ON si.sale_id = venta.sale_id
   AND si.store_id = $1::int
```

Y en `items_agg` y las subconsultas de devoluciones:

```sql
WHERE si.store_id = $1::int
  AND si.sale_id IN (...)
```

Para ello, hacer que `itemsAggCte` reciba `VentaPlaceholders` o al menos el placeholder de tienda.

### MED — La comparación total ignora la regla de períodos parciales

**Archivo:** `src/app/reporte-medida/post-process.ts:411-440`

Las variaciones por fila se anulan para buckets parciales, pero `variacionTotal` siempre se calcula:

```ts
varPct: varPct(actual, anterior)
```

Esto contradice la definición declarada de que los períodos parciales no se comparan. Por ejemplo, “este mes” o un período que incluye hoy puede ocultar la variación de la fila mensual y, simultáneamente, mostrar una variación total.

**Arreglo:** determinar si el período total actual o su contraparte es parcial y devolver `varPct: null`; alternativamente, definir explícitamente que la excepción sólo aplica a filas y corregir el texto del catálogo.

### MED — Protección CSV incompleta ante fórmulas precedidas por LF/espacios

**Archivo:** `src/app/reporte-medida/csv.ts:19-32`

Se protegen `=`, `+`, `-`, `@`, TAB y CR sólo cuando son el primer carácter exacto. Falta al menos LF (`\n`), y algunas hojas de cálculo pueden ignorar espacios o ciertos controles antes de interpretar una fórmula.

Las etiquetas provenientes de productos, usuarios, categorías, etc. son datos capaces de llegar al CSV.

**Arreglo:** para valores textuales, detectar el primer carácter después de espacios y controles, incluyendo LF, y prefijar apóstrofe. Por ejemplo:

```ts
if (/^[\u0000-\u0020]*[=+\-@]/.test(s)) s = `'${s}`;
```

También puede sanearse conservadoramente todo texto que empiece con controles o espacios seguidos de un operador de fórmula.
