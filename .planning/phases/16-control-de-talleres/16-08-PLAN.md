```typescript
import { Injectable, Logger } from '@nestjs/common';
import { Cron } from '@nestjs/schedule';
import { InjectModel } from '@nestjs/sequelize';
import { Op } from 'sequelize';
import { Envio, EnvioStatus } from '../envios/envio.model';
import { computeHealthStatus } from '../envios/envio.service';
import { Vendor } from '../vendors/vendor.model';
import { VendorNotification } from '../../vendor-portal/vendor-notifications/vendor-notification.model';
import { VendorNotificationsService } from '../../vendor-portal/vendor-notifications/vendor-notifications.service';

// Wave 8: LATE envío 자동 알림 크론 — "조기 경보 > 사후 보고" 원칙 (EXTENSION §2 원칙 5)
// 매일 08:00 America/Bogota — LATE 상태(dueDate < today AND pendingQuantity > 0) envío 스캔 → vendor 알림
// 선례: vendor-portal.cron.ts 의 DUE_SOON 크론 (같은 패턴을 LATE 타입으로 복제)
@Injectable()
export class LateEnvioAlertJob {
  private readonly logger = new Logger(LateEnvioAlertJob.name);

  constructor(
    @InjectModel(Envio) private readonly envioModel: typeof Envio,
    @InjectModel(VendorNotification)
    private readonly notificationModel: typeof VendorNotification,
    private readonly notificationsService: VendorNotificationsService,
  ) {}

  @Cron('0 8 * * *', { timeZone: 'America/Bogota' })
  async scanAndNotifyLate(): Promise<void> {
    this.logger.log('LATE 알림 크론 시작 — 지연 envío 스캔');

    try {
      const todayMidnight = new Date();
      todayMidnight.setHours(0, 0, 0, 0);
      const todayIso = todayMidnight.toISOString().split('T')[0];

      // 1) DB 레벨 필터: status ∈ [PENDING, PARTIAL] + dueDate < today + pendingQuantity > 0
      const candidates = await this.envioModel.findAll({
        where: {
          status: { [Op.in]: [EnvioStatus.PENDING, EnvioStatus.PARTIAL] },
          dueDate: { [Op.lt]: todayIso },
          pendingQuantity: { [Op.gt]: 0 },
        },
        include: [{ model: Vendor, attributes: ['id', 'storeId', 'name'] }],
      });

      this.logger.log(`LATE 후보 ${candidates.length}건 — computeHealthStatus 재확인 시작`);

      let created = 0;
      let skippedDuplicate = 0;
      let skippedNotLate = 0;

      for (const envio of candidates) {
        // 2) computeHealthStatus 로 재확인 (단일 진실 원천)
        const health = computeHealthStatus(
          { dueDate: envio.dueDate, pendingQuantity: envio.pendingQuantity },
          todayMidnight,
        );

        if (health !== 'LATE') {
          skippedNotLate++;

          continue;
        }

        // 3) 오늘자 동일 referenceId + type='LATE' 알림 존재 여부 확인 (중복 방지 INV-W8-2)
        const existing = await this.notificationModel.findOne({
          where: {
            vendorId: envio.vendorId,
            type: 'LATE',
            referenceId: envio.id,
            createdAt: { [Op.gte]: todayMidnight },
          },
        });

        if (existing) {
          skippedDuplicate++;

          continue;
        }

        // 4) 신규 LATE 알림 생성
        const daysOverdue = Math.ceil(
          (todayMidnight.getTime() - new Date(envio.dueDate!).getTime()) / (1000 * 60 * 60 * 24),
        );

        await this.notificationsService.createNotification({
          vendorId: envio.vendorId,
          storeId: envio.vendor.storeId,
          type: 'LATE',
          title: `⚠️ Envío atrasado: #${envio.id}`,
          body: `El envío lleva ${daysOverdue} día(s) de atraso. Pendiente: ${envio.pendingQuantity} unidades. Fecha límite original: ${envio.dueDate}.`,
          referenceId: envio.id,
        });

        created++;
      }

      this.logger.log(
        `LATE 알림 크론 완료 — 생성=${created}, 중복스킵=${skippedDuplicate}, 상태아님스킵=${skippedNotLate}, 총후보=${candidates.length}`,
      );
    } catch (error) {
      // INV-W8-5: throw 금지 (이후 스케줄 차단 방지)
      this.logger.error(`LATE 알림 크론 실패: ${error.message}`, error.stack);
    }
  }
}
```

```typescript
import { Injectable, Logger } from '@nestjs/common';
import { InjectModel } from '@nestjs/sequelize';
import { Op, fn, col, literal } from 'sequelize';
import { Sequelize } from 'sequelize-typescript';
import { Envio, EnvioStatus } from '../envios/envio.model';
import { computeHealthStatus } from '../envios/envio.service';
import { Recepcion } from '../recepciones/recepcion.model';
import { QcItem } from '../qc-items/qc-item.model';
import { Vendor } from '../vendors/vendor.model';
import { Etapa } from '../etapas/etapa.model';
import { MemoryCacheService } from 'src/common/cache/memory-cache.service';
import type {
  DashboardV2Response,
  EnviosByStatus,
  TopVendorRow,
  DefectRateTrendPoint,
  BacklogByEtapaRow,
} from './dashboard-v2.types';

// Wave 8: Overview 탭 집계 서비스 — MemoryCache 60s + Promise.all 병렬 + 3 인덱스 활용
@Injectable()
export class TalleresDashboardV2Service {
  private readonly logger = new Logger(TalleresDashboardV2Service.name);
  private static readonly CACHE_TTL_MS = 60_000;

  constructor(
    @InjectModel(Envio) private readonly envioModel: typeof Envio,
    @InjectModel(Recepcion) private readonly recepcionModel: typeof Recepcion,
    @InjectModel(QcItem) private readonly qcItemModel: typeof QcItem,
    @InjectModel(Vendor) private readonly vendorModel: typeof Vendor,
    @InjectModel(Etapa) private readonly etapaModel: typeof Etapa,
    private readonly cacheService: MemoryCacheService,
    private readonly sequelize: Sequelize,
  ) {}

  // GET /api/talleres/dashboard-v2 — 5 집계 병렬 + 60s cache
  async getAggregates(storeId: number): Promise<DashboardV2Response> {
    const cacheKey = `talleres:dashboard-v2:${storeId}`;
    const cached = this.cacheService.get<DashboardV2Response>(cacheKey);

    if (cached) {

      return { ...cached, fromCache: true };
    }

    const t0 = Date.now();
    const [
      enviosByStatus,
      topVendors,
      defectRateTrend,
      backlogByEtapa,
      urgentLateEnvios,
    ] = await Promise.all([
      this.computeEnviosByStatus(storeId),
      this.computeTopVendors(storeId),
      this.computeDefectRateTrend(storeId),
      this.computeBacklogByEtapa(storeId),
      this.computeUrgentLateEnvios(storeId),
    ]);

    const result: DashboardV2Response = {
      enviosByStatus,
      topVendors,
      defectRateTrend,
      backlogByEtapa,
      urgentLateEnvios,
      generatedAt: new Date().toISOString(),
      fromCache: false,
    };

    this.cacheService.set(cacheKey, result, TalleresDashboardV2Service.CACHE_TTL_MS);
    this.logger.log(`[dashboard-v2] miss cached storeId=${storeId} ms=${Date.now() - t0}`);

    return result;
  }

  // (A) enviosByStatus — computeHealthStatus 로 메모리 reduce
  private async computeEnviosByStatus(storeId: number): Promise<EnviosByStatus> {
    const envios = await this.envioModel.findAll({
      where: {
        storeId,
        status: { [Op.in]: [EnvioStatus.PENDING, EnvioStatus.PARTIAL] },
      },
      attributes: ['id', 'dueDate', 'pendingQuantity'],
      raw: true,
    });

    const counts = { ON_TRACK: 0, AT_RISK: 0, LATE: 0, total: envios.length };

    envios.forEach((e) => {
      const h = computeHealthStatus({
        dueDate: (e as any).dueDate,
        pendingQuantity: Number((e as any).pendingQuantity) || 0,
      });
      counts[h]++;
    });

    return counts;
  }

  // (B) topVendors — 최근 30일 volume top 5
  private async computeTopVendors(storeId: number): Promise<TopVendorRow[]> {
    const thirtyDaysAgo = new Date();
    thirtyDaysAgo.setDate(thirtyDaysAgo.getDate() - 30);

    const rows: any[] = await this.envioModel.findAll({
      attributes: [
        'vendorId',
        [fn('SUM', literal('pending_quantity + received_quantity')), 'volume'],
      ],
      where: {
        storeId,
        createdAt: { [Op.gte]: thirtyDaysAgo },
      },
      group: ['vendorId', 'vendor.id', 'vendor.name'],
      include: [{ model: Vendor, attributes: ['name'] }],
      order: [[literal('volume'), 'DESC']],
      limit: 5,
      raw: true,
      nest: true,
    });

    return rows.map((r) => ({
      vendorId: Number(r.vendorId),
      vendorName: r.vendor?.name ?? `Vendor #${r.vendorId}`,
      volume: Number(r.volume) || 0,
    }));
  }

  // (C) defectRateTrend — 12주 주간 루프 (월요일 시작)
  private async computeDefectRateTrend(storeId: number): Promise<DefectRateTrendPoint[]> {
    const points: DefectRateTrendPoint[] = [];
    const now = new Date();

    // 최근 12주 (현재 주 포함) 각 주 Monday 00:00 ~ Sunday 23:59:59
    for (let i = 11; i >= 0; i--) {
      const weekStart = new Date(now);
      weekStart.setDate(now.getDate() - (i * 7 + now.getDay() - 1));  // 월요일
      weekStart.setHours(0, 0, 0, 0);

      const weekEnd = new Date(weekStart);
      weekEnd.setDate(weekStart.getDate() + 7);

      // recepcion.sum(received_quantity) in week
      const recepcionSum = await this.recepcionModel.sum('receivedQuantity', {
        where: {
          storeId,
          recepcionDate: { [Op.gte]: weekStart, [Op.lt]: weekEnd },
        },
      }) ?? 0;

      // qc_items.sum(quantity) in week (storeId 는 recepcion join 필요 — 간단히 created_at 기준)
      const qcSum = await this.qcItemModel.sum('quantity', {
        where: {
          createdAt: { [Op.gte]: weekStart, [Op.lt]: weekEnd },
        },
        include: [{
          association: 'recepcion',
          where: { storeId },
          required: true,
          attributes: [],
        }],
      } as any) ?? 0;

      const defectRate = recepcionSum > 0 ? (qcSum / recepcionSum) * 100 : 0;

      points.push({
        weekStart: weekStart.toISOString().split('T')[0],
        defectRate: Math.round(defectRate * 10) / 10,  // 소수점 1자리
        qcCount: Number(qcSum),
        recepcionCount: Number(recepcionSum),
      });
    }

    return points;
  }

  // (D) backlogByEtapa — etapa 별 pendingQuantity SUM
  private async computeBacklogByEtapa(storeId: number): Promise<BacklogByEtapaRow[]> {
    // 모든 etapa 기본 포함 (backlog 0 인 etapa 도 UI 일관성)
    const etapas = await this.etapaModel.findAll({
      where: { storeId },
      attributes: ['id', 'name', 'order'],
      order: [['order', 'ASC']],
      raw: true,
    });

    const backlogs: any[] = await this.envioModel.findAll({
      attributes: [
        'etapaId',
        [fn('SUM', col('pending_quantity')), 'pendingQuantity'],
      ],
      where: {
        storeId,
        status: { [Op.in]: [EnvioStatus.PENDING, EnvioStatus.PARTIAL] },
      },
      group: ['etapaId'],
      raw: true,
    });

    const backlogMap = new Map<number, number>();
    backlogs.forEach((b) => backlogMap.set(Number(b.etapaId), Number(b.pendingQuantity) || 0));

    return etapas.map((e: any) => ({
      etapaId: e.id,
      etapaName: e.name,
      order: e.order,
      pendingQuantity: backlogMap.get(e.id) ?? 0,
    }));
  }

  // (E) urgentLateEnvios — LATE 상위 5건 (UrgentActionTable 용)
  private async computeUrgentLateEnvios(storeId: number) {
    const todayMidnight = new Date();
    todayMidnight.setHours(0, 0, 0, 0);

    const rows: any[] = await this.envioModel.findAll({
      where: {
        storeId,
        status: { [Op.in]: [EnvioStatus.PENDING, EnvioStatus.PARTIAL] },
        dueDate: { [Op.lt]: todayMidnight },
        pendingQuantity: { [Op.gt]: 0 },
      },
      include: [
        { model: Vendor, attributes: ['name'] },
        { model: Etapa, attributes: ['name'] },
      ],
      order: [['pendingQuantity', 'DESC']],
      limit: 5,
      raw: true,
      nest: true,
    });

    return rows.map((r) => {
      const dueDate = new Date(r.dueDate);
      const daysOverdue = Math.ceil((todayMidnight.getTime() - dueDate.getTime()) / 86_400_000);

      return {
        envioId: Number(r.id),
        vendorName: r.vendor?.name ?? `Vendor #${r.vendorId}`,
        etapaName: r.etapa?.name ?? `Etapa #${r.etapaId}`,
        pendingQuantity: Number(r.pendingQuantity) || 0,
        dueDate: String(r.dueDate).slice(0, 10),
        daysOverdue,
      };
    });
  }
}
```

## 7) `utils/talleres-xlsx-export.ts`

```typescript
// Wave 8: 재사용 가능한 Excel 내보내기 유틸 — dynamic import('xlsx') 로 초기 번들 크기 최적화
export interface ExportColumn<T> {
  key: keyof T
  label: string
  format?: 'date' | 'number' | 'currency'
}

export async function exportTalleresToExcel<T extends Record<string, any>>(
  rows: T[],
  filename: string,
  sheetName: string,
  columns: ExportColumn<T>[],
): Promise<void> {
  if (rows.length === 0) {

    return
  }

  // dynamic import (lazy-load) — 초기 JS 번들에서 xlsx 제외
  const XLSX = await import('xlsx')

  // 컬럼 매핑: row 에서 label 기준으로 재정렬
  const mapped = rows.map(r => {
    const out: Record<string, any> = {}
    columns.forEach(c => {
      const v = r[c.key]
      if (c.format === 'date' && v) {
        out[c.label] = new Date(v as any).toISOString().slice(0, 10)
      } else if (c.format === 'number' && v != null) {
        out[c.label] = Number(v)
      } else if (c.format === 'currency' && v != null) {
        out[c.label] = `$${Number(v).toLocaleString()}`
      } else {
        out[c.label] = v ?? ''
      }
    })

    return out
  })

  const ws = XLSX.utils.json_to_sheet(mapped)
  const wb = XLSX.utils.book_new()
  XLSX.utils.book_append_sheet(wb, ws, sheetName.slice(0, 31))  // Excel 시트명 31자 제한
  XLSX.writeFile(wb, filename.endsWith('.xlsx') ? filename : `${filename}.xlsx`)
}
```

```markdown
# Manual de Talleres — Ventago

**Versión:** 1.0 (Wave 8, 2026-04-24)
**Audiencia:** Administradores y gerentes de marcas de indumentaria que utilizan Ventago para gestionar producción externa (talleres / CMT).

---

## 1. Introducción

Ventago Talleres es un módulo de control de producción externa (CMT — Cut, Make, Trim) inspirado en Zedonk. Permite:

- Planificar cortes (**Cut Ticket**) y simular márgenes (**Cost Sheet**) antes de producir.
- Monitorear el estado de cada **Envío** a taller con semáforo de salud (En curso / En riesgo / Atrasado).
- Registrar recepciones con **Control de Calidad (QC)** estructurado y generar retrabajos automáticos.
- Confirmar **Liquidaciones** con cálculo automático a partir de recepciones y tarifas vigentes.
- Recibir alertas diarias a las 08:00 sobre envíos atrasados.

<!-- screenshot: /docs/img/talleres-overview.png -->

## 2. Overview y KPIs

La pestaña **📊 Overview** es su punto de partida diario.

**4 KPIs principales:**
- **Envíos atrasados** (rojo) — acción inmediata requerida
- **En riesgo** (naranja) — vencen en <2 días
- **En curso** (verde) — dentro del plazo
- **Total activos** — suma global

**4 gráficos:**
- **Donut de estado** — distribución de envíos por salud

**Acción inmediata:** la tabla al pie muestra los 5 envíos más atrasados con drilldown directo a la pestaña Envíos.

<!-- screenshot: /docs/img/talleres-kpi-cards.png -->

## 3. Cut Ticket: creación y matriz de tallas (Wave 9)

El **Cut Ticket** (ticket de corte) es el origen de cada producción. Cada Lote tiene un Cut Ticket numerado (`CT-2026-025`) con:

- **Matriz talla × color** — cantidad por combinación
- **BOM snapshot** — materiales requeridos con precio del momento
- **Routing path** — flujo de etapas (Corte → Confección → Planchado → QC → Empaque)
- **PDF A4 horizontal** para pegar en la pared del taller

<!-- screenshot: /docs/img/cut-ticket-matrix.png -->

### Cómo crear un Cut Ticket

1. Pestaña **Cut Ticket** → seleccione un Lote existente o cree uno nuevo.
2. Complete los 8 campos del header (estilo, temporada, fechas, etc.).
3. Edite la matriz talla × color (input numéricos, totales automáticos).
4. Revise el BOM (badges de stock SUFICIENTE / BAJO / AGOTADO).
5. Click **📄 PDF (re-impresión)** para generar el ticket.

**Atención:** una vez que se establece `cut_date`, la matriz queda congelada (read-only).

## 4. WIP y Kanban semáforo (Wave 5)

La pestaña **Kanban** muestra todas las envíos activos organizados por etapa, con tarjetas de color según salud:

- 🟢 **Verde** (ON_TRACK) — dueDate ≥ hoy + 2 días
- 🟡 **Amarillo** (AT_RISK) — dueDate ∈ [hoy, hoy+2)
- 🔴 **Rojo** (LATE) — dueDate < hoy y quedan unidades pendientes

**Drag & drop:** reordene tarjetas dentro de la misma columna para ajustar prioridades (se guarda automáticamente).

**Filtros:** vendor, producto, "sólo LATE".

<!-- screenshot: /docs/img/kanban-semaforo.png -->

## 5. Control de Calidad (QC) y Retrabajo (Wave 6)

Al recibir un envío, registre QC estructurado:

- **Código de defecto** (costura abierta / mancha / talla incorrecta / etc.)
- **Severidad** (menor / mayor / crítico)
- **Acción:**
  - `ACCEPT` — aceptar con defecto leve
  - `REWORK` — crea envío hijo automático para retrabajo
  - `SCRAP` — descartar (reduce stock del lote)
  - `CLAIM` — descuento a vendor (crea defect con deductionAmount)
- **Foto** — subida con redimensionado cliente a 1280px

**Vendor Scorecard:** defectRate, onTimeRate, reworkRate de los últimos 90 días + sparkline 12 semanas.

<!-- screenshot: /docs/img/qc-reception.png -->

## 6. Liquidaciones: generación automática y confirmación (Wave 7)

Genere liquidaciones (settlements) automáticamente por vendor + período:

1. Pestaña **Liquidaciones** → **Generar borrador** → seleccione vendor + fechas.
2. El sistema calcula: Σ (recepciones netas × tarifa vigente en la fecha) − descuentos por defectos.
3. Revise las líneas (envío → recepción → etapa → qty × unitPrice).
4. Click **Confirmar** — pasa a estado `CONFIRMED` (inmutable).
5. Descargue PDF con CUIT, expedidor, vendor, líneas detalladas.

**Importante:** una vez confirmada, no se puede editar (para cumplimiento fiscal).

**Tarifas históricas:** cada tarifa (VendorEtapa) tiene `effectiveFrom/effectiveTo` — si cambia el precio hoy, las liquidaciones de ayer siguen usando el precio anterior.

<!-- screenshot: /docs/img/liquidacion-draft.png -->

## 7. Cost Sheet: simulación de márgenes (Wave 10)

Simule el margen de un estilo antes de producir:

- **Material cost** — Σ (BOM.quantity × material.standardPrice)
- **CMT cost** — Σ (tarifas vigentes de cada etapa del routing)
- **Overhead** — (material+CMT) × overheadPct + shipping/loteSize
- **Total cost** — suma de los tres
- **Margen** — retailPrice − totalCost, con % objetivo configurable

**MarginCard** (fondo navy + dorado) muestra el % de margen:
- ✅ Verde si alcanza el objetivo (por defecto 50%)
- ⚠️ Rojo si está por debajo

**Edición en vivo:** modifique `retail_price`, `target_margin_pct`, `overhead_pct`, etc. — se recalcula automáticamente con 500ms de debounce.

<!-- screenshot: /docs/img/cost-sheet-margin.png -->

## 8. Alertas y Cron 08:00 (Wave 8)

**Alertas automáticas:** todos los días a las **08:00 (America/Bogota)** el sistema escanea todos los envíos con `dueDate < hoy` y `pendingQuantity > 0` y genera una notificación tipo `LATE` a cada vendor.

Los vendors reciben la alerta en:
- App móvil (Flutter — Portal de Vendors)
- Correo electrónico (si está configurado)
- Dashboard del Portal

**Duplicados:** el cron evita crear la misma alerta dos veces por día (una sola por envío por día).

**Fallas del cron:** si el cron falla (por ejemplo, caída de DB), el error se registra pero el siguiente día se reintenta automáticamente.

**Configuración:** el horario es fijo a las 08:00. Si necesita cambiarlo, contacte soporte.

<!-- screenshot: /docs/img/cron-alert-mobile.png -->

## 9. Exportación a Excel y FAQs

**Excel:** en cada pestaña de lista (Envios / Talleres / Liquidaciones) encontrará el botón **📥 Excel** en la parte superior derecha. Exporta los datos con los filtros actualmente aplicados.

**FAQs:**

**¿Puedo editar una liquidación después de confirmarla?**
No. Una vez confirmada (`CONFIRMED`), la liquidación es inmutable por razones fiscales. Para correcciones, cree una nueva liquidación con ajustes.

**¿Qué pasa si un vendor cambia su tarifa?**
Cree una nueva tarifa con `effectiveFrom = fecha de cambio`. Las liquidaciones anteriores no se ven afectadas.

**¿El Cut Ticket se puede reimprimir?**
Sí, ilimitadamente. El número (CT-YYYY-NNN) es único y permanente.

**¿El Cost Sheet se actualiza solo cuando cambian los precios de los materiales?**
No automáticamente. Debe hacer click en **🔄 Recalcular** o editar cualquier campo para disparar el recálculo.

**¿Cómo saber si el cron de 08:00 funcionó?**
En los logs del servidor busque `LATE 알림 크론 완료`. Los vendors verán las notificaciones en su app móvil.
```
