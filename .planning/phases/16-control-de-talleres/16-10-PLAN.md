```sql
BEGIN;

-- (A) style_cost_sheets — product × store 당 1개의 마진 시뮬레이션 시트
CREATE TABLE IF NOT EXISTS style_cost_sheets (
  id                      SERIAL PRIMARY KEY,
  product_id              INTEGER NOT NULL REFERENCES products(id) ON DELETE CASCADE,
  store_id                INTEGER NOT NULL REFERENCES stores(id)   ON DELETE CASCADE,
  currency                CHAR(3)       NOT NULL DEFAULT 'USD',
  retail_price            NUMERIC(12,2),
  target_margin_pct       NUMERIC(5,2)  NOT NULL DEFAULT 50.00
                            CHECK (target_margin_pct >= 0 AND target_margin_pct <= 100),
  overhead_pct            NUMERIC(5,2)  NOT NULL DEFAULT 11.30
                            CHECK (overhead_pct >= 0 AND overhead_pct <= 100),
  shipping_cost_per_lote  NUMERIC(12,2) NOT NULL DEFAULT 200.00
                            CHECK (shipping_cost_per_lote >= 0),
  lote_size_default       INTEGER       NOT NULL DEFAULT 155
                            CHECK (lote_size_default >= 1),
  material_cost           NUMERIC(12,2),
  cmt_cost                NUMERIC(12,2),
  overhead_cost           NUMERIC(12,2),
  total_cost              NUMERIC(12,2),
  margin_amount           NUMERIC(12,2),
  margin_pct              NUMERIC(5,2),
  calc_snapshot           JSONB,
  last_calculated_at      TIMESTAMPTZ,
  created_at              TIMESTAMPTZ   NOT NULL DEFAULT NOW(),
  updated_at              TIMESTAMPTZ   NOT NULL DEFAULT NOW()
);

-- (B) UNIQUE: (product_id, store_id) — product 당 1개 sheet (Sequelize 멱등 UPSERT 기반)
DO $$
BEGIN
  ALTER TABLE style_cost_sheets
    ADD CONSTRAINT uq_cost_product_store UNIQUE (product_id, store_id);
EXCEPTION WHEN duplicate_object THEN
  NULL;
END $$;

-- (C) index: store 스코프 조회 (list 뷰에서 사용 예정, Wave 10 은 by product 이지만 향후 Dashboard)
CREATE INDEX IF NOT EXISTS idx_cost_sheets_store
  ON style_cost_sheets (store_id);

COMMENT ON TABLE style_cost_sheets IS
  'Wave 10 Cost Sheet — style 별 원가/마진 시뮬레이션 (product × store UNIQUE)';
COMMENT ON COLUMN style_cost_sheets.target_margin_pct IS

  '계산 시점 전체 스냅샷 {materials,cmt,overhead,retailPrice,totalCost,marginAmount,marginPct,computedAt}';

  '간접비 % (기본 11.3%) — (material+cmt) × overhead_pct/100';
COMMENT ON COLUMN style_cost_sheets.shipping_cost_per_lote IS
  '로트당 배송비 — shipping/lote_size 로 단위당 간접비에 반영';

-- (D) 운영 GRANT 누락 방지 (Wave 6/7/9 의 반복 교훈)

GRANT USAGE, SELECT ON SEQUENCE style_cost_sheets_id_seq TO coolsistema;

COMMIT;
```

```typescript
import { Injectable, BadRequestException, InternalServerErrorException, Logger, NotFoundException } from '@nestjs/common';
import { InjectModel } from '@nestjs/sequelize';
import { Transaction } from 'sequelize';
import { Sequelize } from 'sequelize-typescript';
import { MemoryCacheService } from 'src/common/cache/memory-cache.service';
import { StyleCostSheet } from './style-cost-sheet.model';
import { Product } from '../../products/products.model';
import { Bom } from '../../production/bom/bom.model';
import { BomItem } from '../../production/bom/bom-item.model';
import { Material } from '../../production/materials/materials.model';
import { Etapa } from '../etapas/etapa.model';
import { Vendor } from '../vendors/vendor.model';
import { VendorEtapaService } from '../vendor-etapas/vendor-etapa.service';
import { Users } from '../../users/users.model';
import {
  BomLine,
  BomStockStatus,
  CalcSnapshot,
  CmtLine,
  CostSheetDetailResponse,
  OverheadDetail,
  StyleCostSheetDto,
} from './cost-sheet.types';

const CACHE_TTL_MS = 5 * 60 * 1000;

@Injectable()
export class StyleCostSheetService {
  private readonly logger = new Logger(StyleCostSheetService.name);

  constructor(
    @InjectModel(StyleCostSheet) private readonly sheetModel: typeof StyleCostSheet,
    @InjectModel(Product) private readonly productModel: typeof Product,
    @InjectModel(Bom) private readonly bomModel: typeof Bom,
    @InjectModel(BomItem) private readonly bomItemModel: typeof BomItem,
    @InjectModel(Material) private readonly materialModel: typeof Material,
    @InjectModel(Etapa) private readonly etapaModel: typeof Etapa,
    @InjectModel(Vendor) private readonly vendorModel: typeof Vendor,
    private readonly vendorEtapaService: VendorEtapaService,
    private readonly sequelize: Sequelize,
    private readonly cacheService: MemoryCacheService,
  ) {}

  // ===========================================================================
  // compute — POST /cost-sheets/:productId/calculate (핵심)
  // ===========================================================================
  async compute(productId: number, storeId: number, user: Users): Promise<StyleCostSheetDto> {
    this.logger.log(`[compute] productId=${productId} storeId=${storeId} user=${user.id}`);

    return this.sequelize.transaction(async tx => {
      // 1) Product 검증 + storeId 격리 (INV-CS-1)
      const product = await this.productModel.findOne({
        where: { id: productId, storeId },
        transaction: tx,
      });
      if (!product) {
        throw new NotFoundException(`Product #${productId} no encontrado en store ${storeId}`);
      }

      // 2) BOM 스냅샷 (INV-CS-6 graceful)
      const materials = await this.buildMaterialSnapshot(productId, storeId, tx);
      const materialCost = materials.reduce((s, m) => s + m.subtotal, 0);
      this.logger.debug(`[compute] material_cost=${materialCost.toFixed(2)} (lines=${materials.length})`);

      // 3) CMT 스냅샷 (INV-CS-7 graceful)
      const cmt = await this.buildCmtSnapshot(storeId, tx);
      const cmtCost = cmt.reduce((s, c) => s + c.unitPrice, 0);
      this.logger.debug(`[compute] cmt_cost=${cmtCost.toFixed(2)} (lines=${cmt.length})`);

      // 4) 기존 sheet 조회 또는 생성 (편집 가능 필드는 기본값 사용)
      const [sheet] = await this.sheetModel.findOrCreate({
        where: { productId, storeId },
        defaults: {
          productId,
          storeId,
          currency: 'USD',
          targetMarginPct: 50,
          overheadPct: 11.3,
          shippingCostPerLote: 200,
          loteSizeDefault: 155,
        } as any,
        transaction: tx,
      });

      // 5) Overhead 계산
      const overhead: OverheadDetail = this.computeOverhead(
        materialCost,
        cmtCost,
        Number(sheet.overheadPct),
        Number(sheet.shippingCostPerLote),
        sheet.loteSizeDefault,
      );

      // 6) Total + Margin
      const totalCost = materialCost + cmtCost + overhead.total;

      // INV-CS-5: total_cost >= 0 방어
      if (totalCost < 0) {
        this.logger.error(`[compute] total_cost=${totalCost} < 0 — aborting`);
        throw new InternalServerErrorException('total_cost 가 음수 — 계산 오류');
      }

      const retailPrice = sheet.retailPrice != null ? Number(sheet.retailPrice) : null;
      const marginAmount = retailPrice != null ? retailPrice - totalCost : null;
      const marginPct = retailPrice != null && retailPrice > 0
        ? (marginAmount! / retailPrice) * 100
        : null;

      // 7) calc_snapshot 구성
      const snapshot: CalcSnapshot = {
        materials,
        cmt,
        overhead,
        retailPrice,
        totalCost,
        marginAmount,
        marginPct,
        computedAt: new Date().toISOString(),
        notes: this.buildNotes(materials.length === 0, cmt.some(c => c.status === 'NO_RATE')),
      };

      // 8) UPSERT
      await sheet.update(
        {
          materialCost,
          cmtCost,
          overheadCost: overhead.total,
          totalCost,
          marginAmount,
          marginPct,
          calcSnapshot: snapshot,
          lastCalculatedAt: new Date(),
        },
        { transaction: tx },
      );

      // 9) 캐시 무효화 (tx.afterCommit)
      tx.afterCommit(() => {
        this.cacheService.delByPrefix(`talleres:cost-sheet:${storeId}:${productId}`);
        this.logger.log(`[compute] cache invalidated product=${productId} store=${storeId}`);
      });

      this.logger.log(
        `[compute] DONE product=${productId} total=${totalCost.toFixed(2)} margin=${marginPct?.toFixed(2) ?? 'n/a'}%`,
      );

      return this.toDto(sheet);
    });
  }

  // ===========================================================================
  // findByProduct — GET /cost-sheets/:productId
  // hasCostSheet 플래그 (Wave 9 패턴) — 404 대신 200
  // ===========================================================================
  async findByProduct(productId: number, storeId: number): Promise<CostSheetDetailResponse> {
    const cacheKey = `talleres:cost-sheet:${storeId}:${productId}`;
    const cached = this.cacheService.get<CostSheetDetailResponse>(cacheKey);
    if (cached) {
      this.logger.debug(`[findByProduct] cache hit ${cacheKey}`);

      return cached;
    }

    const product = await this.productModel.findOne({
      where: { id: productId, storeId },
    });
    if (!product) {
      throw new NotFoundException(`Product #${productId} no encontrado en store ${storeId}`);
    }

    const sheet = await this.sheetModel.findOne({
      where: { productId, storeId },
    });

    const resp: CostSheetDetailResponse = {
      hasCostSheet: !!sheet,
      product: { id: product.id, name: product.name, code: (product as any).code ?? null, storeId },
      sheet: sheet ? this.toDto(sheet) : null,
    };

    this.cacheService.set(cacheKey, resp, CACHE_TTL_MS);
    this.logger.debug(`[findByProduct] cache set ${cacheKey} hasCostSheet=${resp.hasCostSheet}`);

    return resp;
  }

  // ===========================================================================
  // updateEditableFields — PATCH /cost-sheets/:productId
  // INV-CS-2/3/4 validation
  // ===========================================================================
  async updateEditableFields(
    productId: number,
    storeId: number,
    partial: {
      retailPrice?: number;
      targetMarginPct?: number;
      overheadPct?: number;
      shippingCostPerLote?: number;
      loteSizeDefault?: number;
    },
    user: Users,
  ): Promise<StyleCostSheetDto> {
    this.logger.log(`[updateEditable] productId=${productId} user=${user.id} partial=${JSON.stringify(partial)}`);

    // INV-CS-2/3/4 검증
    if (partial.retailPrice != null && partial.retailPrice < 0) {
      throw new BadRequestException('retailPrice must be >= 0');
    }
    if (partial.targetMarginPct != null && (partial.targetMarginPct < 0 || partial.targetMarginPct > 100)) {
      throw new BadRequestException('targetMarginPct must be in [0, 100]');
    }
    if (partial.overheadPct != null && (partial.overheadPct < 0 || partial.overheadPct > 100)) {
      throw new BadRequestException('overheadPct must be in [0, 100]');
    }
    if (partial.shippingCostPerLote != null && partial.shippingCostPerLote < 0) {
      throw new BadRequestException('shippingCostPerLote must be >= 0');
    }
    if (partial.loteSizeDefault != null && (partial.loteSizeDefault < 1 || !Number.isInteger(partial.loteSizeDefault))) {
      throw new BadRequestException('loteSizeDefault must be integer >= 1');
    }

    return this.sequelize.transaction(async tx => {
      // FOR UPDATE 잠금 — 동시 PATCH race 방지
      let sheet = await this.sheetModel.findOne({
        where: { productId, storeId },
        transaction: tx,
        lock: (tx as any).LOCK.UPDATE,
      });

      if (!sheet) {
        // 편집 시도 시 아직 sheet 없으면 기본값으로 생성
        sheet = await this.sheetModel.create(
          {
            productId, storeId,
            currency: 'USD',
            targetMarginPct: 50,
            overheadPct: 11.3,
            shippingCostPerLote: 200,
            loteSizeDefault: 155,
            ...partial,
          } as any,
          { transaction: tx },
        );
      } else {
        await sheet.update(partial as any, { transaction: tx });
      }

      tx.afterCommit(() => {
        this.cacheService.delByPrefix(`talleres:cost-sheet:${storeId}:${productId}`);
      });

      return this.toDto(sheet);
    });
  }

  // ===========================================================================
  // Helper: buildMaterialSnapshot (BOM → BomLine[])
  // INV-CS-6: BOM 없으면 빈 배열 (에러 아님)
  // ===========================================================================
  private async buildMaterialSnapshot(
    productId: number,
    storeId: number,
    tx: Transaction,
  ): Promise<BomLine[]> {
    const bom = await this.bomModel.findOne({
      where: { productId, storeId, isActive: true },
      include: [{ model: BomItem, include: [Material] }],
      transaction: tx,
    });

    if (!bom || !bom.items || bom.items.length === 0) {
      this.logger.log(`[buildMaterialSnapshot] product=${productId} — no active BOM (graceful empty)`);

      return [];
    }

    return bom.items
      .filter(i => i.material != null)   // subProductId 만 있는 라인은 제외 (Wave 10 scope)
      .map(i => {
        const unitPrice = Number(i.material.standardPrice ?? 0);
        const quantity = Number(i.quantity);
        const subtotal = quantity * unitPrice;
        const stock = Number(i.material.currentStock ?? 0);
        const minStock = Number(i.material.minStock ?? 0);
        const stockStatus: BomStockStatus = stock <= 0 ? 'OUT' : stock <= minStock ? 'LOW' : 'SUFFICIENT';

        return {
          materialId: i.materialId!,
          code: i.material.code,
          name: i.material.name,
          unit: i.unit || i.material.unit || 'unid',
          quantity,
          unitPrice,
          subtotal,
          stockStatus,
        };
      });
  }

  // ===========================================================================
  // Helper: buildCmtSnapshot (talleres_etapas × getRateAt(today))
  // INV-CS-7: rate 없는 etapa → unitPrice=0 + status='NO_RATE'
  // ===========================================================================
  private async buildCmtSnapshot(storeId: number, tx: Transaction): Promise<CmtLine[]> {
    const etapas = await this.etapaModel.findAll({
      where: { storeId },
      order: [['order', 'ASC']],
      transaction: tx,
    });

    if (!etapas.length) {
      this.logger.log(`[buildCmtSnapshot] store=${storeId} — no etapas defined (graceful empty)`);

      return [];
    }

    const today = new Date().toISOString().slice(0, 10);

    // vendor 는 Wave 10 scope 에서 store-wide 평균이 아닌 "primary vendor" 를 찾지 않고,
    // 각 etapa 에 대해 store 내 active rate 중 하나를 선택 (현재 vendor 구조는 etapa↔vendor 다대다).
    // 단순화: 각 etapa 에 대해 store 의 모든 vendor 중 가장 최근 active rate 를 선택.
    const cmtLines: CmtLine[] = await Promise.all(
      etapas.map(async (e, idx) => {
        // vendor-etapa 에서 해당 etapa 에 대한 active rate 중 가장 최근 1건
        const activeRates = await (this.vendorEtapaService as any).vendorEtapaModel.findAll({
          where: {
            etapaId: e.id,
            effectiveFrom: { $lte: today } as any,
          },
          order: [['effectiveFrom', 'DESC']],
          limit: 1,
          transaction: tx,
        });

        const active = activeRates[0];
        if (!active || (active.effectiveTo && active.effectiveTo < today)) {
          return {
            etapaId: e.id,
            etapaName: e.name,
            order: idx + 1,
            vendorId: null,
            vendorName: 'In-house',
            unitPrice: 0,
            status: 'NO_RATE' as const,
          };
        }

        const vendor = active.vendorId
          ? await this.vendorModel.findOne({ where: { id: active.vendorId, storeId }, transaction: tx })
          : null;

        return {
          etapaId: e.id,
          etapaName: e.name,
          order: idx + 1,
          vendorId: active.vendorId ?? null,
          vendorName: vendor?.name || 'In-house',
          unitPrice: Number(active.unitPrice ?? 0),
          status: (vendor ? 'OK' : 'IN_HOUSE') as 'OK' | 'IN_HOUSE',
        };
      }),
    );

    return cmtLines;
  }

  // ===========================================================================
  // Helper: computeOverhead (EXTENSION §3 formula)
  // ===========================================================================
  private computeOverhead(
    materialCost: number,
    cmtCost: number,
    overheadPct: number,
    shippingPerLote: number,
    loteSize: number,
  ): OverheadDetail {
    const overheadBase = materialCost + cmtCost;
    const overheadFromPct = overheadBase * (overheadPct / 100);
    const shippingPerUnit = loteSize > 0 ? shippingPerLote / loteSize : 0;
    const total = overheadFromPct + shippingPerUnit;

    return {
      overheadBase,
      overheadPct,
      overheadFromPct,
      shippingPerLote,
      loteSize,
      shippingPerUnit,
      total,
    };
  }

  private buildNotes(bomEmpty: boolean, hasNoRate: boolean): string | undefined {
    const parts: string[] = [];
    if (bomEmpty) parts.push('BOM not configured');
    if (hasNoRate) parts.push('Some etapas have no active rate');

    return parts.length ? parts.join('; ') : undefined;
  }

  // ===========================================================================
  // toDto — Sequelize model → plain DTO
  // ===========================================================================
  private toDto(s: StyleCostSheet): StyleCostSheetDto {
    return {
      id: s.id,
      productId: s.productId,
      storeId: s.storeId,
      currency: s.currency,
      retailPrice: s.retailPrice != null ? Number(s.retailPrice) : null,
      targetMarginPct: Number(s.targetMarginPct),
      overheadPct: Number(s.overheadPct),
      shippingCostPerLote: Number(s.shippingCostPerLote),
      loteSizeDefault: s.loteSizeDefault,
      materialCost: s.materialCost != null ? Number(s.materialCost) : null,
      cmtCost: s.cmtCost != null ? Number(s.cmtCost) : null,
      overheadCost: s.overheadCost != null ? Number(s.overheadCost) : null,
      totalCost: s.totalCost != null ? Number(s.totalCost) : null,
      marginAmount: s.marginAmount != null ? Number(s.marginAmount) : null,
      marginPct: s.marginPct != null ? Number(s.marginPct) : null,
      calcSnapshot: s.calcSnapshot,
      lastCalculatedAt: s.lastCalculatedAt ? s.lastCalculatedAt.toISOString() : null,
      createdAt: (s as any).createdAt?.toISOString?.() ?? String((s as any).createdAt),
      updatedAt: (s as any).updatedAt?.toISOString?.() ?? String((s as any).updatedAt),
    };
  }
}
```

```typescript
  @Post(':productId/calculate')
```

```typescript
import useSWR from 'swr'
import apiConnector from 'src/services/api.service'
import type { CostSheetDetailResponse } from 'src/views/talleres/cost-sheet/types'

const costSheetFetcher = async (url: string): Promise<CostSheetDetailResponse> => {
  const res = await apiConnector.get(url)
  const payload = (res as any)?.data?.data || (res as any)?.data || res
  return payload
}

export function useCostSheet(productId: number | null) {
  const key = productId ? `/talleres/cost-sheets/${productId}` : null
  const { data, error, isLoading, mutate } = useSWR<CostSheetDetailResponse>(
    key,
    costSheetFetcher,
    { dedupingInterval: 300_000, revalidateOnFocus: false },
  )

  return {
    detail: data,
    hasCostSheet: !!data?.hasCostSheet,
    sheet: data?.sheet ?? null,
    product: data?.product ?? null,
    isLoading,
    error,
    mutate,
  }
}
```

## 7) `CostSheetEditableForm.tsx` — debounce(500ms) PATCH → POST calculate → mutate (≥ 100 lines)

```tsx
import React, { useCallback, useEffect, useRef, useState } from 'react'
import { Box, Card, Grid, TextField, Typography, CircularProgress } from '@mui/material'
import apiConnector from 'src/services/api.service'
import toast from 'react-hot-toast'
import { TALLERES_THEME } from '../../theme/zedonkTheme'
import type { StyleCostSheetDto } from '../types'

interface Props {
  sheet: StyleCostSheetDto
  onSaved: () => void  // parent 는 mutate() 호출
}

type EditField = 'retailPrice' | 'targetMarginPct' | 'overheadPct' | 'shippingCostPerLote' | 'loteSizeDefault'

function CostSheetEditableFormBase({ sheet, onSaved }: Props) {
  const [local, setLocal] = useState({
    retailPrice: sheet.retailPrice ?? 0,
    targetMarginPct: sheet.targetMarginPct,
    overheadPct: sheet.overheadPct,
    shippingCostPerLote: sheet.shippingCostPerLote,
    loteSizeDefault: sheet.loteSizeDefault,
  })
  const [saving, setSaving] = useState(false)
  const debounceRef = useRef<ReturnType<typeof setTimeout> | null>(null)

  // parent prop 변경 시 동기화 (recalculate 완료 후 최신값)
  useEffect(() => {
    setLocal({
      retailPrice: sheet.retailPrice ?? 0,
      targetMarginPct: sheet.targetMarginPct,
      overheadPct: sheet.overheadPct,
      shippingCostPerLote: sheet.shippingCostPerLote,
      loteSizeDefault: sheet.loteSizeDefault,
    })
  }, [sheet.id, sheet.lastCalculatedAt])

  const schedule = useCallback((patch: Partial<typeof local>) => {
    if (debounceRef.current) clearTimeout(debounceRef.current)
    debounceRef.current = setTimeout(async () => {
      setSaving(true)
      try {
        // 1) PATCH 편집 저장
        await apiConnector.instance.patch(`/talleres/cost-sheets/${sheet.productId}`, patch)
        // 2) POST 재계산
        await apiConnector.post(`/talleres/cost-sheets/${sheet.productId}/calculate`, {})
        // 3) SWR revalidate
        onSaved()
      } catch (err: any) {
        toast.error(err?.response?.data?.message || 'Error al guardar')
      } finally {
        setSaving(false)
      }
    }, 500)
  }, [sheet.productId, onSaved])

  const handleChange = (field: EditField) => (e: React.ChangeEvent<HTMLInputElement>) => {
    const raw = e.target.value
    const v = raw === '' ? 0 : parseFloat(raw)
    if (isNaN(v)) return
    const next = { ...local, [field]: v }
    setLocal(next)
    schedule({ [field]: v })
  }

  const input = (field: EditField, label: string, adornment?: string) => (
    <TextField
      size="small"
      fullWidth
      label={label}
      type="number"
      value={local[field]}
      onChange={handleChange(field)}
      InputProps={{ endAdornment: adornment ? <Typography sx={{ fontSize: 12, color: TALLERES_THEME.textLight }}>{adornment}</Typography> : undefined }}
      inputProps={{ min: field === 'loteSizeDefault' ? 1 : 0, step: field === 'loteSizeDefault' ? 1 : 0.01 }}
    />
  )

  return (
    <Card sx={{ p: 2, mb: 2 }}>
      <Box sx={{ display: 'flex', justifyContent: 'space-between', alignItems: 'center', mb: 1.5 }}>
        <Typography sx={{ fontSize: 16, fontWeight: 600, color: TALLERES_THEME.primary }}>
          Parámetros editables
        </Typography>
        <Box sx={{ display: 'flex', alignItems: 'center', gap: 1 }}>
          {saving && <CircularProgress size={14} sx={{ color: TALLERES_THEME.secondary }} />}
          <Typography sx={{ fontSize: 11, color: saving ? TALLERES_THEME.secondary : TALLERES_THEME.textMuted }}>
            {saving ? 'Guardando y recalculando...' : 'Cambios se guardan automáticamente (500ms)'}
          </Typography>
        </Box>
      </Box>
      <Grid container spacing={1.5}>
        <Grid item xs={12} sm={6} md={4}>{input('retailPrice', 'Precio venta', sheet.currency)}</Grid>
        <Grid item xs={12} sm={6} md={4}>{input('targetMarginPct', 'Margen objetivo', '%')}</Grid>
        <Grid item xs={12} sm={6} md={4}>{input('overheadPct', 'Overhead', '%')}</Grid>
        <Grid item xs={12} sm={6} md={4}>{input('shippingCostPerLote', 'Envío por lote', sheet.currency)}</Grid>
        <Grid item xs={12} sm={6} md={4}>{input('loteSizeDefault', 'Tamaño de lote', 'uni')}</Grid>
      </Grid>
    </Card>
  )
}

const CostSheetEditableForm = React.memo(CostSheetEditableFormBase)
export default CostSheetEditableForm
```
