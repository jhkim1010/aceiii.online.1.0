KPI 계산 (클라이언트측):
```typescript
const totalPending = settlements
  .filter((s: any) => s.status === 'OPEN')
  .reduce((sum: number, s: any) => sum + (s.netAmount || 0), 0)

const paidThisMonth = settlements
  .filter((s: any) => {
    if (s.status !== 'PAID') return false
    const paidDate = new Date(s.paidAt || s.updatedAt)
    const now = new Date()

    return paidDate.getMonth() === now.getMonth() && paidDate.getFullYear() === now.getFullYear()
  })
  .reduce((sum: number, s: any) => sum + (s.netAmount || 0), 0)

const avgUnitPrice = settlements.length > 0
  ? settlements.reduce((sum: number, s: any) => sum + (s.netAmount || 0), 0) /
    settlements.reduce((sum: number, s: any) => sum + (s.deliveredQuantity || 1), 0)
  : 0
```
