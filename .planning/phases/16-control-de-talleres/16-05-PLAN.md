```typescript
// 1) 순수 함수 — today 기본값은 new Date() 자정 기준 (time 무시)
private computeHealthStatus(envio: Pick<Envio, 'dueDate' | 'pendingQuantity'>, today: Date = new Date()): 'ON_TRACK' | 'AT_RISK' | 'LATE' {
  // dueDate 없으면 ON_TRACK
  if (!envio.dueDate) return 'ON_TRACK';

  const todayMidnight = new Date(today.getFullYear(), today.getMonth(), today.getDate());
  const due = new Date(envio.dueDate); // DATEONLY → UTC 자정
  const twoDaysFromNow = new Date(todayMidnight);
  twoDaysFromNow.setDate(twoDaysFromNow.getDate() + 2);

  // LATE: 기한 지났고 아직 미반환 수량 존재
  if (due < todayMidnight && envio.pendingQuantity > 0) return 'LATE';

  // AT_RISK: today <= due < today+2일
  if (due >= todayMidnight && due < twoDaysFromNow) return 'AT_RISK';

  return 'ON_TRACK';
}
```

5) **dashboard.service.ts** — `getKanbanBoard` 구현:
```typescript
async getKanbanBoard(storeId: number, branchId?: number) {
  const cacheKey = `talleres:kanban:${storeId}:${branchId ?? 'all'}`;
  const cached = this.cacheService.get<any>(cacheKey);
  if (cached) return cached;

  const today = new Date();

  // Etapas 전체 (order ASC)
  const etapas = await this.etapaModel.findAll({
    where: { storeId, isActive: true },
    order: [['order', 'ASC']],
    attributes: ['id', 'name', 'order'],
  });

  // 활성 envíos: PENDING | PARTIAL
  const envioWhere: any = {
    storeId,
    status: { [Op.in]: ['PENDING', 'PARTIAL'] },
  };

  const envios = await this.envioModel.findAll({
    where: envioWhere,
    include: [
      { model: this.vendorModel, attributes: ['id', 'name'] },
      {
        model: this.loteModel,
        attributes: ['id', 'loteNumber', 'branchId', 'productId'],
        where: branchId ? { branchId } : undefined,
        required: !!branchId, // branchId 필터 시 INNER JOIN
        include: [{ model: this.productModel, attributes: ['name'] }],
      },
      { model: this.etapaModel, attributes: ['id', 'name'] },
    ],
    order: [
      ['priority', 'DESC'],
      ['dueDate', 'ASC'],
    ],
  });

  const enviosByEtapa: Record<number, any[]> = {};
  for (const etapa of etapas) enviosByEtapa[etapa.id] = [];

  for (const e of envios) {
    const etapaBucket = enviosByEtapa[e.etapaId];
    if (!etapaBucket) continue; // 해당 etapa 비활성/삭제된 경우 skip
    etapaBucket.push({
      id: e.id,
      loteId: e.loteId,
      loteNumber: e.lote?.loteNumber ?? '-',
      vendorId: e.vendorId,
      vendorName: e.vendor?.name ?? '-',
      productName: e.lote?.product?.name ?? null,
      quantity: e.quantity,
      pendingQuantity: e.pendingQuantity,
      envioDate: e.envioDate,
      dueDate: e.dueDate,
      status: e.status,
      priority: e.priority ?? 0,
      healthStatus: computeHealthStatus({ dueDate: e.dueDate, pendingQuantity: e.pendingQuantity }, today),
    });
  }

  const result = {
    etapas: etapas.map(e => ({ id: e.id, name: e.name, order: e.order })),
    enviosByEtapa,
    generatedAt: new Date().toISOString(),
  };

  this.cacheService.set(cacheKey, result, 30_000); // 30s TTL
  return result;
}
```
