## 0) 운영 스키마 현실 (주석 근거)

- `talleres_lotes` 는 local PG18 vs 운영 PG10 모두 `status` 가 VARCHAR(20) 로 생성됨 (Sequelize sync 이상). Wave 9 는 **status 를 건드리지 않는다** — 신규 컬럼 7개만 ADD COLUMN IF NOT EXISTS 로 추가.
- ENUM 변경이 없으므로 Wave 7 의 Step2 (autocommit ALTER TYPE) 가 **불필요**. Step1(트랜잭션 DDL) + Step2(검증) 2-file 분리로 충분.
- Wave 6/7 에서 학습한 Sequelize sync 이상 방어: ADD COLUMN / CREATE INDEX / CREATE TABLE / ADD CONSTRAINT 모두 IF NOT EXISTS + DO $$ BEGIN ... EXCEPTION WHEN duplicate_object THEN NULL END $$.

```typescript
async generateCutTicket(
  loteId: number,
  storeId: number,
  user: Users,
  input: {
    styleCode?: string;
    season?: string;
    vendorAssignments?: Array<{ etapaId: number; vendorId: number | null }>;
    initialMatrix?: SizeColorMatrix;
  } = {},
): Promise<CutTicketDetailResponse> {
  this.logger.log(`generateCutTicket lote=${loteId} store=${storeId} user=${user.id}`);

  return this.sequelize.transaction(async tx => {
    // 1) Lote 존재 + storeId 격리 (INV-G)
    const lote = await this.loteModel.findOne({
      where: { id: loteId, storeId },
      include: [{ model: Product }],
      transaction: tx,
      lock: tx.LOCK.UPDATE,
    });
    if (!lote) throw new NotFoundException(`Lote #${loteId} no encontrado`);

    // 2) 멱등 (INV-B): 이미 cut_ticket_number 있으면 카운터 증가 없이 기존 반환
    if (lote.cutTicketNumber) {
      this.logger.log(`Lote #${loteId} already has cut_ticket_number=${lote.cutTicketNumber} — idempotent return`);

      return this.findCutTicketDetail(loteId, storeId);
    }

    // 3) Counter FOR UPDATE — 원자적 시퀀스 증가
    const year = new Date().getUTCFullYear();
    let counter = await this.counterModel.findOne({
      where: { storeId, year },
      transaction: tx,
      lock: tx.LOCK.UPDATE,
    });

    let nextSeq: number;
    if (!counter) {
      counter = await this.counterModel.create(
        { storeId, year, lastSeq: 1 },
        { transaction: tx },
      );
      nextSeq = 1;
    } else {
      nextSeq = counter.lastSeq + 1;
      await counter.update({ lastSeq: nextSeq }, { transaction: tx });
    }

    const cutTicketNumber = `CT-${year}-${String(nextSeq).padStart(3, '0')}`;

    // 4) BOM + routing 스냅샷 빌드
    const bom = await this.buildBomSnapshot(lote.productId, lote.totalQuantity, tx);
    const routing = await this.buildRoutingPath(storeId, input.vendorAssignments, tx);

    // 5) 초기 matrix (없으면 빈 매트릭스 — colors/sizes 빈 배열)
    const matrix: SizeColorMatrix = input.initialMatrix ?? { colors: [], sizes: [], qty: {} };

    // 6) Lote 저장
    await lote.update(
      {
        cutTicketNumber,
        styleCode: input.styleCode ?? null,
        season: input.season ?? null,
        sizeColorMatrix: matrix,
        bomSnapshot: bom,
        routingPath: routing,
      },
      { transaction: tx },
    );

    // 7) 캐시 무효화
    this.cacheService.delByPrefix(`talleres:cut-ticket:${storeId}:`);

    this.logger.log(`Generated ${cutTicketNumber} for lote #${loteId}`);

    return this.findCutTicketDetail(loteId, storeId);
  });
}
```

### 6.3 `buildRoutingPath(storeId, vendorAssignments?, tx?)` — talleres_etapas 기반

```typescript
async buildRoutingPath(
  storeId: number,
  vendorAssignments: Array<{ etapaId: number; vendorId: number | null }> = [],
  tx?: Transaction,
): Promise<RoutingPath> {
  const etapas = await this.etapaModel.findAll({
    where: { storeId },
    order: [['order', 'ASC']],
    transaction: tx,
  });

  const vendorMap = new Map<number, number | null>();
  vendorAssignments.forEach(a => vendorMap.set(a.etapaId, a.vendorId));

  // 필요한 vendor 한번에 조회 (pool 낭비 방지)
  const vendorIds = [...new Set([...vendorMap.values()].filter(v => v != null))] as number[];
  const vendors = vendorIds.length
    ? await this.vendorModel.findAll({ where: { id: vendorIds, storeId }, transaction: tx })
    : [];
  const vendorById = new Map(vendors.map(v => [v.id, v]));

  return etapas.map((e, idx) => {
    const assignedVendorId = vendorMap.has(e.id) ? vendorMap.get(e.id)! : null;
    const vendor = assignedVendorId ? vendorById.get(assignedVendorId) : null;

    return {
      order: idx + 1,
      etapaId: e.id,
      etapaName: e.name,
      vendorId: assignedVendorId,
      vendorName: vendor ? vendor.name : 'In-house',
    };
  });
}
```

### 6.4 `updateSizeColorMatrix(loteId, storeId, matrix, user)` — INV-C

```typescript
async updateSizeColorMatrix(
  loteId: number,
  storeId: number,
  matrix: SizeColorMatrix,
  user: Users,
): Promise<{ lote: Lote; totalQuantity: number }> {
  // matrix 검증 (INV-F)
  if (!matrix || !Array.isArray(matrix.colors) || !Array.isArray(matrix.sizes) || !matrix.qty) {
    throw new BadRequestException('Formato de matrix inválido');
  }

  let total = 0;
  for (const c of matrix.colors) {
    const row = matrix.qty[String(c.id)] || {};
    for (const s of matrix.sizes) {
      const v = row[String(s.id)];
      if (v != null && (!Number.isInteger(v) || v < 0)) {
        throw new BadRequestException(`matrix.qty[${c.id}][${s.id}] debe ser entero >= 0`);
      }
      total += v ?? 0;
    }
  }

  return this.sequelize.transaction(async tx => {
    const lote = await this.loteModel.findOne({
      where: { id: loteId, storeId },
      transaction: tx,
      lock: tx.LOCK.UPDATE,
    });
    if (!lote) throw new NotFoundException(`Lote #${loteId} no encontrado`);

    // INV-C: cut_date NOT NULL → 400
    if (lote.cutDate) {
      throw new CutTicketImmutableException(loteId, `cut_date=${lote.cutDate} 이미 설정됨 — 매트릭스 편집 불가`);
    }

    await lote.update(
      { sizeColorMatrix: matrix, totalQuantity: total },
      { transaction: tx },
    );

    this.cacheService.delByPrefix(`talleres:cut-ticket:${storeId}:`);
    this.logger.log(`Matrix updated lote=${loteId} total=${total} by user=${user.id}`);

    return { lote, totalQuantity: total };
  });
}
```

```typescript
async findCutTicketDetail(loteId: number, storeId: number): Promise<CutTicketDetailResponse> {
  const cacheKey = `talleres:cut-ticket:${storeId}:detail:${loteId}`;
  const cached = this.cacheService.get<CutTicketDetailResponse>(cacheKey);
  if (cached) return cached;

  const lote = await this.loteModel.findOne({
    where: { id: loteId, storeId },
    include: [{ model: Product }],
  });
  if (!lote) throw new NotFoundException(`Lote #${loteId} no encontrado`);
  if (!lote.cutTicketNumber) throw new CutTicketNotGeneratedException(loteId);

  const routing = (lote.routingPath || []) as RoutingPath;
  const assignedSewing = routing.find(r => r.vendorId != null)?.vendorName || 'In-house';

  const resp: CutTicketDetailResponse = {
    lote: {
      id: lote.id, loteNumber: lote.loteNumber, productId: lote.productId,
      productName: lote.product?.name ?? null, storeId: lote.storeId, status: String(lote.status),
    },
    matrix: (lote.sizeColorMatrix || { colors: [], sizes: [], qty: {} }) as SizeColorMatrix,
    bom: (lote.bomSnapshot || []) as BomSnapshot,
    routing,
    meta: {
      cutTicketNumber: lote.cutTicketNumber,
      styleCode: lote.styleCode,
      season: lote.season,
      issuedDate: lote.updatedAt ? String(lote.updatedAt).slice(0, 10) : null,
      dueDate: null,  // Wave 9 scope: null; Wave 10+ 에서 envio.dueDate 최소값으로 확장 가능
      cutDate: lote.cutDate,
      totalQuantity: lote.totalQuantity,
      assignedCut: 'In-house',
      assignedSewing,
    },
  };

  this.cacheService.set(cacheKey, resp, 2 * 60 * 1000);
  return resp;
}
```

```typescript
import { Injectable, Logger } from '@nestjs/common';
import PDFDocument from 'pdfkit';
import { CutTicketDetailResponse } from './cut-ticket.types';

@Injectable()
export class CutTicketPdfService {
  private readonly logger = new Logger(CutTicketPdfService.name);

  // Wave 9: A4 가로 (842 × 595 pt) — 재단실 벽 부착용
  async generate(
    detail: CutTicketDetailResponse,
    store: { name?: string; aliasName?: string; cuit?: string } | null,
  ): Promise<Buffer> {
    return new Promise<Buffer>((resolve, reject) => {
      const doc = new PDFDocument({ size: 'A4', layout: 'landscape', margin: 30 });
      const chunks: Buffer[] = [];
      doc.on('data', (c: Buffer) => chunks.push(c));
      doc.on('end', () => {
        const buf = Buffer.concat(chunks);
        this.logger.log(`PDF generated cut_ticket=${detail.meta.cutTicketNumber} size=${buf.length}B`);
        resolve(buf);
      });
      doc.on('error', reject);

      this.drawHeader(doc, detail, store);
      this.drawMatrix(doc, detail);
      this.drawBom(doc, detail);
      this.drawRouting(doc, detail);
      this.drawQrPlaceholder(doc, detail);

      doc.end();
    });
  }

  private drawHeader(doc: PDFKit.PDFDocument, d: CutTicketDetailResponse, store: any) {
    const storeName = this.sanitize(store?.aliasName || store?.name || 'Tienda');
    doc.fontSize(18).fillColor('#1a1a2e').text(storeName, 30, 30);
    doc.fontSize(22).fillColor('#1a1a2e')
      .text(this.sanitize(d.meta.cutTicketNumber), 30, 55);
    doc.fontSize(11).fillColor('#6b6b8c')
      .text(`Estilo: ${this.sanitize(d.meta.styleCode || '-')}   Temporada: ${this.sanitize(d.meta.season || '-')}`, 30, 82);
    doc.text(`Producto: ${this.sanitize(d.lote.productName || '-')}   Total: ${d.meta.totalQuantity}`, 30, 98);
    doc.text(`Asignado — Corte: ${this.sanitize(d.meta.assignedCut)}   Confección: ${this.sanitize(d.meta.assignedSewing)}`, 30, 114);
    doc.text(`Emitido: ${d.meta.issuedDate || '-'}   Corte iniciado: ${d.meta.cutDate || 'pendiente'}`, 30, 130);

    // gold 4px vertical bar next to title (Zedonk style)
    doc.rect(15, 55, 4, 22).fill('#f5a623');
    doc.moveTo(30, 150).lineTo(810, 150).strokeColor('#d5d7e0').stroke();
  }

  private drawMatrix(doc: PDFKit.PDFDocument, d: CutTicketDetailResponse) {
    const startY = 160;
    doc.fontSize(12).fillColor('#1a1a2e').text('Matriz Talle × Color', 30, startY);
    const { colors, sizes, qty } = d.matrix;
    if (!colors.length || !sizes.length) {
      doc.fontSize(10).fillColor('#8b8ba7').text('(matriz vacía — editar en app)', 30, startY + 18);
      return;
    }
    const colW = 45;
    const rowH = 18;
    let x = 30; let y = startY + 20;
    // Header row
    doc.fontSize(9).fillColor('#6b6b8c').text('Color \\ Talle', x, y, { width: 100 });
    x += 100;
    sizes.forEach(s => { doc.text(this.sanitize(s.name), x, y, { width: colW, align: 'center' }); x += colW; });
    doc.text('Total', x, y, { width: colW, align: 'center' });
    y += rowH;
    doc.moveTo(30, y - 3).lineTo(30 + 100 + sizes.length * colW + colW, y - 3).strokeColor('#eceef3').stroke();
    // Data rows + row totals
    const colTotals = new Array(sizes.length).fill(0);
    let grand = 0;
    colors.forEach(c => {
      x = 30;
      doc.fontSize(9).fillColor('#333').text(this.sanitize(c.name), x, y, { width: 100 });
      x += 100;
      let rowSum = 0;
      sizes.forEach((s, i) => {
        const v = qty[String(c.id)]?.[String(s.id)] ?? 0;
        rowSum += v; colTotals[i] += v;
        doc.text(String(v), x, y, { width: colW, align: 'center' });
        x += colW;
      });
      grand += rowSum;
      doc.fillColor('#1a1a2e').text(String(rowSum), x, y, { width: colW, align: 'center' });
      y += rowH;
    });
    // Column totals
    x = 30;
    doc.fontSize(9).fillColor('#6b6b8c').text('Total', x, y, { width: 100 });
    x += 100;
    colTotals.forEach(t => { doc.text(String(t), x, y, { width: colW, align: 'center' }); x += colW; });
    doc.fillColor('#f5a623').fontSize(10).text(String(grand), x, y, { width: colW, align: 'center' });
  }

  private drawBom(doc: PDFKit.PDFDocument, d: CutTicketDetailResponse) {
    const startY = 370;
    doc.fontSize(12).fillColor('#1a1a2e').text('BOM — Materiales', 30, startY);
    if (!d.bom.length) {
      doc.fontSize(10).fillColor('#8b8ba7').text('(sin BOM — módulo de materiales pendiente — Wave 10 Cost Sheet)', 30, startY + 18);
      return;
    }
    let y = startY + 20;

    ['Material','Uso/unidad','Total','Precio','Subtotal','Stock'].forEach((h, i) => {
      doc.text(h, 30 + i*90, y, { width: 85 });
    });
    y += 14;

    d.bom.forEach(b => {
      doc.fontSize(9).fillColor('#333');
      doc.text(this.sanitize(b.name), 30, y, { width: 85 });

      if (idx < d.routing.length - 1) {

  }

  private drawQrPlaceholder(doc: PDFKit.PDFDocument, d: CutTicketDetailResponse) {
    // QR: pdfkit 내장 없음. qrcode npm 패키지 미설치 — 현재는 텍스트 placeholder.

  private sanitize(raw: string | null | undefined): string {
    return String(raw || '')
      .replace(/[\x00-\x1f\x7f]/g, '')
      .slice(0, 100);
  }
```

```tsx
  useEffect(() => { setMatrix(initialMatrix); latestRef.current = initialMatrix }, [initialMatrix])

  // 행 합계
  const rowTotals = useMemo(() => {
    const out: Record<number, number> = {}
    matrix.colors.forEach(c => {
      out[c.id] = matrix.sizes.reduce((sum, s) => sum + (matrix.qty[String(c.id)]?.[String(s.id)] ?? 0), 0)
    })
    return out
  }, [matrix])

  // 열 합계
  const colTotals = useMemo(() => {
    const out: Record<number, number> = {}
    matrix.sizes.forEach(s => {
      out[s.id] = matrix.colors.reduce((sum, c) => sum + (matrix.qty[String(c.id)]?.[String(s.id)] ?? 0), 0)
    })
    return out
  }, [matrix])

  // 총계
  const grandTotal = useMemo(
    () => Object.values(rowTotals).reduce((a, b) => a + b, 0),
    [rowTotals],
  )

  const saveDebounced = useCallback((newMatrix: SizeColorMatrix) => {
    latestRef.current = newMatrix
    if (debounceRef.current) clearTimeout(debounceRef.current)
    debounceRef.current = setTimeout(async () => {
      setSaving(true)
      try {
        const res = await apiConnector.put(`/talleres/lotes/${loteId}/size-color-matrix`, { matrix: latestRef.current })
        const payload = (res as any)?.data ?? res
        onSaved(payload?.totalQuantity ?? 0)
      } catch (err: any) {
        toast.error(err?.response?.data?.message || 'Error al guardar matriz')
      } finally {
        setSaving(false)
      }
    }, 300)
  }, [loteId, onSaved])

  const handleCellChange = (colorId: number, sizeId: number, raw: string) => {
    const v = parseInt(raw, 10)
    if (raw !== '' && (isNaN(v) || v < 0)) return
    const next: SizeColorMatrix = {
      ...matrix,
      qty: {
        ...matrix.qty,
        [String(colorId)]: {
          ...(matrix.qty[String(colorId)] || {}),
          [String(sizeId)]: raw === '' ? 0 : v,
        },
      },
    }
    setMatrix(next)
    saveDebounced(next)
  }

  if (matrix.colors.length === 0 || matrix.sizes.length === 0) {
    return (
      <Card sx={{ p: 2, mb: 2 }}>
        <Typography sx={{ fontSize: 14, color: TALLERES_THEME.textMuted }}>
          매트릭스가 비어 있습니다. colors/sizes 구성 후 입력하세요.
        </Typography>
      </Card>
    )
  }

  // apiConnector.put — NOT patch (api.service 에 put 만 있는 경우 .put 사용; patch 가 지원되면 apiConnector.patch)
  // NOTE: executor 는 api.service.ts 확인 후 put/patch 선택

  const cellSx = {
    width: 50, textAlign: 'center' as const,
    fontFamily: "'Menlo', monospace", fontSize: 13,
  }

  return (
    <Card sx={{ p: 2, mb: 2 }}>
      <Box sx={{ display: 'flex', justifyContent: 'space-between', alignItems: 'center', mb: 1.5 }}>
        <Typography sx={{ fontSize: 16, fontWeight: 600, color: TALLERES_THEME.primary }}>
          사이즈 × 컬러 매트릭스
        </Typography>
        <Typography sx={{ fontSize: 12, color: saving ? TALLERES_THEME.secondary : TALLERES_THEME.textMuted }}>
          {saving ? '저장 중...' : `총 ${grandTotal}장`}
        </Typography>
      </Box>
      <Box sx={{ overflowX: 'auto' }}>
        <Table size="small">
          <TableHead>
            <TableRow>
              <TableCell sx={{ textTransform: 'uppercase', fontSize: 11, letterSpacing: 0.5, color: TALLERES_THEME.textMuted }}>
                컬러 \\ 사이즈
              </TableCell>
              {matrix.sizes.map(s => (
                <TableCell key={s.id} sx={{ textAlign: 'center', fontSize: 11, color: TALLERES_THEME.textMuted }}>
                  {s.name}
                </TableCell>
              ))}
              <TableCell sx={{ textAlign: 'center', fontSize: 11, color: TALLERES_THEME.textMuted }}>합계</TableCell>
            </TableRow>
          </TableHead>
          <TableBody>
            {matrix.colors.map(c => (
              <TableRow key={c.id}>
                <TableCell sx={{ fontSize: 13 }}>
                  {c.hex && <span style={{ display: 'inline-block', width: 12, height: 12, backgroundColor: c.hex, marginRight: 6, verticalAlign: 'middle', borderRadius: 2 }} />}
                  {c.name}
                </TableCell>
                {matrix.sizes.map(s => (
                  <TableCell key={s.id} sx={{ padding: '4px' }}>
                    {readOnly ? (
                      <Box sx={{ ...cellSx, bgcolor: TALLERES_THEME.bgSoft, py: 0.5 }}>
                        {matrix.qty[String(c.id)]?.[String(s.id)] ?? 0}
                      </Box>
                    ) : (
                      <input
                        type="number"
                        min={0}
                        value={matrix.qty[String(c.id)]?.[String(s.id)] ?? 0}
                        onChange={e => handleCellChange(c.id, s.id, e.target.value)}
                        style={{ ...cellSx, border: `1px solid ${TALLERES_THEME.border}`, borderRadius: 4, padding: 4 }}
                      />
                    )}
                  </TableCell>
                ))}
                <TableCell sx={{ ...cellSx, fontWeight: 600, color: TALLERES_THEME.primary }}>
                  {rowTotals[c.id]}
                </TableCell>
              </TableRow>
            ))}
            <TableRow sx={{ bgcolor: TALLERES_THEME.bgSoft }}>
              <TableCell sx={{ fontWeight: 600 }}>합계</TableCell>
              {matrix.sizes.map(s => (
                <TableCell key={s.id} sx={{ ...cellSx, fontWeight: 600 }}>{colTotals[s.id]}</TableCell>
              ))}
              <TableCell sx={{ ...cellSx, bgcolor: TALLERES_THEME.secondary, color: TALLERES_THEME.primary, fontWeight: 700 }}>
                {grandTotal}
              </TableCell>
            </TableRow>
          </TableBody>
        </Table>
      </Box>
    </Card>
  )
}
```
