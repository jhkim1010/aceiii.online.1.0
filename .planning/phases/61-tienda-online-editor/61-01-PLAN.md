1) **헬퍼 4종** — 파일 내 `clamp()` 바로 아래 배치:
```typescript
// href 화이트리스트 — http(s):// 또는 '/' 시작 상대경로만 허용. javascript:/data: 등은 null 강등.
const HREF_RE = /^(https?:\/\/|\/)/;
// 3rd-party pixel ID — <script> 내부 문자열에 삽입되므로 영숫자/하이픈/언더스코어만 허용(href 규칙과 별개).
const PIXEL_ID_RE = /^[A-Za-z0-9_-]{1,64}$/;

function sanitizeHref(v: unknown): string | null {
  return typeof v === 'string' && HREF_RE.test(v) ? v.slice(0, 500) : null;
}

// 텍스트 길이 강제 (기본 200자). 문자열이 아니면 fallback.
function clampText(v: unknown, maxLen = 200, fallback = ''): string {
  return typeof v === 'string' ? v.slice(0, maxLen) : fallback;
}

// MinIO 파일명 — 경로 구분자/상위 이동 시퀀스 거부(경로 조작 방어).
function sanitizeFileName(v: unknown): string | null {
  if (typeof v !== 'string' || v.length === 0 || v.length > 200) return null;
  if (v.includes('/') || v.includes('\\') || v.includes('..')) return null;

  return v;
}
```
