```typescript
export type CatalogSort = 'newest' | 'price_asc' | 'price_desc' | 'bestseller';

// 정렬 화이트리스트 → 고정 ORDER BY 리터럴. ★ 사용자 입력을 SQL 에 문자열 보간하지 않는다.
const ORDER_BY_MAP: Record<Exclude<CatalogSort, 'bestseller'>, string> = {
  newest: 'p.updated_at DESC',
  price_asc: 'p.price ASC',
  price_desc: 'p.price DESC',
};

function resolveOrderBy(sort: unknown): string {
  const key = String(sort ?? '') as keyof typeof ORDER_BY_MAP;

  return ORDER_BY_MAP[key] ?? ORDER_BY_MAP.newest;
}
```
