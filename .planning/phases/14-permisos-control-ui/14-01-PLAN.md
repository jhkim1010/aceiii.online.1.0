3. **기존 role_functions 레코드에 4개 action backfill:**
   ```bash
   docker exec api_ventago node -e "
   const { Client } = require('pg');
   const c = new Client({host:'dbpostgres',user:'coolsistema',password:'Coo1s1stem4Adm1nPg',database:'ventago'});
   c.connect().then(async () => {
     const actions = ['create', 'read', 'update', 'delete'];
     const rfs = await c.query('SELECT id FROM role_functions');
     console.log('role_functions count:', rfs.rows.length);
     for (const rf of rfs.rows) {
       for (const action of actions) {
         await c.query(
           'INSERT INTO role_function_actions (role_function_id, action, created_at, updated_at) VALUES (\$1, \$2, NOW(), NOW()) ON CONFLICT (role_function_id, action) DO NOTHING',
           [rf.id, action]
         );
       }
     }
     const count = await c.query('SELECT count(*) FROM role_function_actions');
     console.log('role_function_actions backfilled:', count.rows[0].count);
     c.end();
   });
   "
   ```
   기대값: role_function_actions 행 수 = role_functions 행 수 × 4
