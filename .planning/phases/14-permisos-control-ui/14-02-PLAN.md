3. **신규 파일: `api-ventago/src/app/auth/guards/function-permission.guard.ts`**
   ```typescript
   import { CanActivate, ExecutionContext, ForbiddenException, Injectable } from '@nestjs/common';
   import { Reflector } from '@nestjs/core';
   import { FUNCTION_METADATA_KEY } from '../decorators/function-guard.decorator';
   import { Functions } from 'src/app/functions/functions.model';
   import { UserRole } from 'src/app/users/user-role/user-role.model';
   import { RoleFunction } from 'src/app/role/role-function/role-function.model';
   import { RoleFunctionAction } from 'src/app/role/role-function/role-function-action.model';
   import { UserFunction } from 'src/app/users/user-function/user-function.model';
   import { UserFunctionAction } from 'src/app/users/user-function/user-function-action.model';

   @Injectable()
   export class FunctionPermissionGuard implements CanActivate {
     constructor(private reflector: Reflector) {}

     async canActivate(context: ExecutionContext): Promise<boolean> {
       const required = this.reflector.get<{ functionSlug: string; action: string }>(
         FUNCTION_METADATA_KEY,
         context.getHandler(),
       );

       // 메타데이터 없으면 통과
       if (!required) return true;

       const request = context.switchToHttp().getRequest();
       const user = request.user;

       // superadmin은 항상 통과
       if (user?.roles?.some((r: any) => r.slug === 'superadmin')) return true;

       // 경량 쿼리로 권한 확인 — pool 낭비 방지
       const fn = await Functions.findOne({ where: { slug: required.functionSlug }, attributes: ['id'] });
       if (!fn) return true; // function이 DB에 없으면 통과 (아직 등록 안 된 function)

       const userRole = await UserRole.findOne({ where: { userId: user.id }, attributes: ['roleId'] });
       if (!userRole) throw new ForbiddenException('Sin permisos para esta acción');

       // role 기반 권한 확인
       const rf = await RoleFunction.findOne({
         where: { roleId: userRole.roleId, functionId: fn.id, storeId: user.storeId },
         attributes: ['id'],
       });
       if (!rf) throw new ForbiddenException('Sin permisos para esta acción');

       const rfa = await RoleFunctionAction.findOne({
         where: { roleFunctionId: rf.id, action: required.action },
       });
       const baseAllowed = !!rfa;

       // user override 확인
       const uf = await UserFunction.findOne({
         where: { userId: user.id, functionId: fn.id, storeId: user.storeId },
         attributes: ['id'],
       });
       if (!uf) {
         if (!baseAllowed) throw new ForbiddenException('Sin permisos para esta acción');

         return true;
       }

       const ufa = await UserFunctionAction.findOne({
         where: { userFunctionId: uf.id, action: required.action },
       });
       if (!ufa) {
         if (!baseAllowed) throw new ForbiddenException('Sin permisos para esta acción');

         return true;
       }

       if (!ufa.allowed) throw new ForbiddenException('Sin permisos para esta acción');

       return true;
     }
   }
   ```

ESLint 주의: `return` 전에 빈 줄, `//` 주석 전에 빈 줄 필수.

     async updateActionsForRoleFunction(roleFunctionId: number, actions: string[]) {
       const validActions = ['create', 'read', 'update', 'delete'];
       const filtered = actions.filter(a => validActions.includes(a));

       await RoleFunctionAction.destroy({ where: { roleFunctionId } });

       const created = await Promise.all(
         filtered.map(action =>

         )
       );

       return created;

     // 역할의 모든 function-action을 한 번에 업데이트
     async bulkUpdateRoleFunctionActions(roleId: number, storeId: number, data: Array<{ functionId: number; actions: string[] }>) {
       // 1. 기존 role_functions 조회
       const existingRFs = await this.roleFunctionModel.findAll({ where: { roleId, storeId } });
       const existingMap = new Map(existingRFs.map(rf => [rf.functionId, rf]));

       // 2. data에 있는 functionId 처리
       for (const item of data) {
         let rf = existingMap.get(item.functionId);
         if (!rf) {
           rf = await this.createWithActions(roleId, storeId, item.functionId);
         }
         await this.updateActionsForRoleFunction(rf.id, item.actions);
       }

       // 3. data에 없는 기존 function은 삭제 (cascade로 action도 삭제)
       const dataFnIds = new Set(data.map(d => d.functionId));
       for (const [fnId, rf] of existingMap) {
         if (!dataFnIds.has(fnId)) {
           await rf.destroy();
         }
       }
     }
