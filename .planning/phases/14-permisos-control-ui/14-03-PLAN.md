   ```typescript
   export type Subjects = string
   export type Actions = 'manage' | 'create' | 'read' | 'update' | 'delete'
   export type AppAbility = Ability<[Actions, Subjects]> | undefined
   export const AppAbility = Ability as any

   export type PermissionsMap = Record<string, { create: boolean; read: boolean; update: boolean; delete: boolean }>

   export type ACLObj = {
     action: Actions
     subject: string
   }

   const defineRulesFor = (
     roles: string[],
     subject: string,
     permissions?: PermissionsMap | null
   ) => {
     const { can, rules } = new AbilityBuilder(AppAbility)

     // superadmin은 여전히 manage:all (per Pitfall 6)
     if (roles?.includes('superadmin')) {
       can('manage', 'all')

       return rules
     }

     // permissions 맵이 있으면 granular 권한 빌딩
     if (permissions) {
       Object.entries(permissions).forEach(([fnSlug, actions]) => {
         if (actions.create) can('create', fnSlug)
         if (actions.read) can('read', fnSlug)
         if (actions.update) can('update', fnSlug)
         if (actions.delete) can('delete', fnSlug)
       })
     }

     // permissions 없어도 subject에 대해 read는 기본 허용 (fallback — 기존 동작 호환)
     if (!permissions) {
       can('read', subject)
     }

     return rules
   }

   export const buildAbilityFor = (
     roles: string[],
     subject: string,
     permissions?: PermissionsMap | null
   ): AppAbility => {
     return new AppAbility(defineRulesFor(roles, subject, permissions), {
       detectSubjectType: (o: any) => o!.type
     })
   }

   export const defaultACLObj: ACLObj = {
     action: 'read',
     subject: 'all'
   }

   export default defineRulesFor
   ```

```tsx
import { Box, Chip, Tooltip } from '@mui/material'

// CRUD 액션 색상 매핑 (UI-SPEC 참조)
const ACTION_CONFIG = {
  create: { label: 'C', color: 'success' as const, tooltip: 'Crear', ariaAction: 'CREATE' },
  read: { label: 'R', color: 'primary' as const, tooltip: 'Leer', ariaAction: 'READ' },
  update: { label: 'U', color: 'warning' as const, tooltip: 'Editar', ariaAction: 'UPDATE' },
  delete: { label: 'D', color: 'error' as const, tooltip: 'Eliminar', ariaAction: 'DELETE' }
} as const

type ActionKey = 'create' | 'read' | 'update' | 'delete'

interface CrudActionRowProps {
  functionId: number
  actions: Record<ActionKey, boolean>
  roleActions?: Record<ActionKey, boolean>
  onChange: (functionId: number, action: ActionKey, value: boolean) => void
  variant?: 'role' | 'user-override'
  disabled?: boolean
}

const CrudActionRow = ({ functionId, actions, roleActions, onChange, variant = 'role', disabled = false }: CrudActionRowProps) => {

  return (
    <Box sx={{ display: 'flex', gap: 1, alignItems: 'center' }}>
      {(Object.keys(ACTION_CONFIG) as ActionKey[]).map(actionKey => {
        const config = ACTION_CONFIG[actionKey]
        const isEnabled = actions[actionKey]

        // user-override 모드: role 기본값과 다르면 경고 스타일
        const isOverridden = variant === 'user-override' && roleActions && roleActions[actionKey] !== isEnabled
        const tooltipText = isOverridden
          ? 'Sobreescrito manualmente — diferente al rol base'
          : variant === 'user-override' && roleActions
            ? `Hereda del rol`
            : config.tooltip

        return (
          <Tooltip key={actionKey} title={tooltipText} arrow>
            <Chip
              label={config.label}
              size='small'
              color={config.color}
              variant={isEnabled ? 'filled' : 'outlined'}
              onClick={disabled ? undefined : () => onChange(functionId, actionKey, !isEnabled)}
              sx={{
                cursor: disabled ? 'default' : 'pointer',
                minWidth: 32,
                fontWeight: 500,
                ...(isOverridden && {
                  borderWidth: 2,
                  borderStyle: 'solid',
                  borderColor: 'warning.main'
                })
              }}
              aria-label={`Acción ${config.ariaAction} — ${isEnabled ? 'habilitado' : 'deshabilitado'}`}
            />
          </Tooltip>
        )
      })}
    </Box>
  )
}

export default CrudActionRow
```

주의:

- `gap: 1` = 8px (spacing(2)에 해당 — theme.spacing(1)=4px이지만 MUI Box의 gap은 theme.spacing 배수)
  - 정정: `gap: 1`은 spacing(1)=4px. UI-SPEC은 8px이므로 `gap: 2` 사용
- ESLint: `return` 전 빈 줄 필수
