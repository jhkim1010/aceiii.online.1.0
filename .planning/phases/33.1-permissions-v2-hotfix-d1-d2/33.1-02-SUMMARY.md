```typescript
/**
 * 특정 역할의 모든 사용자 캐시 무효화
 * (role_functions / role_function_actions 변경 시 호출 — Phase 33.1 D2-FIX)
 *
 * user_permission_cache 는 user_id 기준 — user_roles JOIN 으로 영향 user 식별.
 * branch 단위 추가 필터링 없음 → 같은 role 의 모든 branch 캐시도 함께 invalidate (의도된 동작).
 */
async invalidateRole(roleId: number): Promise<number> {
  try {
    const sequelize = this.cacheModel.sequelize;
    if (!sequelize) {
      throw new Error('Sequelize instance unavailable');
    }

    const [, meta] = await sequelize.query(
      `DELETE FROM user_permission_cache
       WHERE user_id IN (SELECT user_id FROM user_roles WHERE role_id = :roleId)`,
      { replacements: { roleId } },
    );

    const deleted = (meta as { rowCount?: number })?.rowCount ?? 0;
    this.logger.debug(`invalidateRole(${roleId}) — ${deleted} rows 삭제`);

    return deleted;
  } catch (error) {
    this.logger.error(`invalidateRole(${roleId}) 실패:`, error);

    return 0;
  }
}
```

`invalidateStore` analog 패턴 일관 (try/catch + sequelize.query + meta.rowCount + logger).
