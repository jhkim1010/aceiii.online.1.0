새 본문:

```typescript
// [Phase 33.1 D1-FIX] read-only — RoleFunction 자동 생성/backfill 제거. 빈 결과는 빈 배열 반환.
async ensureRoleFunctions(user: Users, roleId: number): Promise<number[]> {
  const roleFunctions = await RoleFunction.findAll({
    where: { roleId, storeId: user.storeId },
    include: ['function'],
  });

  console.log(
    `[AUTH DEBUG] RoleFunctions found: ${roleFunctions.length} for roleId=${roleId}, storeId=${user.storeId}`,
  );

  const roleFunctionIds = roleFunctions.map((rf: any) => rf.functionId);

  if (!user.storeId) {
    return [];
  }

  const userFunctionService = new UserFunctionService(UserFunction);
  const allowedFunctionIds =
    await userFunctionService.getEffectiveFunctions(
      user.id,
      roleFunctionIds,
      user.storeId,
    );

  console.log(
    `[AUTH DEBUG] Effective FunctionIds: ${allowedFunctionIds.length}`,
  );

  return allowedFunctionIds;
}
```
