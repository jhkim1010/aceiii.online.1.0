```typescript
@Controller('global-clients')
@UseGuards(AuthGuard('jwt'))
export class GlobalClientsController {

  @Get()
  // 주석: list 는 service 에서 ownerGroupId 필터 자동 적용 (req.user.ownerGroupId)
  async findAll(@Req() req: any, @Query() query: any) {
    return this.svc.findAll({ ...query, ownerGroupId: req.user.ownerGroupId });
  }

  @Get(':id')
  @OwnerScope({ globalClientIdParam: 'id' })
  async findOne(@Param('id') id: string) {
    return this.svc.findOne(+id);
  }

  @Get(':id/historial')
  @OwnerScope({ globalClientIdParam: 'id' })
  async getHistorial(@Param('id') id: string, @Req() req: any) {
    // 주석: historial 은 항상 caller.storeId 스코프 — ownerGroup 공유 절대 금지
    return this.svc.getHistorial({ globalClientId: +id, storeId: req.user.storeId });
  }

  @Post()
  async create(@Body() dto: any, @Req() req: any) {
    return this.svc.create({ ...dto, ownerGroupId: req.user.ownerGroupId });
  }

  @Put(':id')
  @OwnerScope({ globalClientIdParam: 'id' })
  async update(@Param('id') id: string, @Body() dto: any) {
    return this.svc.update(+id, dto);
  }

  @Delete(':id')
  @OwnerScope({ globalClientIdParam: 'id' })
  async remove(@Param('id') id: string) {
    return this.svc.softDelete(+id);
  }

  @Get('by-document/:document')
  async findByDocument(@Param('document') document: string, @Req() req: any) {
    return this.svc.findByDocument({ document, ownerGroupId: req.user.ownerGroupId });
  }
}
```

```typescript
// 주석: Phase 25 — 모든 조회는 반드시 ownerGroupId 필터 포함 (T-25-01 IDOR 차단).
async findAll(opts: { ownerGroupId: number; search?: string; limit?: number; offset?: number }) {
  return this.globalClientModel.findAll({
    where: {
      ownerGroupId: opts.ownerGroupId,
      isActive: true,
      // ... search conditions
    },
    limit: opts.limit ?? 50,
    offset: opts.offset ?? 0,
  });
}

async findOne(id: number) {
  // 주석: OwnerScopeGuard 가 이미 ownerGroupId 일치를 검증했으므로 여기서는 PK 조회만 해도 안전
  return this.globalClientModel.findByPk(id);
}

async findByDocument(opts: { document: string; ownerGroupId: number }) {
  return this.globalClientModel.findOne({
    where: { document: opts.document, ownerGroupId: opts.ownerGroupId, isActive: true },
  });
}

async getHistorial(opts: { globalClientId: number; storeId: number }) {
  // 주석: Phase 25 REQ-25-04, REQ-25-07 — sales historial 은 storeId 스코프 강제. ownerGroup 공유 절대 금지.
  return this.salesModel.findAll({

    include: [{
      model: StoreClient,
      where: { globalClientId: opts.globalClientId },
    }],
    order: [['createdAt', 'DESC']],
    limit: 100,
  });
}

async create(payload: any) {
  // ownerGroupId 는 controller 에서 req.user.ownerGroupId 를 주입받음
  return this.globalClientModel.create(payload);
}

async update(id: number, patch: any) {
  // 주석: D1-06 demotion 금지 — document 를 null 로 만들 수 없음
  if (patch.document === null || patch.document === '') {
    throw new BadRequestException('Cannot remove document from a global client (D1-06)');
  }

  return this.globalClientModel.update(patch, { where: { id } });
}

async softDelete(id: number) {
  return this.globalClientModel.update({ isActive: false }, { where: { id } });
}
```
