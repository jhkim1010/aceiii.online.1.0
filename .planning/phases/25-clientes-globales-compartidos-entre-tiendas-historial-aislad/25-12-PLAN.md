3. **Replace** existing `handleUpload` with the new implementation:

```typescript
const handleUpload = async () => {
  if (classifiedRows.length === 0) return
  setUploading(true)

  try {
    const chunkSize = 500
    let totalCreated = 0
    let totalUpdated = 0
    let totalSkipped = 0
    const allErrors: ImportErrorRow[] = []
    let lastClientImportId = 0

    for (let i = 0; i < classifiedRows.length; i += chunkSize) {
      const chunk = classifiedRows.slice(i, i + chunkSize)

      const payload = {
        fileName: uploadedFileName || 'import.csv',
        missingDocPolicy,
        rows: chunk.map(r => {
          const effectiveOverride = ['skip', 'update', 'link'].includes((r.override ?? '') as any) ? (r.override as any) : undefined
          const effectiveBucket = effectiveOverride
            ? (effectiveOverride === 'skip' ? 'Skip' : 'Global')
            : (r.override as any ?? r.bucket)

          return {
            document: r.document,
            fullname: r.fullname,
            email: r.email,
            phone: r.phone,
            address: r.address,
            provinceId: r.provinceId,
            city: r.city,
            birthdate: r.birthdate,
            notes: r.notes,
            bucket: effectiveBucket,
            override: effectiveOverride,
          }
        }),
      }

      const result: UploadResult = await apiConnector.post('/clients/import', payload)

      totalCreated += result.createdCount ?? 0
      totalUpdated += result.updatedCount ?? 0
      totalSkipped += result.skippedCount ?? 0
      allErrors.push(...(result.errors ?? []))
      lastClientImportId = result.clientImportId
    }

    setUploadResult({
      clientImportId: lastClientImportId,
      totalRows: classifiedRows.length,
      createdCount: totalCreated,
      updatedCount: totalUpdated,
      skippedCount: totalSkipped,
      errorCount: allErrors.length,
      errors: allErrors,
    })
    setStep(2)
  } catch (err: any) {
    // eslint-disable-next-line no-console
    console.error('Error uploading clients:', err)
    const msg = err?.response?.data?.message ?? 'Error al cargar clientes'
    toast.error(msg)
  } finally {
    setUploading(false)
  }
}
```

```typescript
const downloadFailureCsv = (errors: ImportErrorRow[]) => {
  const csv = Papa.unparse({
    fields: ['row_index', 'error_code', 'error_message', 'fullname', 'document'],
    data: errors.map(e => [e.rowIndex, e.errorCode, e.errorMessage, e.fullname ?? '', e.document ?? '']),
  })

  const blob = new Blob(['\uFEFF' + csv], { type: 'text/csv;charset=utf-8;' })   // BOM for Excel
  const url = URL.createObjectURL(blob)
  const a = document.createElement('a')
  a.href = url
  a.download = `errores_importacion_${new Date().toISOString().slice(0, 10)}.csv`
  a.click()
  URL.revokeObjectURL(url)
}
```
