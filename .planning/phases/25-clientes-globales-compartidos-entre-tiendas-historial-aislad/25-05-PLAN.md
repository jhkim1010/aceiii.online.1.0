```typescript
import { Injectable, Logger } from '@nestjs/common';
import { InjectModel } from '@nestjs/sequelize';
import { Store } from '../../store/store.model';
import { GlobalClient } from '../../shared/global-clients/global-clients.model';

// 주석: Phase 25 D3-01/D3-04 — storeId → ownerGroupId 해석 서비스.
// In-memory 캐시 (기본 TTL 5분) — pool 낭비 방지 (CLAUDE.md 성능 규약).
@Injectable()
export class OwnerScopeService {
  private readonly logger = new Logger(OwnerScopeService.name);
  private readonly cache = new Map<number, { ownerGroupId: number; ts: number }>();
  private readonly ttlMs = 5 * 60 * 1000;

  constructor(
    @InjectModel(Store) private readonly storeModel: typeof Store,
    @InjectModel(GlobalClient) private readonly globalClientModel: typeof GlobalClient,
  ) {}

  async resolveStoreOwnerGroup(storeId: number): Promise<number | null> {
    if (!storeId) return null;

    const cached = this.cache.get(storeId);
    if (cached && Date.now() - cached.ts < this.ttlMs) {
      return cached.ownerGroupId;
    }

    const row = await this.storeModel.findByPk(storeId, {
      attributes: ['id', 'ownerGroupId'],
    });
    if (!row) return null;

    this.cache.set(storeId, { ownerGroupId: row.ownerGroupId, ts: Date.now() });

    return row.ownerGroupId;
  }

  async resolveGlobalClientOwnerGroup(globalClientId: number): Promise<number | null> {
    if (!globalClientId) return null;

    const row = await this.globalClientModel.findByPk(globalClientId, {
      attributes: ['id', 'ownerGroupId'],
    });

    return row ? row.ownerGroupId : null;
  }

  invalidateStore(storeId: number): void {
    this.cache.delete(storeId);
  }
}

import {
  CanActivate,
  ExecutionContext,
  ForbiddenException,
  Injectable,
  Logger,
  NotFoundException,
} from '@nestjs/common';
import { Reflector } from '@nestjs/core';
import { InjectModel } from '@nestjs/sequelize';
import { OWNER_SCOPE_KEY, OwnerScopeMeta } from '../decorators/owner-scope.decorator';
import { OwnerScopeService } from '../services/owner-scope.service';
import { ClientAccessAudit } from '../models/client-access-audit.model';

// 주석: Phase 25 D3-04 — 다른 ownerGroup 의 store / GlobalClient 접근 시 403 + audit.
@Injectable()
export class OwnerScopeGuard implements CanActivate {
  private readonly logger = new Logger(OwnerScopeGuard.name);

  constructor(
    private readonly reflector: Reflector,
    private readonly ownerScopeService: OwnerScopeService,
    @InjectModel(ClientAccessAudit) private readonly auditModel: typeof ClientAccessAudit,
  ) {}

  async canActivate(context: ExecutionContext): Promise<boolean> {
    const meta = this.reflector.get<OwnerScopeMeta>(OWNER_SCOPE_KEY, context.getHandler());

    if (!meta) return true;

    const request = context.switchToHttp().getRequest();
    const user = request.user;

    if (!user) {
      throw new ForbiddenException('No user context');
    }

    // superadmin bypass (FunctionGuard 와 동일 패턴)
    if (user?.roles?.some((r: any) => r.slug === 'superadmin')) {
      return true;
    }

    const callerStoreId = user.storeId;
    const callerOwnerGroupId = await this.ownerScopeService.resolveStoreOwnerGroup(callerStoreId);

    if (!callerOwnerGroupId) {
      throw new ForbiddenException('Caller store has no ownerGroupId');
    }

    // Resolve target ownerGroupId from meta configuration
    let targetOwnerGroupId: number | null = null;
    let targetStoreId: number | undefined;
    let targetGlobalClientId: number | undefined;

    if (meta.storeIdParam) {
      targetStoreId = Number(request.params[meta.storeIdParam]);
    } else if (meta.storeIdField) {
      targetStoreId = Number(request.body?.[meta.storeIdField] ?? user.storeId);
    }

    if (targetStoreId) {
      targetOwnerGroupId = await this.ownerScopeService.resolveStoreOwnerGroup(targetStoreId);
    }

    if (meta.globalClientIdParam) {
      targetGlobalClientId = Number(request.params[meta.globalClientIdParam]);
    } else if (meta.globalClientIdField) {
      targetGlobalClientId = Number(request.body?.[meta.globalClientIdField]);
    }

    if (targetGlobalClientId) {
      const gcOwnerGroupId = await this.ownerScopeService.resolveGlobalClientOwnerGroup(targetGlobalClientId);
      if (gcOwnerGroupId === null) {
        throw new NotFoundException('GlobalClient not found');
      }

      targetOwnerGroupId = gcOwnerGroupId;
    }

    // Default: no explicit target means caller's own scope — pass
    if (targetOwnerGroupId === null || targetOwnerGroupId === undefined) {
      return true;
    }

    if (callerOwnerGroupId !== targetOwnerGroupId) {
      // audit 기록 — non-blocking but awaited (small row)
      await this.auditModel.create({
        userId: user.id,
        callerStoreId,
        callerOwnerGroupId,
        targetStoreId: targetStoreId ?? null,
        targetOwnerGroupId,
        targetGlobalClientId: targetGlobalClientId ?? null,
        endpoint: request.originalUrl || request.url,
        method: request.method,
        ipAddress: request.ip || request.headers['x-forwarded-for'] || null,
        userAgent: request.headers['user-agent'] || null,
      });

      throw new ForbiddenException('CROSS_OWNER_GROUP_FORBIDDEN');
    }

    return true;
  }
}
```
