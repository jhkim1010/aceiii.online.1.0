```typescript
// Before:
async getKpis({ startDate, endDate, storeId }: Filters) {
  return this.sequelize.query(`SELECT ... FROM sales WHERE store_id = :storeId`, { replacements: { storeId } });
}

// After:
async getKpis({ startDate, endDate, storeId }: Filters, ctx: { ownerGroupId: number }) {
  let storeClause = '';
  let replacements: any = { startDate, endDate };
  if (storeId) {
    storeClause = 'AND store_id = :storeId';
    replacements.storeId = storeId;
  } else {
    const stores = await this.ownerScopeService.resolveStoresForOwnerGroup(ctx.ownerGroupId);
    storeClause = 'AND store_id = ANY(:stores)';
    replacements.stores = stores;
  }

  return this.sequelize.query(
    `SELECT ... FROM sales WHERE 1=1 ${storeClause} AND ...`,
    { replacements },
  );
}
```

Note: If ctx.ownerGroupId isn't already threaded through, the controller must extract it from req.user.ownerGroupId and pass down.
