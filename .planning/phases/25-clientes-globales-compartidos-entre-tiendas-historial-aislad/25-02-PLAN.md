```sql
-- ===========================================================================
-- Phase 25 Wave 1 Step 3: sales.store_client_id nullable FK (PG10 호환)
-- 실행: sudo -u postgres psql -d ventago -v ON_ERROR_STOP=1 -1 -f 20260424-phase25-step3-sales-store-client.sql
-- 목적: D2-01 dual-FK 전략 — 신규 sales 는 store_client_id 사용, 레거시 client_id 유지
-- ===========================================================================

BEGIN;

ALTER TABLE sales
  ADD COLUMN IF NOT EXISTS store_client_id INTEGER NULL;

-- FK 제약 idempotent 추가
DO $$
BEGIN
  ALTER TABLE sales
    ADD CONSTRAINT fk_sales_store_client
    FOREIGN KEY (store_client_id)
    REFERENCES store_clients(id)
    ON DELETE SET NULL
    ON UPDATE CASCADE;
EXCEPTION WHEN duplicate_object THEN
  NULL;
END $$;

-- 조회 가속 — storeClientId 기반 historial 조회 시 사용
CREATE INDEX IF NOT EXISTS idx_sales_store_client_id
  ON sales (store_client_id)
  WHERE store_client_id IS NOT NULL;

-- 통합 scope 쿼리용 (store_id + store_client_id) — 보고서에서 사용
CREATE INDEX IF NOT EXISTS idx_sales_store_id_store_client_id
  ON sales (store_id, store_client_id)
  WHERE store_client_id IS NOT NULL;

COMMIT;
```
