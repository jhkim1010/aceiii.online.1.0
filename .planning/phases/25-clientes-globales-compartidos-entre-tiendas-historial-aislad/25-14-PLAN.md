```tsx
import React, { useMemo, useState, useEffect } from 'react'
import {
  Dialog,
  DialogTitle,
  DialogContent,
  DialogActions,
  Button,
  Alert,
  Grid,
  FormControl,
  FormLabel,
  RadioGroup,
  FormControlLabel,
  Radio,
  Chip,
  Typography,
  CircularProgress,
  Box,
} from '@mui/material'
import { useTranslation } from 'react-i18next'
import Icon from 'src/@core/components/icon'

// 주석: Phase 25 D1-04 — promote / merge 결정 다이얼로그.
// mode='promote-confirm': 단일 확인
// mode='merge-resolve': 9 필드 중 document 비활성, 나머지 8 필드에 radio 선택 필수

type PromoteMergeMode = 'promote-confirm' | 'merge-resolve'

export interface PromoteMergeClient {
  id: number
  fullname: string
  document: string
  email?: string
  phone?: string
  address?: string
  provincia?: string
  localidad?: string
  fechaNacimiento?: string
  notas?: string
}

export interface ExistingGlobalHit extends PromoteMergeClient {
  ownerGroupId: number
  linkedStoreCount?: number
  updatedAt: string   // ISO timestamp — opt-lock token
}

export interface PromoteMergeResolution {
  action: 'promote' | 'merge' | 'cancel'
  fieldPicks?: Record<string, 'local' | 'global'>
  winnerUpdatedAt?: string
}

export interface PromoteMergeDialogProps {
  open: boolean
  onClose: () => void
  mode: PromoteMergeMode
  localClient: PromoteMergeClient
  existingGlobalClient?: ExistingGlobalHit
  onConfirm: (resolution: PromoteMergeResolution) => Promise<void>
}

// 주석: document 는 merge 대상에서 제외 (D1-04 — 매칭 키이므로 양쪽 값이 동일해야 함).
const MERGE_FIELDS: Array<keyof PromoteMergeClient> = [
  'fullname', 'phone', 'email', 'address',
  'provincia', 'localidad', 'fechaNacimiento', 'notas',
]

const PromoteMergeDialog: React.FC<PromoteMergeDialogProps> = ({
  open, onClose, mode, localClient, existingGlobalClient, onConfirm,
}) => {
  const { t } = useTranslation()
  const [fieldPicks, setFieldPicks] = useState<Record<string, 'local' | 'global'>>({})
  const [loading, setLoading] = useState(false)
  const [errorMsg, setErrorMsg] = useState<string | null>(null)

  useEffect(() => {
    if (open) {
      setFieldPicks({})
      setErrorMsg(null)
      setLoading(false)
    }
  }, [open])

  const allFieldsPicked = useMemo(
    () => MERGE_FIELDS.every(f => fieldPicks[f] === 'local' || fieldPicks[f] === 'global'),
    [fieldPicks],
  )

  const handlePromote = async () => {
    setLoading(true)
    setErrorMsg(null)
    try {
      await onConfirm({ action: 'promote' })
    } catch (err: any) {
      setErrorMsg(err?.response?.data?.message ?? err?.message ?? 'Error')
      setLoading(false)
    }
  }

  const handleMerge = async () => {
    setLoading(true)
    setErrorMsg(null)
    try {
      await onConfirm({
        action: 'merge',
        fieldPicks,
        winnerUpdatedAt: existingGlobalClient?.updatedAt,
      })
    } catch (err: any) {
      setErrorMsg(err?.response?.data?.message ?? err?.message ?? 'Error')
      setLoading(false)
    }
  }

  const handleCancel = async () => {
    await onConfirm({ action: 'cancel' })
    onClose()
  }

  const renderFieldPairRow = (field: keyof PromoteMergeClient) => {
    const lv = localClient[field] as any
    const gv = existingGlobalClient ? (existingGlobalClient[field] as any) : undefined

    return (
      <Grid item xs={12} md={6} key={field as string}>
        <FormControl component='fieldset' fullWidth sx={{ border: 1, borderColor: 'divider', borderRadius: 1, p: 2 }}>
          <FormLabel component='legend' sx={{ px: 1 }}>{field as string}</FormLabel>
          <RadioGroup
            value={fieldPicks[field as string] ?? ''}
            onChange={(e) => setFieldPicks(prev => ({ ...prev, [field as string]: e.target.value as 'local' | 'global' }))}
          >
            <FormControlLabel
              value='global'
              control={<Radio size='small' />}
              label={
                <Box sx={{ display: 'flex', alignItems: 'center', gap: 1 }}>
                  <Chip label='Global' color='primary' size='small' variant='outlined' />
                  <Typography variant='body2'>{gv || <em>{t('cliente_merge_field_empty_placeholder', '(vacío)')}</em>}</Typography>
                </Box>
              }
            />
            <FormControlLabel
              value='local'
              control={<Radio size='small' />}
              label={
                <Box sx={{ display: 'flex', alignItems: 'center', gap: 1 }}>
                  <Chip label='Local' size='small' variant='outlined' />
                  <Typography variant='body2'>{lv || <em>{t('cliente_merge_field_empty_placeholder', '(vacío)')}</em>}</Typography>
                </Box>
              }
            />
          </RadioGroup>
        </FormControl>
      </Grid>
    )
  }

  if (mode === 'promote-confirm') {
    return (
      <Dialog open={open} onClose={onClose} maxWidth='sm' fullWidth>
        <DialogTitle>
          <Box sx={{ display: 'flex', alignItems: 'center', gap: 1 }}>
            <Icon icon='mdi:account-arrow-up-outline' />
            {t('cliente_promote_dialog_title', 'Promover a cliente global')}
          </Box>
        </DialogTitle>
        <DialogContent dividers>
          {errorMsg && <Alert severity='error' sx={{ mb: 2 }}>{errorMsg}</Alert>}
          <Alert severity='info' sx={{ mb: 2 }}>
            {t('cliente_promote_dialog_body', 'Al guardar el DNI/CUIT, este cliente pasará a ser visible en las demás tiendas del mismo grupo propietario. El historial de ventas permanece aislado por tienda.')}
          </Alert>
          <Typography variant='subtitle2'>{localClient.fullname}</Typography>
          <Typography variant='caption' color='text.secondary'>DNI/CUIT: {localClient.document}</Typography>
        </DialogContent>
        <DialogActions>
          <Button variant='outlined' onClick={handleCancel} disabled={loading}>
            {t('cliente_merge_cancel_destructive', 'Cancelar promoción')}
          </Button>
          <Button variant='contained' color='primary' onClick={handlePromote} disabled={loading}>
            {loading
              ? <><CircularProgress size={16} sx={{ mr: 1 }} />{t('common_processing', 'Procesando…')}</>
              : t('cliente_promote_confirm_button', 'Promover')}
          </Button>
        </DialogActions>
      </Dialog>
    )
  }

  // merge-resolve mode
  return (
    <Dialog open={open} onClose={onClose} maxWidth='md' fullWidth>
      <DialogTitle>
        <Box sx={{ display: 'flex', alignItems: 'center', gap: 1 }}>
          <Icon icon='mdi:call-merge' />
          {t('cliente_merge_dialog_title', 'Cliente ya existe en otra tienda del grupo')}
        </Box>
      </DialogTitle>
      <DialogContent dividers>
        {errorMsg && <Alert severity='error' sx={{ mb: 2 }}>{errorMsg}</Alert>}
        <Alert severity='warning' sx={{ mb: 2 }}>
          {t('cliente_merge_dialog_body', 'Se encontró un cliente global con el mismo {{documentType}}. Elige qué valor conservar para cada campo.', {
            documentType: localClient.document?.length === 11 ? 'CUIT' : 'DNI',
          })}
        </Alert>

        {existingGlobalClient?.linkedStoreCount !== undefined && (
          <Typography variant='caption' color='text.secondary' sx={{ mb: 2, display: 'block' }}>
            Global actual está vinculado a {existingGlobalClient.linkedStoreCount} tienda(s) del grupo.
          </Typography>
        )}

        {/* Document field — disabled, authoritative by match */}
        <Grid container spacing={2} sx={{ mb: 2 }}>
          <Grid item xs={12}>
            <FormControl component='fieldset' fullWidth sx={{ border: 1, borderColor: 'divider', borderRadius: 1, p: 2, opacity: 0.7 }} disabled>
              <FormLabel component='legend' sx={{ px: 1 }}>document</FormLabel>
              <Typography variant='body2' sx={{ fontFamily: 'ui-monospace, Menlo, monospace' }}>
                {localClient.document} — {t('cliente_merge_document_authoritative', 'por coincidencia obligatoria')}
              </Typography>
            </FormControl>
          </Grid>

          {MERGE_FIELDS.map(renderFieldPairRow)}
        </Grid>
      </DialogContent>
      <DialogActions>
        <Button variant='outlined' onClick={handleCancel} disabled={loading}>
          {t('cliente_merge_cancel_destructive', 'Cancelar promoción')}
        </Button>
        <Button
          variant='contained'
          color='primary'
          onClick={handleMerge}
          disabled={!allFieldsPicked || loading}
        >
          {loading
            ? <><CircularProgress size={16} sx={{ mr: 1 }} />{t('common_processing', 'Procesando…')}</>
            : t('cliente_merge_confirm_button', 'Combinar y promover')}
        </Button>
      </DialogActions>
    </Dialog>
  )
}

export default PromoteMergeDialog
```

const handlePromoteMergeConfirm = async (resolution: PromoteMergeResolution) => {
  if (!pendingLocalClient) return

  if (resolution.action === 'cancel') {
    skipPromotionFor.add(pendingLocalClient.id)
    setPromoteMergeOpen(false)
    setPendingLocalClient(null)
    setPendingExistingGlobal(null)

    return
  }

  if (resolution.action === 'promote') {
    // 이미 POST /promote 호출 완료된 상태의 confirm (promote-confirm 모드) — 단순 닫기 + 리프레시
    setPromoteMergeOpen(false)
    toast.success(t('cliente_promote_success', 'Cliente promovido correctamente.'))
    await refreshTable()

    return
  }

  if (resolution.action === 'merge' && pendingExistingGlobal) {
    try {
      await apiConnector.post('/clients/merge', {
        localClientId: pendingLocalClient.id,
        winnerGlobalClientId: pendingExistingGlobal.id,
        fieldPicks: resolution.fieldPicks,
        winnerUpdatedAt: pendingExistingGlobal.updatedAt,
      })
      toast.success(t('cliente_merge_success', 'Clientes combinados correctamente.'))
      setPromoteMergeOpen(false)
      setPendingLocalClient(null)
      setPendingExistingGlobal(null)
      await refreshTable()
    } catch (err: any) {
      // Let dialog's errorMsg state show it
      throw err
    }
  }
}
