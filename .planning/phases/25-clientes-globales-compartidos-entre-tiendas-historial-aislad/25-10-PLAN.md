```typescript
async importBatch(rows: ImportRow[], storeId: number, ownerGroupId: number) {
  const transaction = await this.sequelize.transaction();
  try {
    const chunkSize = 500;
    for (let i = 0; i < rows.length; i += chunkSize) {
      const chunk = rows.slice(i, i + chunkSize);
      const globalRows = chunk.filter(r => r.bucket === 'Global');
      const localRows  = chunk.filter(r => r.bucket === 'Local');

      // --- Global path ---
      const docs = globalRows.map(r => r.document);
      const existing = await GlobalClient.findAll({
        where: { ownerGroupId, document: { [Op.in]: docs } },
        transaction,
      });
      const existingMap = new Map(existing.map(e => [e.document, e]));

      for (const row of globalRows) {
        const found = existingMap.get(row.document);
        if (found) {
          await StoreClient.findOrCreate({
            where: { globalClientId: found.id, storeId },
            defaults: { globalClientId: found.id, storeId },
            transaction,
          });
          if (row.override === 'update') {
            await found.update(pickNonEmpty(row), { transaction });
          }
        } else {
          const created = await GlobalClient.create({ ...row, ownerGroupId, createdByStoreId: storeId }, { transaction });
          await StoreClient.create({ globalClientId: created.id, storeId }, { transaction });
        }
      }

      // --- Local path ---
      if (localRows.length) {
        await Clients.bulkCreate(localRows.map(r => ({ ...r, storeId })), { transaction });
      }
    }
    await transaction.commit();
  } catch (err) {
    await transaction.rollback();
    throw err;
  }
}
```

```typescript
import { Injectable, Logger } from '@nestjs/common';
import { InjectModel, InjectConnection } from '@nestjs/sequelize';
import { Op, Transaction } from 'sequelize';
import { Sequelize } from 'sequelize-typescript';
import { GlobalClient } from '../shared/global-clients/global-clients.model';
import { StoreClient } from '../shared/store-clients/store-clients.model';
import { Clients } from '../clients/clients.model';
import { ClientImport } from './models/client-import.model';
import { ImportRowDto, Bucket } from './dto/import-row.dto';
import { isValidCuit, normalizeCuit } from './validators/cuit.validator';
import { isValidDni, normalizeDni } from './validators/dni.validator';

const CHUNK_SIZE = 500;

interface BatchContext {
  userId: number;
  storeId: number;
  ownerGroupId: number;
  fileName: string;
  missingDocPolicy: 'local' | 'skip';
}

interface RowError {
  rowIndex: number;
  errorCode: string;
  errorMessage: string;
  document?: string;
  fullname?: string;
}

@Injectable()
export class ClientImportService {
  private readonly logger = new Logger(ClientImportService.name);

  constructor(
    @InjectModel(GlobalClient) private readonly globalClientModel: typeof GlobalClient,
    @InjectModel(StoreClient) private readonly storeClientModel: typeof StoreClient,
    @InjectModel(Clients) private readonly clientsModel: typeof Clients,
    @InjectModel(ClientImport) private readonly auditModel: typeof ClientImport,
    @InjectConnection() private readonly sequelize: Sequelize,
  ) {}

  // 주석: Phase 25 REQ-25-10, REQ-25-11, REQ-25-16, REQ-25-17, REQ-25-18, REQ-25-21.
  async importBatch(rows: ImportRowDto[], ctx: BatchContext) {
    const errors: RowError[] = [];
    let createdCount = 0;
    let updatedCount = 0;
    let skippedCount = 0;
    const totalRows = rows.length;

    const transaction: Transaction = await this.sequelize.transaction();
    try {
      for (let i = 0; i < rows.length; i += CHUNK_SIZE) {
        const chunk = rows.slice(i, i + CHUNK_SIZE);

        // Validate + partition
        const validated = chunk.map((r, idx) => this.validateRow(r, i + idx, ctx.missingDocPolicy, errors));
        const globals = validated.filter(v => v && v.bucket === 'Global') as any[];
        const locals  = validated.filter(v => v && v.bucket === 'Local')  as any[];
        skippedCount += validated.filter(v => v && v.bucket === 'Skip').length;

        // --- Global bucket: batch-lookup + upsert + link ---
        if (globals.length > 0) {
          const docs = globals.map(r => r.document).filter(Boolean);
          const existing = docs.length > 0
            ? await this.globalClientModel.findAll({
                where: { ownerGroupId: ctx.ownerGroupId, document: { [Op.in]: docs } },
                transaction,
              })
            : [];

          const existingMap = new Map(existing.map((e: any) => [e.document, e]));

          for (const row of globals) {
            try {
              const found = existingMap.get(row.document);
              if (found) {
                // D4-04: default override = 'link' (link to current tienda only).
                // UI may override to 'update' (overwrite) or 'skip'.
                if (row.override === 'skip') {
                  skippedCount += 1;
                  continue;
                }

                await this.storeClientModel.findOrCreate({
                  where: { globalClientId: found.id, storeId: ctx.storeId },
                  defaults: { globalClientId: found.id, storeId: ctx.storeId },
                  transaction,
                });

                if (row.override === 'update') {
                  await found.update(this.pickUpdateFields(row), { transaction });
                  updatedCount += 1;
                } else {
                  // 'link' (default) — just the StoreClient link, no GlobalClient field write
                  updatedCount += 1;   // count as touched (linked)
                }
              } else {
                const created = await this.globalClientModel.create(
                  {
                    ownerGroupId: ctx.ownerGroupId,
                    document: row.document,
                    fullname: row.fullname,
                    email: row.email ?? null,
                    phone: row.phone ?? null,
                    address: row.address ?? null,
                    provinceId: row.provinceId ?? null,
                    city: row.city ?? null,
                    birthdate: row.birthdate ? new Date(row.birthdate) : null,
                    notes: row.notes ?? null,
                    createdByStoreId: ctx.storeId,
                    isActive: true,
                  },
                  { transaction },
                );

                await this.storeClientModel.create(
                  { globalClientId: created.id, storeId: ctx.storeId },
                  { transaction },
                );
                createdCount += 1;
              }
            } catch (rowErr: any) {
              errors.push({
                rowIndex: row.__rowIndex,
                errorCode: 'GLOBAL_UPSERT_FAILED',
                errorMessage: rowErr.message ?? String(rowErr),
                document: row.document,
                fullname: row.fullname,
              });
            }
          }
        }

        // --- Local bucket: bulkCreate into legacy clients ---
        if (locals.length > 0) {
          try {
            const rowsForInsert = locals.map(r => ({
              storeId: ctx.storeId,
              fullname: r.fullname,
              document: r.document ?? null,
              email: r.email ?? null,
              phone: r.phone ?? null,
              address: r.address ?? null,
              provinceId: r.provinceId ?? null,
              city: r.city ?? null,
              birthdate: r.birthdate ? new Date(r.birthdate) : null,
              notes: r.notes ?? null,
            }));
            const createdLocals = await this.clientsModel.bulkCreate(rowsForInsert, { transaction });
            createdCount += createdLocals.length;
          } catch (localErr: any) {
            for (const row of locals) {
              errors.push({
                rowIndex: row.__rowIndex,
                errorCode: 'LOCAL_INSERT_FAILED',
                errorMessage: localErr.message ?? String(localErr),
                document: row.document,
                fullname: row.fullname,
              });
            }
          }
        }
      }

      // Write audit row
      const audit = await this.auditModel.create(
        {
          userId: ctx.userId,
          storeId: ctx.storeId,
          fileName: ctx.fileName,
          totalRows,
          createdCount,
          updatedCount,
          skippedCount,
          errorCount: errors.length,
          executedAt: new Date(),
          missingDocPolicy: ctx.missingDocPolicy,
        },
        { transaction },
      );

      await transaction.commit();

      return {
        clientImportId: audit.id,
        totalRows,
        createdCount,
        updatedCount,
        skippedCount,
        errorCount: errors.length,
        errors,
      };
    } catch (err) {
      await transaction.rollback();
      throw err;
    }
  }

  private validateRow(raw: ImportRowDto, rowIndex: number, missingDocPolicy: 'local' | 'skip', errors: RowError[]) {
    const row: any = { ...raw, __rowIndex: rowIndex };
    // Normalize document
    const docRaw = row.document?.trim() || '';
    const normalized = docRaw.replace(/\D/g, '');
    const hasDoc = normalized.length > 0;

    if (!row.fullname || row.fullname.trim().length === 0) {
      errors.push({ rowIndex, errorCode: 'MISSING_FULLNAME', errorMessage: 'fullname required', document: docRaw });

      return null;
    }

    if (!hasDoc) {
      // No doc → bucket comes from missingDocPolicy (or per-row override already set to Skip/Local)
      if (row.bucket === 'Global') {
        errors.push({ rowIndex, errorCode: 'MISSING_DOCUMENT', errorMessage: 'Global bucket requires DNI or CUIT', fullname: row.fullname });

        return null;
      }

      return row;  // Local or Skip stays as-is
    }

    // Has a document — validate
    if (normalized.length === 11) {
      if (!isValidCuit(normalized)) {
        errors.push({ rowIndex, errorCode: 'INVALID_CUIT', errorMessage: 'CUIT fails mod 11 checksum', document: normalized, fullname: row.fullname });

        return null;
      }
    } else if (normalized.length === 7 || normalized.length === 8) {
      if (!isValidDni(normalized)) {
        errors.push({ rowIndex, errorCode: 'INVALID_DNI', errorMessage: 'DNI must be 7-8 digits', document: normalized, fullname: row.fullname });

        return null;
      }
    } else {
      errors.push({ rowIndex, errorCode: 'INVALID_DOCUMENT_LENGTH', errorMessage: 'Document must be 7-8 digits (DNI) or 11 digits (CUIT)', document: normalized, fullname: row.fullname });

      return null;
    }

    row.document = normalized;

    return row;
  }

  private pickUpdateFields(row: any) {
    const pick: any = {};
    for (const f of ['fullname', 'email', 'phone', 'address', 'provinceId', 'city', 'birthdate', 'notes']) {
      if (row[f] !== null && row[f] !== undefined && row[f] !== '') {
        pick[f] = row[f];
      }
    }

    return pick;
  }
}
```

```typescript
import { Body, Controller, Post, Req, UseGuards, HttpCode, HttpStatus } from '@nestjs/common';
import { AuthGuard } from '@nestjs/passport';
import { FunctionGuard } from '../auth/decorators/function-guard.decorator';
import { ClientImportService } from './client-import.service';
import { ImportBatchDto } from './dto/import-batch.dto';

// 주석: Phase 25 REQ-25-10, REQ-25-19 — admin/superadmin 전용 대량 import.
@Controller('clients')
@UseGuards(AuthGuard('jwt'))
export class ClientImportController {
  constructor(private readonly svc: ClientImportService) {}

  @Post('import')
  @HttpCode(HttpStatus.OK)
  @FunctionGuard('manage-clientes-import', 'create')
  async import(@Body() dto: ImportBatchDto, @Req() req: any) {
    return this.svc.importBatch(dto.rows, {
      userId: req.user.id,
      storeId: req.user.storeId,
      ownerGroupId: req.user.ownerGroupId,
      fileName: dto.fileName,
      missingDocPolicy: dto.missingDocPolicy,
    });
  }
}
```

```typescript
    expect(res.errorCount).toBe(0);
    expect(gcModel.create).toHaveBeenCalledTimes(2);

    expect(tx.commit).toHaveBeenCalled();
  });

  it('links existing global for hit rows (override=link default D4-04)', async () => {
```
