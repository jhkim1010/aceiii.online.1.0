```typescript
// 주석: Phase 25 REQ-25-14 — AR CUIT 서버사이드 검증.
// 근거: RESEARCH.md Example 2 + Pitfall 5 (check==10 → invalid, AFIP 는 발행하지 않음).
// Weights: https://lookuptax.com/docs/tax-identification-number/argentina-tax-id-guide

/**
 * CUIT mod 11 weights applied to the first 10 digits.
 * Weights sum * digit[i] ; rem = sum % 11 ; calc = 11 - rem.
 * - calc == 11 → check digit = 0
 * - calc == 10 → INVALID (AFIP does not issue — reject)
 * - otherwise → check digit = calc
 */
const CUIT_WEIGHTS = [5, 4, 3, 2, 7, 6, 5, 4, 3, 2] as const;

/**
 * Strips every non-digit character from the input and returns the digit-only string.
 * Returns empty string for null/undefined.
 */
export function normalizeCuit(raw: string | null | undefined): string {
  if (raw === null || raw === undefined) return '';

  return String(raw).replace(/\D/g, '');
}

/**
 * Validates an Argentine CUIT by format + mod 11 checksum.
 *
 * @param raw - the raw input (may contain dashes, dots, spaces).
 * @returns true if the input normalizes to 11 digits and passes AFIP mod 11.
 */
export function isValidCuit(raw: string | null | undefined): boolean {
  const digits = normalizeCuit(raw);
  if (digits.length !== 11) return false;

  const ns = digits.split('').map(Number);

  let sum = 0;
  for (let i = 0; i < 10; i++) {
    sum += ns[i] * CUIT_WEIGHTS[i];
  }

  const rem = sum % 11;
  const calc = 11 - rem;
  let expected: number;
  if (calc === 11) {
    expected = 0;
  } else if (calc === 10) {
    return false;  // AFIP 은 발행하지 않음 (Pitfall 5)
  } else {
    expected = calc;
  }

  return expected === ns[10];
}
```

```typescript
// 주석: Phase 25 REQ-25-14 — AR DNI 서버사이드 검증 (7-8 자리 숫자).

/**
 * Strips every non-digit character; returns empty for null/undefined.
 */
export function normalizeDni(raw: string | null | undefined): string {
  if (raw === null || raw === undefined) return '';

  return String(raw).replace(/\D/g, '');
}

/**
 * Validates an Argentine DNI (7 or 8 digits after normalization).
 */
export function isValidDni(raw: string | null | undefined): boolean {
  const digits = normalizeDni(raw);

  return /^\d{7,8}$/.test(digits);
}
```
