-- ===========================================================================
-- Phase 25 Wave 1 Step 5: 레거시 clients → global_clients + store_clients + sales 재매핑
-- 실행: sudo -u postgres psql -d ventago -v ON_ERROR_STOP=1 -1 -f 20260424-phase25-step5-data-migration.sql
-- 목적: D2-02 일회성 bulk 이관. document 가 DNI(7-8자리) 또는 CUIT(11자리) 형식을 통과한 레거시 클라이언트만 승격.
-- 멱등성: ON CONFLICT DO NOTHING + WHERE NOT EXISTS 로 재실행 안전.
-- PG10 호환: SERIAL (GENERATED AS IDENTITY 금지).
-- ===========================================================================

BEGIN;

-- (A) 후보 레거시 고객 임시 추출 — document 가 DNI/CUIT 형식 통과한 행만
-- 주석: mod 11 체크섬은 SQL 에서 구현 불가능 → Wave 3 promote service 에서 재검증
CREATE TEMP TABLE IF NOT EXISTS tmp_legacy_promote AS
SELECT
  c.id                       AS legacy_client_id,
  c.store_id,
  c.fullname,
  regexp_replace(c.document, '\D', '', 'g') AS normalized_document,
  c.email,
  c.phone,
  c.address,
  c.province_id,
  c.city,
  c.birthdate,
  c.notes,
  c.created_at,
  c.updated_at
FROM clients c
WHERE c.document IS NOT NULL
  AND (
    regexp_replace(c.document, '\D', '', 'g') ~ '^\d{7,8}$'     -- DNI
    OR regexp_replace(c.document, '\D', '', 'g') ~ '^\d{11}$'   -- CUIT (mod 11 은 Wave 3 에서)
  );

-- (B) 승격 대상이 없으면 조용히 종료
DO $$
DECLARE
  cnt INTEGER;
BEGIN
  SELECT COUNT(*) INTO cnt FROM tmp_legacy_promote;
  RAISE NOTICE 'Phase 25 Step 5: % legacy clients qualify for promotion.', cnt;
END $$;

-- (C) global_clients 에 upsert (ownerGroupId=1 — 모든 기존 매장은 그룹 1)
INSERT INTO global_clients (
  owner_group_id, document, fullname, email, phone, address,
  province_id, city, birthdate, notes, created_by_store_id, is_active,
  created_at, updated_at
)
SELECT
  1                          AS owner_group_id,
  t.normalized_document      AS document,
  t.fullname,
  t.email,
  t.phone,
  t.address,
  t.province_id,
  t.city,
  t.birthdate,
  t.notes,
  t.store_id                 AS created_by_store_id,
  TRUE                       AS is_active,
  COALESCE(t.created_at, NOW()),
  NOW()
FROM tmp_legacy_promote t
ON CONFLICT (owner_group_id, document) WHERE document IS NOT NULL
  DO NOTHING;

-- (D) store_clients 연결 (globalClientId, storeId) 유일해야 함 — findOrCreate 패턴
INSERT INTO store_clients (global_client_id, store_id, created_at, updated_at)
SELECT
  gc.id          AS global_client_id,
  t.store_id     AS store_id,
  NOW(),
  NOW()
FROM tmp_legacy_promote t
JOIN global_clients gc
  ON gc.owner_group_id = 1
 AND gc.document = t.normalized_document
WHERE NOT EXISTS (
  SELECT 1 FROM store_clients sc
  WHERE sc.global_client_id = gc.id
    AND sc.store_id = t.store_id
);

-- (E) sales.store_client_id 재매핑 — legacy client_id 가 이관된 레거시 클라이언트를 가리키면 새 store_client.id 로 업데이트
UPDATE sales s
SET store_client_id = sc.id
FROM tmp_legacy_promote t
JOIN global_clients gc
  ON gc.owner_group_id = 1
 AND gc.document = t.normalized_document
JOIN store_clients sc
  ON sc.global_client_id = gc.id
 AND sc.store_id = t.store_id
WHERE s.client_id = t.legacy_client_id
  AND s.store_id  = t.store_id
  AND s.store_client_id IS NULL;  -- 이미 매핑된 행은 건드리지 않음 (멱등)

-- (F) 요약 로그
DO $$
DECLARE
  promoted INTEGER;
  linked INTEGER;
  remapped INTEGER;
BEGIN
  SELECT COUNT(*) INTO promoted FROM global_clients WHERE owner_group_id = 1;
  SELECT COUNT(*) INTO linked   FROM store_clients sc JOIN global_clients gc ON gc.id = sc.global_client_id WHERE gc.owner_group_id = 1;
  SELECT COUNT(*) INTO remapped FROM sales WHERE store_client_id IS NOT NULL;
  RAISE NOTICE 'Phase 25 Step 5 complete: global_clients=%, store_clients=%, sales_remapped=%.', promoted, linked, remapped;
END $$;

DROP TABLE tmp_legacy_promote;

COMMIT;
```

-- ===========================================================================
-- Phase 25 Wave 1 Step 6: 읽기 전용 검증 쿼리 (idempotent, 상태 변경 없음)
-- 실행: sudo -u postgres psql -d ventago -v ON_ERROR_STOP=1 -1 -f 20260424-phase25-step6-verify.sql
-- 목적: Step 1-5 적용 후 일관성 검증. 모든 SELECT 결과는 수동으로 리뷰.
-- ===========================================================================

\echo '=== 1. stores.owner_group_id 컬럼 & 데이터 ==='
SELECT id, name, owner_group_id FROM stores ORDER BY id;

\echo ''
\echo '=== 2. global_clients.owner_group_id 컬럼 & UNIQUE 인덱스 ==='
SELECT COUNT(*) AS global_clients_total,
       COUNT(DISTINCT (owner_group_id, document)) AS unique_pairs
  FROM global_clients WHERE document IS NOT NULL;

SELECT indexname
  FROM pg_indexes
 WHERE tablename = 'global_clients'
 ORDER BY indexname;

\echo ''
\echo '=== 3. 레거시 idx_global_clients_name_phone 제거 확인 ==='
SELECT CASE WHEN COUNT(*) = 0 THEN 'DROPPED (OK)' ELSE 'STILL PRESENT (FAIL)' END AS legacy_index_status
  FROM pg_indexes
 WHERE tablename='global_clients'
   AND indexname='idx_global_clients_name_phone';

\echo ''
\echo '=== 4. store_clients 생성 확인 ==='
SELECT COUNT(*) AS store_clients_total,
       COUNT(DISTINCT global_client_id) AS distinct_globals,
       COUNT(DISTINCT store_id) AS distinct_stores
  FROM store_clients;

\echo ''
\echo '=== 5. sales.store_client_id FK + 매핑 현황 ==='
SELECT COUNT(*) AS sales_total,
       COUNT(store_client_id) AS sales_remapped,
       COUNT(*) - COUNT(store_client_id) AS sales_still_legacy_only
  FROM sales;

\echo ''
\echo '=== 6. sales.store_id vs store_clients.store_id 일관성 (owner scope 위반 가능성) ==='
SELECT s.id AS sale_id, s.store_id AS sale_store, sc.store_id AS sc_store
  FROM sales s
  JOIN store_clients sc ON sc.id = s.store_client_id
 WHERE s.store_id <> sc.store_id
 LIMIT 20;

\echo ''
\echo '=== 7. 레거시 clients 중 승격되지 않은 행 (document 무효 또는 NULL) ==='
SELECT COUNT(*) AS unpromoted_legacy_count
  FROM clients c
 WHERE NOT EXISTS (
   SELECT 1 FROM global_clients gc
    WHERE gc.owner_group_id = 1
      AND gc.document = regexp_replace(COALESCE(c.document, ''), '\D', '', 'g')
      AND gc.document <> ''
 );

\echo ''
\echo '=== 8. FK 일관성 검증 (orphan 여부) ==='
SELECT
  (SELECT COUNT(*) FROM sales s LEFT JOIN store_clients sc ON sc.id = s.store_client_id WHERE s.store_client_id IS NOT NULL AND sc.id IS NULL) AS orphan_sales,
  (SELECT COUNT(*) FROM store_clients sc LEFT JOIN global_clients gc ON gc.id = sc.global_client_id WHERE gc.id IS NULL) AS orphan_store_clients;

\echo ''
\echo '=== VERIFICATION COMPLETE ==='
```
