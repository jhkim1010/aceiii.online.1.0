- Every value must be 'local' OR 'global'
- Document is NOT allowed in fieldPicks (winner's document is authoritative — D1-04 UI also disables document radio)

```typescript
// 주석: Phase 25 D1-04 — field-by-field merge 커밋. 옵티미스틱 락 via winnerUpdatedAt.
async merge(
  payload: {
    localClientId?: number | null;
    winnerGlobalClientId: number;
    fieldPicks: Record<string, 'local' | 'global'>;
    winnerUpdatedAt: string;
  },
  ctx: { userId: number; storeId: number; ownerGroupId: number },
): Promise<{
  status: 'merged';
  globalClientId: number;
  storeClientId: number;
  salesRemappedCount: number;
  clientMergeAuditId: number;
}> {
  // Validate fieldPicks keys
  for (const k of Object.keys(payload.fieldPicks || {})) {
    if (k === 'document') {
      throw new BadRequestException('Cannot override document in merge — document is authoritative on winner');
    }

    if (!ClientsService.MERGE_ALLOWED_FIELDS.includes(k)) {
      throw new BadRequestException(`Invalid merge field: ${k}`);
    }

    if (!['local', 'global'].includes(payload.fieldPicks[k])) {
      throw new BadRequestException(`Invalid pick value for ${k}: must be 'local' or 'global'`);
    }
  }

  // Fetch winner with ownerGroup check
  const winner = await this.globalClientModel.findOne({
    where: { id: payload.winnerGlobalClientId, ownerGroupId: ctx.ownerGroupId },
  });
  if (!winner) throw new NotFoundException('Winner GlobalClient not found in caller ownerGroup');

  // Optimistic lock check
  const winnerTs = winner.updatedAt instanceof Date ? winner.updatedAt.toISOString() : String(winner.updatedAt);
  if (winnerTs !== payload.winnerUpdatedAt) {
    throw new ConflictException('STALE_MERGE');
  }

  // Fetch local (if provided) — required if any pick === 'local'
  let local: any = null;
  if (payload.localClientId) {
    local = await this.clientsModel.findOne({
      where: { id: payload.localClientId, storeId: ctx.storeId },
    });
    if (!local) throw new NotFoundException('Local client not found in caller store');
  }

  const transaction = await this.sequelize.transaction();
  try {
    // Build patch from field picks
    const patch: any = {};
    for (const [field, side] of Object.entries(payload.fieldPicks)) {
      if (side === 'local') {
        if (!local) {
          throw new BadRequestException(`Cannot pick 'local' for field ${field} — localClientId not provided`);
        }

        patch[field] = local[field] ?? null;
      } else {
        patch[field] = (winner as any)[field] ?? null;
      }
    }

    // Apply patch
    if (Object.keys(patch).length > 0) {
      await winner.update(patch, { transaction });
    }

    // Ensure StoreClient link for caller store
    const [sc] = await this.storeClientModel.findOrCreate({
      where: { globalClientId: winner.id, storeId: ctx.storeId },
      defaults: { globalClientId: winner.id, storeId: ctx.storeId },
      transaction,
    });

    // Remap sales for caller's store that reference local client (if provided)
    let salesRemappedCount = 0;
    if (payload.localClientId) {
      const [n] = await this.salesModel.update(
        { storeClientId: sc.id },
        {
          where: { clientId: payload.localClientId, storeId: ctx.storeId, storeClientId: null },
          transaction,
        },
      );
      salesRemappedCount = n;
    }

    // Write audit row
    const audit = await this.clientMergeModel.create(
      {
        userId: ctx.userId,
        storeId: ctx.storeId,
        winnerGlobalClientId: winner.id,
        loserGlobalClientId: null,                  // promote-from-local has no "loser" GlobalClient
        localClientId: payload.localClientId ?? null,
        fieldPicks: payload.fieldPicks,
        mergeReason: payload.localClientId ? 'promote_conflict' : 'manual_merge',
        mergedAt: new Date(),
      },
      { transaction },
    );

    await transaction.commit();

    return {
      status: 'merged',
      globalClientId: winner.id,
      storeClientId: sc.id,
      salesRemappedCount,
      clientMergeAuditId: audit.id,
    };
  } catch (err) {
    await transaction.rollback();
    throw err;
  }
}
```

```typescript
import { Test } from '@nestjs/testing';
import { getModelToken, getConnectionToken } from '@nestjs/sequelize';
import { BadRequestException, ConflictException, NotFoundException } from '@nestjs/common';
import { ClientsService } from './clients.service';
import { Clients } from './clients.model';
import { GlobalClient } from '../shared/global-clients/global-clients.model';
import { StoreClient } from '../shared/store-clients/store-clients.model';
import { Sale } from '../sales/sales.model';
import { ClientMerge } from './models/client-merge.model';

describe('ClientsService.merge', () => {
  let svc: ClientsService;
  let clientsModel: any, globalClientModel: any, storeClientModel: any, salesModel: any, mergeAudit: any, sequelize: any;
  const tx = { commit: jest.fn(), rollback: jest.fn() };

  const ctx = { userId: 1, storeId: 3, ownerGroupId: 1 };
  const WINNER_TS = '2026-04-23T12:00:00.000Z';

  const mockWinner = () => ({
    id: 99, ownerGroupId: 1, fullname: 'Juan G', email: 'g@g.com',
    updatedAt: new Date(WINNER_TS),
    update: jest.fn().mockResolvedValue(undefined),
  });

  beforeEach(async () => {
    clientsModel = { findOne: jest.fn() };
    globalClientModel = { findOne: jest.fn() };
    storeClientModel = { findOrCreate: jest.fn() };
    salesModel = { update: jest.fn() };
    mergeAudit = { create: jest.fn() };
    sequelize = { transaction: jest.fn().mockResolvedValue(tx) };

    const moduleRef = await Test.createTestingModule({
      providers: [
        ClientsService,
        { provide: getModelToken(Clients), useValue: clientsModel },
        { provide: getModelToken(GlobalClient), useValue: globalClientModel },
        { provide: getModelToken(StoreClient), useValue: storeClientModel },
        { provide: getModelToken(Sale), useValue: salesModel },
        { provide: getModelToken(ClientMerge), useValue: mergeAudit },
        { provide: getConnectionToken(), useValue: sequelize },
      ],
    }).compile();

    svc = moduleRef.get(ClientsService);
    tx.commit.mockClear();
    tx.rollback.mockClear();
  });

  it('merges successfully with valid payload', async () => {
    const winner = mockWinner();
    globalClientModel.findOne.mockResolvedValue(winner);
    clientsModel.findOne.mockResolvedValue({ id: 10, storeId: 3, fullname: 'Juan L', email: 'l@l.com' });
    storeClientModel.findOrCreate.mockResolvedValue([{ id: 200 }]);
    salesModel.update.mockResolvedValue([2]);
    mergeAudit.create.mockResolvedValue({ id: 1000 });

    const res = await svc.merge(
      {
        localClientId: 10,
        winnerGlobalClientId: 99,
        fieldPicks: { fullname: 'local', email: 'global' },
        winnerUpdatedAt: WINNER_TS,
      },
      ctx,
    );

    expect(res).toEqual({
      status: 'merged', globalClientId: 99, storeClientId: 200, salesRemappedCount: 2, clientMergeAuditId: 1000,
    });
    expect(winner.update).toHaveBeenCalledWith({ fullname: 'Juan L', email: 'g@g.com' }, expect.anything());
    expect(mergeAudit.create).toHaveBeenCalledWith(expect.objectContaining({ mergeReason: 'promote_conflict' }), expect.anything());
    expect(tx.commit).toHaveBeenCalled();
  });

  it('throws ConflictException on stale winnerUpdatedAt', async () => {
    globalClientModel.findOne.mockResolvedValue(mockWinner());
    await expect(svc.merge({
      localClientId: 10, winnerGlobalClientId: 99,
      fieldPicks: { fullname: 'local' }, winnerUpdatedAt: '1999-01-01T00:00:00.000Z',
    }, ctx)).rejects.toThrow(ConflictException);
  });

  it('throws BadRequest on invalid field in fieldPicks', async () => {
    await expect(svc.merge({
      localClientId: 10, winnerGlobalClientId: 99,
      fieldPicks: { evil_field: 'local' }, winnerUpdatedAt: WINNER_TS,
    }, ctx)).rejects.toThrow(BadRequestException);
  });

  it('throws BadRequest when document in fieldPicks (D1-04)', async () => {
    await expect(svc.merge({
      localClientId: 10, winnerGlobalClientId: 99,
      fieldPicks: { document: 'local' }, winnerUpdatedAt: WINNER_TS,
    }, ctx)).rejects.toThrow(BadRequestException);
  });

  it('rolls back when audit create fails', async () => {
    const winner = mockWinner();
    globalClientModel.findOne.mockResolvedValue(winner);
    clientsModel.findOne.mockResolvedValue({ id: 10, storeId: 3, fullname: 'X' });
    storeClientModel.findOrCreate.mockResolvedValue([{ id: 200 }]);
    salesModel.update.mockResolvedValue([0]);
    mergeAudit.create.mockRejectedValue(new Error('audit failure'));

    await expect(svc.merge({
      localClientId: 10, winnerGlobalClientId: 99,
      fieldPicks: { fullname: 'local' }, winnerUpdatedAt: WINNER_TS,
    }, ctx)).rejects.toThrow('audit failure');
    expect(tx.rollback).toHaveBeenCalled();
    expect(tx.commit).not.toHaveBeenCalled();
  });

  it('throws NotFound when winner not in ownerGroup', async () => {
    globalClientModel.findOne.mockResolvedValue(null);
    await expect(svc.merge({
      localClientId: 10, winnerGlobalClientId: 9999,
      fieldPicks: { fullname: 'local' }, winnerUpdatedAt: WINNER_TS,
    }, ctx)).rejects.toThrow(NotFoundException);
  });
});
```
