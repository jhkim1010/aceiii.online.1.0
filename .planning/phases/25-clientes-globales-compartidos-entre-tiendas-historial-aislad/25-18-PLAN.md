`api-ventago/src/app/clients/clients.model.ts` 에 hook 추가:
```typescript
import { AfterCreate, AfterBulkCreate, ... } from 'sequelize-typescript';
import { Logger } from '@nestjs/common';
import { getClientsSyncSingleton } from '../shared/clients-sync/clients-sync.service';

@Table({ timestamps: true })
export class Clients extends Model {
  // ... existing columns

  @AfterCreate
  static async syncToGlobalAfterCreate(instance: Clients, options: any) {
    const sync = getClientsSyncSingleton();
    if (!sync) return;  // bootstrap 전이거나 standalone 스크립트 — skip
    if (options?._skipGlobalSync) return;  // 명시적 escape hatch (backfill 등)

    try {
      await sync.syncFromLegacy(
        instance,
        { ownerGroupId: null },  // service 가 storeId 로 보강
        options?.transaction,
      );
    } catch (err) {
      // 주석: hook 은 caller 의 INSERT 를 절대 막지 않는다 — sync 실패 시 legacy row 는 보존,
      // partial UNIQUE 가 final backstop. 에러는 운영 로그에 남겨 모니터링.
      Logger.warn(
        `[Clients.afterCreate] syncFromLegacy 실패 — legacy row 는 보존됨. clientId=${instance.id}, err=${String(err)}`,
        'ClientsHook',
      );
    }
  }

  @AfterBulkCreate
  static async syncToGlobalAfterBulkCreate(instances: Clients[], options: any) {
    const sync = getClientsSyncSingleton();
    if (!sync) return;
    if (options?._skipGlobalSync) return;

    for (const instance of instances) {
      try {
        await sync.syncFromLegacy(instance, { ownerGroupId: null }, options?.transaction);
      } catch (err) {
        Logger.warn(`[Clients.afterBulkCreate] syncFromLegacy 실패. clientId=${instance.id}`, 'ClientsHook');
      }
    }
  }
}
```

```typescript
import { attachClientsHook } from '../shared/clients-sync/clients-sync.service';

describe('Clients afterCreate hook', () => {
  let mockSync: any;

  beforeEach(() => {
    mockSync = { syncFromLegacy: jest.fn().mockResolvedValue(null) };
    attachClientsHook(mockSync);
  });

  afterEach(() => {
    attachClientsHook(null as any);  // reset
  });

  it('invokes syncFromLegacy with the new instance and options.transaction', async () => {
    const fakeTransaction: any = { id: 'tx-1' };
    // simulate hook directly (모델 메서드 정적 호출)
    await (Clients as any).syncToGlobalAfterCreate(
      { id: 1, storeId: 9, document: '20123456789' } as any,
      { transaction: fakeTransaction },
    );
    expect(mockSync.syncFromLegacy).toHaveBeenCalledWith(
      expect.objectContaining({ id: 1, storeId: 9, document: '20123456789' }),
      { ownerGroupId: null },
      fakeTransaction,
    );
  });

  it('skips when _skipGlobalSync option is true', async () => {
    await (Clients as any).syncToGlobalAfterCreate(
      { id: 2, storeId: 9, document: '20123456789' } as any,
      { _skipGlobalSync: true },
    );
    expect(mockSync.syncFromLegacy).not.toHaveBeenCalled();
  });

  it('skips when sync singleton is null', async () => {
    attachClientsHook(null as any);
    await (Clients as any).syncToGlobalAfterCreate(
      { id: 3, storeId: 9, document: '20123456789' } as any,
      {},
    );
    // no throw
  });

  it('does not throw when syncFromLegacy rejects', async () => {
    mockSync.syncFromLegacy.mockRejectedValue(new Error('DB down'));
    await expect(
      (Clients as any).syncToGlobalAfterCreate(
        { id: 4, storeId: 9, document: '20123456789' } as any,
        {},
      ),
    ).resolves.not.toThrow();
  });

  it('afterBulkCreate iterates instances', async () => {
    await (Clients as any).syncToGlobalAfterBulkCreate(
      [
        { id: 5, storeId: 9, document: '20123456789' },
        { id: 6, storeId: 9, document: '20987654321' },
      ] as any,
      {},
    );
    expect(mockSync.syncFromLegacy).toHaveBeenCalledTimes(2);
  });
});
```

| Hook 실행 시점 | Sequelize transaction 안 (caller transaction 존재 시) — caller commit 직전. Hook 실패가 transaction rollback 을 트리거하지 않음 (graceful) |
| Bridge singleton | bootstrap 전 standalone 환경 (jest, migration) 에서 null — graceful skip 으로 호환성 보장 |
| 중복 호출 | service-layer + hook 동시 호출 가능 — findOrCreate 라 idempotent, 단 추가 쿼리 1회 발생 (성능 trace 모니터링) |
