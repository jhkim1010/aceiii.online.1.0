```typescript
type Bucket = 'Global' | 'Local' | 'Skip';

interface ClassifiedRow extends ParsedClient {
  bucket: Bucket;
  errors: string[];
  override?: Bucket | 'link' | 'update';
  existingGlobalHit?: boolean;   // flag when a Global match found (Plan 12 expands this)
}

function classifyRow(row: ParsedClient, defaultNoDoc: 'Local' | 'Skip'): ClassifiedRow {
  const errors: string[] = [];
  const hasDoc = row.document && row.document.trim().length > 0;
  if (!hasDoc) {
    return { ...row, bucket: defaultNoDoc, errors };
  }
  const digits = row.document.replace(/\D/g, '');
  if (digits.length === 11) {
    if (!isValidCuit(digits)) errors.push('CUIT checksum inválido');
  } else if (digits.length === 7 || digits.length === 8) {
    // DNI OK
  } else {
    errors.push('Document debe ser DNI (7-8 dígitos) o CUIT (11 dígitos)');
  }
  if (row.email && !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(row.email)) {
    errors.push('Email inválido');
  }
  return {
    ...row,
    bucket: errors.length > 0 ? 'Skip' : 'Global',
    errors,
  };
}
```

```typescript
// 주석: Phase 25 — AR CUIT 프론트엔드 검증 (Next.js 는 api-ventago 에서 import 불가 → 의도적 복제).
// 원본: api-ventago/src/app/client-import/validators/cuit.validator.ts

const CUIT_WEIGHTS = [5, 4, 3, 2, 7, 6, 5, 4, 3, 2] as const

export function normalizeCuit(raw: string | null | undefined): string {
  if (raw === null || raw === undefined) return ''

  return String(raw).replace(/\D/g, '')
}

export function isValidCuit(raw: string | null | undefined): boolean {
  const digits = normalizeCuit(raw)
  if (digits.length !== 11) return false

  const ns = digits.split('').map(Number)

  let sum = 0
  for (let i = 0; i < 10; i++) {
    sum += ns[i] * CUIT_WEIGHTS[i]
  }

  const rem = sum % 11
  const calc = 11 - rem
  let expected: number
  if (calc === 11) {
    expected = 0
  } else if (calc === 10) {
    return false
  } else {
    expected = calc
  }

  return expected === ns[10]
}
```

```typescript
function classifyRow(
  row: any,
  rowIndex: number,
  missingDocPolicy: 'local' | 'skip',
): ClassifiedRow {
  const errors: string[] = []
  const docRaw = (row.document ?? '').toString().trim()
  const digits = docRaw.replace(/\D/g, '')
  const hasDoc = digits.length > 0

  if (!row.fullname || row.fullname.toString().trim().length === 0) {
    errors.push('Falta nombre')
  }

  if (!hasDoc) {
    return {
      ...row,
      document: undefined,
      bucket: missingDocPolicy === 'skip' ? 'Skip' : 'Local',
      errors,
      __rowIndex: rowIndex,
    }
  }

  if (digits.length === 11) {
    if (!isValidCuit(digits)) errors.push('CUIT inválido (fallo checksum mod-11)')
  } else if (digits.length === 7 || digits.length === 8) {
    if (!isValidDni(digits)) errors.push('DNI inválido (se esperan 7-8 dígitos)')
  } else {
    errors.push('Document debe ser DNI (7-8) o CUIT (11)')
  }

  if (row.email && !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(row.email)) {
    errors.push('Formato de email inválido')
  }

  return {
    ...row,
    document: digits,
    bucket: errors.length > 0 ? 'Skip' : 'Global',
    errors,
    __rowIndex: rowIndex,
  }
}
```
