```typescript
import { Injectable, Logger } from '@nestjs/common';
import { InjectModel } from '@nestjs/sequelize';
import { Transaction } from 'sequelize';

import {
  isValidCuit,
  normalizeCuit,
} from '../../client-import/validators/cuit.validator';
import { isValidDni } from '../../client-import/validators/dni.validator';
import { GlobalClient } from '../global-clients/global-clients.model';
import { StoreClient } from '../store-clients/store-clients.model';
import { Store } from '../../store/store.model';

interface SyncableClient {
  id: number;
  storeId: number;
  document: string | null;
  fullname?: string | null;
  email?: string | null;
  phone?: string | null;
  address?: string | null;
  location?: string | null;
  provinceId?: number | null;
  transport?: string | null;
  resIva?: string | null;
  nameFantasy?: string | null;
}

interface SyncContext {
  ownerGroupId?: number | null;
}

interface SyncResult {
  globalClientId: number;
  storeClientId: number;
  alreadyExisted: boolean;
}

@Injectable()
export class ClientsSyncService {
  private readonly logger = new Logger(ClientsSyncService.name);

  constructor(
    @InjectModel(GlobalClient)
    private readonly globalClientModel: typeof GlobalClient,
    @InjectModel(StoreClient)
    private readonly storeClientModel: typeof StoreClient,
    @InjectModel(Store) private readonly storeModel: typeof Store,
  ) {}

  /**
   * 레거시 clients row 1건을 GlobalClient + StoreClient 로 idempotent sync.
   * - document 가 invalid/temp 이면 null 반환 (skip)
   * - ownerGroupId 가 ctx 에 없으면 storeId 로 Store 조회해 보강
   * - 보강 실패 시 null 반환 (anonymous owner 매장은 글로벌 풀에 절대 진입 불가)
   */
  async syncFromLegacy(
    legacy: SyncableClient,
    ctx: SyncContext,
    transaction?: Transaction,
  ): Promise<SyncResult | null> {
    // 주석: D1-02 — valid DNI/CUIT 없으면 글로벌/스토어 매핑 절대 금지
    const doc = normalizeCuit(legacy.document ?? '');
    if (!doc) return null;
    if (!isValidDni(doc) && !isValidCuit(doc)) return null;

    // ownerGroupId 보강
    let ownerGroupId = ctx.ownerGroupId ?? null;
    if (!ownerGroupId && legacy.storeId) {
      const store = await this.storeModel.findByPk(legacy.storeId, {
        attributes: ['id', 'ownerGroupId'],
        transaction,
      });
      ownerGroupId = (store as any)?.ownerGroupId ?? null;
    }
    if (!ownerGroupId) return null;

    // 1. 같은 owner 그룹의 동일 document GlobalClient 조회/생성
    const [gc, gcCreated] = await this.globalClientModel.findOrCreate({
      where: { ownerGroupId, document: doc },
      defaults: {
        ownerGroupId,
        document: doc,
        fullname: legacy.fullname ?? null,
        nameFantasy: legacy.nameFantasy ?? null,
        email: legacy.email ?? null,
        phone: legacy.phone ?? null,
        address: legacy.address ?? null,
        location: legacy.location ?? null,
        provinceId: legacy.provinceId ?? null,
        transport: legacy.transport ?? null,
        resIva: legacy.resIva ?? null,
        createdByStoreId: legacy.storeId,
        isActive: true,
      } as any,
      transaction,
    });

    // 2. (globalClientId, storeId) 매핑 — store_clients findOrCreate
    const [sc] = await this.storeClientModel.findOrCreate({
      where: { globalClientId: gc.id, storeId: legacy.storeId },
      defaults: {
        globalClientId: gc.id,
        storeId: legacy.storeId,
        isActive: true,
      } as any,
      transaction,
    });

    return {
      globalClientId: gc.id,
      storeClientId: sc.id,
      alreadyExisted: !gcCreated,
    };
  }
}
```
