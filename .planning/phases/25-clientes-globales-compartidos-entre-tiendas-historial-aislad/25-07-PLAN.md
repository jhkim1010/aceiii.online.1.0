```typescript
// 주석: Phase 25 REQ-25-05 + D1-03 — 로컬 client 가 DNI/CUIT 를 가지면 global 승격.
// 같은 (ownerGroupId, document) GlobalClient 가 존재하면 409 + merge payload 반환 (Plan 08 에서 commit).
async promote(
  localClientId: number,
  ctx: { userId: number; storeId: number; ownerGroupId: number },
): Promise<
  | { status: 'promoted'; globalClientId: number; storeClientId: number; salesRemappedCount: number }
  | { status: 'merge_required'; existingGlobalClient: any; localClient: any; conflictFields: any[] }
> {
  const local = await this.clientsModel.findOne({
    where: { id: localClientId, storeId: ctx.storeId },
  });
  if (!local) throw new NotFoundException('Local client not found in caller store');

  const doc = this.normalizeDocument(local.document);
  if (!doc) throw new BadRequestException('Promotion requires valid DNI or CUIT');

  if (!this.isValidDni(doc) && !this.isValidCuit(doc)) {
    throw new BadRequestException('Document must be valid DNI (7-8 digits) or CUIT (11 digits + mod11 checksum)');
  }

  // Check for existing GlobalClient in same ownerGroup
  const existingGlobal = await this.globalClientModel.findOne({
    where: { ownerGroupId: ctx.ownerGroupId, document: doc },
  });

  if (existingGlobal) {
    // Build field-by-field conflict report
    const fields = ['fullname', 'email', 'phone', 'address', 'provinceId', 'city', 'birthdate', 'notes'];
    const conflictFields = fields
      .filter(f => {
        const lv = (local as any)[f];
        const gv = (existingGlobal as any)[f];
        // Treat empty-string, null, undefined as "empty" and equivalent
        const lEmpty = lv === null || lv === undefined || lv === '';
        const gEmpty = gv === null || gv === undefined || gv === '';
        if (lEmpty && gEmpty) return false;

        return String(lv ?? '') !== String(gv ?? '');
      })
      .map(f => ({ field: f, localValue: (local as any)[f], globalValue: (existingGlobal as any)[f] }));

    return {
      status: 'merge_required',
      existingGlobalClient: existingGlobal.toJSON(),
      localClient: local.toJSON(),
      conflictFields,
    };
  }

  // Atomic happy path: create GlobalClient + StoreClient link + remap sales
  const transaction: Transaction = await this.sequelize.transaction();
  try {
    const created = await this.globalClientModel.create(
      {
        ownerGroupId: ctx.ownerGroupId,
        document: doc,
        fullname: local.fullname,
        email: local.email,
        phone: local.phone,
        address: local.address,
        provinceId: local.provinceId,
        city: local.city,
        birthdate: local.birthdate,
        notes: local.notes,
        createdByStoreId: ctx.storeId,
        isActive: true,
      },
      { transaction },
    );

    const sc = await this.storeClientModel.create(
      { globalClientId: created.id, storeId: ctx.storeId },
      { transaction },
    );

    const [remapped] = await this.salesModel.update(
      { storeClientId: sc.id },
      {
        where: { clientId: localClientId, storeId: ctx.storeId, storeClientId: null },
        transaction,
      },
    );

    await transaction.commit();

    return {
      status: 'promoted',
      globalClientId: created.id,
      storeClientId: sc.id,
      salesRemappedCount: remapped,
    };
  } catch (err) {
    await transaction.rollback();
    throw err;
  }
}

// 주석: 인라인 validator — Plan 09 에서 공용 validator 파일로 이관 예정.
private normalizeDocument(raw?: string | null): string | null {
  if (!raw) return null;
  const digits = raw.replace(/\D/g, '');
  return digits || null;
}

private isValidDni(digits: string): boolean {
  return /^\d{7,8}$/.test(digits);
}

private isValidCuit(digits: string): boolean {
  if (!/^\d{11}$/.test(digits)) return false;
  const weights = [5, 4, 3, 2, 7, 6, 5, 4, 3, 2];
  const ns = digits.split('').map(Number);
  let sum = 0;
  for (let i = 0; i < 10; i++) sum += ns[i] * weights[i];
  const rem = sum % 11;
  const calc = 11 - rem;
  let expected: number;
  if (calc === 11) expected = 0;
  else if (calc === 10) return false;   // AFIP rejects — RESEARCH.md Pitfall 5
  else expected = calc;

  return expected === ns[10];
}
```

```typescript
import { Test } from '@nestjs/testing';
import { getModelToken, getConnectionToken } from '@nestjs/sequelize';

describe('ClientsService.promote', () => {
  let svc: ClientsService;
  let clientsModel: any;
  let globalClientModel: any;
  let storeClientModel: any;
  let salesModel: any;
  let sequelize: any;

    globalClientModel = { findOne: jest.fn(), create: jest.fn() };

      providers: [
        ClientsService,
        { provide: getModelToken(Clients), useValue: clientsModel },
        { provide: getModelToken(GlobalClient), useValue: globalClientModel },
        { provide: getModelToken(StoreClient), useValue: storeClientModel },
        { provide: getModelToken(Sale), useValue: salesModel },
        { provide: getConnectionToken(), useValue: sequelize },
      ],
    }).compile();

    svc = moduleRef.get(ClientsService);
  });

  const ctx = { userId: 1, storeId: 3, ownerGroupId: 1 };

  it('promotes when no existing global match (happy path)', async () => {
    clientsModel.findOne.mockResolvedValue({
      id: 10, storeId: 3, fullname: 'Juan', document: '20304050', email: null, phone: '1111',
      toJSON: () => ({ id: 10 }),
    });
    globalClientModel.findOne.mockResolvedValue(null);
    globalClientModel.create.mockResolvedValue({ id: 100 });
    storeClientModel.create.mockResolvedValue({ id: 200 });
    salesModel.update.mockResolvedValue([3]);

    const res = await svc.promote(10, ctx);
    expect(res).toEqual({ status: 'promoted', globalClientId: 100, storeClientId: 200, salesRemappedCount: 3 });
    expect(tx.commit).toHaveBeenCalled();
    expect(tx.rollback).not.toHaveBeenCalled();
  });

  it('returns merge_required when conflict in same ownerGroup', async () => {
    clientsModel.findOne.mockResolvedValue({
      id: 10, storeId: 3, fullname: 'Juan A', document: '20304050', email: 'a@a.com',

  });
```
