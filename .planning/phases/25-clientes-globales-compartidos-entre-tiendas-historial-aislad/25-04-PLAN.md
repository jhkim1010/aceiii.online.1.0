```sql
-- ===========================================================================
-- Phase 25 Wave 1 Step 4: audit 테이블 3종 생성 (PG10 호환)
-- 실행: sudo -u postgres psql -d ventago -v ON_ERROR_STOP=1 -1 -f 20260424-phase25-step4-audit-tables.sql
-- 목적: D4-06 (client_imports), D1-04 (client_merges), D3-04 (client_access_audits) audit 기록.
-- 근거: RESEARCH.md Pitfall 2 — audit_logs.action ENUM 확장은 PG10 트랜잭션 외부에서만 가능.
-- ===========================================================================

BEGIN;

-- (1) client_imports — 일괄 import 한 번당 한 행
CREATE TABLE IF NOT EXISTS client_imports (
  id                SERIAL PRIMARY KEY,
  user_id           INTEGER NOT NULL,
  store_id          INTEGER NOT NULL,
  file_name         VARCHAR(255) NOT NULL,
  total_rows        INTEGER NOT NULL DEFAULT 0,
  created_count     INTEGER NOT NULL DEFAULT 0,
  updated_count     INTEGER NOT NULL DEFAULT 0,
  skipped_count     INTEGER NOT NULL DEFAULT 0,
  error_count       INTEGER NOT NULL DEFAULT 0,
  executed_at       TIMESTAMP WITH TIME ZONE NOT NULL DEFAULT NOW(),
  missing_doc_policy VARCHAR(20) NOT NULL DEFAULT 'local',  -- 'local' | 'skip' (D4-03)
  created_at        TIMESTAMP WITH TIME ZONE NOT NULL DEFAULT NOW(),
  updated_at        TIMESTAMP WITH TIME ZONE NOT NULL DEFAULT NOW()
);

DO $$
BEGIN
  ALTER TABLE client_imports
    ADD CONSTRAINT fk_client_imports_user
    FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE RESTRICT;
EXCEPTION WHEN duplicate_object THEN NULL;
END $$;

DO $$
BEGIN
  ALTER TABLE client_imports
    ADD CONSTRAINT fk_client_imports_store
    FOREIGN KEY (store_id) REFERENCES stores(id) ON DELETE RESTRICT;
EXCEPTION WHEN duplicate_object THEN NULL;
END $$;

CREATE INDEX IF NOT EXISTS idx_client_imports_store_exec
  ON client_imports (store_id, executed_at DESC);
CREATE INDEX IF NOT EXISTS idx_client_imports_user_exec
  ON client_imports (user_id, executed_at DESC);

GRANT ALL PRIVILEGES ON TABLE client_imports TO coolsistema;
GRANT USAGE, SELECT ON SEQUENCE client_imports_id_seq TO coolsistema;


-- (2) client_merges — merge 작업 1건당 한 행 (D1-04 field-by-field 선택 기록)
CREATE TABLE IF NOT EXISTS client_merges (
  id                     SERIAL PRIMARY KEY,
  user_id                INTEGER NOT NULL,
  store_id               INTEGER NOT NULL,
  winner_global_client_id INTEGER NOT NULL,   -- 살아남는 GlobalClient
  loser_global_client_id  INTEGER,            -- 승격 시 아직 없을 수 있음 (null 허용)
  local_client_id        INTEGER,             -- promote-from-local 경로일 때의 원본 legacy client
  field_picks            JSONB NOT NULL DEFAULT '{}'::jsonb,   -- { fullname: 'local', email: 'global', ... }
  merge_reason           VARCHAR(50) NOT NULL DEFAULT 'promote_conflict',  -- 'promote_conflict' | 'manual_merge'
  merged_at              TIMESTAMP WITH TIME ZONE NOT NULL DEFAULT NOW(),
  created_at             TIMESTAMP WITH TIME ZONE NOT NULL DEFAULT NOW(),
  updated_at             TIMESTAMP WITH TIME ZONE NOT NULL DEFAULT NOW()
);

DO $$
BEGIN
  ALTER TABLE client_merges
    ADD CONSTRAINT fk_client_merges_user
    FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE RESTRICT;
EXCEPTION WHEN duplicate_object THEN NULL;
END $$;

DO $$
BEGIN
  ALTER TABLE client_merges
    ADD CONSTRAINT fk_client_merges_store
    FOREIGN KEY (store_id) REFERENCES stores(id) ON DELETE RESTRICT;
EXCEPTION WHEN duplicate_object THEN NULL;
END $$;

DO $$
BEGIN
  ALTER TABLE client_merges
    ADD CONSTRAINT fk_client_merges_winner
    FOREIGN KEY (winner_global_client_id) REFERENCES global_clients(id) ON DELETE RESTRICT;
EXCEPTION WHEN duplicate_object THEN NULL;
END $$;

CREATE INDEX IF NOT EXISTS idx_client_merges_winner
  ON client_merges (winner_global_client_id);
CREATE INDEX IF NOT EXISTS idx_client_merges_merged_at
  ON client_merges (merged_at DESC);

GRANT ALL PRIVILEGES ON TABLE client_merges TO coolsistema;
GRANT USAGE, SELECT ON SEQUENCE client_merges_id_seq TO coolsistema;


-- (3) client_access_audits — cross-ownerGroup 403 시도 1건당 한 행 (D3-04)
CREATE TABLE IF NOT EXISTS client_access_audits (
  id                      SERIAL PRIMARY KEY,
  user_id                 INTEGER NOT NULL,
  caller_store_id         INTEGER NOT NULL,
  caller_owner_group_id   INTEGER NOT NULL,
  target_store_id         INTEGER,
  target_owner_group_id   INTEGER,
  target_global_client_id INTEGER,
  endpoint                TEXT NOT NULL,
  method                  VARCHAR(10) NOT NULL,
  ip_address              VARCHAR(45),  -- IPv6 safe
  user_agent              TEXT,
  denied_at               TIMESTAMP WITH TIME ZONE NOT NULL DEFAULT NOW(),
  created_at              TIMESTAMP WITH TIME ZONE NOT NULL DEFAULT NOW(),
  updated_at              TIMESTAMP WITH TIME ZONE NOT NULL DEFAULT NOW()
);

DO $$
BEGIN
  ALTER TABLE client_access_audits
    ADD CONSTRAINT fk_client_access_audits_user
    FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE RESTRICT;
EXCEPTION WHEN duplicate_object THEN NULL;
END $$;

CREATE INDEX IF NOT EXISTS idx_client_access_audits_user_denied
  ON client_access_audits (user_id, denied_at DESC);
CREATE INDEX IF NOT EXISTS idx_client_access_audits_caller_store
  ON client_access_audits (caller_store_id, denied_at DESC);

GRANT ALL PRIVILEGES ON TABLE client_access_audits TO coolsistema;
GRANT USAGE, SELECT ON SEQUENCE client_access_audits_id_seq TO coolsistema;

COMMIT;
```

```typescript
import {
  Table, Column, Model, DataType, ForeignKey,
  BelongsTo, CreatedAt, UpdatedAt, PrimaryKey, AutoIncrement,
} from 'sequelize-typescript';
import { User } from '../../users/user.model';   // 경로는 실제 파일 확인 필수
import { Store } from '../../store/store.model';

// 주석: Phase 25 D4-06 — CSV/Excel import 1건당 1행 기록.
@Table({ tableName: 'client_imports', timestamps: true })
export class ClientImport extends Model {
  @PrimaryKey
  @AutoIncrement
  @Column(DataType.INTEGER)
  id!: number;

  @ForeignKey(() => User)
  @Column({ field: 'user_id', type: DataType.INTEGER, allowNull: false })
  userId!: number;

  @ForeignKey(() => Store)
  @Column({ field: 'store_id', type: DataType.INTEGER, allowNull: false })
  storeId!: number;

  @Column({ field: 'file_name', type: DataType.STRING(255), allowNull: false })
  fileName!: string;

  @Column({ field: 'total_rows', type: DataType.INTEGER, allowNull: false, defaultValue: 0 })
  totalRows!: number;

  @Column({ field: 'created_count', type: DataType.INTEGER, allowNull: false, defaultValue: 0 })
  createdCount!: number;

  @Column({ field: 'updated_count', type: DataType.INTEGER, allowNull: false, defaultValue: 0 })
  updatedCount!: number;

  @Column({ field: 'skipped_count', type: DataType.INTEGER, allowNull: false, defaultValue: 0 })
  skippedCount!: number;

  @Column({ field: 'error_count', type: DataType.INTEGER, allowNull: false, defaultValue: 0 })
  errorCount!: number;

  @Column({ field: 'executed_at', type: DataType.DATE, allowNull: false, defaultValue: DataType.NOW })
  executedAt!: Date;

  @Column({ field: 'missing_doc_policy', type: DataType.STRING(20), allowNull: false, defaultValue: 'local' })
  missingDocPolicy!: 'local' | 'skip';

  @BelongsTo(() => User, { foreignKey: 'userId' })
  user?: User;

  @BelongsTo(() => Store, { foreignKey: 'storeId' })
  store?: Store;
}
```

```typescript
import {
  Table, Column, Model, DataType, ForeignKey,
  BelongsTo, PrimaryKey, AutoIncrement,

import { User } from '../../users/user.model';
import { Store } from '../../store/store.model';

// 주석: Phase 25 D1-04 — merge 작업 1건당 1행. fieldPicks 는 JSONB (PG).
@Table({ tableName: 'client_merges', timestamps: true })
export class ClientMerge extends Model {
  @PrimaryKey
  @AutoIncrement
  @Column(DataType.INTEGER)
  id!: number;

  @ForeignKey(() => User)
  @Column({ field: 'user_id', type: DataType.INTEGER, allowNull: false })
  userId!: number;

  @ForeignKey(() => Store)
  @Column({ field: 'store_id', type: DataType.INTEGER, allowNull: false })
  storeId!: number;

  @ForeignKey(() => GlobalClient)
  @Column({ field: 'winner_global_client_id', type: DataType.INTEGER, allowNull: false })
  winnerGlobalClientId!: number;

  @Column({ field: 'loser_global_client_id', type: DataType.INTEGER, allowNull: true })
  loserGlobalClientId?: number;

  @Column({ field: 'local_client_id', type: DataType.INTEGER, allowNull: true })
  localClientId?: number;

  @Column({ field: 'field_picks', type: DataType.JSONB, allowNull: false, defaultValue: {} })
  fieldPicks!: Record<string, 'local' | 'global'>;

  @Column({ field: 'merge_reason', type: DataType.STRING(50), allowNull: false, defaultValue: 'promote_conflict' })
  mergeReason!: 'promote_conflict' | 'manual_merge';

  @Column({ field: 'merged_at', type: DataType.DATE, allowNull: false, defaultValue: DataType.NOW })
  mergedAt!: Date;

  @BelongsTo(() => User, { foreignKey: 'userId' })
  user?: User;

  @BelongsTo(() => GlobalClient, { foreignKey: 'winnerGlobalClientId' })

}
```
