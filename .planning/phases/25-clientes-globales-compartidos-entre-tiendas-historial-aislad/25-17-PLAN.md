Standalone NestJS script pattern (api-ventago/scripts/backfill-clients-to-global.ts):
```typescript
import { NestFactory } from '@nestjs/core';
import { Sequelize } from 'sequelize-typescript';
import { Op } from 'sequelize';

import { AppModule } from '../src/app.module';
import { ClientsSyncService } from '../src/app/shared/clients-sync/clients-sync.service';
import { Clients } from '../src/app/clients/clients.model';
import { Sale } from '../src/app/sales/sales.model';

interface BackfillArgs {
  dryRun: boolean;     // --dry-run
  batchSize: number;   // --batch=100
  storeId?: number;    // --store=9 (optional filter for staged rollout)
}

async function main() {
  const args = parseArgs();
  const app = await NestFactory.createApplicationContext(AppModule, { logger: ['log', 'warn', 'error'] });
  const syncService = app.get(ClientsSyncService);
  const sequelize = app.get(Sequelize);

  const candidates = await Clients.findAll({
    where: {
      document: { [Op.ne]: null },
      ...(args.storeId ? { storeId: args.storeId } : {}),
    },
    order: [['id', 'ASC']],
  });

  const stats = { skipped_invalid: 0, skipped_temp: 0, synced_new: 0, synced_existing: 0, sales_remapped: 0, errors: [] as any[] };

  for (const legacy of candidates) {
    if (args.dryRun) {
      // dry-run: classify only, no writes
      // ...
      continue;
    }
    const t = await sequelize.transaction();
    try {
      const result = await syncService.syncFromLegacy(legacy, { ownerGroupId: null }, t);
      if (!result) {
        stats.skipped_invalid += 1;
        await t.commit();
        continue;
      }
      // sales remap
      const [n] = await Sale.update(
        { storeClientId: result.storeClientId },
        {
          where: {
            clientId: legacy.id,
            storeId: legacy.storeId,
            storeClientId: null,
          },
          transaction: t,
        },
      );
      stats.sales_remapped += n;
      result.alreadyExisted ? stats.synced_existing++ : stats.synced_new++;
      await t.commit();
    } catch (err) {
      await t.rollback();
      stats.errors.push({ clientId: legacy.id, document: legacy.document, error: String(err) });
    }
  }

  console.log(JSON.stringify({ args, stats }, null, 2));
  await app.close();
  process.exit(stats.errors.length > 0 ? 1 : 0);
}

main().catch((err) => { console.error(err); process.exit(2); });
```

Dry-run SQL (read-only stats — pre-execute 운영 사전 점검):

```sql
-- Phase 25 Plan 17 dry-run stats
WITH legacy AS (
  SELECT id, store_id, document, fullname,
    REGEXP_REPLACE(COALESCE(document, ''), '[^0-9]', '', 'g') AS doc_norm
  FROM clients
  WHERE document IS NOT NULL AND TRIM(document) <> ''
)
SELECT
  COUNT(*) FILTER (WHERE doc_norm = '') AS empty_after_norm,
  COUNT(*) FILTER (WHERE doc_norm ~ '^S[0-9]+-') AS temp_doc,
  COUNT(*) FILTER (WHERE LENGTH(doc_norm) = 7 OR LENGTH(doc_norm) = 8) AS likely_dni,
  COUNT(*) FILTER (WHERE LENGTH(doc_norm) = 11) AS likely_cuit,
  COUNT(*) FILTER (WHERE LENGTH(doc_norm) NOT IN (7, 8, 11)) AS invalid_length,
  COUNT(*) AS total
FROM legacy;
-- Plus per-store breakdown
SELECT store_id, COUNT(*) FROM legacy GROUP BY store_id ORDER BY store_id;
-- Plus already-mapped count
SELECT COUNT(*) AS already_mapped
  FROM legacy l
  JOIN global_clients gc ON gc.document = l.doc_norm
  JOIN store_clients sc ON sc.global_client_id = gc.id AND sc.store_id = l.store_id;
-- Sales remap candidate count
SELECT COUNT(*) AS sales_remap_candidates
  FROM sales s
  JOIN clients c ON c.id = s.client_id
 WHERE s.store_client_id IS NULL
   AND c.document IS NOT NULL
   AND TRIM(c.document) <> '';
```
