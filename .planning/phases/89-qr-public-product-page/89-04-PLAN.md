```typescript
// [Phase 89] QR 공개 상품 페이지가 받는 전부.
//
// ★ 여기 없는 것은 나가지 않는다. 특히:

//   OAuth 토큰을 실어 보낸 전례가 있다.

export type QrPublicMode = 'detail' | 'shop_redirect' | 'closed';

export interface QrPublicProductDto {
  id: number;
  name: string;
  /** MinIO 오브젝트 키 또는 절대 URL. 없으면 null — 사진 없음은 예외가 아니라 주 경로다
   *  (운영 실측: store 9 부모상품 15개 중 사진 0개, store 6 은 20개 중 7개). */
  imageUrl: string | null;
  price: number;
  /** 인쇄 당시 가격유형의 이름. fallback 이면 null. */
  priceLabel: string | null;
  /** 'exact' = 인쇄 기록을 찾아 그 가격유형의 현재 가격을 보여준다(라벨과 일치).

   *  화면이 이 사실을 숨기면 안 된다(D-01 전제 4). */
  priceSource: 'exact' | 'fallback';
}

export interface QrPublicDto {
  mode: QrPublicMode;
  /** closed 에서는 null — 닫힘은 «아무것도 안 보인다» 이다(D-04). */
  storeName: string | null;
  /** MinIO 오브젝트 키. 프론트가 {API}/minio/{key} 로 조립한다. */
  logoUrl: string | null;
  /** ?b= 로 받은 지점이 **이 매장 것일 때만** 이름을 준다. 없으면 null. */
  branchName: string | null;
  /** shop_redirect 에서만 채워진다. 완성된 절대 URL. */
  shopUrl: string | null;
  product: QrPublicProductDto | null;
}
```

**a) 상품 + 매장 단건 조회 — 테넌트 강제**
```sql
SELECT p.id            AS product_id,
       p.name          AS product_name,
       p.price         AS base_price,
       p.image_url     AS image_url,
       p.image_urls    AS image_urls,
       s.id            AS store_id,
       COALESCE(NULLIF(s.alias_name, ''), s.name) AS store_name,
       s.logo_url      AS logo_url,
       s.slug          AS slug,
       COALESCE(c.qr_precio_publico, false)       AS qr_on
  FROM products p
  JOIN stores s          ON s.id = p.store_id
  LEFT JOIN store_configs c ON c.store_id = s.id
 WHERE p.id = $1
   AND p.store_id = $2
   AND COALESCE(p.is_active, TRUE) = TRUE
   AND COALESCE(s.is_active, TRUE) = TRUE
 LIMIT 1
```
- bind `[productId, storeId]`.
- 0행이면 `throw new NotFoundException('No encontrado')`.

    그 차이 자체가 «그 매장에 이 id 가 있다» 는 정보다.
