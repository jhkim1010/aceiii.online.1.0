   SELECT p.id AS product_id, p.store_id AS owner,

   SELECT count(*) FROM qr_print_log l
     JOIN products p ON p.id = l.product_id
     JOIN branches b ON b.id = l.branch_id
     LEFT JOIN price_types pt ON pt.id = l.price_type_id
