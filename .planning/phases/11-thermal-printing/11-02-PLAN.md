## Step 2: `preload.js` — contextBridge IPC (0.2일)

renderer ↔ main process 사이의 보안 브릿지.
renderer는 Node.js API에 직접 접근 불가 — preload를 통해서만 통신.

```javascript
// print-agent/preload.js
const { contextBridge, ipcRenderer } = require('electron');

contextBridge.exposeInMainWorld('printAgent', {
  // 설정 관련
  getConfig:    ()       => ipcRenderer.invoke('config:get'),
  saveConfig:   (cfg)    => ipcRenderer.invoke('config:save', cfg),

  // 프린터 탐색
  discoverPrinters: ()   => ipcRenderer.invoke('printer:discover'),
  testPrinter:  (cfg)    => ipcRenderer.invoke('printer:test', cfg),

  // 연결 상태
  getStatus:    ()       => ipcRenderer.invoke('status:get'),
  reconnect:    ()       => ipcRenderer.invoke('ws:reconnect'),

  // 이벤트 수신 (main → renderer)
  onLog:        (cb)     => ipcRenderer.on('log', (_e, msg) => cb(msg)),
  onStatus:     (cb)     => ipcRenderer.on('status', (_e, st) => cb(st)),
  onPrinterList:(cb)     => ipcRenderer.on('printerList', (_e, list) => cb(list)),
});
```

## Step 3: `main.js` 스켈레톤 (0.5일)

Wave 2에서는 WebSocket 실 연결 없이 구조만 잡는다.
실제 WebSocket 루프는 Wave 3에서 구현.

```javascript
// print-agent/main.js
'use strict';

const { app, BrowserWindow, Tray, Menu, ipcMain, nativeImage } = require('electron');
const path   = require('path');
const Store  = require('electron-store');

// ── electron-store 스키마 정의 ─────────────────────────────────────────────
const store = new Store({
  schema: {
    serverUrl:  { type: 'string', default: '' },
    apiKey:     { type: 'string', default: '' },
    branchId:   { type: 'number', default: 0 },
    printer:    {
      type: 'object',
      properties: {
        type:      { type: 'string', enum: ['usb', 'network'] },
        vendorId:  { type: 'string' },
        productId: { type: 'string' },
        host:      { type: 'string' },
        port:      { type: 'number', default: 9100 },
      },
    },
    setupDone:  { type: 'boolean', default: false },
  },
});

let mainWin  = null;
let tray     = null;
let wsClient = null;  // Wave 3에서 초기화

// ── 앱 준비 ───────────────────────────────────────────────────────────────
app.whenReady().then(() => {
  createTray();

  if (!store.get('setupDone')) {
    openSetupWizard();
  } else {
    openMainWindow();
    // Wave 3: initWebSocket(store.get('serverUrl'), store.get('apiKey'));
  }
});

// ── 트레이 ────────────────────────────────────────────────────────────────
function createTray() {
  const icon = nativeImage.createFromPath(
    path.join(__dirname, 'renderer/assets/logo.png')
  );
  tray = new Tray(icon.resize({ width: 16, height: 16 }));

  const menu = Menu.buildFromTemplate([
    { label: 'Configuración',  click: openMainWindow },
    { type: 'separator' },
    { label: 'Salir',          click: () => app.quit() },
  ]);

  tray.setContextMenu(menu);
  tray.setToolTip('VentaGO Print Agent');
  tray.on('double-click', openMainWindow);
}

// ── 메인 설정 창 ──────────────────────────────────────────────────────────
function openMainWindow() {
  if (mainWin) { mainWin.focus(); return; }

  mainWin = new BrowserWindow({
    width: 680, height: 520,
    resizable: false,
    webPreferences: {
      preload: path.join(__dirname, 'preload.js'),
      contextIsolation: true,
      nodeIntegration: false,
    },
    title: 'VentaGO Print Agent',
    show: false,
  });

  mainWin.loadFile('renderer/index.html');
  mainWin.once('ready-to-show', () => mainWin.show());
  mainWin.on('closed', () => { mainWin = null; });
}

// ── 셋업 마법사 창 ─────────────────────────────────────────────────────────
function openSetupWizard() {
  const wizWin = new BrowserWindow({
    width: 560, height: 480,
    resizable: false,
    webPreferences: {
      preload: path.join(__dirname, 'preload.js'),
      contextIsolation: true,
      nodeIntegration: false,
    },
    title: 'VentaGO — Configuración Inicial',
    show: false,
  });

  wizWin.loadFile('renderer/setup-wizard.html');
  wizWin.once('ready-to-show', () => wizWin.show());

  // 마법사 완료 시 메인 창으로 전환
  ipcMain.once('wizard:done', () => {
    wizWin.close();
    openMainWindow();
    // Wave 3: initWebSocket(...)
  });
}

// ── IPC 핸들러 ────────────────────────────────────────────────────────────
ipcMain.handle('config:get',  ()      => store.store);
ipcMain.handle('config:save', (_, cfg) => {
  Object.assign(store.store, cfg);

  return { ok: true };
});

ipcMain.handle('status:get',  () => ({
  connected: wsClient?.connected ?? false,
  serverUrl: store.get('serverUrl'),
  branchId:  store.get('branchId'),
}));

ipcMain.handle('printer:discover', async () => {
  // Wave 3: const { discoverPrinters } = require('./src/printer-discovery');
  // return discoverPrinters();
  return [];  // Wave 2 stub
});

ipcMain.handle('printer:test', async (_, cfg) => {
  const { testConnection } = require('./src/printer');

  return testConnection(cfg);
});

ipcMain.handle('ws:reconnect', async () => {
  // Wave 3: wsClient.disconnect(); initWebSocket(...);
  return { ok: true };
});

// 모든 창이 닫혀도 앱 유지 (트레이에서 실행 중)
app.on('window-all-closed', (e) => e.preventDefault());
```
