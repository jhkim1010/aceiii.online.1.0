## Step 1: `src/printer-discovery.js` — 프린터 자동 탐색 (0.3일)

```javascript
// print-agent/src/printer-discovery.js
'use strict';

const net = require('net');
const os  = require('os');

// 선택적 USB 지원 (libusb 없는 환경도 graceful skip)
let USB = null;
try { USB = require('escpos-usb'); } catch (_e) {}

/**
 * USB + 네트워크 프린터 탐색
 * @returns {Promise<Array<{ type, label, config }>>}
 */
async function discoverPrinters() {
  const results = [];

  // ── USB 탐색 ─────────────────────────────────────────────────────────────
  if (USB) {
    try {
      const devices = USB.findPrinter();

      devices.forEach((d) => {
        const vid = d.deviceDescriptor.idVendor.toString(16).toUpperCase().padStart(4, '0');
        const pid = d.deviceDescriptor.idProduct.toString(16).toUpperCase().padStart(4, '0');

        results.push({
          type:   'usb',
          label:  `USB — VID:${vid} PID:${pid}`,
          config: {
            type:      'usb',
            vendorId:  '0x' + vid,
            productId: '0x' + pid,
          },
        });
      });
    } catch (_e) {
      // libusb 없는 환경에서 graceful skip
    }
  }

  // ── 네트워크 탐색: 로컬 서브넷 포트 9100 스캔 ─────────────────────────────
  const subnet = detectLocalSubnet();

  if (subnet) {
    const hosts = await scanSubnet(subnet, 9100, 300); // timeout 300ms/host

    hosts.forEach((host) => {
      results.push({
        type:   'network',
        label:  `Red — ${host}:9100`,
        config: { type: 'network', host, port: 9100 },
      });
    });
  }

  return results;
}

/**
 * 로컬 IPv4 서브넷 감지 (예: '192.168.1')
 */
function detectLocalSubnet() {
  const ifaces = os.networkInterfaces();

  for (const iface of Object.values(ifaces)) {
    for (const addr of iface) {
      if (addr.family === 'IPv4' && !addr.internal) {
        const parts = addr.address.split('.');

        if (parts[0] !== '169') { // APIPA 제외
          return parts.slice(0, 3).join('.');
        }
      }
    }
  }

  return null;
}

/**
 * 서브넷 내 특정 포트 스캔
 * @param {string} subnet   '192.168.1'
 * @param {number} port     9100
 * @param {number} timeout  ms per host
 * @returns {Promise<string[]>} 응답한 IP 목록
 */
async function scanSubnet(subnet, port, timeout = 300) {
  const probes = [];

  for (let i = 1; i <= 254; i++) {
    const host = `${subnet}.${i}`;

    probes.push(
      new Promise((resolve) => {
        const sock = new net.Socket();
        let done = false;

        const finish = (ok) => {
          if (done) return;
          done = true;
          sock.destroy();
          resolve(ok ? host : null);
        };

        sock.setTimeout(timeout);
        sock.connect(port, host, () => finish(true));
        sock.on('error',   () => finish(false));
        sock.on('timeout', () => finish(false));
      })
    );
  }

  const results = await Promise.all(probes);

  return results.filter(Boolean);
}

module.exports = { discoverPrinters };
```

## Step 2: `src/fiscal-formatter.js` — AFIP 영수증 HTML 포맷터 (0.4일)

`formatInvoiceHtml()`를 기반으로 AFIP 필드를 추가한다.

```javascript
// print-agent/src/fiscal-formatter.js
'use strict';

const { formatInvoiceHtml } = require('./formatter');

/**
 * AFIP Comprobante Fiscal HTML 생성
 *
 * @param {object} data  컨트롤 티켓 데이터 + AFIP 추가 필드
 *   data.afip {
 *     tipo:       'Factura B'  // InvoiceType 문자열
 *     puntoVenta: '00005'
 *     numero:     '00000042'
 *     cae:        '12345678901234'
 *     vtoCae:     '2026-04-16'   // YYYYMMDD 또는 ISO
 *     qrUrl:      'https://www.afip.gob.ar/fe/qr/?p=...'
 *   }
 */
const formatFiscalHtml = (data) => {
  // 기본 컨트롤 티켓 HTML 생성 후 AFIP 블록 삽입
  const baseHtml = formatInvoiceHtml({
    ...data,
    // 배너 변경: "DOCUMENTO NO VÁLIDO" → 세금계산서 타입
    _fiscalBanner: data.afip?.tipo || 'Comprobante Fiscal',
  });

  // AFIP 블록 HTML
  const afip = data.afip || {};
  const vtoCaeStr = afip.vtoCae
    ? new Date(afip.vtoCae).toLocaleDateString('es-AR', {
        day: '2-digit', month: '2-digit', year: 'numeric',
      })
    : '—';

  const afipBlock = `
    <div class="afip-section">
      <div class="afip-title">COMPROBANTE ELECTRÓNICO AFIP</div>
      <table class="afip-table">
        <tr>
          <td class="afip-label">Tipo</td>
          <td class="afip-val">${afip.tipo || '—'}</td>
        </tr>
        <tr>
          <td class="afip-label">Pto. Venta</td>
          <td class="afip-val">${afip.puntoVenta || '—'}</td>
        </tr>
        <tr>
          <td class="afip-label">Número</td>
          <td class="afip-val">${afip.numero || '—'}</td>
        </tr>
        <tr>
          <td class="afip-label">CAE</td>
          <td class="afip-val afip-cae">${afip.cae || '—'}</td>
        </tr>
        <tr>
          <td class="afip-label">Vto. CAE</td>
          <td class="afip-val">${vtoCaeStr}</td>
        </tr>
      </table>
      ${afip.qrUrl ? `
      <div class="afip-qr-url">
        <div class="afip-qr-label">Verificar en AFIP:</div>
        <div class="afip-qr-text">${afip.qrUrl}</div>
      </div>` : ''}
    </div>`;

  const afipCss = `
    <style>
      .afip-section {
        border-top: 3px solid #1a1a1a;
        padding: 10px 14px 8px;
        margin-top: 4px;
      }
      .afip-title {
        font-size: 14px;
        font-weight: bold;
        letter-spacing: 1px;
        text-transform: uppercase;
        color: #888;
        text-align: center;
        margin-bottom: 6px;
      }
      table.afip-table { width: 100%; border-collapse: collapse; font-size: 18px; }
      table.afip-table td { padding: 3px 0; }
      .afip-label { color: #666; width: 100px; }
      .afip-val   { font-weight: bold; }
      .afip-cae   { font-size: 16px; letter-spacing: 1px; }
      .afip-qr-url {
        margin-top: 8px;
        padding: 6px 0;
        border-top: 1px dashed #ccc;
      }
      .afip-qr-label { font-size: 14px; color: #888; }
      .afip-qr-text  {
        font-size: 14px;
        word-break: break-all;
        color: #333;
        margin-top: 2px;
      }
    </style>`;

  // baseHtml의 </body> 직전에 AFIP 블록 + CSS 삽입
  return baseHtml
    .replace('</body>', `${afipCss}${afipBlock}</body>`)
    .replace(
      'DOCUMENTO NO VÁLIDO COMO FACTURA',
      (afip.tipo || 'Comprobante Fiscal').toUpperCase()
    );
};

module.exports = { formatFiscalHtml };
```

## Step 3: `main.js` WebSocket 루프 구현 (0.5일)

Wave 2의 stub을 실제 동작으로 교체.

```javascript
// main.js에 추가할 함수들

const { io }            = require('socket.io-client');
const { printTicket }   = require('./src/print-pipeline');
const { formatFiscalHtml } = require('./src/fiscal-formatter');
const { renderHtmlToPng }  = require('./src/renderer-engine');
const { printImage }       = require('./src/printer');

let reconnectTimer = null;

/**
 * WebSocket 연결 초기화 및 이벤트 루프
 */
function initWebSocket(serverUrl, apiKey) {
  if (wsClient) wsClient.disconnect();

  wsClient = io(serverUrl, {
    auth:              { token: apiKey },
    reconnection:      true,
    reconnectionDelay: 3000,
    timeout:           10000,
  });

  // ── 연결 이벤트 ──────────────────────────────────────────────────────────
  wsClient.on('connect', () => {
    broadcastStatus({ connected: true });
    broadcastLog('✅ Conectado al servidor');

    wsClient.emit('agent_online', {
      branchId: store.get('branchId'),
      version:  app.getVersion(),
      ts:       Date.now(),
    });
  });

  wsClient.on('disconnect', (reason) => {
    broadcastStatus({ connected: false });
    broadcastLog(`⚠️ Desconectado: ${reason}`);
  });

  // ── 컨트롤 티켓 출력 ─────────────────────────────────────────────────────
  wsClient.on('print_invoice', async (payload) => {
    const printerCfg = store.get('printer');
    const start      = Date.now();

    broadcastLog(`🖨 print_invoice #${payload.invoice?.number} — imprimiendo...`);

    try {
      await printTicket(payload, printerCfg);

      const elapsed = Date.now() - start;

      broadcastLog(`✅ print_invoice #${payload.invoice?.number} — OK (${elapsed}ms)`);
      wsClient.emit('print_ack', {
        invoiceId: payload.invoiceId,
        status:    'ok',
        ts:        Date.now(),
      });
    } catch (err) {
      broadcastLog(`❌ print_invoice #${payload.invoice?.number} — ${err.message}`);
      wsClient.emit('print_ack', {
        invoiceId: payload.invoiceId,
        status:    'error',
        error:     err.message,
        ts:        Date.now(),
      });
    }
  });

  // ── AFIP 영수증 출력 ──────────────────────────────────────────────────────
  wsClient.on('print_fiscal', async (payload) => {
    const printerCfg = store.get('printer');
    const start      = Date.now();

    broadcastLog(`🖨 print_fiscal CAE:${payload.afip?.cae?.slice(-6)} — imprimiendo...`);

    try {
      const html = formatFiscalHtml(payload);
      const png  = await renderHtmlToPng(html, 576);

      await printImage(png, printerCfg);

      const elapsed = Date.now() - start;

      broadcastLog(`✅ print_fiscal — OK (${elapsed}ms)`);
      wsClient.emit('print_ack', {
        invoiceId: payload.invoiceId,
        status:    'ok',
        ts:        Date.now(),
      });
    } catch (err) {
      broadcastLog(`❌ print_fiscal — ${err.message}`);
      wsClient.emit('print_ack', {
        invoiceId: payload.invoiceId,
        status:    'error',
        error:     err.message,
        ts:        Date.now(),
      });
    }
  });
}

// ── 유틸 ─────────────────────────────────────────────────────────────────
function broadcastStatus(st) {
  if (mainWin) mainWin.webContents.send('status', st);
  tray?.setToolTip(st.connected
    ? 'VentaGO Print Agent — Conectado'
    : 'VentaGO Print Agent — Desconectado'
  );
}

function broadcastLog(msg) {
  const line = `${new Date().toLocaleTimeString('es-AR')}  ${msg}`;

  if (mainWin) mainWin.webContents.send('log', line);
  console.log(line);
}
```
