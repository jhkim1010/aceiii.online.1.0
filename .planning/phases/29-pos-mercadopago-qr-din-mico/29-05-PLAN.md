       import { Injectable, Logger } from '@nestjs/common';
       import { InjectModel, InjectConnection } from '@nestjs/sequelize';
       import { QueryTypes, Transaction } from 'sequelize';
       import { Sequelize } from 'sequelize-typescript';

       import { MpWallet } from '../models/mp-wallet.model';
       import { MpMovement } from '../models/mp-movement.model';
       import { MpPaymentIntent } from '../models/mp-payment-intent.model';

       interface MpPaymentMinimal {
         id: string | number;
         transaction_amount: number;
       }

       @Injectable()
       export class MpWalletService {
         private readonly logger = new Logger(MpWalletService.name);

         constructor(
           @InjectModel(MpWallet) private readonly walletModel: typeof MpWallet,
           @InjectModel(MpMovement) private readonly movementModel: typeof MpMovement,
           @InjectConnection() private readonly sequelize: Sequelize,
         ) {}

         // 결제 webhook 트랜잭션 안에서 호출 — 같은 TX 객체 t 를 받아 모든 변경을 원자화
         async creditOnSale(intent: MpPaymentIntent, mpPayment: MpPaymentMinimal, t: Transaction): Promise<void> {
           const wallet = await this.walletModel.findOne({
             where: { mpAccountId: intent.mpAccountId },
             transaction: t,
             lock: t.LOCK.UPDATE, // wallet row lock — concurrent webhooks can't double-credit
           });

           if (!wallet) {
             this.logger.error(`mp_wallets row missing for mp_account ${intent.mpAccountId} — credit skipped`);

             return;
           }
           const amount = Number(intent.amount);

           await this.movementModel.create(
             {
               mpWalletId: wallet.id,
               type: 'credit',
               amount,
               saleId: null, // sale은 frontend가 생성 — 나중에 back-fill 가능
               mpPaymentId: String(mpPayment.id),
               note: `MP QR cobro intent=${intent.id}`,
             },
             { transaction: t },
           );
           await wallet.update(
             { balance: Number(wallet.balance) + amount, lastSyncedAt: new Date() },
             { transaction: t },
           );
         }

         async debitOnRefund(intent: MpPaymentIntent, amount: number, t: Transaction): Promise<void> {
           const wallet = await this.walletModel.findOne({
             where: { mpAccountId: intent.mpAccountId },
             transaction: t,
             lock: t.LOCK.UPDATE,
           });

           if (!wallet) {
             this.logger.error(`mp_wallets row missing for mp_account ${intent.mpAccountId} — debit skipped`);

             return;
           }
           await this.movementModel.create(
             {
               mpWalletId: wallet.id,
               type: 'refund_debit',
               amount,
               saleId: null,
               mpPaymentId: intent.paymentId,
               note: `MP refund intent=${intent.id}`,
             },
             { transaction: t },
           );
           await wallet.update(
             { balance: Number(wallet.balance) - amount, lastSyncedAt: new Date() },
             { transaction: t },
           );
         }

         async computeBalance(walletId: number): Promise<number> {
           try {
             // snake_case raw SQL — CLAUDE.md rule
             const [row] = await this.sequelize.query<{ balance: string }>(
               `
               SELECT
                 COALESCE(SUM(CASE WHEN type = 'credit' THEN amount END), 0)
                 - COALESCE(SUM(CASE WHEN type IN ('debit','transfer_out','refund_debit') THEN amount END), 0)
                 AS balance
               FROM mp_movements
               WHERE mp_wallet_id = :walletId
               `,
               { replacements: { walletId }, type: QueryTypes.SELECT },
             );

             return Number(row?.balance ?? 0);
           } catch (err) {
             this.logger.error(`computeBalance failed for wallet ${walletId}: ${(err as Error).message}`);

             return 0;
           }
         }
       }

       import { Injectable, Logger } from '@nestjs/common';
       import { InjectModel, InjectConnection } from '@nestjs/sequelize';
       import { Sequelize } from 'sequelize-typescript';

       import { MpPaymentIntent } from '../models/mp-payment-intent.model';
       import { MpAccount } from '../models/mp-account.model';
       import { MpApiClientService } from '../api-client/mp-api-client.service';
       import { MpTokenCryptoService } from '../crypto/mp-token-crypto.service';
       import { MpWalletService } from '../wallet/mp-wallet.service';
       import { MpWebhookDto } from '../dto/mp-webhook.dto';
       import { WebsocketService } from '../../../common/socket/websocket.service';

       interface MpPaymentResponse {
         id: string | number;
         status: string;
         status_detail?: string;
         transaction_amount: number;
         external_reference?: string;
         date_approved?: string;
         collector_id?: number;
       }

       @Injectable()
       export class MpWebhookService {
         private readonly logger = new Logger(MpWebhookService.name);

         constructor(
           @InjectModel(MpPaymentIntent) private readonly intentModel: typeof MpPaymentIntent,
           @InjectModel(MpAccount) private readonly mpAccountModel: typeof MpAccount,
           @InjectConnection() private readonly sequelize: Sequelize,
           private readonly mpApi: MpApiClientService,
           private readonly crypto: MpTokenCryptoService,
           private readonly wallets: MpWalletService,
           private readonly websocket: WebsocketService,
         ) {}

         async handleNotification(
           body: MpWebhookDto,
           _signature: string | undefined,
           requestId: string | undefined,
           accountIdFromQuery?: number,
         ): Promise<void> {
           // QR 결제만 처리 — type=payment 외에는 무시 (RESEARCH §Open Q3 — merchant_order는 중복)
           if (body.type !== 'payment') {
             this.logger.debug(`Ignoring webhook type=${body.type} (req=${requestId})`);

             return;
           }
           const mpPaymentId = body.data?.id ? String(body.data.id) : undefined;

           if (!mpPaymentId) {
             this.logger.warn(`Webhook missing data.id (req=${requestId})`);

             return;
           }

           // 1) account 결정 — query param 우선, 없으면 re-fetch 후 collector_id 로 검색 (Assumption A9)
           let account: MpAccount | null = null;

           if (accountIdFromQuery) {
             account = await this.mpAccountModel.findByPk(accountIdFromQuery);
           }

           // 2) MP /v1/payments/{id} re-fetch (canonical truth — webhook signature 없음)
           // account 가 없으면 모든 active mp_account 토큰을 시도하지 않고, 우선 fallback로 user_id 매칭
           if (!account && body.user_id) {
             account = await this.mpAccountModel.findOne({
               where: { mpUserId: String(body.user_id), disconnectedAt: null },
             });
           }
           if (!account) {
             this.logger.warn(`Cannot resolve mp_account for webhook payment ${mpPaymentId} (req=${requestId})`);

             return;
           }

           let payment: MpPaymentResponse;

           try {
             const accessToken = this.crypto.decrypt(account.accessToken);

             payment = await this.mpApi.get<MpPaymentResponse>(
               `/v1/payments/${mpPaymentId}`,
               accessToken,
               { baseUrl: this.mpApi.hostFor(account.environment) },
             );
           } catch (err: any) {
             this.logger.error(`MP re-fetch failed for payment ${mpPaymentId}: ${err?.message}`);

             return;
           }

           if (payment.status !== 'approved') {
             this.logger.log(`Payment ${mpPaymentId} status=${payment.status} — skipping`);

             return;
           }

           const intentId = Number(payment.external_reference);

           if (!intentId || Number.isNaN(intentId)) {
             this.logger.warn(`Payment ${mpPaymentId} missing or invalid external_reference (req=${requestId})`);

             return;
           }

           // 3) 작은 critical section — SELECT FOR UPDATE intent + check + UPDATE + wallet credit
           let processedIntent: MpPaymentIntent | null = null;

           await this.sequelize.transaction(async (t) => {
             const intent = await this.intentModel.findByPk(intentId, {
               lock: t.LOCK.UPDATE, // 명시적 SELECT FOR UPDATE (RESEARCH §Pitfall 9)
               transaction: t,
             });

             if (!intent) {
               this.logger.warn(`Intent ${intentId} not found for payment ${mpPaymentId}`);

               return;
             }
             if (intent.paymentId) {
               // 이미 처리됨 — webhook 또는 polling 두 번째 도착 (멱등 — Pitfall 5)
               this.logger.log(`Intent ${intentId} already has payment_id, skip duplicate webhook`);

               return;
             }
             // mp_account 일치 검증 (cross-store 차단)
             if (intent.mpAccountId !== account!.id) {
               this.logger.warn(
                 `Intent ${intentId} mp_account_id=${intent.mpAccountId} but webhook mp_account=${account!.id} — rejecting`,
               );

               return;
             }
             await intent.update(
               {
                 paymentId: mpPaymentId,
                 status: 'approved',
                 approvedAt: payment.date_approved ? new Date(payment.date_approved) : new Date(),
               },
               { transaction: t },
             );
             await this.wallets.creditOnSale(intent, payment, t);
             processedIntent = intent;
           });

           // 4) Socket.io emit — 트랜잭션 밖 (실패해도 polling이 회복)
           if (processedIntent) {
             try {
               this.websocket.emitToTerminal(processedIntent.terminalId, 'mercadopago:approved', {
                 intentId: processedIntent.id,
                 paymentId: mpPaymentId,
                 amount: Number(processedIntent.amount),
                 capturedAt: payment.date_approved,
                 pendingVentaId: processedIntent.pendingVentaId,
               });
             } catch (err: any) {
               this.logger.error(`emitToTerminal failed for intent ${processedIntent.id}: ${err?.message}`);
             }
           }
         }
       }
