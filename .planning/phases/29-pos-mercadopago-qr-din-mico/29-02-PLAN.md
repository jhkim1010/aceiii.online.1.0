    ```sql
    -- =============================================================================
    -- Phase 29 Wave 1 — Step 1: mp_accounts 테이블 (OAuth 토큰 + scope)
    -- =============================================================================
    -- 목적: 매장/지점 단위 MP OAuth 계정 저장 (access_token/refresh_token AES-GCM 암호화)
    -- 실행 순서: 가장 먼저 실행 (mp_payment_intents/mp_wallets/mp_refunds 모두 이 테이블에 의존)
    -- Idempotency: CREATE TABLE IF NOT EXISTS + DO 블록으로 CHECK/INDEX 보호
    -- PG10/15 호환:
    --   SERIAL 사용 (NOT GENERATED AS IDENTITY)
    --   부분 UNIQUE 인덱스 2개로 (store_id, branch_id|NULL) 격리 (COALESCE-in-UNIQUE 회피)
    --   gen_random_uuid 미사용 (PG10 호환)
    -- =============================================================================

    BEGIN;

    CREATE TABLE IF NOT EXISTS mp_accounts (
      id              SERIAL PRIMARY KEY,
      store_id        INTEGER NOT NULL REFERENCES stores(id) ON DELETE CASCADE,
      branch_id       INTEGER NULL REFERENCES branches(id) ON DELETE CASCADE,
      mp_user_id      VARCHAR(64) NOT NULL,
      access_token    TEXT NOT NULL,                                    -- AES-256-GCM (iv_b64:tag_b64:ct_b64)
      refresh_token   TEXT NOT NULL,                                    -- AES-256-GCM
      public_key      VARCHAR(255) NULL,
      environment     VARCHAR(16) NOT NULL DEFAULT 'sandbox',
      external_pos_id VARCHAR(60) NULL,                                 -- RESEARCH §Pitfall 3
      expires_at      TIMESTAMPTZ NULL,
      connected_at    TIMESTAMPTZ NOT NULL DEFAULT NOW(),
      disconnected_at TIMESTAMPTZ NULL,
      created_at      TIMESTAMPTZ NOT NULL DEFAULT NOW(),
      updated_at      TIMESTAMPTZ NOT NULL DEFAULT NOW()
    );

    -- environment ENUM 가드 (CHECK constraint)
    DO $$
    BEGIN
      IF NOT EXISTS (
        SELECT 1 FROM pg_constraint
         WHERE conname = 'chk_mp_accounts_environment' AND conrelid = 'mp_accounts'::regclass
      ) THEN
        ALTER TABLE mp_accounts
          ADD CONSTRAINT chk_mp_accounts_environment
          CHECK (environment IN ('sandbox', 'production'));
      END IF;
    END$$;

    -- 부분 UNIQUE 인덱스 1: store-level (branch_id IS NULL) — 매장당 1개만
    CREATE UNIQUE INDEX IF NOT EXISTS uniq_mp_accounts_store_only
      ON mp_accounts (store_id)
      WHERE branch_id IS NULL AND disconnected_at IS NULL;

    -- 부분 UNIQUE 인덱스 2: branch-level (branch_id IS NOT NULL) — 지점당 1개만
    CREATE UNIQUE INDEX IF NOT EXISTS uniq_mp_accounts_store_branch
      ON mp_accounts (store_id, branch_id)
      WHERE branch_id IS NOT NULL AND disconnected_at IS NULL;

    -- 조회 성능
    CREATE INDEX IF NOT EXISTS idx_mp_accounts_store
      ON mp_accounts (store_id, environment)
      WHERE disconnected_at IS NULL;

    COMMIT;
    ```

    ```sql
    -- =============================================================================
    -- Phase 29 Wave 1 — Step 2: mp_payment_intents (QR + 결제 추적, 멱등성 핵심)
    -- =============================================================================
    -- 목적: QR 발급 시점부터 결제 완료/취소/만료까지 결제 의도 라이프사이클 추적
    -- 실행 순서: step 1 (mp_accounts) 이후
    -- 멱등성 핵심: payment_id UNIQUE — webhook 중복 도착 시 두 번째는 INSERT 충돌 → 무시
    -- =============================================================================

    BEGIN;

    CREATE TABLE IF NOT EXISTS mp_payment_intents (
      id               SERIAL PRIMARY KEY,
      mp_account_id    INTEGER NOT NULL REFERENCES mp_accounts(id) ON DELETE CASCADE,
      store_id         INTEGER NOT NULL REFERENCES stores(id) ON DELETE CASCADE,
      branch_id        INTEGER NULL REFERENCES branches(id) ON DELETE SET NULL,
      terminal_id      INTEGER NOT NULL REFERENCES terminals(id) ON DELETE CASCADE,
      pending_venta_id INTEGER NOT NULL,
      amount           NUMERIC(14, 2) NOT NULL,
      mp_order_id      VARCHAR(64) NULL,
      qr_data          TEXT NULL,
      payment_id       VARCHAR(32) NULL,
      status           VARCHAR(20) NOT NULL DEFAULT 'pending',
      expires_at       TIMESTAMPTZ NOT NULL,
      approved_at      TIMESTAMPTZ NULL,
      created_at       TIMESTAMPTZ NOT NULL DEFAULT NOW(),
      updated_at       TIMESTAMPTZ NOT NULL DEFAULT NOW()
    );

    -- payment_id UNIQUE — 멱등성 핵심 (webhook 중복 차단)
    CREATE UNIQUE INDEX IF NOT EXISTS uniq_mp_intents_payment_id
      ON mp_payment_intents (payment_id)
      WHERE payment_id IS NOT NULL;

    -- status enum 가드
    DO $$
    BEGIN
      IF NOT EXISTS (
        SELECT 1 FROM pg_constraint
         WHERE conname = 'chk_mp_intents_status' AND conrelid = 'mp_payment_intents'::regclass
      ) THEN
        ALTER TABLE mp_payment_intents
          ADD CONSTRAINT chk_mp_intents_status
          CHECK (status IN ('pending', 'approved', 'cancelled', 'expired', 'failed'));
      END IF;
    END$$;

    -- expired/pending 조회 (cron + polling)
    CREATE INDEX IF NOT EXISTS idx_mp_intents_status_expires
      ON mp_payment_intents (status, expires_at)
      WHERE status = 'pending';

    -- terminal 별 조회
    CREATE INDEX IF NOT EXISTS idx_mp_intents_terminal
      ON mp_payment_intents (terminal_id, status);

    -- pending_venta_id 역참조
    CREATE INDEX IF NOT EXISTS idx_mp_intents_pending_venta
      ON mp_payment_intents (pending_venta_id);

    COMMIT;
    ```

       ```typescript
       import { Injectable } from '@nestjs/common';
       import { createCipheriv, createDecipheriv, randomBytes } from 'crypto';

       const ALGO = 'aes-256-gcm';
       const IV_LEN = 12;     // 96-bit, GCM standard (NEVER reuse with same key)
       const TAG_LEN = 16;
       const KEY_HEX_LEN = 64; // 32 bytes hex

       @Injectable()
       export class MpTokenCryptoService {
         private readonly key: Buffer;

         constructor() {
           const hex = process.env.MP_TOKEN_ENCRYPTION_KEY;

           if (!hex || hex.length !== KEY_HEX_LEN) {
             throw new Error(
               'MP_TOKEN_ENCRYPTION_KEY must be 32 bytes hex (64 hex chars)',
             );
           }
           this.key = Buffer.from(hex, 'hex');

           if (this.key.length !== 32) {
             throw new Error(
               'MP_TOKEN_ENCRYPTION_KEY decoded length must be 32 bytes (invalid hex)',
             );
           }
         }

         // 평문 → "iv_b64:tag_b64:ct_b64"
         encrypt(plaintext: string): string {
           // 96-bit IV — 호출마다 신규 생성 (GCM 보안의 핵심)
           const iv = randomBytes(IV_LEN);
           const cipher = createCipheriv(ALGO, this.key, iv);
           const ct = Buffer.concat([cipher.update(plaintext, 'utf8'), cipher.final()]);
           const tag = cipher.getAuthTag();

           return [
             iv.toString('base64'),
             tag.toString('base64'),
             ct.toString('base64'),
           ].join(':');
         }

         // "iv_b64:tag_b64:ct_b64" → 평문
         decrypt(encoded: string): string {
           const parts = encoded.split(':');

           if (parts.length !== 3) {
             throw new Error('Invalid encrypted token format');
           }
           const [ivB64, tagB64, ctB64] = parts;

           if (!ivB64 || !tagB64 || !ctB64) {
             throw new Error('Invalid encrypted token format');
           }
           const iv = Buffer.from(ivB64, 'base64');
           const tag = Buffer.from(tagB64, 'base64');
           const ct = Buffer.from(ctB64, 'base64');

           if (iv.length !== IV_LEN) {
             throw new Error(`IV length must be ${IV_LEN} bytes`);
           }
           if (tag.length !== TAG_LEN) {
             throw new Error(`Auth tag length must be ${TAG_LEN} bytes`);
           }
           const decipher = createDecipheriv(ALGO, this.key, iv);

           decipher.setAuthTag(tag);
           const pt = Buffer.concat([decipher.update(ct), decipher.final()]);

           return pt.toString('utf8');
         }
       }
       ```
