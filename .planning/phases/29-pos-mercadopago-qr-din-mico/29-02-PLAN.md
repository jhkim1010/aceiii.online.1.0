    ```sql
    -- =============================================================================
    -- Phase 29 Wave 1 — Step 1: mp_accounts 테이블 (OAuth 토큰 + scope)
    -- =============================================================================
    -- 목적: 매장/지점 단위 MP OAuth 계정 저장 (access_token/refresh_token AES-GCM 암호화)
    -- 실행 순서: 가장 먼저 실행 (mp_payment_intents/mp_wallets/mp_refunds 모두 이 테이블에 의존)
    -- Idempotency: CREATE TABLE IF NOT EXISTS + DO 블록으로 CHECK/INDEX 보호
    -- PG10/15 호환:
    --   SERIAL 사용 (NOT GENERATED AS IDENTITY)
    --   부분 UNIQUE 인덱스 2개로 (store_id, branch_id|NULL) 격리 (COALESCE-in-UNIQUE 회피)
    --   gen_random_uuid 미사용 (PG10 호환)
    -- =============================================================================

    BEGIN;

    CREATE TABLE IF NOT EXISTS mp_accounts (
      id              SERIAL PRIMARY KEY,
      store_id        INTEGER NOT NULL REFERENCES stores(id) ON DELETE CASCADE,
      branch_id       INTEGER NULL REFERENCES branches(id) ON DELETE CASCADE,
      mp_user_id      VARCHAR(64) NOT NULL,
      access_token    TEXT NOT NULL,                                    -- AES-256-GCM (iv_b64:tag_b64:ct_b64)
      refresh_token   TEXT NOT NULL,                                    -- AES-256-GCM
      public_key      VARCHAR(255) NULL,
      environment     VARCHAR(16) NOT NULL DEFAULT 'sandbox',
      external_pos_id VARCHAR(60) NULL,                                 -- RESEARCH §Pitfall 3
      expires_at      TIMESTAMPTZ NULL,
      connected_at    TIMESTAMPTZ NOT NULL DEFAULT NOW(),
      disconnected_at TIMESTAMPTZ NULL,
      created_at      TIMESTAMPTZ NOT NULL DEFAULT NOW(),
      updated_at      TIMESTAMPTZ NOT NULL DEFAULT NOW()
    );

    -- environment ENUM 가드 (CHECK constraint)
    DO $$
    BEGIN
      IF NOT EXISTS (
        SELECT 1 FROM pg_constraint
         WHERE conname = 'chk_mp_accounts_environment' AND conrelid = 'mp_accounts'::regclass
      ) THEN
        ALTER TABLE mp_accounts
          ADD CONSTRAINT chk_mp_accounts_environment
          CHECK (environment IN ('sandbox', 'production'));
      END IF;
    END$$;

    -- 부분 UNIQUE 인덱스 1: store-level (branch_id IS NULL) — 매장당 1개만
    CREATE UNIQUE INDEX IF NOT EXISTS uniq_mp_accounts_store_only
      ON mp_accounts (store_id)
      WHERE branch_id IS NULL AND disconnected_at IS NULL;

    -- 부분 UNIQUE 인덱스 2: branch-level (branch_id IS NOT NULL) — 지점당 1개만
    CREATE UNIQUE INDEX IF NOT EXISTS uniq_mp_accounts_store_branch
      ON mp_accounts (store_id, branch_id)
      WHERE branch_id IS NOT NULL AND disconnected_at IS NULL;

    -- 조회 성능
    CREATE INDEX IF NOT EXISTS idx_mp_accounts_store
      ON mp_accounts (store_id, environment)
      WHERE disconnected_at IS NULL;

    COMMIT;
    ```
