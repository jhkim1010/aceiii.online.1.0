       ```typescript
       import { Controller, Get, Query, ParseIntPipe } from '@nestjs/common';
       import { InjectModel } from '@nestjs/sequelize';

       import { Auth } from '../auth/decorators/auth.decorator';
       import { ValidRoles } from '../auth/interfaces/valid-roles';
       import { MpAccount } from './models/mp-account.model';
       import { Branch } from '../branch/branch.model';

       interface McdpgAccountResponse {
         id: number;
         storeId: number;
         branchId: number | null;
         branchName: string | null;
         mpUserId: string;
         environment: 'sandbox' | 'production';
         externalPosId: string | null;
         expiresAt: string | null;
         connectedAt: string;
         disconnectedAt: string | null;
         daysUntilExpire: number | null;
       }

       function toResponse(row: any): McdpgAccountResponse {
         const expiresAt = row.expiresAt as Date | null;
         const daysUntilExpire = expiresAt
           ? Math.floor((new Date(expiresAt).getTime() - Date.now()) / 86_400_000)
           : null;

         return {
           id: row.id,
           storeId: row.storeId,
           branchId: row.branchId,
           branchName: row.branch?.name ?? null,
           mpUserId: row.mpUserId,
           environment: row.environment,
           externalPosId: row.externalPosId,
           expiresAt: expiresAt ? new Date(expiresAt).toISOString() : null,
           connectedAt: row.connectedAt ? new Date(row.connectedAt).toISOString() : '',
           disconnectedAt: row.disconnectedAt ? new Date(row.disconnectedAt).toISOString() : null,
           daysUntilExpire,
         };
       }

       @Controller('mercadopago')
       export class MercadopagoController {
         constructor(
           @InjectModel(MpAccount) private readonly mpAccountModel: typeof MpAccount,
         ) {}

         // 매장 admin 만 — 토큰 컬럼 절대 응답에 포함 안 함 (RESEARCH §V8)
         @Get('accounts')
         @Auth(ValidRoles.admin, ValidRoles.superadmin)
         async listAccounts(@Query('storeId', ParseIntPipe) storeId: number): Promise<McdpgAccountResponse[]> {
           const rows = await this.mpAccountModel.findAll({
             where: { storeId },
             attributes: [
               'id', 'storeId', 'branchId', 'mpUserId', 'environment',
               'externalPosId', 'expiresAt', 'connectedAt', 'disconnectedAt',
             ],
             include: [{ model: Branch, attributes: ['id', 'name'] }],
             order: [['branchId', 'ASC NULLS FIRST'] as any],
           });

           return rows.map(toResponse);
         }
       }
       ```

       import { useMemo } from 'react'

       import Box from '@mui/material/Box'
       import Card from '@mui/material/Card'
       import CardContent from '@mui/material/CardContent'
       import Chip from '@mui/material/Chip'
       import Stack from '@mui/material/Stack'
       import Switch from '@mui/material/Switch'
       import Typography from '@mui/material/Typography'

       import McdpgEnvironmentBadge from './McdpgEnvironmentBadge'
       import type { McdpgAccount } from 'src/types/mercadopago'

       interface Branch {
         id: number
         name: string
       }

       interface Props {
         storeId: number
         branches: Branch[]
         accounts: McdpgAccount[]
       }

       // UI-SPEC Surface 1 §Branch toggle table — row per branch
       const McdpgBranchToggleTable = ({ storeId, branches, accounts }: Props) => {
         const accountsByBranch = useMemo(() => {
           const map = new Map<number, McdpgAccount>()

           accounts.forEach((a) => { if (a.branchId !== null && !a.disconnectedAt) map.set(a.branchId, a) })

           return map
         }, [accounts])

         const startBranchOAuth = (branchId: number) => {
           const account = accountsByBranch.get(branchId)
           const env: 'sandbox' | 'production' = account?.environment ?? 'sandbox'
           const url = `/api/mercadopago/oauth/start?storeId=${storeId}&branchId=${branchId}&environment=${env}`

           window.location.assign(url)
         }

         const branchCount = branches.length
         const ownAccountCount = accountsByBranch.size

         return (
           <Card sx={{ mb: 3 }}>
             <CardContent>
               <Stack direction='row' justifyContent='space-between' alignItems='center' sx={{ mb: 2 }}>
                 <Typography variant='h6'>🏬 Configuración por sucursal</Typography>
                 <Chip size='small' label={`${ownAccountCount} / ${branchCount} con cuenta propia`} />
               </Stack>
               <Stack divider={<Box sx={{ borderBottom: 1, borderColor: 'divider' }} />}>
                 {branches.map((b) => {
                   const account = accountsByBranch.get(b.id)
                   const expireWarn = account && (account.daysUntilExpire ?? 999) < 7

                   return (
                     <Box
                       key={b.id}
                       sx={{
                         display: 'grid',
                         gridTemplateColumns: 'minmax(180px,1fr) auto auto auto',
                         alignItems: 'center',
                         gap: 2,
                         py: '14px',
                         px: 1,
                       }}
                     >
                       <Box>
                         <Typography variant='body2' sx={{ fontWeight: 500 }}>{b.name}</Typography>
                         {account && (
                           <Typography variant='caption' sx={{ color: 'text.secondary', fontFamily: 'JetBrains Mono' }}>
                             MP user_id: {account.mpUserId}
                           </Typography>
                         )}
                       </Box>
                       <Box>
                         {account ? <McdpgEnvironmentBadge environment={account.environment} /> : null}
                       </Box>
                       <Box>
                         {expireWarn && account && (
                           <Typography variant='caption' sx={{ color: 'error.main' }}>
                             ⚠️ renueva en {account.daysUntilExpire} días
                           </Typography>
                         )}
                       </Box>
                       <Box sx={{ textAlign: 'right' }}>
                         <Switch
                           checked={!!account}
                           onChange={() => startBranchOAuth(b.id)}
                           inputProps={{ 'aria-label': `${b.name} cuenta propia` }}
                         />
                         <Typography variant='caption' sx={{ display: 'block', color: 'text.secondary' }}>
                           {account ? 'cuenta propia' : 'usa tienda'}
                         </Typography>
                       </Box>
                     </Box>
                   )
                 })}
               </Stack>
             </CardContent>
           </Card>
         )
       }

       export default McdpgBranchToggleTable
