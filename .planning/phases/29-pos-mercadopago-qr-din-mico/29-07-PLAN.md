       ```typescript
       import { useEffect, useRef } from 'react'
       import { io, Socket } from 'socket.io-client'

       import type { McdpgApprovedPayload } from 'src/types/mercadopago'

       // 환경별 WS URL — team-chat/TeamChatPanel.tsx 패턴 따름
       const WS_URL =
         process.env.NEXT_PUBLIC_WS_URL ||
         (process.env.NODE_ENV === 'development'
           ? 'http://localhost:5002/realtime'
           : 'https://newapi.coolsistema.com/realtime')

       export function useMpApprovedSocket(
         terminalId: number | null,
         onApproved: (payload: McdpgApprovedPayload) => void,
       ) {
         const callbackRef = useRef(onApproved)

         // ref-stable callback — onApproved 변경에 socket 재연결 안 함
         useEffect(() => { callbackRef.current = onApproved }, [onApproved])

         useEffect(() => {
           if (!terminalId) return undefined

           const socket: Socket = io(WS_URL, { transports: ['websocket'] })

           socket.on('connect', () => {
             socket.emit('register_terminal', { terminalId })
           })

           socket.on('mercadopago:approved', (payload: McdpgApprovedPayload) => {
             callbackRef.current(payload)
           })

           // cleanup — 메모리 누수 방지 (PATTERNS.md cleanup 강조)
           return () => {
             socket.off('mercadopago:approved')
             socket.disconnect()
           }
         }, [terminalId])
       }
       ```

       ```tsx
       import { useEffect, useState } from 'react'
       import { QRCodeSVG } from 'qrcode.react'

       import Alert from '@mui/material/Alert'
       import Box from '@mui/material/Box'
       import Button from '@mui/material/Button'
       import Chip from '@mui/material/Chip'
       import Stack from '@mui/material/Stack'
       import Typography from '@mui/material/Typography'

       export type McdpgQrStatus = 'idle' | 'generating' | 'waiting' | 'approved' | 'expired' | 'cancelled' | 'error'

       interface Props {
         qrData: string | null
         expiresAt: Date | null
         amount: number
         environment: 'sandbox' | 'production'
         status: McdpgQrStatus
         errorMessage?: string
         onCancel: () => void
         onRetry: () => void
       }

       // 통화 포맷 (AR) — '$ 30.000,00'
       function formatARS(n: number): string {
         return `$ ${n.toLocaleString('es-AR', { minimumFractionDigits: 2, maximumFractionDigits: 2 })}`
       }

       // UI-SPEC Surface 2 §QR side panel — winner B layout, 320px wide
       const McdpgQrPanel = ({ qrData, expiresAt, amount, environment, status, errorMessage, onCancel, onRetry }: Props) => {
         const isSandbox = environment === 'sandbox'
         const borderColor = isSandbox ? 'warning.main' : 'info.main'
         const glow = isSandbox
           ? '0 0 30px rgba(245,166,35,0.15)'
           : '0 0 30px rgba(0,177,234,0.15)'

         // Countdown — refresh every second
         const [now, setNow] = useState(Date.now())

         useEffect(() => {
           if (status !== 'waiting') return undefined
           const id = setInterval(() => setNow(Date.now()), 1000)

           return () => clearInterval(id)
         }, [status])

         const secondsLeft = expiresAt ? Math.max(0, Math.floor((expiresAt.getTime() - now) / 1000)) : 0
         const mm = String(Math.floor(secondsLeft / 60)).padStart(2, '0')
         const ss = String(secondsLeft % 60).padStart(2, '0')
         const countdownColor = secondsLeft < 30 ? 'error.main' : 'warning.main'

         return (
           <Box
             sx={{
               width: 320,
               bgcolor: 'background.default',
               borderLeft: 2,
               borderLeftColor: borderColor,
               boxShadow: glow,
               p: 3,
               display: 'flex',
               flexDirection: 'column',
               gap: 2,
             }}
           >
             <Stack direction='row' spacing={1} alignItems='center'>
               <Chip size='small' color={isSandbox ? 'warning' : 'info'} label='QR DINÁMICO' sx={{ fontWeight: 600, letterSpacing: '0.04em' }} />
               <Chip size='small' color={isSandbox ? 'warning' : 'info'} label={isSandbox ? '🧪 SANDBOX' : '🌐 PRODUCCIÓN'} />
             </Stack>

             {status === 'generating' && (
               <Typography variant='body2' sx={{ textAlign: 'center', py: 4 }}>Generando QR…</Typography>
             )}

             {status === 'error' && (
               <>
                 <Alert severity='error'>
                   No se pudo generar el QR ({errorMessage ?? 'error'}). Probá de nuevo o seleccioná otro medio de pago.
                 </Alert>
                 <Button onClick={onRetry} variant='contained' color='primary' size='small'>Reintentar</Button>
               </>
             )}

             {status === 'waiting' && qrData && (
               <>
                 <Typography variant='caption' sx={{ color: 'text.secondary' }}>Monto a cobrar</Typography>
                 <Typography sx={{ fontFamily: 'JetBrains Mono', fontSize: 24, fontWeight: 700 }}>{formatARS(amount)}</Typography>

                 <Box sx={{ bgcolor: '#fff', p: 1.5, borderRadius: 2, display: 'flex', justifyContent: 'center' }}>
                   <QRCodeSVG value={qrData} size={180} level='M' />
                 </Box>

                 <Typography variant='caption' sx={{ textAlign: 'center', color: 'text.secondary' }}>
                   Escaneá con la app MP
                 </Typography>

                 <Stack direction='row' spacing={1} alignItems='center' justifyContent='center'>
                   <Box sx={{ width: 8, height: 8, borderRadius: '50%', bgcolor: 'warning.main', animation: 'pulse 1.4s ease-in-out infinite' }} />
                   <Typography variant='caption'>Esperando pago…</Typography>
                 </Stack>

                 <Box sx={{ textAlign: 'center', bgcolor: 'action.hover', borderRadius: 1, py: 1, px: 2, fontFamily: 'JetBrains Mono', fontSize: 18, fontWeight: 600, color: countdownColor }}>
                   {mm}:{ss}
                 </Box>

                 <Button onClick={onCancel} variant='outlined' color='warning' size='small' fullWidth>
                   Cancelar QR
                 </Button>
               </>
             )}

             {status === 'approved' && (
               <Box sx={{ textAlign: 'center', py: 5, bgcolor: 'success.main', borderRadius: 2, color: '#fff' }}>
                 <Typography variant='h6'>✓ Pago confirmado</Typography>
                 <Typography variant='caption'>Generando venta…</Typography>
               </Box>
             )}

             {status === 'expired' && (
               <>
                 <Typography variant='h6' sx={{ textAlign: 'center', color: 'error.main' }}>QR expirado</Typography>
                 <Typography variant='caption' sx={{ textAlign: 'center' }}>El cliente no completó el pago en 3 minutos.</Typography>
                 <Button onClick={onRetry} variant='contained' color='primary' size='small' fullWidth>Generar nuevo QR</Button>
               </>
             )}

             {status === 'cancelled' && (
               <>
                 <Typography variant='caption' sx={{ textAlign: 'center', color: 'text.secondary' }}>QR cancelado</Typography>
                 <Button onClick={onRetry} variant='contained' color='primary' size='small' fullWidth>Generar nuevo QR</Button>
               </>
             )}

             <style jsx global>{`
               @keyframes pulse {
                 0%, 100% { opacity: 1; transform: scale(1); }
                 50% { opacity: 0.5; transform: scale(1.3); }
               }
             `}</style>
           </Box>
         )
       }

       export default McdpgQrPanel
       ```
