       // Phase 29 — MP REST mock helper for Jest unit/integration tests.
       // Mocks axios at the api-client layer so MpApiClientService.get/post return fixtures.
       import axios from 'axios';

       jest.mock('axios');

       const mockedAxios = axios as jest.Mocked<typeof axios>;

       export interface MpMockEntry {
         method: 'GET' | 'POST' | 'PUT' | 'DELETE';
         pathPattern: RegExp;
         response: any;
         status?: number;
       }

       const registry: MpMockEntry[] = [];

       export function resetMpMocks(): void {
         registry.length = 0;
         mockedAxios.get.mockReset();
         mockedAxios.post.mockReset();
         mockedAxios.put.mockReset();
         mockedAxios.delete.mockReset();
       }

       export function mockMpGet(pathPattern: RegExp, response: any, status = 200): void {
         registry.push({ method: 'GET', pathPattern, response, status });
         mockedAxios.get.mockImplementation((url: string) => {
           const hit = registry.find((e) => e.method === 'GET' && e.pathPattern.test(url));

           if (hit) return Promise.resolve({ data: hit.response, status: hit.status ?? 200 });

           return Promise.reject(new Error(`Unmocked MP GET: ${url}`));
         });
       }

       export function mockMpPost(pathPattern: RegExp, response: any, status = 200): void {
         registry.push({ method: 'POST', pathPattern, response, status });
         mockedAxios.post.mockImplementation((url: string) => {
           const hit = registry.find((e) => e.method === 'POST' && e.pathPattern.test(url));

           if (hit) return Promise.resolve({ data: hit.response, status: hit.status ?? 200 });

           return Promise.reject(new Error(`Unmocked MP POST: ${url}`));
         });
       }

       export function mockMpFailure(method: 'GET' | 'POST', pathPattern: RegExp, status: number, body: any): void {
         registry.push({ method, pathPattern, response: body, status });
         const fn = method === 'GET' ? mockedAxios.get : mockedAxios.post;

         fn.mockImplementation((url: string) => {
           if (pathPattern.test(url)) {
             const err: any = new Error(`MP ${method} ${url} failed`);

             err.response = { status, data: body };

             return Promise.reject(err);
           }

           return Promise.reject(new Error(`Unmocked MP ${method}: ${url}`));
         });
       }
