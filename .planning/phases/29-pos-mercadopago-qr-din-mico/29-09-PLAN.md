       import { BadRequestException, Injectable, Logger, NotFoundException } from '@nestjs/common';
       import { InjectModel, InjectConnection } from '@nestjs/sequelize';
       import { Sequelize } from 'sequelize-typescript';

       import { MpRefund } from '../models/mp-refund.model';
       import { MpRefundAttempt } from '../models/mp-refund-attempt.model';
       import { MpPaymentIntent } from '../models/mp-payment-intent.model';
       import { MpAccount } from '../models/mp-account.model';
       import { MpApiClientService } from '../api-client/mp-api-client.service';
       import { MpTokenCryptoService } from '../crypto/mp-token-crypto.service';
       import { MpWalletService } from '../wallet/mp-wallet.service';

       interface MpRefundResponse {
         id: number | string;
         status: string;
         amount: number;
       }

       interface PerPaymentResult {
         mpPaymentId: string;
         success: boolean;
         attemptId: number;
         refundId?: string;
         errorMessage?: string;
       }

       @Injectable()
       export class MpRefundService {
         private readonly logger = new Logger(MpRefundService.name);

         constructor(
           @InjectModel(MpRefund) private readonly refundModel: typeof MpRefund,
           @InjectModel(MpRefundAttempt) private readonly attemptModel: typeof MpRefundAttempt,
           @InjectModel(MpPaymentIntent) private readonly intentModel: typeof MpPaymentIntent,
           @InjectModel(MpAccount) private readonly mpAccountModel: typeof MpAccount,
           @InjectConnection() private readonly sequelize: Sequelize,
           private readonly mpApi: MpApiClientService,
           private readonly crypto: MpTokenCryptoService,
           private readonly walletSvc: MpWalletService,
         ) {}

         // 호출 from sales-create.service.ts nullifySale extension
         // sale.paymentMethods 배열을 iterate, slug==='mercadopago'+mpPaymentId 인 entry마다 retryAttempt
         async refundForSale(sale: { id: number; paymentMethods: any[] }): Promise<{ perPayment: PerPaymentResult[] }> {
           const mpEntries = (sale.paymentMethods ?? []).filter((p: any) =>
             p?.slug === 'mercadopago' && (p.mpPaymentId || p.mp_payment_id),
           );
           const perPayment: PerPaymentResult[] = [];

           for (const entry of mpEntries) {
             const mpPaymentId = String(entry.mpPaymentId || entry.mp_payment_id);
             const amount = Number(entry.amount);
             const result = await this.retryAttempt(sale.id, mpPaymentId, amount);

             perPayment.push(result);
           }

           return { perPayment };
         }

         // 단일 mp_payment에 대한 1회 시도 — attempt_no 자동 증가
         async retryAttempt(saleId: number, mpPaymentId: string, amount: number): Promise<PerPaymentResult> {
           // 1) intent 조회 — full vs partial 결정 + mp_account 결정
           const intent = await this.intentModel.findOne({ where: { paymentId: mpPaymentId } });

           if (!intent) {
             throw new NotFoundException(`mp_payment_intent for paymentId=${mpPaymentId} not found`);
           }
           const account = await this.mpAccountModel.findByPk(intent.mpAccountId);

           if (!account) throw new NotFoundException(`mp_account ${intent.mpAccountId} not found`);

           // 2) 새 attempt_no 결정
           const lastAttempt = await this.attemptModel.findOne({
             where: { saleId, mpPaymentId },
             order: [['attemptNo', 'DESC']],
           });
           const attemptNo = (lastAttempt?.attemptNo ?? 0) + 1;
           const attempt = await this.attemptModel.create({
             saleId,
             mpPaymentId,
             attemptNo,
             status: 'pending',
             attemptedAt: new Date(),
           });

           // 3) MP API call with X-Idempotency-Key (RESEARCH §Pitfall 10)
           const accessToken = this.crypto.decrypt(account.accessToken);
           const isFull = Math.abs(amount - Number(intent.amount)) < 0.01;
           const body: any = isFull ? {} : { amount };
           const idempotencyKey = `refund-${saleId}-${attemptNo}`;

           try {
             const refund = await this.mpApi.post<MpRefundResponse>(
               `/v1/payments/${mpPaymentId}/refunds`,
               body,
               accessToken,
               { idempotencyKey, baseUrl: this.mpApi.hostFor(account.environment) },
             );

             // 4) Success TX — INSERT mp_refunds + walletSvc.debitOnRefund + UPDATE attempt
             await this.sequelize.transaction(async (t) => {
               await this.refundModel.create(
                 {
                   saleId,
                   mpPaymentId,
                   refundId: String(refund.id),
                   amount,
                   status: refund.status,
                 },
                 { transaction: t },
               );
               await this.walletSvc.debitOnRefund(intent, amount, t);
               await attempt.update({ status: 'success' }, { transaction: t });
             });

             return { mpPaymentId, success: true, attemptId: attempt.id, refundId: String(refund.id) };
           } catch (err: any) {
             // 5) Failure — attempt 만 mark failed, sale는 nullified 유지
             const errorMessage = (err?.response?.data?.message || err?.message || 'unknown').slice(0, 500);

             await attempt.update({ status: 'failed', errorMessage });
             this.logger.warn(`MP refund failed sale=${saleId} payment=${mpPaymentId} attempt=${attemptNo}: ${errorMessage}`);

             return { mpPaymentId, success: false, attemptId: attempt.id, errorMessage };
           }
         }

         async listAttempts(saleId: number): Promise<MpRefundAttempt[]> {
           return this.attemptModel.findAll({
             where: { saleId },
             order: [['attemptNo', 'ASC']],
           });
         }
       }
