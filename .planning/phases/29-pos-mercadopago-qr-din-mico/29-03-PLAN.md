       import { BadRequestException, Injectable, Logger } from '@nestjs/common';
       import axios, { AxiosRequestConfig } from 'axios';

       const MP_API_HOST = 'https://api.mercadopago.com';
       const MP_AUTH_HOST = 'https://auth.mercadopago.com.ar';

       @Injectable()
       export class MpApiClientService {
         private readonly logger = new Logger(MpApiClientService.name);

         hostFor(_env: 'sandbox' | 'production'): string {
           // MP API host는 sandbox/prod 동일 — 차이는 credentials + token 환경 (RESEARCH §State of the Art)
           return MP_API_HOST;
         }

         authHostFor(_env: 'sandbox' | 'production'): string {
           return MP_AUTH_HOST;
         }

         credentialsFor(env: 'sandbox' | 'production'): { clientId: string; clientSecret: string } {
           const prefix = env === 'sandbox' ? 'MP_SANDBOX' : 'MP_PRODUCTION';
           const clientId = process.env[`${prefix}_CLIENT_ID`];
           const clientSecret = process.env[`${prefix}_CLIENT_SECRET`];

           if (!clientId || !clientSecret) {
             throw new Error(
               `Missing ${prefix}_CLIENT_ID / ${prefix}_CLIENT_SECRET env vars (env=${env})`,
             );
           }

           return { clientId, clientSecret };
         }

         async get<T>(
           path: string,
           accessToken: string,
           opts?: { baseUrl?: string },
         ): Promise<T> {
           return this.request<T>('GET', path, undefined, accessToken, opts);
         }

         async post<T>(
           path: string,
           body: any,
           accessToken: string,
           opts?: { idempotencyKey?: string; baseUrl?: string },
         ): Promise<T> {
           return this.request<T>('POST', path, body, accessToken, opts);
         }

         async put<T>(
           path: string,
           body: any,
           accessToken: string,
           opts?: { baseUrl?: string },
         ): Promise<T> {
           return this.request<T>('PUT', path, body, accessToken, opts);
         }

         async delete<T>(
           path: string,
           accessToken: string,
           opts?: { baseUrl?: string },
         ): Promise<T> {
           return this.request<T>('DELETE', path, undefined, accessToken, opts);
         }

         private async request<T>(
           method: 'GET' | 'POST' | 'PUT' | 'DELETE',
           path: string,
           body: any,
           accessToken: string,
           opts?: { idempotencyKey?: string; baseUrl?: string },
         ): Promise<T> {
           const baseUrl = opts?.baseUrl ?? MP_API_HOST;
           const headers: Record<string, string> = {
             Authorization: `Bearer ${accessToken}`,
             'Content-Type': 'application/json',
           };

           if (opts?.idempotencyKey) {
             headers['X-Idempotency-Key'] = opts.idempotencyKey;
           }
           const config: AxiosRequestConfig = { headers, timeout: 10_000 };

           try {
             let res;

             switch (method) {
               case 'GET': res = await axios.get<T>(`${baseUrl}${path}`, config); break;
               case 'POST': res = await axios.post<T>(`${baseUrl}${path}`, body, config); break;
               case 'PUT': res = await axios.put<T>(`${baseUrl}${path}`, body, config); break;
               case 'DELETE': res = await axios.delete<T>(`${baseUrl}${path}`, config); break;
             }

             return res.data;
           } catch (err: any) {
             const status = err?.response?.status;
             const mpError = err?.response?.data;
             // 토큰 마스킹 — 로그에 평문 토큰 노출 금지 (RESEARCH §V8 Data Protection)
             const tokenMask = `${accessToken.slice(0, 12)}…(${accessToken.length})`;

             this.logger.error(
               `MP ${method} ${path} failed (token=${tokenMask}): ${status} ${JSON.stringify(mpError)}`,
             );
             throw new BadRequestException({
               message: 'Mercadopago API error',
               mpStatus: status,
               mpError,
             });
           }
         }
       }

       import { createHmac, timingSafeEqual } from 'crypto';

       const DEFAULT_MAX_AGE_MS = 10 * 60 * 1000; // 10 min — RESEARCH §Open Q6

       export interface OAuthStatePayload {
         storeId: number;
         branchId: number | null;
         environment: 'sandbox' | 'production';
         nonce: string;
         ts: number;
       }

       function b64url(buf: Buffer): string {
         return buf.toString('base64').replace(/\+/g, '-').replace(/\//g, '_').replace(/=+$/, '');
       }

       function fromB64url(s: string): Buffer {
         const pad = s.length % 4 === 0 ? '' : '='.repeat(4 - (s.length % 4));

         return Buffer.from(s.replace(/-/g, '+').replace(/_/g, '/') + pad, 'base64');
       }

       // Format: "{base64url(jsonPayload)}.{base64url(hmac)}"
       export function signOAuthState(payload: OAuthStatePayload, secret: string): string {
         const json = JSON.stringify(payload);
         const data = b64url(Buffer.from(json, 'utf8'));
         const sig = createHmac('sha256', secret).update(data).digest();

         return `${data}.${b64url(sig)}`;
       }

       export function verifyOAuthState(
         state: string,
         secret: string,
         maxAgeMs = DEFAULT_MAX_AGE_MS,
       ): OAuthStatePayload {
         const parts = state.split('.');

         if (parts.length !== 2) throw new Error('Invalid OAuth state format');
         const [data, sigB64] = parts;
         const expected = createHmac('sha256', secret).update(data).digest();
         const provided = fromB64url(sigB64);

         if (provided.length !== expected.length || !timingSafeEqual(provided, expected)) {
           throw new Error('OAuth state HMAC mismatch');
         }
         const payload = JSON.parse(fromB64url(data).toString('utf8')) as OAuthStatePayload;

         if (typeof payload.ts !== 'number' || Date.now() - payload.ts > maxAgeMs) {
           throw new Error('OAuth state expired (>10 min)');
         }
         if (!['sandbox', 'production'].includes(payload.environment)) {
           throw new Error('OAuth state has invalid environment');
         }

         return payload;
       }

       import { BadRequestException, Injectable, Logger } from '@nestjs/common';
       import { InjectConnection, InjectModel } from '@nestjs/sequelize';
       import { Sequelize } from 'sequelize-typescript';
       import { randomBytes } from 'crypto';

       import { MpAccount } from '../models/mp-account.model';
       import { MpWallet } from '../models/mp-wallet.model';
       import { MpTokenCryptoService } from '../crypto/mp-token-crypto.service';
       import { MpApiClientService } from '../api-client/mp-api-client.service';
       import { MpStorePosService } from '../api-client/mp-store-pos.service';
       import { signOAuthState, verifyOAuthState, OAuthStatePayload } from './mp-oauth-state.util';

       interface OAuthTokenResponse {
         access_token: string;
         refresh_token: string;
         public_key: string;
         user_id: number | string;
         expires_in: number;
         scope: string;
       }

       @Injectable()
       export class MpOAuthService {
         private readonly logger = new Logger(MpOAuthService.name);

         constructor(
           @InjectModel(MpAccount) private readonly mpAccountModel: typeof MpAccount,
           @InjectModel(MpWallet) private readonly mpWalletModel: typeof MpWallet,
           @InjectConnection() private readonly sequelize: Sequelize,
           private readonly crypto: MpTokenCryptoService,
           private readonly mpApi: MpApiClientService,
           private readonly storePos: MpStorePosService,
         ) {}

         private get stateSecret(): string {
           const s = process.env.MP_OAUTH_STATE_SECRET;

           if (!s) throw new Error('MP_OAUTH_STATE_SECRET env var missing');

           return s;
         }

         private get notificationBaseUrl(): string {
           return process.env.MP_NOTIFICATION_BASE_URL || 'https://newapi.coolsistema.com/api';
         }

         buildAuthorizeUrl(input: { storeId: number; branchId: number | null; environment: 'sandbox' | 'production' }): string {
           const { clientId } = this.mpApi.credentialsFor(input.environment);
           const payload: OAuthStatePayload = {
             ...input,
             nonce: randomBytes(16).toString('hex'),
             ts: Date.now(),
           };
           const state = signOAuthState(payload, this.stateSecret);
           const redirectUri = `${this.notificationBaseUrl}/mercadopago/oauth/callback`;
           const params = new URLSearchParams({
             client_id: clientId,
             response_type: 'code',
             platform_id: 'mp',
             state,
             redirect_uri: redirectUri,
           });

           return `${this.mpApi.authHostFor(input.environment)}/authorization?${params.toString()}`;
         }

         async handleCallback(code: string, state: string): Promise<{ accountId: number; walletId: number; environment: string }> {
           // 1) HMAC state 검증 (anti-CSRF + 10 min 만료)
           const payload = verifyOAuthState(state, this.stateSecret);

           // 2) Code → tokens 교환 (MP /oauth/token)
           const { clientId, clientSecret } = this.mpApi.credentialsFor(payload.environment);
           const redirectUri = `${this.notificationBaseUrl}/mercadopago/oauth/callback`;
           const tokenRes = await this.mpApi.post<OAuthTokenResponse>(
             '/oauth/token',
             {
               client_id: clientId,
               client_secret: clientSecret,
               grant_type: 'authorization_code',
               code,
               redirect_uri: redirectUri,
             },
             '', // /oauth/token doesn't need bearer
           );

           // 3) Store + POS 등록 (RESEARCH §Pitfall 3) — 실패해도 멱등이라 throw 안 함
           const { externalPosId } = await this.storePos.registerStoreAndPos({
             accessToken: tokenRes.access_token,
             mpUserId: String(tokenRes.user_id),
             storeId: payload.storeId,
             branchId: payload.branchId,
             environment: payload.environment,
           });

           // 4) DB에 token + wallet 저장 (단일 TX)
           const expiresAt = new Date(Date.now() + tokenRes.expires_in * 1000);
           const result = await this.sequelize.transaction(async (t) => {
             const existing = await this.mpAccountModel.findOne({
               where: { storeId: payload.storeId, branchId: payload.branchId },
               transaction: t,
             });

             let account: MpAccount;

             if (existing) {
               // Re-OAuth — UPDATE existing row, clear disconnected_at
               await existing.update(
                 {
                   mpUserId: String(tokenRes.user_id),
                   accessToken: this.crypto.encrypt(tokenRes.access_token),
                   refreshToken: this.crypto.encrypt(tokenRes.refresh_token),
                   publicKey: tokenRes.public_key,
                   environment: payload.environment,
                   externalPosId,
                   expiresAt,
                   connectedAt: new Date(),
                   disconnectedAt: null,
                 },
                 { transaction: t },
               );
               account = existing;
             } else {
               account = await this.mpAccountModel.create(
                 {
                   storeId: payload.storeId,
                   branchId: payload.branchId,
                   mpUserId: String(tokenRes.user_id),
                   accessToken: this.crypto.encrypt(tokenRes.access_token),
                   refreshToken: this.crypto.encrypt(tokenRes.refresh_token),
                   publicKey: tokenRes.public_key,
                   environment: payload.environment,
                   externalPosId,
                   expiresAt,
                   connectedAt: new Date(),
                 },
                 { transaction: t },
               );
             }

             // mp_wallets — 1:1 with account
             let wallet = await this.mpWalletModel.findOne({
               where: { mpAccountId: account.id },
               transaction: t,
             });

             if (!wallet) {
               wallet = await this.mpWalletModel.create(
                 {
                   mpAccountId: account.id,
                   storeId: payload.storeId,
                   branchId: payload.branchId,
                   balance: 0,
                   currency: 'ARS',
                 },
                 { transaction: t },
               );
             }

             return { accountId: account.id, walletId: wallet.id, environment: payload.environment };
           });

           return result;
         }

         // RESEARCH §Pitfall 2 — refresh response includes NEW refresh_token, MUST re-save both
         async refreshAccount(accountId: number): Promise<MpAccount> {
           const account = await this.mpAccountModel.findByPk(accountId);

           if (!account) throw new BadRequestException(`mp_account ${accountId} not found`);
           const refreshToken = this.crypto.decrypt(account.refreshToken);
           const { clientId, clientSecret } = this.mpApi.credentialsFor(account.environment);
           const tokenRes = await this.mpApi.post<OAuthTokenResponse>(
             '/oauth/token',
             {
               client_id: clientId,
               client_secret: clientSecret,
               grant_type: 'refresh_token',
               refresh_token: refreshToken,
             },
             '',
           );
           const expiresAt = new Date(Date.now() + tokenRes.expires_in * 1000);

           await this.sequelize.transaction(async (t) => {
             await account.update(
               {
                 accessToken: this.crypto.encrypt(tokenRes.access_token),
                 refreshToken: this.crypto.encrypt(tokenRes.refresh_token),
                 expiresAt,
               },
               { transaction: t },
             );
           });

           return account;
         }

         async disconnect(accountId: number, _userId: number): Promise<void> {
           const account = await this.mpAccountModel.findByPk(accountId);

           if (!account) throw new BadRequestException(`mp_account ${accountId} not found`);
           await account.update({ disconnectedAt: new Date() });
           this.logger.log(`mp_account ${accountId} disconnected by user ${_userId}`);
         }
       }

       import { BadRequestException, Body, Controller, Get, Param, ParseIntPipe, Post, Query, Res } from '@nestjs/common';
       import { Response } from 'express';

       import { Audit } from 'src/common/decorators/audit.decorator';
       import { Auth } from '../../auth/decorators/auth.decorator';
       import { Public } from '../../auth/decorators/public.decorator';
       import { ValidRoles } from '../../auth/interfaces/valid-roles';
       import { GetUser } from '../../auth/decorators/get-user.decorator';
       import { Users } from '../../users/users.model';

       import { MpOAuthService } from './mp-oauth.service';

       const FRONTEND_BASE = process.env.FRONTEND_BASE_URL || 'https://app.coolsistema.com';

       // OAuth 시작/콜백/연결해제 — 매장 owner 단독 동작 (admin)
       @Controller('mercadopago/oauth')
       export class MpOAuthController {
         constructor(private readonly svc: MpOAuthService) {}

         // start: 매장 admin 가 클릭 → MP authorize URL 로 redirect
         @Get('start')
         @Auth(ValidRoles.admin, ValidRoles.superadmin)
         start(
           @Query('storeId', ParseIntPipe) storeId: number,
           @Query('branchId') branchIdRaw: string | undefined,
           @Query('environment') environment: 'sandbox' | 'production',
           @Res() res: Response,
         ) {
           if (!['sandbox', 'production'].includes(environment)) {
             throw new BadRequestException('environment must be sandbox|production');
           }
           const branchId = branchIdRaw && branchIdRaw !== 'null' ? Number(branchIdRaw) : null;

           if (branchId !== null && Number.isNaN(branchId)) {
             throw new BadRequestException('branchId must be integer or "null"');
           }
           const url = this.svc.buildAuthorizeUrl({ storeId, branchId, environment });

           // 브라우저 redirect — JSON 응답이 아님
           res.redirect(302, url);
         }

         // callback: MP가 호출 — JWT 없음 (HMAC state 가 anti-CSRF)
         @Public()
         @Get('callback')
         async callback(
           @Query('code') code: string,
           @Query('state') state: string,
           @Res() res: Response,
         ) {
           if (!code || !state) {
             return res.redirect(302, `${FRONTEND_BASE}/configuracion/mercadopago?error=missing-code-or-state`);
           }
           try {
             const result = await this.svc.handleCallback(code, state);

             return res.redirect(302, `${FRONTEND_BASE}/configuracion/mercadopago?ok=1&accountId=${result.accountId}&env=${result.environment}`);
           } catch (err: any) {
             const msg = encodeURIComponent(err?.message?.slice(0, 200) ?? 'unknown');

             return res.redirect(302, `${FRONTEND_BASE}/configuracion/mercadopago?error=${msg}`);
           }
         }

         // disconnect: admin이 명시적으로 끊음
         @Post('disconnect/:accountId')
         @Auth(ValidRoles.admin, ValidRoles.superadmin)
         @Audit({
           entityType: 'McdpgAccount',
           action: 'delete',
           getDescription: (params: any) => `mp_account ${params?.accountId} disconnected`,
         })
         async disconnect(
           @Param('accountId', ParseIntPipe) accountId: number,
           @GetUser() user: Users,
         ): Promise<{ ok: true }> {
           await this.svc.disconnect(accountId, user.id);

           return { ok: true };
         }
       }
