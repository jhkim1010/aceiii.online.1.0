       import { BadRequestException, Injectable, Logger, NotFoundException } from '@nestjs/common';
       import { InjectModel, InjectConnection } from '@nestjs/sequelize';
       import { Sequelize } from 'sequelize-typescript';

       import { MpWallet } from '../models/mp-wallet.model';
       import { MpMovement } from '../models/mp-movement.model';
       import { MpTransfer } from '../models/mp-transfer.model';
       import { Box } from '../../box/box.model';
       import { TransferMpToCashDto } from '../dto/transfer-mp-to-cash.dto';

       @Injectable()
       export class MpTransferService {
         private readonly logger = new Logger(MpTransferService.name);

         constructor(
           @InjectModel(MpWallet) private readonly walletModel: typeof MpWallet,
           @InjectModel(MpMovement) private readonly movementModel: typeof MpMovement,
           @InjectModel(MpTransfer) private readonly transferModel: typeof MpTransfer,
           @InjectModel(Box) private readonly boxModel: typeof Box,
           @InjectConnection() private readonly sequelize: Sequelize,
         ) {}

         async transfer(
           input: TransferMpToCashDto,
           userId: number,
         ): Promise<{ transferId: number; mpWalletBalanceAfter: number; boxBalanceAfter: number }> {
           if (input.amount <= 0) {
             throw new BadRequestException('Monto debe ser mayor a 0');
           }

           // 단일 트랜잭션 — 모든 변경을 원자적으로 (Pattern E + RESEARCH Pattern 3)
           const result = await this.sequelize.transaction(async (t) => {
             // 1) MP wallet — SELECT FOR UPDATE
             const wallet = await this.walletModel.findByPk(input.mpWalletId, {
               lock: t.LOCK.UPDATE,
               transaction: t,
             });

             if (!wallet) throw new NotFoundException(`mp_wallet ${input.mpWalletId} not found`);
             const currentBalance = Number(wallet.balance);

             if (currentBalance < input.amount) {
               throw new BadRequestException({
                 message: `Saldo insuficiente en Caja MP. Disponible: $ ${currentBalance.toFixed(2)}.`,
                 code: 'MP_INSUFFICIENT_BALANCE',
                 available: currentBalance,
               });
             }

             // 2) Target box — SELECT FOR UPDATE
             const box = await this.boxModel.findByPk(input.targetBoxId, {
               lock: t.LOCK.UPDATE,
               transaction: t,
             });

             if (!box) throw new NotFoundException(`box ${input.targetBoxId} not found`);
             // 매장 격리 검증 — wallet.storeId === box.storeId
             if (Number((box as any).storeId) !== Number(wallet.storeId)) {
               throw new BadRequestException('Caja destino no pertenece a este matrimonio store');
             }

             // 3) mp_movements debit (transfer_out)
             const debitMovement = await this.movementModel.create(
               {
                 mpWalletId: wallet.id,
                 type: 'transfer_out',
                 amount: input.amount,
                 saleId: null,
                 mpPaymentId: null,
                 note: input.note?.slice(0, 200) || `Transferencia a caja física box=${box.id}`,
               },
               { transaction: t },
             );

             // 4) mp_transfers row
             const transfer = await this.transferModel.create(
               {
                 mpWalletId: wallet.id,
                 targetBoxId: box.id,
                 amount: input.amount,
                 userId,
                 note: input.note?.slice(0, 255) || null,
                 transferredAt: new Date(),
               },
               { transaction: t },
             );

             // 5) link mp_movements → mp_transfers
             await debitMovement.update({ transferId: transfer.id }, { transaction: t });

             // 6) update mp_wallets.balance
             const mpWalletBalanceAfter = currentBalance - input.amount;

             await wallet.update(
               { balance: mpWalletBalanceAfter, lastSyncedAt: new Date() },
               { transaction: t },
             );

             // 7) box.balance += amount  — 기존 box 시스템 패턴은 box.balance 캐시 컬럼 직접 update가 안전
             // (box 시스템은 cash_register / box_operations 별도 테이블 — 여기선 box.balance만 업데이트, 상세 movement audit는 mp_transfers 가 담당)
             const boxBalanceAfter = Number((box as any).balance ?? 0) + input.amount;

             await box.update({ balance: boxBalanceAfter } as any, { transaction: t });

             this.logger.log(
               `Transfer #${transfer.id}: mp_wallet ${wallet.id} -${input.amount} → box ${box.id} +${input.amount} by user ${userId}`,
             );

             return { transferId: transfer.id, mpWalletBalanceAfter, boxBalanceAfter };
           });

           return result;
         }
       }

         constructor(

           private readonly walletSvc: MpWalletService,
         ) {}

         // 매일 03:00 — balance 캐시 vs movements SUM 정합성 검증
         @Cron(CronExpression.EVERY_DAY_AT_3AM)
         async reconcile() {
           const wallets = await this.walletModel.findAll();
           let drifted = 0;

           for (const w of wallets) {
             const computed = await this.walletSvc.computeBalance(w.id);
             const cached = Number(w.balance);

             if (Math.abs(computed - cached) > 0.01) {
               drifted++;
               this.logger.warn(
                 `Wallet ${w.id}: cached=${cached} computed=${computed} drift=${(computed - cached).toFixed(2)} — updating cache`,
               );
               await w.update({ balance: computed, lastSyncedAt: new Date() });
             }
           }
           this.logger.log(`mp_wallet reconcile: ${wallets.length} wallets, ${drifted} drifted (corrected)`);
         }
