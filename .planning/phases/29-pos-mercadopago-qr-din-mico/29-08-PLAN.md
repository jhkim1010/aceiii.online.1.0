       ```typescript
       import { Injectable, Logger } from '@nestjs/common';
       import { Cron, CronExpression } from '@nestjs/schedule';
       import { InjectModel } from '@nestjs/sequelize';
       import { Op } from 'sequelize';

       import { MpWallet } from '../models/mp-wallet.model';
       import { MpPaymentIntent } from '../models/mp-payment-intent.model';
       import { MpWalletService } from '../wallet/mp-wallet.service';

       @Injectable()
       export class MpWalletReconcileCron {
         private readonly logger = new Logger(MpWalletReconcileCron.name);

         constructor(
           @InjectModel(MpWallet) private readonly walletModel: typeof MpWallet,
           @InjectModel(MpPaymentIntent) private readonly intentModel: typeof MpPaymentIntent,
           private readonly walletSvc: MpWalletService,
         ) {}

         @Cron(CronExpression.EVERY_DAY_AT_3AM)
         async reconcile() {
           // 1) wallet 잔액 정합성
           const wallets = await this.walletModel.findAll();
           let drifted = 0;

           for (const w of wallets) {
             const computed = await this.walletSvc.computeBalance(w.id);
             const cached = Number(w.balance);

             if (Math.abs(computed - cached) > 0.01) {
               drifted++;
               this.logger.warn(
                 `Wallet ${w.id}: cached=${cached} computed=${computed} drift=${(computed - cached).toFixed(2)} — updating cache`,
               );
               await w.update({ balance: computed, lastSyncedAt: new Date() });
             }
           }
           this.logger.log(`mp_wallet reconcile: ${wallets.length} wallets, ${drifted} drifted (corrected)`);

           // 2) 만료된 pending intent 정리 (Plan 04 createIntent 실패 / 사용자 abandon 경로)
           const [, expiredCount] = await this.intentModel.update(
             { status: 'expired' },
             {
               where: {
                 status: 'pending',
                 expiresAt: { [Op.lt]: new Date() },
               },
             },
           ) as unknown as [number, number];
           this.logger.log(`mp_payment_intents stale sweep: ${expiredCount ?? 0} pending → expired`);
         }
       }
       ```
