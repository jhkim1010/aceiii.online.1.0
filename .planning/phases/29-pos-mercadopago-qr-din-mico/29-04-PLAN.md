       ```typescript
       import { BadRequestException, Injectable, Logger, NotFoundException } from '@nestjs/common';
       import { InjectModel } from '@nestjs/sequelize';

       import { MpPaymentIntent } from '../models/mp-payment-intent.model';
       import { MpAccount } from '../models/mp-account.model';
       import { MpAccountResolverService } from '../mp-account-resolver.service';
       import { MpApiClientService } from '../api-client/mp-api-client.service';
       import { MpTokenCryptoService } from '../crypto/mp-token-crypto.service';
       import { CreateMpQrDto } from '../dto/create-mp-qr.dto';

       const QR_TTL_MS = 3 * 60 * 1000;        /* 3 min — per SPEC.md MP-POS-02 — DO NOT MOVE TO ENV (locked timeout) */
       const MP_QR_PATH_TPL = (userId: string, posId: string) =>
         `/instore/orders/qr/seller/collectors/${userId}/pos/${encodeURIComponent(posId)}/qrs`;
       const MP_QR_CANCEL_PATH_TPL = (userId: string, posId: string) =>
         `/instore/qr/seller/collectors/${userId}/pos/${encodeURIComponent(posId)}/orders`;

       interface MpQrResponse {
         qr_data: string;
         in_store_order_id: string;
       }

       @Injectable()
       export class MpQrService {
         private readonly logger = new Logger(MpQrService.name);

         constructor(
           @InjectModel(MpPaymentIntent) private readonly intentModel: typeof MpPaymentIntent,
           @InjectModel(MpAccount) private readonly mpAccountModel: typeof MpAccount,
           private readonly resolver: MpAccountResolverService,
           private readonly mpApi: MpApiClientService,
           private readonly crypto: MpTokenCryptoService,
         ) {}

         private get notificationBaseUrl(): string {
           return process.env.MP_NOTIFICATION_BASE_URL || 'https://newapi.coolsistema.com/api';
         }

         async createIntent(input: CreateMpQrDto): Promise<{ intentId: number; qrData: string; expiresAt: Date }> {
           // 1) Resolve mp_account (branch-first → store-level)
           const account = await this.resolver.resolveForScope(input.storeId, input.branchId ?? null);

           if (!account) {
             throw new BadRequestException({
               message: 'No hay cuenta Mercadopago vinculada para esta sucursal/tienda. Vinculá MP en Configuración › Mercadopago.',
               code: 'MP_NO_ACCOUNT',
             });
           }
           if (account.disconnectedAt) {
             throw new BadRequestException({
               message: 'La cuenta Mercadopago fue desconectada. Re-vinculá desde Configuración.',
               code: 'MP_ACCOUNT_DISCONNECTED',
             });
           }
           if (!account.externalPosId) {
             throw new BadRequestException({
               message: 'MP Store/POS no registrado — re-vinculá la cuenta MP.',
               code: 'MP_NO_EXTERNAL_POS_ID',
             });
           }

           // 2) Insert intent (status=pending) — get the intent.id BEFORE calling MP (external_reference)
           const expiresAt = new Date(Date.now() + QR_TTL_MS);
           const intent = await this.intentModel.create({
             mpAccountId: account.id,
             storeId: input.storeId,
             branchId: input.branchId ?? null,
             terminalId: input.terminalId,
             pendingVentaId: input.pendingVentaId,
             amount: input.amount,
             status: 'pending',
             expiresAt,
           });

           // 3) Call MP /instore/orders/qr/seller/collectors/{user_id}/pos/{external_pos_id}/qrs
           const accessToken = this.crypto.decrypt(account.accessToken);
           const path = MP_QR_PATH_TPL(account.mpUserId, account.externalPosId);
           // Per RESEARCH §Assumption A9, embed accountId in notification_url query (in case MP keeps query params)
           const notificationUrl = `${this.notificationBaseUrl}/mercadopago/webhook?accountId=${account.id}`;
           const description = input.description?.slice(0, 200) || `Venta Ventago #${input.pendingVentaId}`;

           let mpResponse: MpQrResponse;

           try {
             mpResponse = await this.mpApi.post<MpQrResponse>(
               path,
               {
                 external_reference: String(intent.id),
                 notification_url: notificationUrl,
                 total_amount: Number(input.amount),
                 description,
                 // 1개 묶음 아이템으로 보냄 (Phase 29는 라인별 분해 안 함)
                 items: [
                   {
                     sku_number: `intent-${intent.id}`,
                     category: 'marketplace',
                     title: description,
                     description,
                     unit_price: Number(input.amount),
                     quantity: 1,
                     unit_measure: 'unit',
                     total_amount: Number(input.amount),
                   },
                 ],
               },
               accessToken,
               { baseUrl: this.mpApi.hostFor(account.environment) },
             );
           } catch (err: any) {
             // MP API 실패 — intent 를 즉시 'failed' 로 마킹 (rollback path, per checker WARNING 8)
             // 대안 1 (DELETE)도 가능하지만 audit 보존 위해 status 마커 채택
             this.logger.error(`MP QR create failed for intent ${intent.id}: ${err?.message}`);
             try {
               await intent.update({ status: 'failed' });
             } catch (markErr: any) {
               // mark 실패 시에도 원본 에러는 throw (cron 의 stale-pending sweep 가 백업 cleanup)
               this.logger.error(`Also failed to mark intent ${intent.id} as failed: ${markErr?.message}`);
             }
             throw err;
           }

           // 4) Update intent with QR data
           await intent.update({
             qrData: mpResponse.qr_data,
             mpOrderId: mpResponse.in_store_order_id,
           });

           return {
             intentId: intent.id,
             qrData: mpResponse.qr_data,
             expiresAt,
           };
         }

         async cancelIntent(intentId: number, userId: number): Promise<{ ok: true }> {
           const intent = await this.intentModel.findByPk(intentId);

           if (!intent) throw new NotFoundException(`mp_payment_intent ${intentId} not found`);

           if (intent.status === 'approved') {
             this.logger.warn(`Intent ${intentId} already approved, cannot cancel (user=${userId})`);

             return { ok: true };
           }
           if (intent.status === 'cancelled' || intent.status === 'expired') {
             return { ok: true }; // already terminal
           }

           const account = await this.mpAccountModel.findByPk(intent.mpAccountId);

           if (account && account.externalPosId && !account.disconnectedAt) {
             try {
               const accessToken = this.crypto.decrypt(account.accessToken);
               const path = MP_QR_CANCEL_PATH_TPL(account.mpUserId, account.externalPosId);

               // MP는 PUT으로 현재 활성 order 를 clear (DELETE 도 일부 버전 가능 — 최신 PUT 권장)
               await this.mpApi.put(path, { external_reference: '' }, accessToken, {
                 baseUrl: this.mpApi.hostFor(account.environment),
               });
             } catch (err: any) {
               // MP cancel 실패해도 intent 는 cancel 처리 (UX 일관성)
               this.logger.warn(`MP cancel failed for intent ${intentId}: ${err?.message}`);
             }
           }

           await intent.update({ status: 'cancelled' });

           return { ok: true };
         }
       }
       ```
