```ts
const MRU_KEY = (userId: number, storeId: number) => `expense_category_mru_user_${userId}_store_${storeId}`;
const MRU_LIMIT = 5;

export function readMru(userId: number, storeId: number): number[] {
  try {
    const raw = localStorage.getItem(MRU_KEY(userId, storeId));

    return raw ? JSON.parse(raw) : [];
  } catch {
    return [];
  }
}

export function pushMru(userId: number, storeId: number, categoryId: number): void {
  const current = readMru(userId, storeId);
  const next = [categoryId, ...current.filter((id) => id !== categoryId)].slice(0, MRU_LIMIT);
  localStorage.setItem(MRU_KEY(userId, storeId), JSON.stringify(next));
}
```

```tsx
import { useEffect, useMemo, useRef, useState } from 'react';
import { Box, Button, Dialog, DialogActions, DialogContent, DialogTitle, IconButton, List, ListItemButton, ListItemText, Popover, TextField, Typography } from '@mui/material';
import { Icon } from '@iconify/react';
import { useExpenseCategoryTree } from 'src/hooks/api/useExpenseCategoryTree';
import { buildTreeFromFlat, multiKeywordPathMatch, TreeNode } from 'src/views/configuracion/categorias-gastos/utils/buildTree';
import { apiConnector } from 'src/services/api.service';
import { readMru, pushMru } from './CategoryTreeSelector.utils';

interface Props {
  value: number | null;
  onChange: (id: number | null, path: string | null) => void;
  userId: number;
  storeId: number;
  required?: boolean;
  label?: string;
  disabled?: boolean;
}

export const CategoryTreeSelector = ({ value, onChange, userId, storeId, required, label = 'Categoría', disabled }: Props) => {
  const [anchor, setAnchor] = useState<HTMLElement | null>(null);
  const [search, setSearch] = useState('');
  const { data: flat, mutate } = useExpenseCategoryTree(false);
  const inputRef = useRef<HTMLInputElement>(null);

  // MRU
  const mruIds = useMemo(() => readMru(userId, storeId), [userId, storeId, anchor /* 패널 열 때마다 재읽음 */]);
  const mruNodes = useMemo(
    () => mruIds.map((id) => flat.find((n) => n.id === id)).filter((n): n is NonNullable<typeof n> => !!n),
    [mruIds, flat],
  );

  // 검색 매칭 노드 (flat)
  const matched = useMemo(() => {
    if (!search.trim()) return null;

    return flat.filter((n) => {
      const haystack = `${n.path} ${n.name}`.toLowerCase();
      const tokens = search.toLowerCase().split(/\s+/).filter(Boolean);

      return tokens.every((t) => haystack.includes(t));
    });
  }, [search, flat]);

  const selectedNode = useMemo(() => flat.find((n) => n.id === value) ?? null, [flat, value]);

  // 선택 핸들러
  const handleSelect = (id: number, path: string) => {
    pushMru(userId, storeId, id);
    onChange(id, path);
    setAnchor(null);
    setSearch('');
  };

  // Inline create 다이얼로그
  const [createOpen, setCreateOpen] = useState(false);
  const [createParent, setCreateParent] = useState<number | null>(null);

  const handleInlineCreate = async (name: string, parentId: number | null) => {
    const res: any = await apiConnector.post('/expense-categories', { name, parentId });
    const newId = res?.data?.id ?? res?.id;
    const newPath = res?.data?.path ?? res?.path ?? name;
    await mutate();
    handleSelect(newId, newPath);
    setCreateOpen(false);
  };

  return (
    <>
      {/* 입력 표시 */}
      <Box
        sx={{
          border: '1px solid',
          borderColor: 'divider',
          borderRadius: 1,
          p: 1,
          cursor: disabled ? 'default' : 'pointer',
          minHeight: 40,
          display: 'flex',
          alignItems: 'center',
          opacity: disabled ? 0.6 : 1,
        }}
        onClick={(e) => !disabled && setAnchor(e.currentTarget)}
      >
        {selectedNode ? (
          <Typography variant="body2">{selectedNode.path}</Typography>
        ) : (
          <Typography variant="body2" color="text.secondary">
            {label}{required ? ' *' : ''}
          </Typography>
        )}
      </Box>

      {/* 패널 */}
      <Popover
        open={!!anchor}
        anchorEl={anchor}
        onClose={() => setAnchor(null)}
        anchorOrigin={{ vertical: 'bottom', horizontal: 'left' }}
        PaperProps={{ sx: { width: 360, maxHeight: 480 } }}
      >
        <Box sx={{ p: 1 }}>
          <TextField
            inputRef={inputRef}
            size="small"
            fullWidth
            autoFocus
            placeholder="Buscar (multi-palabra AND)..."
            value={search}
            onChange={(e) => setSearch(e.target.value)}
            onKeyDown={(e) => {
              if (e.nativeEvent.isComposing || e.keyCode === 229) return;
            }}
            InputProps={{ startAdornment: <Icon icon="tabler:search" width={16} /> }}
          />
        </Box>

        {/* MRU (검색 비어 있을 때만) */}
        {!search.trim() && mruNodes.length > 0 && (
          <>
            <Typography variant="caption" sx={{ px: 2, color: 'text.secondary' }}>
              Recientes
            </Typography>
            <List dense>
              {mruNodes.map((n) => (
                <ListItemButton key={`mru-${n.id}`} onClick={() => handleSelect(n.id, n.path)}>
                  <Icon icon="tabler:clock" width={14} style={{ marginRight: 8 }} />
                  <ListItemText primary={n.path} />
                </ListItemButton>
              ))}
            </List>
          </>
        )}

        {/* 검색 결과 또는 전체 트리 (평면 path 리스트) */}
        <Typography variant="caption" sx={{ px: 2, color: 'text.secondary' }}>
          {search.trim() ? 'Coincidencias' : 'Todas las categorías'}
        </Typography>
        <List dense sx={{ maxHeight: 280, overflow: 'auto' }}>
          {(matched ?? flat).map((n) => (
            <ListItemButton key={n.id} onClick={() => handleSelect(n.id, n.path)}>
              <Box sx={{ display: 'flex', alignItems: 'center', gap: 1, width: '100%' }}>
                <Box sx={{ width: 8, height: 8, borderRadius: '50%', bgcolor: n.color ?? '#cccccc' }} />
                <ListItemText primary={n.path} />
              </Box>
            </ListItemButton>
          ))}
        </List>

        {/* Inline create 버튼 */}
        {search.trim() && (
          <Button
            fullWidth
            startIcon={<Icon icon="tabler:plus" />}
            onClick={() => { setCreateParent(null); setCreateOpen(true); }}
            sx={{ borderTop: '1px solid', borderTopColor: 'divider', borderRadius: 0 }}
          >
            Crear &quot;{search}&quot; en...
          </Button>
        )}
      </Popover>

      {/* Inline create 다이얼로그 */}
      <Dialog open={createOpen} onClose={() => setCreateOpen(false)} fullWidth maxWidth="xs">
        <DialogTitle>Crear &quot;{search}&quot;</DialogTitle>
        <DialogContent>
          <Typography variant="body2" sx={{ mb: 1 }}>
            Seleccione el padre (o vacío para raíz):
          </Typography>
          <List dense sx={{ maxHeight: 280, overflow: 'auto' }}>
            <ListItemButton selected={createParent === null} onClick={() => setCreateParent(null)}>
              <ListItemText primary="(Raíz)" />
            </ListItemButton>
            {flat.filter((n) => n.depth < 5).map((n) => (
              <ListItemButton
                key={`parent-${n.id}`}
                selected={createParent === n.id}
                onClick={() => setCreateParent(n.id)}
              >
                <ListItemText primary={n.path} secondary={`depth ${n.depth} → ${n.depth + 1}`} />
              </ListItemButton>
            ))}
          </List>
        </DialogContent>
        <DialogActions>
          <Button onClick={() => setCreateOpen(false)}>Cancelar</Button>
          <Button variant="contained" onClick={() => handleInlineCreate(search, createParent)}>
            Crear
          </Button>
        </DialogActions>
      </Dialog>
    </>
  );
};
```

2) **Service 메서드** (`reportsGasto.service.ts` 에 추가):

```ts
async getByCategoryRollup(
  storeId: number,
  startDate: string,
  endDate: string,
  branchId: number | null,
  depth: string,
): Promise<Array<{ categoryId: number; categoryPath: string; totalAmount: number; expenseCount: number; depth: number }>> {
  // depth='all' → 5 와 동일하게 처리 (leaf 그대로)
  const depthLimit = depth === 'all' ? 5 : parseInt(depth, 10);

  const sql = `
    WITH RECURSIVE
    ec AS (
      SELECT id, parent_id, name, path, depth
        FROM expense_categories
       WHERE store_id = :storeId AND status = 1
    ),
    ancestor_chain AS (
      SELECT id AS leaf_id, id AS anc_id, depth AS anc_depth, path AS anc_path, name AS anc_name
        FROM ec
      UNION ALL
      SELECT ac.leaf_id, p.id, p.depth, p.path, p.name
        FROM ancestor_chain ac
        JOIN ec p ON p.id = (SELECT parent_id FROM ec WHERE id = ac.anc_id)
       WHERE ac.anc_depth > 0
    ),
    rollup_map AS (
      SELECT DISTINCT ON (leaf_id) leaf_id, anc_id AS rollup_id, anc_path AS rollup_path, anc_depth AS rollup_depth
        FROM ancestor_chain
       WHERE anc_depth <= :depthLimit
       ORDER BY leaf_id, anc_depth DESC
    )
    SELECT rm.rollup_id      AS "categoryId",
           rm.rollup_path    AS "categoryPath",
           rm.rollup_depth   AS "depth",
           COALESCE(SUM(e.amount), 0)::numeric AS "totalAmount",
           COUNT(*)::int     AS "expenseCount"
      FROM expenses e
      JOIN rollup_map rm ON rm.leaf_id = e.category_id
     WHERE e.store_id = :storeId
       AND (:branchId::int IS NULL OR e.branch_id = :branchId::int)
       AND e.date >= :startDate::date
       AND e.date <  (:endDate::date + INTERVAL '1 day')
     GROUP BY rm.rollup_id, rm.rollup_path, rm.rollup_depth
     ORDER BY "totalAmount" DESC
  `;

  const rows = await this.sequelize.query(sql, {
    replacements: { storeId, startDate, endDate, branchId, depthLimit },
    type: QueryTypes.SELECT,
  }) as any[];

  return rows.map((r) => ({
    categoryId: Number(r.categoryId),
    categoryPath: r.categoryPath,
    depth: Number(r.depth),
    totalAmount: Number(r.totalAmount),
    expenseCount: Number(r.expenseCount),
  }));
}
```

⚠️ 카테고리 없는 expenses (category_id = NULL) 는 이 결과에서 제외됨 (JOIN 으로 빠짐). 별도 row 로 표시하려면 추가 UNION 또는 클라이언트에서 따로 fetch — 일단 이번 wave 는 제외.

```tsx
import { useState } from 'react';
import useSWR from 'swr';
import { Card, CardContent, Typography, Box, Table, TableHead, TableBody, TableRow, TableCell } from '@mui/material';
import { apiConnector } from 'src/services/api.service';
import { DepthSelector } from './components/DepthSelector';

interface Row {
  categoryId: number;
  categoryPath: string;
  depth: number;
  totalAmount: number;
  expenseCount: number;
}

interface Props {
  startDate: string;
  endDate: string;
  branchId?: number | null;
}

export const GastosByCategoryReport = ({ startDate, endDate, branchId }: Props) => {
  const [depth, setDepth] = useState('2');
  const url = `/reports/gastos/by-category?startDate=${startDate}&endDate=${endDate}&depth=${depth}${branchId ? `&branchId=${branchId}` : ''}`;
  const { data, isLoading } = useSWR<Row[]>(url, (u: string) => apiConnector.get(u).then((r: any) => r.data ?? r));

  const total = (data ?? []).reduce((acc, r) => acc + Number(r.totalAmount), 0);

  return (
    <Card>
      <CardContent>
        <Box sx={{ display: 'flex', alignItems: 'center', gap: 2, mb: 2 }}>
          <Typography variant="h6" sx={{ flex: 1 }}>
            Gastos por categoría
          </Typography>
          <DepthSelector value={depth} onChange={setDepth} />
        </Box>

        {isLoading ? (
          <Typography>Cargando...</Typography>
        ) : (
          <Table size="small">
            <TableHead>
              <TableRow>
                <TableCell>Categoría</TableCell>
                <TableCell align="right">Cantidad</TableCell>
                <TableCell align="right">Total</TableCell>
                <TableCell align="right">%</TableCell>
              </TableRow>
            </TableHead>
            <TableBody>
              {(data ?? []).map((r) => (
                <TableRow key={r.categoryId}>
                  <TableCell sx={{ pl: 2 + r.depth * 2 }}>{r.categoryPath}</TableCell>
                  <TableCell align="right">{r.expenseCount}</TableCell>
                  <TableCell align="right">${Number(r.totalAmount).toLocaleString()}</TableCell>
                  <TableCell align="right">
                    {total ? ((Number(r.totalAmount) / total) * 100).toFixed(1) : 0}%
                  </TableCell>
                </TableRow>
              ))}
              <TableRow>
                <TableCell><strong>Total</strong></TableCell>
                <TableCell />
                <TableCell align="right"><strong>${total.toLocaleString()}</strong></TableCell>
                <TableCell align="right"><strong>100%</strong></TableCell>
              </TableRow>
            </TableBody>
          </Table>
        )}
      </CardContent>
    </Card>
  );
};
```
