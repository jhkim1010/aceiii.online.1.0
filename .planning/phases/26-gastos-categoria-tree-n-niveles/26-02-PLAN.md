```ts
@Injectable()
export class ExpenseCategoryService {
  constructor(
    @InjectModel(ExpenseCategory) private model: typeof ExpenseCategory,
    @InjectConnection() private sequelize: Sequelize,
  ) {}

  async getTree(storeId: number, includeArchived = false): Promise<ExpenseCategory[]> {
    return this.model.findAll({
      where: { storeId, ...(includeArchived ? {} : { status: 1 }) },
      order: [['depth', 'ASC'], ['sortOrder', 'ASC'], ['name', 'ASC']],
    });
  }

  async create(storeId: number, dto: CreateExpenseCategoryDto): Promise<ExpenseCategory> {
    // 부모 검증
    let parentDepth = -1;
    if (dto.parentId) {
      const parent = await this.model.findOne({ where: { id: dto.parentId, storeId } });
      if (!parent) throw new BadRequestException('parentId 가 존재하지 않거나 다른 매장 소속');
      if (parent.status === 0) throw new BadRequestException('archived 부모 밑에 자식 추가 불가');
      parentDepth = parent.depth;
      if (parentDepth + 1 > 5) throw new BadRequestException('depth 5 초과 — 새 자식 추가 불가');
    }

    // 동명 형제 사전 검증
    const sibling = await this.model.findOne({
      where: {
        storeId,
        parentId: dto.parentId ?? null,
        name: dto.name,
      },
    });
    if (sibling) throw new BadRequestException('같은 부모 밑에 동명 카테고리가 이미 존재');

    try {
      return await this.model.create({
        storeId,
        parentId: dto.parentId ?? null,
        name: dto.name,
        sortOrder: dto.sortOrder ?? 0,
        color: dto.color ?? null,
        icon: dto.icon ?? null,
        status: 1,
      } as any);
    } catch (e: any) {
      // UNIQUE 위반 등 트리거/제약 에러는 400 으로 변환
      throw new BadRequestException(e?.message ?? '카테고리 생성 실패');
    }
  }

  // rename, sort 동일 패턴

  async move(storeId: number, id: number, newParentId: number | null): Promise<ExpenseCategory> {
    const node = await this.model.findOne({ where: { id, storeId } });
    if (!node) throw new NotFoundException('노드 없음');

    // 1) 사이클 검증
    if (newParentId !== null) {
      if (newParentId === id) throw new BadRequestException('자기 자신을 부모로 지정 불가');
      const target = await this.model.findOne({ where: { id: newParentId, storeId } });
      if (!target) throw new BadRequestException('새 부모가 존재하지 않거나 다른 매장');
      // descendant 인지: target.path LIKE node.path || ' > %'
      if (target.path === node.path || target.path.startsWith(node.path + ' > ')) {
        throw new BadRequestException('자기 자손을 부모로 지정 불가 (사이클)');
      }
    }

    // 2) 깊이 검증
    const newOwnDepth = newParentId === null ? 0 : (await this.model.findByPk(newParentId))!.depth + 1;
    // 자신 + 자손의 최대 depth 확장
    const subtree = await this.model.findAll({
      where: {
        storeId,
        path: { [Op.or]: [node.path, { [Op.like]: node.path + ' > %' }] },
      },
      attributes: ['depth'],
    });
    const maxOldDepth = Math.max(...subtree.map((n) => n.depth));
    const shift = newOwnDepth - node.depth;
    if (maxOldDepth + shift > 5) {
      throw new BadRequestException(`subtree 이동 시 깊이 5 초과 (예상 max depth ${maxOldDepth + shift})`);
    }

    // 3) UPDATE — 트리거가 path 자동 갱신 + 자손 캐스케이드
    node.parentId = newParentId;
    await node.save();

    return node.reload();
  }

  async softDelete(storeId: number, id: number, policy: 'promote' | 'move' | 'cascade', moveTo?: number) {
    const t = await this.sequelize.transaction();
    try {
      const node = await this.model.findOne({ where: { id, storeId }, transaction: t });
      if (!node) throw new NotFoundException();

      const children = await this.model.findAll({ where: { storeId, parentId: id, status: 1 }, transaction: t });

      if (policy === 'cascade') {
        // 자신 + 모든 자손 status=0
        await this.model.update(
          { status: 0 },
          {
            where: {
              storeId,
              [Op.or]: [{ id }, { path: { [Op.like]: node.path + ' > %' } }],
            },
            transaction: t,
          },
        );
      } else if (policy === 'promote') {
        // 자식들의 parentId = node.parentId (grandparent). 트리거가 path 자동 재계산.
        for (const child of children) {
          // 깊이 검증: node.parentId 의 depth + child subtree max - child.depth 가 5 초과면 거부
          // 단순화: 한 단계 promote 는 무조건 안전 (depth -1)
          child.parentId = node.parentId;
          await child.save({ transaction: t });
        }
        await node.update({ status: 0 }, { transaction: t });
      } else if (policy === 'move') {
        if (!moveTo) throw new BadRequestException('policy=move 일 때 moveTo 필수');
        // moveTo 가 자신 또는 자손이면 거부
        const target = await this.model.findOne({ where: { id: moveTo, storeId }, transaction: t });
        if (!target) throw new BadRequestException('moveTo 노드 없음');
        if (target.path === node.path || target.path.startsWith(node.path + ' > ')) {
          throw new BadRequestException('자손을 moveTo 로 지정 불가');
        }
        // 깊이 검증 후 자식들 모두 target 으로 이동
        for (const child of children) {
          const newDepth = target.depth + 1;
          const childSubtree = await this.model.findAll({
            where: { storeId, path: { [Op.or]: [child.path, { [Op.like]: child.path + ' > %' }] } },
            attributes: ['depth'],
            transaction: t,
          });
          const maxDepth = Math.max(...childSubtree.map((n) => n.depth));
          const shift = newDepth - child.depth;
          if (maxDepth + shift > 5) {
            throw new BadRequestException('자식 이동 시 깊이 5 초과');
          }
          child.parentId = target.id;
          await child.save({ transaction: t });
        }
        await node.update({ status: 0 }, { transaction: t });
      }

      await t.commit();
    } catch (e) {
      await t.rollback();
      throw e;
    }
  }

  async restore(storeId: number, id: number) {
    const node = await this.model.findOne({ where: { id, storeId } });
    if (!node) throw new NotFoundException();
    // 부모가 archived 면 거부 (orphan 방지)
    if (node.parentId) {
      const parent = await this.model.findByPk(node.parentId);
      if (!parent || parent.status === 0) {
        throw new BadRequestException('부모가 archived 상태 — 먼저 부모 복원 필요');
      }
    }
    await node.update({ status: 1 });

    return node;
  }

  async getInUseCount(storeId: number, categoryId: number): Promise<{ count: number; recentMonth: number }> {
    const [{ count }] = await this.sequelize.query(
      'SELECT COUNT(*)::int AS count FROM expenses WHERE store_id = :storeId AND category_id = :categoryId',
      { replacements: { storeId, categoryId }, type: QueryTypes.SELECT },
    ) as any;

    const [{ count: recentMonth }] = await this.sequelize.query(
      `SELECT COUNT(*)::int AS count FROM expenses
        WHERE store_id = :storeId AND category_id = :categoryId
          AND date >= date_trunc('month', NOW())`,
      { replacements: { storeId, categoryId }, type: QueryTypes.SELECT },
    ) as any;

    return { count, recentMonth };
  }
}
```

```ts
import { Injectable } from '@nestjs/common';
import { InjectModel } from '@nestjs/sequelize';
import { Transaction } from 'sequelize';
import { ExpenseCategory } from './expense-category.model';

const DEFAULT_CATEGORIES = [
  { name: 'Servicios',   color: '#3498db', icon: 'tabler:bolt' },
  { name: 'Comida',      color: '#e67e22', icon: 'tabler:tools-kitchen-2' },
  { name: 'Transporte',  color: '#2ecc71', icon: 'tabler:bus' },
  { name: 'Insumos',     color: '#9b59b6', icon: 'tabler:package' },
  { name: 'Sueldos',     color: '#f1c40f', icon: 'tabler:cash' },
  { name: 'Otros',       color: '#95a5a6', icon: 'tabler:dots' },
];

@Injectable()
export class ExpenseCategorySeedService {
  constructor(@InjectModel(ExpenseCategory) private model: typeof ExpenseCategory) {}

  async seedDefaults(storeId: number, transaction?: Transaction): Promise<void> {
    const existing = await this.model.count({ where: { storeId }, transaction });

    // 이미 카테고리가 하나라도 있으면 시드하지 않음 (idempotent)
    if (existing > 0) {
      console.log(`ℹ️ store ${storeId} 에 이미 expense_categories 가 ${existing}개 존재 — 시드 skip`);

      return;
    }

    for (let i = 0; i < DEFAULT_CATEGORIES.length; i += 1) {
      const cat = DEFAULT_CATEGORIES[i];
      await this.model.create(
        {
          storeId,
          parentId: null,
          name: cat.name,
          sortOrder: i,
          color: cat.color,
          icon: cat.icon,
          status: 1,
        } as any,
        { transaction },
      );
    }

    console.log(`✅ store ${storeId} 에 6개 기본 expense_categories 시드 완료`);
  }
}
```
