// ventago-app/src/views/configuracion/categorias-gastos/utils/buildTree.ts
import { ExpenseCategoryDto } from 'src/hooks/api/useExpenseCategoryTree';

export interface TreeNode {
  id: string;            // react-arborist 가 string 권장 — `${dto.id}` 변환
  name: string;
  path: string;
  depth: number;
  sortOrder: number;
  color: string | null;
  icon: string | null;
  status: number;
  parentId: number | null;
  rawId: number;         // 백엔드 호출용 원본 INT id
  children?: TreeNode[];
}

// flat 배열 → 중첩 트리 (O(N))
export function buildTreeFromFlat(flat: ExpenseCategoryDto[]): TreeNode[] {
  const byId = new Map<number, TreeNode>();
  flat.forEach((dto) => {
    byId.set(dto.id, {
      id: String(dto.id),
      name: dto.name,
      path: dto.path,
      depth: dto.depth,
      sortOrder: dto.sortOrder,
      color: dto.color,
      icon: dto.icon,
      status: dto.status,
      parentId: dto.parentId,
      rawId: dto.id,
      children: [],
    });
  });

  const roots: TreeNode[] = [];
  byId.forEach((node) => {
    if (node.parentId === null) {
      roots.push(node);
    } else {
      const parent = byId.get(node.parentId);
      if (parent) {
        parent.children!.push(node);
      }
    }
  });

  // 자식 정렬 (서버에서 이미 정렬했지만 안전하게)
  const sortRecursive = (nodes: TreeNode[]) => {
    nodes.sort((a, b) => a.sortOrder - b.sortOrder || a.name.localeCompare(b.name));
    nodes.forEach((n) => n.children && sortRecursive(n.children));
  };
  sortRecursive(roots);

  return roots;
}

// 다중 키워드 AND 매칭 (path 전체 + name)
export function multiKeywordPathMatch(node: { data: TreeNode }, term: string): boolean {
  if (!term.trim()) return true;
  const haystack = `${node.data.path} ${node.data.name}`.toLowerCase();
  const tokens = term.toLowerCase().split(/\s+/).filter(Boolean);

  return tokens.every((t) => haystack.includes(t));
}

// 깊이 5 초과 후보 필터 (Move dialog 에서 사용)
export function getValidMoveTargets(
  flat: ExpenseCategoryDto[],
  movingNodeId: number,
): ExpenseCategoryDto[] {
  const movingNode = flat.find((n) => n.id === movingNodeId);
  if (!movingNode) return [];

  // 자손 path 수집
  const descendantPaths = flat
    .filter((n) => n.path === movingNode.path || n.path.startsWith(movingNode.path + ' > '))
    .map((n) => n.path);

  // movingNode 의 subtree max depth 계산
  const subtreeMaxDepth = Math.max(...flat
    .filter((n) => n.path === movingNode.path || n.path.startsWith(movingNode.path + ' > '))
    .map((n) => n.depth));

  return flat.filter((target) => {
    if (target.id === movingNodeId) return false;
    if (descendantPaths.includes(target.path)) return false;
    if (target.status !== 1) return false;
    // target.depth + 1 + (subtreeMaxDepth - movingNode.depth) <= 5 인지
    const projectedMaxDepth = target.depth + 1 + (subtreeMaxDepth - movingNode.depth);

    return projectedMaxDepth <= 5;
  });
}

```tsx
import { useMemo, useState } from 'react';
import { Tree, NodeApi } from 'react-arborist';
import { Box, Card, CardContent, FormControlLabel, Switch, TextField, Typography, Button } from '@mui/material';
import { Icon } from '@iconify/react';
import { useExpenseCategoryTree } from 'src/hooks/api/useExpenseCategoryTree';
import { buildTreeFromFlat, multiKeywordPathMatch, TreeNode } from './utils/buildTree';
import { CategoryNode } from './components/CategoryNode';
import { CreateNodeDialog } from './components/CreateNodeDialog';
import { MoveNodeDialog } from './components/MoveNodeDialog';
import { DeleteNodeDialog } from './components/DeleteNodeDialog';
import { apiConnector } from 'src/services/api.service';

export const CategoriasGastosTreeView = () => {
  const [showArchived, setShowArchived] = useState(false);
  const [search, setSearch] = useState('');
  const { data: flat, isLoading, mutate } = useExpenseCategoryTree(showArchived);

  const treeData = useMemo(() => buildTreeFromFlat(flat), [flat]);

  // Dialog 상태
  const [createOpen, setCreateOpen] = useState(false);
  const [createParent, setCreateParent] = useState<TreeNode | null>(null);
  const [moveOpen, setMoveOpen] = useState(false);
  const [moveTarget, setMoveTarget] = useState<TreeNode | null>(null);
  const [deleteOpen, setDeleteOpen] = useState(false);
  const [deleteTarget, setDeleteTarget] = useState<TreeNode | null>(null);

  // react-arborist onMove → 같은 부모 내에서만 호출됨 (disableDrop 가드)
  // dragIds 의 새 sortOrder 를 PUT /:id/sort 로 일괄 호출
  const handleMove = async ({ dragIds, parentId, index }: any) => {
    // 같은 부모의 자식들 순서대로 sortOrder 재할당
    // index = 새 위치
    // 형제들 fetch → 새 순서 계산 → 각 노드에 PUT /:id/sort
    for (let i = 0; i < dragIds.length; i += 1) {
      const dragId = dragIds[i];
      await apiConnector.put(`/expense-categories/${dragId}/sort`, { sortOrder: index + i });
    }
    await mutate();
  };

  const handleRename = async ({ id, name }: { id: string; name: string }) => {
    await apiConnector.put(`/expense-categories/${id}`, { name });
    await mutate();
  };

  return (
    <Box>
      <Card sx={{ borderTop: '4px solid', borderTopColor: 'primary.main' }}>
        <CardContent>
          <Box sx={{ display: 'flex', alignItems: 'center', gap: 2, mb: 2 }}>
            <Typography variant="h6" sx={{ flex: 1 }}>
              Categorías de Gastos
            </Typography>
            <Button
              startIcon={<Icon icon="tabler:plus" />}
              variant="contained"
              onClick={() => { setCreateParent(null); setCreateOpen(true); }}
            >
              Nueva categoría raíz
            </Button>
          </Box>

          <Box sx={{ display: 'flex', alignItems: 'center', gap: 2, mb: 2 }}>
            <TextField
              size="small"
              placeholder="Buscar (multi-palabra AND)..."
              value={search}
              onChange={(e) => setSearch(e.target.value)}
              onKeyDown={(e) => {
                // 한글 IME 합성 중에는 검색 트리거 무시 (텍스트 변화는 onChange 가 처리)
                if (e.nativeEvent.isComposing || e.keyCode === 229) return;
              }}
              sx={{ flex: 1 }}
              InputProps={{ startAdornment: <Icon icon="tabler:search" width={16} /> }}
            />
            <FormControlLabel
              control={<Switch checked={showArchived} onChange={(e) => setShowArchived(e.target.checked)} />}
              label="Mostrar archivados"
            />
          </Box>

          {isLoading ? (
            <Typography>Cargando...</Typography>
          ) : (
            <Tree<TreeNode>
              data={treeData}
              openByDefault={false}
              width="100%"
              height={600}
              rowHeight={36}
              indent={24}
              searchTerm={search}
              searchMatch={multiKeywordPathMatch}
              disableDrop={({ parentNode, dragNodes }) => {
                // 같은 부모 내에서만 정렬 허용 — 다른 부모로 D&D 차단
                if (!parentNode) return true;
                const dragParentId = dragNodes[0].parent?.id ?? null;
                const dropParentId = parentNode.id;

                return dragParentId !== dropParentId;
              }}
              onMove={handleMove}
              onRename={handleRename}
            >
              {(props) => (
                <CategoryNode
                  {...props}
                  onAddChild={(parent) => { setCreateParent(parent); setCreateOpen(true); }}
                  onRequestMove={(node) => { setMoveTarget(node); setMoveOpen(true); }}
                  onRequestDelete={(node) => { setDeleteTarget(node); setDeleteOpen(true); }}
                  onRequestRestore={async (node) => {
                    await apiConnector.put(`/expense-categories/${node.rawId}/restore`, {});
                    await mutate();
                  }}
                />
              )}
            </Tree>
          )}
        </CardContent>
      </Card>

      {/* Dialogs */}
      <CreateNodeDialog
        open={createOpen}
        parent={createParent}
        onClose={() => setCreateOpen(false)}
        onCreated={() => { mutate(); setCreateOpen(false); }}
      />
      {moveTarget && (
        <MoveNodeDialog
          open={moveOpen}
          node={moveTarget}
          allFlatNodes={flat}
          onClose={() => setMoveOpen(false)}
          onMoved={() => { mutate(); setMoveOpen(false); }}
        />
      )}
      {deleteTarget && (
        <DeleteNodeDialog
          open={deleteOpen}
          node={deleteTarget}
          allFlatNodes={flat}
          onClose={() => setDeleteOpen(false)}
          onDeleted={() => { mutate(); setDeleteOpen(false); }}
        />
      )}
    </Box>
  );
};
```
