```
  INSERT INTO expense_categories (store_id, parent_id, name, sort_order, status)
  SELECT store_id, NULL, name, 0,
         CASE WHEN COALESCE(status, 1) = 0 THEN 0 ELSE 1 END
    FROM expenses_categories

     SELECT 1 FROM expense_categories ec
      WHERE ec.store_id = expenses_categories.store_id
```

RESEARCH §3.4 의 5개 검증 쿼리를 그대로 SQL 파일로 만든다. 모두 SELECT only — 쓰기 없음.

내용:
```
-- 1) 루트 카테고리 수가 기존 categories 와 일치
SELECT 'roots_match' AS check_name,
  (SELECT COUNT(*) FROM expenses_categories) AS old_count,
  (SELECT COUNT(*) FROM expense_categories WHERE parent_id IS NULL) AS new_count;

-- 2) 자식 노드 수가 기존 subcategories 와 일치
SELECT 'children_match' AS check_name,
  (SELECT COUNT(*) FROM expenses_subcategories) AS old_count,
  (SELECT COUNT(*) FROM expense_categories WHERE parent_id IS NOT NULL) AS new_count;

-- 3) orphan expenses (category_id가 가리키는 노드가 없는 경우) — 0 이어야 함
SELECT 'orphan_expenses' AS check_name, COUNT(*) AS orphan_count
  FROM expenses e
  LEFT JOIN expense_categories ec ON ec.id = e.category_id
 WHERE e.category_id IS NOT NULL AND ec.id IS NULL;

-- 4) path/depth 무결성: depth 범위 + parent NULL ↔ depth=0 일관성 — 0 행이어야 함
SELECT 'path_depth_integrity' AS check_name, COUNT(*) AS bad_rows
  FROM expense_categories
 WHERE depth > 5 OR depth < 0
    OR (parent_id IS NULL AND depth <> 0)
    OR (parent_id IS NOT NULL AND depth = 0)
    OR path = '';

-- 5) 사이클 검출: 모든 노드가 루트로부터 도달 가능
WITH RECURSIVE walk(id, parent_id, depth_walked) AS (
  SELECT id, parent_id, 0 FROM expense_categories WHERE parent_id IS NULL
  UNION ALL
  SELECT c.id, c.parent_id, w.depth_walked + 1
    FROM expense_categories c JOIN walk w ON c.parent_id = w.id
   WHERE w.depth_walked < 10
)
SELECT 'cycle_check' AS check_name, COUNT(*) AS unreachable
  FROM expense_categories WHERE id NOT IN (SELECT id FROM walk);

-- 6) 매핑 테이블이 모든 옛 카테고리/서브카테고리를 포함하는지
SELECT 'map_coverage' AS check_name,
  (SELECT COUNT(*) FROM expenses_categories) +
```
