```typescript
async completeWorkOrder(
  workOrderId: number,
  producedQuantity: number,
  storeId?: number | null,
  isSuperAdmin = false,
  defectQuantity = 0,
  operatorId?: number,
): Promise<ProductionResult> {
  return this.sequelize.transaction(async (t) => {
    // [Phase 64 W4] 판매 경로와 동일한 대기 상한
    await this.sequelize.query(
      `SET LOCAL lock_timeout = '3s'; SET LOCAL statement_timeout = '15s'`,
      { transaction: t },
    );

    // [Phase 64 W4] ★ work order 를 잠그고 트랜잭션 안에서 상태를 재검사한다.
    // 종전에는 락 없이 status 를 봐서 동시 2회 완료가 원자재를 두 번 차감할 수 있었다.
    const workOrder = await this.workOrderModel.findByPk(workOrderId, {
      include: [{ model: Product, as: 'product' }],
      lock: Transaction.LOCK.UPDATE,
      transaction: t,
    });
    if (!workOrder) throw new NotFoundException(...);
    if (!isSuperAdmin && storeId && workOrder.storeId !== storeId) throw new NotFoundException(...);
    if (workOrder.status === WorkOrderStatus.COMPLETED) {
      throw new BadRequestException('La orden de trabajo ya está completada');
    }

    const branchId = await this.resolveProductionBranchId(workOrder, t);  // Task 1 규칙

    const bom = await this.bomModel.findOne({
      where: { productId: workOrder.productId, isActive: true },
      include: [{ model: BomItem, as: 'items' }],
      transaction: t,
    });
    if (!bom) throw new NotFoundException(`BOM no encontrado para el producto ${workOrder.productId}`);

    // ── 소비 대상 수집 후 productId 오름차순 정렬 (판매 경로와 같은 락 순서 = 교착 없음) ──
    const consumptions: { productId: number; qty: number }[] = [];
    for (const item of bom.items) { /* materialId → sku 매핑 / subProductId */ }
    consumptions.sort((a, b) => a.productId - b.productId);

    // ── 원자재 출고: 원장 INSERT + 잔액 캐시 감소 ──
    for (const c of consumptions) {
      const pb = await this.findOrCreateProductBranch(c.productId, branchId, t);
      await this.stockModel.create(
        { productBranchId: pb.id, stock: -Math.abs(c.qty), type: 'produccion',
          note: `consumo OT#${workOrder.id}` },
        { transaction: t },
      );
      await this.productModel.decrement('stock', { by: c.qty, where: { id: c.productId }, transaction: t });
    }

    // ── 완제품 입고: 원장 INSERT + 잔액 캐시 증가 ──
    const pbOut = await this.findOrCreateProductBranch(workOrder.productId, branchId, t);
    await this.stockModel.create(
      { productBranchId: pbOut.id, stock: Math.abs(producedQuantity), type: 'produccion',
        note: `ingreso OT#${workOrder.id}` },
      { transaction: t },
    );
    await this.productModel.increment('stock', {
      by: producedQuantity, where: { id: workOrder.productId }, transaction: t,
    });

    const productionResult = await this.productionResultModel.create({ /* 기존 필드 */ }, { transaction: t });
    await workOrder.update(
      { status: WorkOrderStatus.COMPLETED, completedDate: new Date() },
      { transaction: t },
    );

    return productionResult;
  });
}
```
