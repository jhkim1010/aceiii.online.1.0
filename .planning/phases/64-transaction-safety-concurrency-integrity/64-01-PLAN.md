CREATE TABLE IF NOT EXISTS sale_idempotency_keys (
  id              SERIAL       PRIMARY KEY,
  store_id        INTEGER      NOT NULL REFERENCES stores(id) ON DELETE CASCADE,

```typescript
// 판매 요청 멱등 처리 (Phase 64 W1).
// claim 은 반드시 판매 트랜잭션과 같은 transaction 으로 호출한다 —
// 그래야 "판매는 커밋됐는데 키 행이 없다"가 구조적으로 불가능해진다.
async claim(
  storeId: number,
  key: string,
  requestHash: string,
  t: Transaction,
): Promise<ClaimResult> {
  const [rows] = await this.sequelize.query(
    `INSERT INTO sale_idempotency_keys (store_id, idempotency_key, request_hash, status)
     VALUES ($1, $2, $3, 'processing')
     ON CONFLICT (store_id, idempotency_key) DO NOTHING
     RETURNING id`,
    { bind: [storeId, key, requestHash], transaction: t },
  );

  // 삽입 성공 = 이 요청이 최초 → 그대로 판매 생성 진행
  if ((rows as unknown[]).length > 0) return { outcome: 'proceed' };

  // 삽입 실패 = 이미 같은 키가 있음 → 기존 행 상태로 판단
  const existing = await this.model.findOne({
    where: { storeId, idempotencyKey: key },
    transaction: t,
  });
  if (!existing) return { outcome: 'proceed' }; // 만료 정리와 경합 — 재시도 허용

  if (existing.requestHash !== requestHash) return { outcome: 'conflict' };
  if (existing.status === 'completed') {
    return { outcome: 'replay', responseBody: existing.responseBody, saleId: existing.saleId };
  }
  if (existing.status === 'failed') return { outcome: 'proceed' };

  return { outcome: 'in_flight' };
}
```

`complete(storeId, key, saleId, responseBody)` — 커밋 후 호출, `status='completed'` + 본문 저장.
`markFailed(storeId, key)` — 판매 트랜잭션 롤백 시 호출(같은 키 재시도 허용).
`hashRequest(dto)` — 정규화(키 정렬) 후 SHA-256. `Idempotency-Key`/타임스탬프성 필드는 해시 대상에서 제외.
`purgeExpired()` — `DELETE FROM sale_idempotency_keys WHERE expires_at < NOW()`.
