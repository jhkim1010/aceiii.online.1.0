```typescript
// [Phase 64 W7] 원장은 추가만 된다(append-only).
// 종전에는 행을 UPDATE/DELETE 해서 products.stock 캐시와 영구히 어긋났다.
// 잘못된 이동은 반대 부호 보정 행으로 상쇄하고, 캐시도 같은 트랜잭션에서 맞춘다.
async adjust(stockId: number, reason: string, userId: number): Promise<Stocks> {
  return this.sequelize.transaction(async (t) => {
    const origin = await this.stockModel.findByPk(stockId, {
      lock: Transaction.LOCK.UPDATE,
      transaction: t,
    });
    if (!origin) throw new NotFoundException('Stock movement not found');

    const pb = await ProductBranch.findByPk(origin.productBranchId, { transaction: t });
    const compensation = await this.stockModel.create(
      {
        productBranchId: origin.productBranchId,
        stock: -Number(origin.stock),
        type: 'ajuste',
        note: `ajuste de mov#${origin.id} por user#${userId}: ${reason}`.slice(0, 500),
      },
      { transaction: t },
    );

    // 잔액 캐시도 같은 트랜잭션에서 조정 — 원장과 캐시가 어긋나지 않는다.
    await this.productModel.increment('stock', {
      by: -Number(origin.stock), where: { id: pb.productId }, transaction: t,
    });

    return compensation;
  });
}
```
