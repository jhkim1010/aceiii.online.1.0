```typescript
// [Phase 64 W6] 수신 원장 착지 = 멱등 게이트.
// 종전에는 findOne 으로 존재를 확인한 뒤 create 했다 — 그 사이 같은 uuid 가 들어오면
// UNIQUE 위반이 그대로 500 이 됐다(edge 는 성공/실패를 구분 못 함).
// ON CONFLICT DO NOTHING 은 "삽입에 성공한 요청만 적용을 수행한다"는 규칙을 한 문장으로 만든다.
const [inserted] = await this.sequelize.query<{ id: number }>(
  `INSERT INTO offline_sync_ops
     (op_uuid, agent_id, branch_id, store_id, op_type, payload, status, original_at, offline_number, received_at)
   VALUES ($1, $2, $3, $4, $5, $6::jsonb, 'received', $7, $8, NOW())
   ON CONFLICT (op_uuid) DO NOTHING
   RETURNING id`,
  { bind: [...], type: QueryTypes.SELECT },
);

let record: OfflineSyncOp;
if (inserted?.id) {
  record = (await this.opModel.findByPk(inserted.id)) as OfflineSyncOp;
} else {
  // 삽입 실패 = 이미 수신된 uuid
  const existing = await this.opModel.findOne({ where: { opUuid: op.uuid } });
  if (!existing) {
    // 극히 드문 경합(정리 작업 등) — 재시도 가능한 오류로 응답
    return { uuid: op.uuid, status: 'error', resultId: null, error: 'conflicto transitorio, reintente' };
  }

  if (existing.status === 'applied') {
    return { uuid: op.uuid, status: 'duplicate', resultId: existing.resultId ? Number(existing.resultId) : null };
  }
  if (existing.status === 'error') {
    return { uuid: op.uuid, status: 'error', resultId: null, error: existing.error || undefined };
  }

  // [Phase 64 W6] ★ 'received' 는 종결 상태가 아니다 — 적용이 끝나지 않았다는 뜻이다.
  // 종전에는 이걸 duplicate 로 돌려줘서, insert 후 크래시한 판매가 영원히 생성되지 않았다.
  // 재적용은 안전하다: 판매 생성이 offline:{uuid} 멱등키를 쓰므로 판매는 1건을 넘지 않는다.
  record = existing;
}
```
이후 기존 `:129`~ 적용 로직으로 진행하되, 성공/실패 시 `record.update({ status, resultId, error })` 로 상태를 확정한다.

주의:
- `QueryTypes` / `sequelize` 주입이 없으면 추가한다.
- 컬럼명은 **snake_case**(raw SQL). 모델 속성명(camelCase)과 혼동 금지.
- `payload` 는 `$6::jsonb` 로 캐스팅.
- 배치 상한 50, 순차 처리, per-op try/catch 는 **그대로 유지**한다.
