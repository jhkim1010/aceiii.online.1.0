```typescript
async create(dto: CreateSuspendedSaleDto, userId: number): Promise<SuspendedSale | null> {
  const resolvedBranchId = await this.resolveBranchId(dto.branchId, dto.storeId); // 읽기 — 트랜잭션 전

  const suspendedId = await this.sequelize.transaction(async (t) => {
    // [Phase 64 W3] 판매 생성과 동일한 대기 상한 (락 대기가 pgbouncer 슬롯을 먹지 않게)
    await this.sequelize.query(
      `SET LOCAL lock_timeout = '3s'; SET LOCAL statement_timeout = '15s'`,
      { transaction: t },
    );

    const suspended = await this.suspendedSaleModel.create({ /* 기존 필드 */ }, { transaction: t });
    if (dto.items?.length > 0) {
      await this.itemModel.bulkCreate(/* 기존 매핑 */, { transaction: t });
      await this.recordReservationMoves(dto.items, resolvedBranchId, -1, t);
    }
    if (dto.discounts?.length) await this.discountModel.bulkCreate(..., { transaction: t });
    if (dto.recharges?.length) await this.rechargeModel.bulkCreate(..., { transaction: t });

    return suspended.id;
  });

  return this.findOne(suspendedId);   // 커밋 후 조회 — 응답 형식 불변
}
```

**A. `update`**:
```typescript
async update(id: number, dto: UpdateSuspendedSaleDto, userId: number) {
  await this.sequelize.transaction(async (t) => {
    await this.sequelize.query(
      `SET LOCAL lock_timeout = '3s'; SET LOCAL statement_timeout = '15s'`,
      { transaction: t },
    );

    // [Phase 64 W3] ★ 헤더를 잠근다 — 같은 보류 판매를 두 창에서 동시에 수정/삭제하는 경합을 직렬화.
    const existing = await this.suspendedSaleModel.findByPk(id, {
      lock: Transaction.LOCK.UPDATE,
      transaction: t,
    });
    if (!existing) throw new NotFoundException(...);

    // 구 아이템 조회 → 예약 +1 복원 → 헤더 update → destroy ×3 → 재생성 → 예약 -1
    // 전부 { transaction: t }
  });

  return this.findOne(id);
}
```
