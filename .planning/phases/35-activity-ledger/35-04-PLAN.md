```typescript
// Phase 35 — ventaVista Resumen 테이블 데이터 source SWR 훅
// 백엔드: GET /sales/daily-stats?startDate=Y-m-d&endDate=Y-m-d
// 응답: { date, perBranch[], total, movBalance }
// 캐시: 5분 dedup (CLAUDE.md SWR 규약)

import useSWR from 'swr'
import apiConnector from 'src/services/api.service'

export interface DailyStatsBranchRow {
  branchId: number
  branchName: string
  ventas: { count: number; amount: number }
  prendas: number
  descuento: number
  movIn: number
  movOut: number
  fallados: number
  neto: number
}

export interface DailyStatsResponse {
  date: string
  perBranch: DailyStatsBranchRow[]
  total: {
    ventas: { count: number; amount: number }
    prendas: number
    descuento: number
    movIn: number
    movOut: number
    fallados: number
    neto: number
  }
  movBalance: { in: number; out: number; balanced: boolean }
}

const fetcher = async (key: string): Promise<DailyStatsResponse> => {
  // key 포맷: "/sales/daily-stats?startDate=...&endDate=..."
  const [path, qs] = key.split('?')
  const params: Record<string, string> = {}
  if (qs) {
    new URLSearchParams(qs).forEach((v, k) => { params[k] = v })
  }

  return apiConnector.get(path, params) as Promise<DailyStatsResponse>
}

interface UseDailySalesStatsArgs {
  startDate: string | null
  endDate: string | null
  enabled?: boolean
}

export const useDailySalesStats = ({
  startDate,
  endDate,
  enabled = true,
}: UseDailySalesStatsArgs) => {
  const key =
    enabled && startDate && endDate
      ? `/sales/daily-stats?startDate=${startDate}&endDate=${endDate}`
      : null

  return useSWR<DailyStatsResponse>(key, fetcher, {
    dedupingInterval: 300000, // 5분
    revalidateOnFocus: false,
    revalidateOnReconnect: false,
  })
}
```

```tsx
// Phase 35 — ventaVista 상단 Resumen 테이블 (KPI strip 대체)
// =========================================================================
//
// 컬럼: SUCURSAL · VENTAS(건·금액) · PRENDAS · DESC · MOV+ · MOV− · FAL · NETO
//
// 동작:
//   - VENTAS / PRENDAS / DESC: activity_type='sale' 만 (Anulado 제외) — 백엔드 처리
//   - MOV+: 그 지점이 target_branch_id 인 행의 prendas 합
//   - MOV−: 그 지점이 origin_branch_id 인 행의 prendas 합
//   - FAL : 그 지점이 origin 인 fallado 행의 prendas 합
//   - NETO: MOV+ − MOV− − FAL
//
// 드릴다운:
//   - 행 클릭 → onRowClick({ branchId, branchName })
//   - 셀 클릭 → onCellClick({ branchId, type:'movido'|'fallado', direction:'in'|'out' })
//
// 단일 지점 사용자: perBranch.length === 1 이면 TOTAL 행 숨김 (UAT U11)
// movBalance.balanced === false 시 TOTAL 행 셀에 ⚠ + tooltip (UAT U14)

import { useMemo, useCallback } from 'react'
import {
  Box, Table, TableBody, TableCell, TableContainer, TableHead, TableRow,
  Typography, Paper, Skeleton, Tooltip, Stack,
} from '@mui/material'
import { Icon } from '@iconify/react'
import { useDailySalesStats, DailyStatsBranchRow } from 'src/hooks/api/useDailySalesStats'
import { formatPriceNoDecimals } from 'src/configs/utils'

export interface ResumenRowClick {
  branchId: number | null  // null = TOTAL 행
  branchName: string
}

export interface ResumenCellClick {
  branchId: number | null
  branchName: string
  type: 'movido' | 'fallado'
  direction?: 'in' | 'out'
}

interface Props {
  startDate: string
  endDate: string
  onRowClick?: (e: ResumenRowClick) => void
  onCellClick?: (e: ResumenCellClick) => void
}

const SalesResumenTable = ({ startDate, endDate, onRowClick, onCellClick }: Props) => {
  const { data, isLoading } = useDailySalesStats({ startDate, endDate })

  const perBranch = data?.perBranch ?? []
  const total = data?.total
  const movBalance = data?.movBalance
  const showTotalRow = perBranch.length > 1

  // ─── 셀 클릭 핸들러 (useCallback 으로 안정화) ───────────────────────────
  const handleCellMovIn = useCallback(
    (row: DailyStatsBranchRow) => onCellClick?.({
      branchId: row.branchId, branchName: row.branchName,
      type: 'movido', direction: 'in',
    }),
    [onCellClick],
  )
  const handleCellMovOut = useCallback(
    (row: DailyStatsBranchRow) => onCellClick?.({
      branchId: row.branchId, branchName: row.branchName,
      type: 'movido', direction: 'out',
    }),
    [onCellClick],
  )
  const handleCellFal = useCallback(
    (row: DailyStatsBranchRow) => onCellClick?.({
      branchId: row.branchId, branchName: row.branchName,
      type: 'fallado',
    }),
    [onCellClick],
  )
  const handleRowClickInternal = useCallback(
    (row: DailyStatsBranchRow) => onRowClick?.({ branchId: row.branchId, branchName: row.branchName }),
    [onRowClick],
  )

  // ─── 빈 상태 / 로딩 ────────────────────────────────────────────────────
  if (isLoading) {
    return (
      <Box sx={{ p: 1 }}>
        <Skeleton variant="rectangular" height={120} />
      </Box>
    )
  }

  if (!data || perBranch.length === 0) {
    return (
      <Box sx={{ p: 2, textAlign: 'center', color: 'text.secondary' }}>
        <Typography variant="caption">오늘 활동 없음</Typography>
      </Box>
    )
  }

  // ─── 셀 렌더링 헬퍼 ────────────────────────────────────────────────────
  const renderMovIn = (n: number) => (
    n > 0
      ? <Typography variant="body2" sx={{ color: 'info.main', fontWeight: 600 }}>+{n}</Typography>
      : <Typography variant="caption" sx={{ color: 'text.disabled' }}>—</Typography>
  )
  const renderMovOut = (n: number) => (
    n > 0
      ? <Typography variant="body2" sx={{ color: 'info.main', fontWeight: 600 }}>−{n}</Typography>
      : <Typography variant="caption" sx={{ color: 'text.disabled' }}>—</Typography>
  )
  const renderFal = (n: number) => (
    n > 0
      ? <Typography variant="body2" sx={{ color: 'error.main', fontWeight: 600 }}>{n}</Typography>
      : <Typography variant="caption" sx={{ color: 'text.disabled' }}>—</Typography>
  )
  const renderNeto = (n: number) => {
    const sign = n > 0 ? '+' : n < 0 ? '−' : ''
    const color = n > 0 ? 'success.main' : n < 0 ? 'error.main' : 'text.primary'

    return (
      <Typography variant="body2" sx={{ color, fontWeight: 700 }}>
        {sign}{Math.abs(n)}
      </Typography>
    )
  }

  const balanceWarning = movBalance && !movBalance.balanced

  return (
    <TableContainer component={Paper} sx={{ maxHeight: 200, overflow: 'auto' }}>
      <Table size="small" stickyHeader>
        <TableHead>
          <TableRow>
            <TableCell sx={{ fontWeight: 700, position: 'sticky', left: 0, bgcolor: 'background.paper', zIndex: 2 }}>SUCURSAL</TableCell>
            <TableCell align="right" sx={{ fontWeight: 700 }}>VENTAS</TableCell>
            <TableCell align="right" sx={{ fontWeight: 700 }}>PRENDAS</TableCell>
            <TableCell align="right" sx={{ fontWeight: 700 }}>DESC.</TableCell>
            <TableCell align="right" sx={{ fontWeight: 700 }}>MOV+</TableCell>
            <TableCell align="right" sx={{ fontWeight: 700 }}>MOV−</TableCell>
            <TableCell align="right" sx={{ fontWeight: 700 }}>FAL</TableCell>
            <TableCell align="right" sx={{ fontWeight: 700 }}>NETO</TableCell>
          </TableRow>
        </TableHead>
        <TableBody>
          {perBranch.map((row) => (
            <TableRow
              key={row.branchId}
              hover
              sx={{ cursor: onRowClick ? 'pointer' : 'default' }}
              onClick={() => handleRowClickInternal(row)}
            >
              <TableCell sx={{ fontWeight: 600, position: 'sticky', left: 0, bgcolor: 'background.paper' }}>
                {row.branchName}
              </TableCell>
              <TableCell align="right">
                <Stack direction="row" justifyContent="flex-end" alignItems="baseline" spacing={0.5}>
                  <Typography variant="body2">{row.ventas.count}</Typography>
                  <Typography variant="caption" color="text.secondary">·</Typography>
                  <Typography variant="body2" sx={{ fontWeight: 600 }}>
                    {formatPriceNoDecimals(row.ventas.amount)}
                  </Typography>
                </Stack>
              </TableCell>
              <TableCell align="right">{row.prendas}</TableCell>
              <TableCell align="right">
                {row.descuento > 0
                  ? <Typography variant="body2" sx={{ color: 'warning.main' }}>−{formatPriceNoDecimals(row.descuento)}</Typography>
                  : <Typography variant="caption" sx={{ color: 'text.disabled' }}>—</Typography>}
              </TableCell>
              <TableCell
                align="right"
                onClick={(e) => { e.stopPropagation(); handleCellMovIn(row) }}
                sx={{ cursor: onCellClick ? 'pointer' : 'default' }}
              >
                {renderMovIn(row.movIn)}
              </TableCell>
              <TableCell
                align="right"
                onClick={(e) => { e.stopPropagation(); handleCellMovOut(row) }}
                sx={{ cursor: onCellClick ? 'pointer' : 'default' }}
              >
                {renderMovOut(row.movOut)}
              </TableCell>
              <TableCell
                align="right"
                onClick={(e) => { e.stopPropagation(); handleCellFal(row) }}
                sx={{ cursor: onCellClick ? 'pointer' : 'default' }}
              >
                {renderFal(row.fallados)}
              </TableCell>
              <TableCell align="right">{renderNeto(row.neto)}</TableCell>
            </TableRow>
          ))}

          {showTotalRow &&  total && (
            <TableRow sx={{ bgcolor: 'action.hover' }}>
              <TableCell sx={{ fontWeight: 700, position: 'sticky', left: 0, bgcolor: 'action.hover' }}>
                Σ TOTAL
              </TableCell>
              <TableCell align="right">
                <Stack direction="row" justifyContent="flex-end" alignItems="baseline" spacing={0.5}>
                  <Typography variant="body2" sx={{ fontWeight: 700 }}>{total.ventas.count}</Typography>
                  <Typography variant="caption" color="text.secondary">·</Typography>
                  <Typography variant="body2" sx={{ fontWeight: 700 }}>{formatPriceNoDecimals(total.ventas.amount)}</Typography>
                </Stack>
              </TableCell>
              <TableCell align="right" sx={{ fontWeight: 700 }}>{total.prendas}</TableCell>
              <TableCell align="right" sx={{ fontWeight: 700 }}>
                {total.descuento > 0 ? `−${formatPriceNoDecimals(total.descuento)}` : '—'}
              </TableCell>
              <TableCell align="right" sx={{ fontWeight: 700 }}>
                <Stack direction="row" justifyContent="flex-end" alignItems="center" spacing={0.5}>
                  {balanceWarning && (
                    <Tooltip title="동일 store 내 이동인데 IN/OUT 합이 다릅니다 — 데이터 점검 필요">
                      <Box component="span" sx={{ color: 'warning.main', display: 'inline-flex' }}>
                        <Icon icon="mdi:alert-circle-outline" />
                      </Box>
                    </Tooltip>
                  )}
                  {renderMovIn(total.movIn)}
                </Stack>
              </TableCell>
              <TableCell align="right" sx={{ fontWeight: 700 }}>
                <Stack direction="row" justifyContent="flex-end" alignItems="center" spacing={0.5}>
                  {balanceWarning && (
                    <Tooltip title="동일 store 내 이동인데 IN/OUT 합이 다릅니다 — 데이터 점검 필요">
                      <Box component="span" sx={{ color: 'warning.main', display: 'inline-flex' }}>
                        <Icon icon="mdi:alert-circle-outline" />
                      </Box>
                    </Tooltip>
                  )}
                  {renderMovOut(total.movOut)}
                </Stack>
              </TableCell>
              <TableCell align="right" sx={{ fontWeight: 700 }}>{renderFal(total.fallados)}</TableCell>
              <TableCell align="right" sx={{ fontWeight: 700 }}>{renderNeto(total.neto)}</TableCell>
            </TableRow>
          )}
        </TableBody>
      </Table>
    </TableContainer>
  )
}

export default SalesResumenTable
```
