```typescript
const renderTipoChip = (activityType?: string) => {

// Phase 35: Cliente 컬럼 dual-purpose 렌더러 (sale / movido / fallado 분기)
const renderClienteOrRoute = (row: any) => {
  const at = row?.activityType
  if (at === 'movido') {
    const origin = row?.originBranch?.name || `#${row?.originBranchId ?? '?'}`
    const target = row?.targetBranch?.name || `#${row?.targetBranchId ?? '?'}`

    return (
      <Stack direction="row" spacing={0.5} alignItems="center">
        <Typography variant="body2" sx={{ fontWeight: 600, color: 'info.main' }}>{origin}</Typography>
        <Box component="span" sx={{ color: 'info.main', display: 'inline-flex', fontSize: 14 }}>→</Box>
        <Typography variant="body2" sx={{ fontWeight: 600, color: 'info.main' }}>{target}</Typography>
      </Stack>
    )
  }
  if (at === 'fallado') {
    const origin = row?.originBranch?.name || `#${row?.originBranchId ?? '?'}`

    return (
      <Typography variant="body2" sx={{ fontWeight: 600, color: 'error.main' }}>
        FAL · {origin}
      </Typography>
    )
  }

  // sale: 기존 cliente 이름
  return (
    <Typography variant="body2" sx={{ fontWeight: 600, whiteSpace: 'nowrap', overflow: 'hidden', textOverflow: 'ellipsis' }}>
      {row?.client?.fullname || row?.storeClient?.fullname || 'Cliente Indefinido'}
    </Typography>
  )
}

// Phase 35: movido/fallado 시 의미 없는 컬럼은 '—' 처리
const isNonSale = (row: any): boolean => {
```

```typescript
  // Phase 35: 행별 스타일 (movido/fallado tint).
  // getRowSx 가 sx-like 객체를 반환하지만, 본 컴포넌트에서는 CSSProperties 만 사용 (AG Grid 호환).
  const getRowStyleCb = useCallback(
    (params: any) => {
      if (!getRowSx) return undefined
      const sx = getRowSx(params.data)
      if (!sx) return undefined

      // sx 의 backgroundColor / borderLeft 등 CSS 호환 키만 추출 (안전 매핑)
      const style: React.CSSProperties = {}
      if (sx.backgroundColor) style.backgroundColor = sx.backgroundColor
      if (sx.bgcolor) style.backgroundColor = sx.bgcolor // sx alias
      if (sx.borderLeft) style.borderLeft = sx.borderLeft
      if (sx.color) style.color = sx.color

      return style
    },
    [getRowSx],
  )
```

```typescript
const router = useRouter()

// URL → state (mount 또는 URL 변경 시)
useEffect(() => {
  const q = router.query
  setFilters((prev) => ({
    ...prev,
    activityType: (q.activityType as any) || prev.activityType,
    originBranchId: q.originBranchId ? Number(q.originBranchId) : null,
    targetBranchId: q.targetBranchId ? Number(q.targetBranchId) : null,
    direction: (q.direction as any) || null,
    branchChipLabel: (q.branchLabel as string) || null,
  }))

// state → URL (filter 변경 시)
const updateUrlQuery = useCallback((next: Partial<Phase35Filters>) => {
  const merged = { ...router.query }
  if (next.activityType && next.activityType !== 'sale') merged.activityType = next.activityType
  else delete merged.activityType
  if (next.originBranchId) merged.originBranchId = String(next.originBranchId)
  else delete merged.originBranchId
  if (next.targetBranchId) merged.targetBranchId = String(next.targetBranchId)
  else delete merged.targetBranchId
  if (next.direction) merged.direction = next.direction
  else delete merged.direction
  if (next.branchChipLabel) merged.branchLabel = next.branchChipLabel
  else delete merged.branchLabel

  router.replace({ pathname: router.pathname, query: merged }, undefined, { shallow: true })
}, [router])
```
