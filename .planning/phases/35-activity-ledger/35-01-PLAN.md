```sql
-- Phase 35 — Activity Ledger
-- ============================================================================
-- 목적: sales 테이블을 unified transaction ledger 로 확장
--   - movido / fallado 활동을 별도 테이블 아닌 sales 행으로 1급 처리
--   - 기존 sales.activity_type IS NULL 행은 모두 'sale' default 로 채워짐 (DEFAULT)
-- 호환성: PostgreSQL 10 (운영) + 15 (로컬 docker) 양쪽 동작 검증.
--   - GENERATED AS IDENTITY 미사용
--   - native ENUM TYPE 대신 VARCHAR + CHECK 사용 (alter 용이성 + PG10 호환)
--   - 모든 DDL idempotent (IF NOT EXISTS) — 재실행 안전
-- ============================================================================

BEGIN;

-- 1) 신규 컬럼 3개 (idempotent)
ALTER TABLE sales
  ADD COLUMN IF NOT EXISTS activity_type   VARCHAR(16) NOT NULL DEFAULT 'sale',
  ADD COLUMN IF NOT EXISTS origin_branch_id INTEGER NULL,
  ADD COLUMN IF NOT EXISTS target_branch_id INTEGER NULL;

-- 2) CHECK 제약 (idempotent — DO 블록으로 중복 생성 방지)
DO $$
BEGIN
  IF NOT EXISTS (
    SELECT 1 FROM pg_constraint
    WHERE conname = 'chk_sales_activity_type' AND conrelid = 'sales'::regclass
  ) THEN
    ALTER TABLE sales
      ADD CONSTRAINT chk_sales_activity_type
      CHECK (activity_type IN ('sale','movido','fallado'));
  END IF;
END $$;

-- 3) FK 제약 (idempotent — DO 블록)
DO $$
BEGIN
  IF NOT EXISTS (
    SELECT 1 FROM pg_constraint
    WHERE conname = 'fk_sales_origin_branch' AND conrelid = 'sales'::regclass
  ) THEN
    ALTER TABLE sales
      ADD CONSTRAINT fk_sales_origin_branch
      FOREIGN KEY (origin_branch_id) REFERENCES branches(id);
  END IF;

  IF NOT EXISTS (
    SELECT 1 FROM pg_constraint
    WHERE conname = 'fk_sales_target_branch' AND conrelid = 'sales'::regclass
  ) THEN
    ALTER TABLE sales
      ADD CONSTRAINT fk_sales_target_branch
      FOREIGN KEY (target_branch_id) REFERENCES branches(id);
  END IF;
END $$;

-- 4) 인덱스 3개 (partial — PG10 호환)
CREATE INDEX IF NOT EXISTS idx_sales_activity_date
  ON sales(activity_type, sale_date);

CREATE INDEX IF NOT EXISTS idx_sales_origin_branch
  ON sales(origin_branch_id) WHERE origin_branch_id IS NOT NULL;

CREATE INDEX IF NOT EXISTS idx_sales_target_branch
  ON sales(target_branch_id) WHERE target_branch_id IS NOT NULL;

-- 5) 컬럼 주석 (디버깅 + 향후 작업자 가이드)
COMMENT ON COLUMN sales.activity_type IS
  'Phase 35: sale | movido | fallado. 기존 매출 쿼리는 모두 activity_type = ''sale'' 필터 필요.';
COMMENT ON COLUMN sales.origin_branch_id IS
  'Phase 35: movido/fallado 의 출발(out) 지점. sale 행은 NULL.';
COMMENT ON COLUMN sales.target_branch_id IS
  'Phase 35: movido 의 도착(in) 지점. sale/fallado 행은 NULL.';

COMMIT;

-- ============================================================================
-- 검증 쿼리 (dry-run 후 수동 실행):
--   SELECT activity_type, COUNT(*) FROM sales GROUP BY activity_type;
--     기대: 모든 기존 행이 activity_type='sale' (DEFAULT 적용)
--   SELECT conname FROM pg_constraint WHERE conrelid='sales'::regclass
--     AND conname IN ('chk_sales_activity_type','fk_sales_origin_branch','fk_sales_target_branch');
--     기대: 3행
--   SELECT indexname FROM pg_indexes WHERE tablename='sales'
--     AND indexname IN ('idx_sales_activity_date','idx_sales_origin_branch','idx_sales_target_branch');
--     기대: 3행
-- ============================================================================
```
