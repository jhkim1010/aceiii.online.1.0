5) **신규 메서드 `getDailyStats` 추가** (sales.service.ts 클래스 본체 끝, 적절한 위치):
```typescript
  // ─── Phase 35: ventaVista Resumen 테이블 데이터 source ──────────────────────
  // 응답 shape: { date, perBranch:[{branchId,branchName,ventas:{count,amount},prendas,descuento,movIn,movOut,fallados,neto}],
  //              total: {...}, movBalance:{in,out,balanced} }
  //
  // 정책:
  //  - VENTAS / PRENDAS / DESC: activity_type='sale' 만 (sales.user_id → users.branch_id 경유)
  //  - MOV+: target_branch_id 가 그 지점인 movido 행의 prendas
  //  - MOV−: origin_branch_id 가 그 지점인 movido 행의 prendas
  //  - FAL : origin_branch_id 가 그 지점인 fallado 행의 prendas
  //  - NETO: MOV+ − MOV− − FAL
  //
  // PG pool 절약: 단일 raw SQL 로 perBranch 집계 + JS 에서 total/movBalance 계산.
  async getDailyStats(opts: {
    storeId: number;
    startDate: string;
    endDate: string;
  }): Promise<any> {
    const tz = await this.getStoreTimezone(opts.storeId);

    // perBranch 집계: branches LEFT JOIN sales(sale) + LEFT JOIN sales(movido in/out) + LEFT JOIN sales(fallado)
    // sales 자체에 branch_id 없으므로 sale 활동은 user.branch_id 로 매핑.
    const sequelize = this.saleModel.sequelize!;
    const rows = await sequelize.query<any>(
      `
      WITH
        sale_per_branch AS (
          SELECT u.branch_id, COUNT(*) AS ventas_count,
                 COALESCE(SUM(s.total_amount), 0)::numeric AS ventas_amount,
                 COALESCE(SUM(s.discount_amount), 0)::numeric AS descuento,
                 COALESCE(SUM(si_q.q), 0)::int AS prendas
          FROM sales s
          JOIN users u ON u.id = s.user_id
          LEFT JOIN LATERAL (SELECT COALESCE(SUM(quantity),0)::int AS q FROM sale_items WHERE sale_id = s.id) si_q ON TRUE
          WHERE s.store_id = :storeId
            AND s.activity_type = 'sale'
            AND s.status NOT IN ('Anulado','Anulación')
            AND DATE(s.sale_date AT TIME ZONE :tz) BETWEEN :startDate AND :endDate
          GROUP BY u.branch_id
        ),
        mov_in AS (
          SELECT s.target_branch_id AS branch_id, COALESCE(SUM(si_q.q),0)::int AS prendas
          FROM sales s
          LEFT JOIN LATERAL (SELECT COALESCE(SUM(quantity),0)::int AS q FROM sale_items WHERE sale_id = s.id) si_q ON TRUE
          WHERE s.store_id = :storeId AND s.activity_type = 'movido'
            AND DATE(s.sale_date AT TIME ZONE :tz) BETWEEN :startDate AND :endDate
          GROUP BY s.target_branch_id
        ),
        mov_out AS (
          SELECT s.origin_branch_id AS branch_id, COALESCE(SUM(si_q.q),0)::int AS prendas
          FROM sales s
          LEFT JOIN LATERAL (SELECT COALESCE(SUM(quantity),0)::int AS q FROM sale_items WHERE sale_id = s.id) si_q ON TRUE
          WHERE s.store_id = :storeId AND s.activity_type = 'movido'
            AND DATE(s.sale_date AT TIME ZONE :tz) BETWEEN :startDate AND :endDate
          GROUP BY s.origin_branch_id
        ),
        fal AS (
          SELECT s.origin_branch_id AS branch_id, COALESCE(SUM(si_q.q),0)::int AS prendas
          FROM sales s
          LEFT JOIN LATERAL (SELECT COALESCE(SUM(quantity),0)::int AS q FROM sale_items WHERE sale_id = s.id) si_q ON TRUE
          WHERE s.store_id = :storeId AND s.activity_type = 'fallado'
            AND DATE(s.sale_date AT TIME ZONE :tz) BETWEEN :startDate AND :endDate
          GROUP BY s.origin_branch_id
        )
      SELECT b.id AS branch_id, b.name AS branch_name,
             COALESCE(spb.ventas_count, 0)::int AS ventas_count,
             COALESCE(spb.ventas_amount, 0)::numeric AS ventas_amount,
             COALESCE(spb.prendas, 0)::int AS prendas,
             COALESCE(spb.descuento, 0)::numeric AS descuento,
             COALESCE(mi.prendas, 0)::int AS mov_in,
             COALESCE(mo.prendas, 0)::int AS mov_out,
             COALESCE(f.prendas, 0)::int AS fallados
      FROM branches b
      LEFT JOIN sale_per_branch spb ON spb.branch_id = b.id
      LEFT JOIN mov_in mi  ON mi.branch_id = b.id
      LEFT JOIN mov_out mo ON mo.branch_id = b.id
      LEFT JOIN fal f      ON f.branch_id = b.id
      WHERE b.store_id = :storeId
      ORDER BY b.id ASC
      `,
      {
        replacements: { storeId: opts.storeId, startDate: opts.startDate, endDate: opts.endDate, tz },
        type: QueryTypes.SELECT,
      },
    );

    const perBranch = rows.map((r: any) => {
      const movIn = Number(r.mov_in || 0);
      const movOut = Number(r.mov_out || 0);
      const fallados = Number(r.fallados || 0);

      return {
        branchId: Number(r.branch_id),
        branchName: r.branch_name,
        ventas: { count: Number(r.ventas_count || 0), amount: Number(r.ventas_amount || 0) },
        prendas: Number(r.prendas || 0),
        descuento: Number(r.descuento || 0),
        movIn, movOut, fallados,
        neto: movIn - movOut - fallados,
      };
    });

    const total = perBranch.reduce(
      (acc: any, b: any) => ({
        ventas: { count: acc.ventas.count + b.ventas.count, amount: acc.ventas.amount + b.ventas.amount },
        prendas: acc.prendas + b.prendas,
        descuento: acc.descuento + b.descuento,
        movIn: acc.movIn + b.movIn,
        movOut: acc.movOut + b.movOut,
        fallados: acc.fallados + b.fallados,
        neto: acc.neto + b.neto,
      }),
      { ventas: { count: 0, amount: 0 }, prendas: 0, descuento: 0, movIn: 0, movOut: 0, fallados: 0, neto: 0 },
    );

    const movBalance = { in: total.movIn, out: total.movOut, balanced: total.movIn === total.movOut };

    return { date: opts.endDate, perBranch, total, movBalance };
  }
```
