-- Phase 35 — stock.movement Permission
-- ============================================================================
-- 목적: ventaVista special mode (movidos/fallados) 등록을 위한 신규 권한 등록.
--   - permission_slug: 'stock.movement'
--   - 기본 부여 (role): store_owner, store_admin, gerente, admin, superadmin (매장 책임자 레벨)
--   - 일반 vendedor 는 기본 미부여 (사용자가 권한 페이지에서 명시적 활성화)
--   - 기존 사용자 자동 부여: 최근 90일간 movido/fallado 등록한 user 에게 user_functions 직접 부여
--     (역호환 — Phase 35 적용 후 갑자기 등록 불가 되는 vendedor/encargado 방지)
-- ============================================================================

BEGIN;

-- 1) functions 테이블에 stock.movement 등록 (이미 존재하는 'movidos-fallados' 또는 'movimiento-stock'
--    function 행을 찾아 permission_slug 채움. 없으면 신규 생성.)
DO $$
DECLARE
  func_id INTEGER;
  recent_user_count INTEGER;
BEGIN
  -- 기존 function 찾기 (slug 후보 여러 개 시도)
  SELECT id INTO func_id FROM functions
  WHERE slug IN ('movidos-fallados','movimiento-stock','stock-movement')
  LIMIT 1;

  IF func_id IS NULL THEN
    -- 신규 function 생성. module_id = stocks 모듈 (legacy fallback: 1 = root)
    INSERT INTO functions (slug, name, permission_slug, module_id, created_at, updated_at)
    SELECT 'stock-movement', 'Movimiento de stock (movidos/fallados)', 'stock.movement',
           COALESCE((SELECT id FROM modules WHERE slug='stocks' LIMIT 1), 1),
           NOW(), NOW()
    WHERE NOT EXISTS (SELECT 1 FROM functions WHERE slug='stock-movement');

    SELECT id INTO func_id FROM functions WHERE slug='stock-movement';
  ELSE
    -- 기존 row 에 permission_slug 채움 (idempotent)
    UPDATE functions SET permission_slug = 'stock.movement', updated_at=NOW()
    WHERE id = func_id AND (permission_slug IS NULL OR permission_slug <> 'stock.movement');
  END IF;

  -- 2) 기본 role 부여 (store_owner / store_admin / gerente / admin / superadmin)
  --    role_functions(role_id, function_id) 매핑 — 중복 INSERT 방지 NOT EXISTS
  INSERT INTO role_functions (role_id, function_id, created_at, updated_at)
  SELECT r.id, func_id, NOW(), NOW()
  FROM roles r
  WHERE r.slug IN ('store_owner','store_admin','gerente','admin','superadmin')
    AND NOT EXISTS (
      SELECT 1 FROM role_functions rf WHERE rf.role_id = r.id AND rf.function_id = func_id
    );

  -- 3) 기존 사용자 자동 부여 (역호환):
  --    최근 90일간 movido/fallado 를 등록한 user 에게 user_functions 직접 부여.
  --    sales 테이블의 activityType 컬럼은 Phase 35 적용 후에 채워지지만, stocks(note LIKE 'movido%' OR 'fallado%')
  --    에 user 흔적은 없음 (stocks 자체 user_id 없음). 따라서 cash_register 시간대 + branch 일치로 추정 불가.
  --    대신: 일반 매장 운영 패턴에서 vendedor/encargado 가 movido 등록하는 경우가 흔하므로
  --    "vendedor + encargado role 보유 사용자" 에게 90일 활동 이력 조건으로 부여.
  --    조건: 사용자가 sales 활동 (어떤 형태든) 이 90일 내 1건 이상 → movido 등록 가능성 있음.
  INSERT INTO user_functions (user_id, function_id, created_at, updated_at)
  SELECT DISTINCT u.id, func_id, NOW(), NOW()
  FROM users u
  WHERE u.id IN (
    -- 최근 90일간 sale 활동 이력이 있는 user (vendedor/encargado 추정)
    SELECT DISTINCT s.user_id FROM sales s
    WHERE s.sale_date >= NOW() - INTERVAL '90 days'
      AND s.user_id IS NOT NULL
  )
    AND u.id IN (
      -- vendedor/encargado role 보유 (privileged 가 아닌 일반 영업)
      SELECT DISTINCT ur.user_id FROM user_roles ur
      JOIN roles r ON r.id = ur.role_id
      WHERE r.slug IN ('vendedor','encargado')
    )
    AND NOT EXISTS (
      SELECT 1 FROM user_functions uf
      WHERE uf.user_id = u.id AND uf.function_id = func_id
    );

  GET DIAGNOSTICS recent_user_count = ROW_COUNT;
  RAISE NOTICE 'Phase 35: stock.movement 권한 자동 부여된 사용자 수 = %', recent_user_count;
END $$;

-- 4) 검증 쿼리 출력 (마이그레이션 실행 후 수동 확인)
-- SELECT f.slug, f.permission_slug, r.slug AS role_slug
-- FROM functions f
-- JOIN role_functions rf ON rf.function_id = f.id
-- JOIN roles r ON r.id = rf.role_id
-- WHERE f.permission_slug = 'stock.movement'
-- ORDER BY r.slug;
-- 기대: store_owner, store_admin, gerente, admin, superadmin 각 1행 (총 5행)
--
-- SELECT COUNT(*) AS direct_grants FROM user_functions uf

-- 기대: 최근 90일 활동 vendedor/encargado 수 (실데이터 기준)

COMMIT;
```

      `[MOVIDOS-DEBUG] ${dbgId} 트랜잭션 START — items 처리 시작 (${items.length}개)`,
    );

    try {
      await this.sequelize.transaction(async (tx) => {
        // ─── Phase 35: sales 행 INSERT (활동의 헤더) ──────────────────────
        const sale = await Sale.create(
          {
            activityType:
              type === 'movido'
                ? SaleActivityType.MOVIDO
                : SaleActivityType.FALLADO,
            storeId: data.storeId ?? null,
            userId: data.userId ?? null,
            status: SaleStatus.PAID, // 'Pagado' — 거래 완료 표시
            subtotal: 0,
            discountAmount: 0,
            totalAmount: 0,
            originBranchId: Number(originBranchId),
            targetBranchId:
              type === 'movido' && targetBranchId
                ? Number(targetBranchId)
                : null,
            saleDate: new Date(),
            notes: baseNote,
          } as any,
          { transaction: tx },
        );
        createdSaleId = (sale as any).id;
        this.logger.log(
          `[MOVIDOS-DEBUG] ${dbgId} Sale INSERT saleId=${createdSaleId} activityType=${type}`,
        );

        // ─── Phase 35: sale_items 일괄 INSERT (활동의 라인) ────────────────
        // unit_price=0, subtotal=0 — D-10: 매출 sum 쿼리는 sales.activity_type='sale'
        // 필터로 차단되므로 통계 오염 없음.
        const validItems = items
          .map((it) => ({
            productId: Number(it.productId),
            quantity: Math.abs(Number(it.quantity) || 0),
          }))
          .filter((it) => it.productId > 0 && it.quantity > 0);

        if (validItems.length > 0) {
          await SaleItem.bulkCreate(
            validItems.map((it) => ({
              saleId: createdSaleId,
              productId: it.productId,
              quantity: it.quantity,
              price: 0,
              subtotal: 0,
              discountAmount: 0,
            })) as any[],
            { transaction: tx },
          );
          this.logger.log(
            `[MOVIDOS-DEBUG] ${dbgId} SaleItem.bulkCreate saleId=${createdSaleId} count=${validItems.length}`,
          );
        }

        // ─── 기존 stocks INSERT 로직 (origin debit, target credit) ─────────
        // 동일 트랜잭션 tx 사용. ProductBranch findOrCreate 패턴 유지.
        for (const [idx, item] of items.entries()) {
          const productId = Number(item.productId);
          const rawQty = Number(item.quantity);
          const qty = Math.abs(rawQty || 0);
          if (qty === 0) {
            this.logger.warn(
              `[MOVIDOS-DEBUG] ${dbgId} item[${idx}] productId=${productId} qty=0 → SKIP`,
            );
            continue;
          }

          // origin: 항상 -qty
          const originPb = await this.findOrCreateProductBranch(
            productId,
            Number(originBranchId),
            tx,
          );
          const originStockRow = await this.stockModel.create(
            {
              productBranchId: originPb.id,
              stock: -qty,
              type: 'adjust',
              note:
                type === 'fallado'
                  ? `fallado: ${baseNote}`
                  : `movido(out→${targetBranchId}): ${baseNote}`,
            } as any,
            { transaction: tx },
          );
          insertedRows++;
          this.logger.log(
            `[MOVIDOS-DEBUG] ${dbgId} item[${idx}] origin Stocks INSERT stockId=${(originStockRow as any)?.id} delta=${-qty}`,
          );

          // movido: target 에 +qty
          if (type === 'movido' && targetBranchId) {
            const targetPb = await this.findOrCreateProductBranch(
              productId,
              Number(targetBranchId),
              tx,
            );
            const targetStockRow = await this.stockModel.create(
              {
                productBranchId: targetPb.id,
                stock: qty,
                type: 'adjust',
                note: `movido(in←${originBranchId}): ${baseNote}`,
              } as any,
              { transaction: tx },
            );
            insertedRows++;
            this.logger.log(
              `[MOVIDOS-DEBUG] ${dbgId} item[${idx}] target Stocks INSERT stockId=${(targetStockRow as any)?.id} delta=+${qty}`,
            );
          }
        }
      });

      this.logger.log(
        `[MOVIDOS-DEBUG] ${dbgId} 트랜잭션 COMMIT — saleId=${createdSaleId} insertedRows=${insertedRows}`,
      );
    } catch (txErr: any) {
      this.logger.error(
        `[MOVIDOS-DEBUG] ${dbgId} 트랜잭션 ROLLBACK — ${txErr?.name}: ${txErr?.message}`,
        txErr?.stack,
      );
      throw txErr;
    }

    return {
      success: true,
      saleId: createdSaleId,
      type,
      itemCount: items.length,
      insertedRows,
    };
```

import { CanActivate, ExecutionContext, Injectable } from '@nestjs/common';

@Injectable()
class InjectBranchIdFromOriginGuard implements CanActivate {
  canActivate(context: ExecutionContext): boolean {
    const req = context.switchToHttp().getRequest();
    if (req?.body && !req.body.branchId && req.body.originBranchId) {
      // PermissionGuard 가 body.branchId 만 읽으므로,
      // originBranchId 를 사전 매핑하여 비-privileged 사용자도 통과 가능하게 함.
      req.body.branchId = req.body.originBranchId;
    }
    return true;
  }
}

    action: 'create',

        body.type === 'movido'
          ? `branch ${body.originBranchId} → ${body.targetBranchId}`
          : `branch ${body.originBranchId}`;

    },

    @Body() body: CreateStockMovementDto,
    @GetUser() user: Users,

    this.logger.log(
      `[MOVIDOS-DEBUG] ${correlationId} controller IN type=${body?.type} ` +
        `origin=${body?.originBranchId} target=${body?.targetBranchId} ` +
        `items=${Array.isArray(body?.items) ? body.items.length : 'N/A'} ` +
        `userId=${user?.id} storeId=${body?.storeId} note="${body?.notes ?? ''}"`,

    ]);
    const userRoles = (user?.roles ?? []).map((r: any) =>
      typeof r === 'string' ? r : r?.slug,
    );
    const isPrivileged = userRoles.some((s: string) => privilegedRoles.has(s));

    if (!isPrivileged) {
      const userBranchId = (user as any)?.branchId ?? null;
      const origin = Number(body?.originBranchId);
      const target = body?.targetBranchId ? Number(body.targetBranchId) : null;
      const allowed =
        userBranchId !== null &&
        (userBranchId === origin || (target !== null && userBranchId === target));

      if (!allowed) {
        this.logger.warn(
          `[MOVIDOS-DEBUG] ${correlationId} 403 branch 제약 위반 — userBranch=${userBranchId} origin=${origin} target=${target}`,
        );
        throw new ForbiddenException(
          'No tiene permiso para registrar movimientos entre estas sucursales',
        );
      }
    }

    // userId 를 service 로 전달 (sales.user_id 채움 용도)
    const enrichedBody: CreateStockMovementDto = {
      ...body,
      userId: user?.id,
      storeId: user?.storeId ?? body?.storeId ?? null,
    };

    try {
      const result = await this.stockService.createStockMovement(enrichedBody);
      this.logger.log(

          `saleId=${result?.saleId} insertedRows=${result?.insertedRows} itemCount=${result?.itemCount}`,
      );

      return result;
    } catch (err: any) {
      this.logger.error(
        `[MOVIDOS-DEBUG] ${correlationId} controller ERROR ${err?.name}: ${err?.message}`,
        err?.stack,
      );
      throw err;
    }
  }
