```sql
-- Phase 35 — Backfill historical movidos/fallados to sales
-- ============================================================================
-- 목적: Phase 35 이전에 stocks(type='adjust', note LIKE 'movido(%' OR 'fallado%') 행
--       으로만 기록된 활동을, sales/sale_items 1급 시민으로 backfill.
--
-- 전략:
--   1) backfill_failures 테이블 생성 (parsing 실패 행 격리)
--   2) movido out 행을 그룹화 → sales(activity_type='movido') 1개 + sale_items N개 INSERT
--   3) fallado 행을 그룹화 → sales(activity_type='fallado') 1개 + sale_items N개 INSERT
--   4) note 파싱 실패 행 → backfill_failures 에 기록 (sales 생성 안 함)
--
-- 그룹화 키: (DATE_TRUNC('second', stocks.created_at), product_branch.branch_id, note baseNote 부분)
--   같은 createStockMovement 호출의 모든 행이 동일 그룹.
--
-- Idempotent: 이미 backfill 된 stocks 행을 markCol 컬럼으로 추적 (backfill_processed_sale_id).
-- ============================================================================

BEGIN;

-- ─── STEP 1: backfill 실패 행 격리 테이블 ─────────────────────────────────────
CREATE TABLE IF NOT EXISTS backfill_failures (
  id BIGSERIAL PRIMARY KEY,
  source_table VARCHAR(50) NOT NULL,
  source_row_id BIGINT NOT NULL,
  reason VARCHAR(255) NOT NULL,
  raw_note TEXT,
  created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE INDEX IF NOT EXISTS idx_backfill_failures_source ON backfill_failures(source_table, source_row_id);

-- ─── STEP 2: stocks 에 backfill 추적 컬럼 (idempotent) ────────────────────────
ALTER TABLE stocks
  ADD COLUMN IF NOT EXISTS backfill_processed_sale_id INTEGER NULL;
COMMENT ON COLUMN stocks.backfill_processed_sale_id IS
  'Phase 35 backfill: 이 stocks 행이 어느 sales.id 로 backfill 되었는지 추적 (NULL = 아직 처리 안 됨).';

-- ─── STEP 3: movido out 그룹 backfill ─────────────────────────────────────────
-- note 패턴: 'movido(out→{targetBranchId}): {baseNote}'
-- 그룹화: (origin_branch_id, target_branch_id, time_bucket, base_note)
WITH movido_out_groups AS (
  SELECT
    DATE_TRUNC('second', s.created_at) AS time_bucket,
    pb.branch_id AS origin_branch_id,
    -- target_branch_id 를 note 에서 추출. 'movido(out→{N}): {note}' → N
    NULLIF(
      (regexp_match(s.note, 'movido\(out→([0-9]+)\)'))[1],
      ''
    )::integer AS target_branch_id,
    -- baseNote: ': ' 뒤 부분
    COALESCE(NULLIF(SPLIT_PART(s.note, ': ', 2), ''), 'Movimiento entre sucursales') AS base_note,
    p.store_id,
    ARRAY_AGG(s.id ORDER BY s.id) AS stock_ids,
    ARRAY_AGG(pb.product_id ORDER BY s.id) AS product_ids,
    ARRAY_AGG(ABS(s.stock) ORDER BY s.id) AS quantities,
    MIN(s.created_at) AS min_created_at
  FROM stocks s
  JOIN "ProductBranch" pb ON pb.id = s.product_branch_id
  JOIN products p ON p.id = pb.product_id
  WHERE s.type = 'adjust'
    AND s.note LIKE 'movido(out→%'
    AND s.backfill_processed_sale_id IS NULL  -- idempotent
  GROUP BY 1, 2, 3, 4, 5
)
INSERT INTO sales (
  activity_type, store_id, status, subtotal, discount_amount, total_amount,
  origin_branch_id, target_branch_id, sale_date, notes,
  created_at, updated_at, daily_number, print_count, source
)
SELECT
  'movido', store_id, 'Pagado', 0, 0, 0,
  origin_branch_id, target_branch_id, min_created_at,
  '[Backfill Phase 35] ' || base_note,
  min_created_at, NOW(), 0, 0, 'pos'
FROM movido_out_groups
WHERE target_branch_id IS NOT NULL  -- 파싱 실패한 행은 다음 STEP 에서 별도 처리
RETURNING id, origin_branch_id, target_branch_id, sale_date;

-- (위 INSERT 의 returning 결과를 다음 단계에서 사용하려면 PL/pgSQL 함수 또는 DO 블록 필요)
-- → 단순화: 별도 DO 블록으로 순차 처리.

-- ─── STEP 4: PL/pgSQL 로 sale_items + stocks.backfill_processed_sale_id 채움 ─
DO $$
DECLARE
  grp RECORD;
  new_sale_id INTEGER;
  i INTEGER;
BEGIN
  FOR grp IN
    SELECT
      DATE_TRUNC('second', s.created_at) AS time_bucket,
      pb.branch_id AS origin_branch_id,
      NULLIF((regexp_match(s.note, 'movido\(out→([0-9]+)\)'))[1], '')::integer AS target_branch_id,
      COALESCE(NULLIF(SPLIT_PART(s.note, ': ', 2), ''), 'Movimiento entre sucursales') AS base_note,
      p.store_id,
      ARRAY_AGG(s.id ORDER BY s.id) AS stock_ids,
      ARRAY_AGG(pb.product_id ORDER BY s.id) AS product_ids,
      ARRAY_AGG(ABS(s.stock) ORDER BY s.id) AS quantities,
      MIN(s.created_at) AS min_created_at
    FROM stocks s
    JOIN "ProductBranch" pb ON pb.id = s.product_branch_id
    JOIN products p ON p.id = pb.product_id
    WHERE s.type = 'adjust'
      AND s.note LIKE 'movido(out→%'
      AND s.backfill_processed_sale_id IS NULL
    GROUP BY 1, 2, 3, 4, 5
  LOOP
    -- 파싱 실패 행 → backfill_failures + 다음 그룹
    IF grp.target_branch_id IS NULL THEN
      INSERT INTO backfill_failures (source_table, source_row_id, reason, raw_note)
      SELECT 'stocks', stock_id, 'movido_out: target_branch_id 파싱 실패', grp.base_note
      FROM UNNEST(grp.stock_ids) AS stock_id;
      CONTINUE;
    END IF;

    -- sales 1행 INSERT
    INSERT INTO sales (
      activity_type, store_id, status, subtotal, discount_amount, total_amount,
      origin_branch_id, target_branch_id, sale_date, notes,
      created_at, updated_at, daily_number, print_count, source
    )
    VALUES (
      'movido', grp.store_id, 'Pagado', 0, 0, 0,
      grp.origin_branch_id, grp.target_branch_id, grp.min_created_at,
      '[Backfill Phase 35] ' || grp.base_note,
      grp.min_created_at, NOW(), 0, 0, 'pos'
    )
    RETURNING id INTO new_sale_id;

    -- sale_items N행 INSERT
    FOR i IN 1..array_length(grp.product_ids, 1) LOOP
      INSERT INTO sale_items (sale_id, product_id, quantity, price, subtotal, discount_amount, created_at, updated_at)
      VALUES (new_sale_id, grp.product_ids[i], grp.quantities[i], 0, 0, 0, grp.min_created_at, NOW());
    END LOOP;

    -- 처리된 stocks 행 마킹
    UPDATE stocks SET backfill_processed_sale_id = new_sale_id
    WHERE id = ANY(grp.stock_ids);
  END LOOP;
END $$;

-- ─── STEP 5: fallado 그룹 backfill (동일 패턴) ─────────────────────────────────
DO $$
DECLARE
  grp RECORD;
  new_sale_id INTEGER;
  i INTEGER;
BEGIN
  FOR grp IN
    SELECT
      DATE_TRUNC('second', s.created_at) AS time_bucket,
      pb.branch_id AS origin_branch_id,
      COALESCE(NULLIF(SPLIT_PART(s.note, ': ', 2), ''), 'Fallado') AS base_note,
      p.store_id,
      ARRAY_AGG(s.id ORDER BY s.id) AS stock_ids,
      ARRAY_AGG(pb.product_id ORDER BY s.id) AS product_ids,
      ARRAY_AGG(ABS(s.stock) ORDER BY s.id) AS quantities,
      MIN(s.created_at) AS min_created_at
    FROM stocks s
    JOIN "ProductBranch" pb ON pb.id = s.product_branch_id
    JOIN products p ON p.id = pb.product_id
    WHERE s.type = 'adjust'
      AND s.note LIKE 'fallado:%'
      AND s.backfill_processed_sale_id IS NULL
    GROUP BY 1, 2, 3, 4
  LOOP
    INSERT INTO sales (
      activity_type, store_id, status, subtotal, discount_amount, total_amount,
      origin_branch_id, sale_date, notes,
      created_at, updated_at, daily_number, print_count, source
    )
    VALUES (
      'fallado', grp.store_id, 'Pagado', 0, 0, 0,
      grp.origin_branch_id, grp.min_created_at,
      '[Backfill Phase 35] ' || grp.base_note,
      grp.min_created_at, NOW(), 0, 0, 'pos'
    )
    RETURNING id INTO new_sale_id;

    FOR i IN 1..array_length(grp.product_ids, 1) LOOP
      INSERT INTO sale_items (sale_id, product_id, quantity, price, subtotal, discount_amount, created_at, updated_at)
      VALUES (new_sale_id, grp.product_ids[i], grp.quantities[i], 0, 0, 0, grp.min_created_at, NOW());
    END LOOP;

    UPDATE stocks SET backfill_processed_sale_id = new_sale_id
    WHERE id = ANY(grp.stock_ids);
  END LOOP;
END $$;

-- ─── STEP 6: movido in 행 (origin 의 짝) 도 backfill_processed_sale_id 채움 ──
-- movido(in←{originBranchId}) 행은 동일 그룹의 out 행과 연결됨.
-- 위 STEP 4 에서 out 만 처리했으므로 in 행은 별도 매칭 필요.
UPDATE stocks s_in
SET backfill_processed_sale_id = s_in_match.sale_id
FROM (
  SELECT s_in.id AS in_id, sa.id AS sale_id
  FROM stocks s_in
  JOIN "ProductBranch" pb_in ON pb_in.id = s_in.product_branch_id
  JOIN sales sa ON sa.activity_type = 'movido'
    AND sa.target_branch_id = pb_in.branch_id
    AND DATE_TRUNC('second', sa.sale_date) = DATE_TRUNC('second', s_in.created_at)
  WHERE s_in.type = 'adjust'
    AND s_in.note LIKE 'movido(in←%'
    AND s_in.backfill_processed_sale_id IS NULL
) AS s_in_match
WHERE s_in.id = s_in_match.in_id;

-- ─── 검증 쿼리 (수동 실행 — dry-run 시 사용) ───────────────────────────────────

COMMIT;
```

```bash
#!/usr/bin/env bash
# ============================================================================
# Phase 35 — Backfill dry-run 검증 스크립트
# ============================================================================
# 사용:
#   ./api-ventago/migrations/phase35-backfill-dry-run.sh [dev|prod]
#
# 동작:
#   1) BEGIN; 후 backfill SQL 적용
#   2) sales(activity_type='movido'|'fallado') 생성 수, backfill_failures 수,
#      stocks(backfill_processed_sale_id IS NOT NULL) 수 출력
#   3) ROLLBACK (실제 변경 없음)
#
# 환경:
#   dev (default) : 로컬 docker dbpostgres PG15
#   prod          : 운영 PG10 (sudo -u postgres psql) — 별도 ssh 필요
# ============================================================================

set -euo pipefail

ENV="${1:-dev}"
SQL_FILE="$(dirname "$0")/phase35-backfill-movidos-to-sales.sql"

if [[ ! -f "$SQL_FILE" ]]; then
  echo "ERROR: SQL file not found: $SQL_FILE" >&2
  exit 1
fi

echo "=== Phase 35 Backfill Dry-Run ($ENV) ==="
echo "SQL file: $SQL_FILE"
echo

# SQL 의 COMMIT 을 ROLLBACK 으로 치환한 임시 SQL 작성
TMP_SQL="$(mktemp -t phase35-dry-run-XXXXXX.sql)"
trap "rm -f $TMP_SQL" EXIT

# 마지막 COMMIT 만 ROLLBACK 으로 (안전하게 awk 로)
awk '{
  if ($0 ~ /^COMMIT;$/) print "ROLLBACK;";
  else print $0;
```
