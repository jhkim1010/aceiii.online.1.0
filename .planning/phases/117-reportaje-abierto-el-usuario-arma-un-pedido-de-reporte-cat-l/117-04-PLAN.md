- export function periodoComparacion(desde: string, hasta: string, tipo: 'periodo_anterior' | 'anio_anterior', alineacion: 'semanal' | 'calendario'): { desde: string; hasta: string }
    · semanal: previous = shift both ends −ceil(n/7)*7 days; year ago = −364 days (D-31)
    · calendario: previous = shift both ends back m months, m = the smallest number of months such that (hasta − m months) < desde (so the ranges never overlap and month buckets keep the same date cuts); if hasta is the last day of its month the shifted hasta is the last day of the target month; days clamped to month end; year ago = same dates −1 year (D-31a)
