## Step 1: Fiscal Config Tab (0.5일)

Phase 9의 `StoresUnifiedView` 상세 페이지 탭에 **Configuración Fiscal** 탭을 추가.

### `FiscalConfigTab.tsx`

```tsx
// ventago-app/src/views/admin/stores/tabs/FiscalConfigTab.tsx
import { useState, useEffect } from 'react'
import { useForm } from 'react-hook-form'
import { yupResolver } from '@hookform/resolvers/yup'
import * as Yup from 'yup'
import {
  Box, TextField, Button, CircularProgress,
  Typography, Alert, Divider, Grid,
} from '@mui/material'
import { apiConnector } from 'src/services/api.service'

interface FiscalConfigFormData {
  cuit: string
  razonSocial: string
  domicilio: string
  iibb: string
  condicionIva: string
  inicioActividades: string
  puntoVenta: number
  relayClientId: string
}

const schema = Yup.object({
  cuit: Yup.string()
    .matches(/^\d{2}-\d{8}-\d$/, 'Formato: 20-12345678-9')
    .required('Requerido'),
  razonSocial: Yup.string().required('Requerido'),
  domicilio: Yup.string().required('Requerido'),
  iibb: Yup.string().required('Requerido'),
  condicionIva: Yup.string().required('Requerido'),
  inicioActividades: Yup.string().required('Requerido'),
  puntoVenta: Yup.number().min(1).max(9999).required('Requerido'),
  relayClientId: Yup.string().required('Requerido'),
})

interface Props {
  storeId: number
}

export default function FiscalConfigTab({ storeId }: Props) {
  const [loading, setLoading] = useState(true)
  const [saving, setSaving] = useState(false)
  const [success, setSuccess] = useState(false)
  const [error, setError] = useState<string | null>(null)

  const {
    register,
    handleSubmit,
    reset,
    formState: { errors },
  } = useForm<FiscalConfigFormData>({ resolver: yupResolver(schema) })

  useEffect(() => {
    // 기존 설정 로드
    apiConnector
      .get(`/facturacion/fiscal-config/${storeId}`)
      .then((res: any) => {
        if (res.data) {
          reset(res.data)
        }
      })
      .catch(() => {
        // 미설정 상태면 빈 폼
      })
      .finally(() => setLoading(false))
  }, [storeId, reset])

  const onSubmit = async (data: FiscalConfigFormData) => {
    setSaving(true)
    setError(null)
    setSuccess(false)

    try {
      await apiConnector.put(`/facturacion/fiscal-config/${storeId}`, data)
      setSuccess(true)
    } catch (err: any) {
      setError(err?.response?.data?.message ?? 'Error al guardar')
    } finally {
      setSaving(false)
    }
  }

  if (loading) {

    return (
      <Box display="flex" justifyContent="center" py={4}>
        <CircularProgress />
      </Box>
    )
  }

  return (
    <Box component="form" onSubmit={handleSubmit(onSubmit)} p={2}>
      <Typography variant="h6" mb={2}>
        Configuración Fiscal AFIP
      </Typography>

      {success && (
        <Alert severity="success" sx={{ mb: 2 }}>
          Configuración guardada correctamente
        </Alert>
      )}

      {error && (
        <Alert severity="error" sx={{ mb: 2 }}>
          {error}
        </Alert>
      )}

      <Grid container spacing={2}>
        <Grid item xs={12} sm={6}>
          <TextField
            label="CUIT (20-12345678-9)"
            fullWidth
            {...register('cuit')}
            error={!!errors.cuit}
            helperText={errors.cuit?.message}
          />
        </Grid>
        <Grid item xs={12} sm={6}>
          <TextField
            label="Razón Social"
            fullWidth
            {...register('razonSocial')}
            error={!!errors.razonSocial}
            helperText={errors.razonSocial?.message}
          />
        </Grid>
        <Grid item xs={12}>
          <TextField
            label="Domicilio"
            fullWidth
            {...register('domicilio')}
            error={!!errors.domicilio}
            helperText={errors.domicilio?.message}
          />
        </Grid>
        <Grid item xs={12} sm={4}>
          <TextField
            label="Ingresos Brutos"
            fullWidth
            {...register('iibb')}
            error={!!errors.iibb}
            helperText={errors.iibb?.message}
          />
        </Grid>
        <Grid item xs={12} sm={4}>
          <TextField
            label="Condición frente al IVA"
            fullWidth
            placeholder="Responsable Inscripto"
            {...register('condicionIva')}
            error={!!errors.condicionIva}
            helperText={errors.condicionIva?.message}
          />
        </Grid>
        <Grid item xs={12} sm={4}>
          <TextField
            label="Inicio de Actividades"
            type="date"
            fullWidth
            InputLabelProps={{ shrink: true }}
            {...register('inicioActividades')}
            error={!!errors.inicioActividades}
            helperText={errors.inicioActividades?.message}
          />
        </Grid>

        <Grid item xs={12}>
          <Divider />
        </Grid>

        <Grid item xs={12} sm={4}>
          <TextField
            label="Punto de Venta"
            type="number"
            fullWidth
            {...register('puntoVenta', { valueAsNumber: true })}
            error={!!errors.puntoVenta}
            helperText={errors.puntoVenta?.message}
          />
        </Grid>
        <Grid item xs={12} sm={8}>
          <TextField
            label="Relay Client ID"
            fullWidth
            placeholder="ID asignado por el administrador del relay"
            {...register('relayClientId')}
            error={!!errors.relayClientId}
            helperText={errors.relayClientId?.message}
          />
        </Grid>

        <Grid item xs={12}>
          <Button
            type="submit"
            variant="contained"
            disabled={saving}
            startIcon={saving ? <CircularProgress size={16} /> : null}
          >
            {saving ? 'Guardando...' : 'Guardar Configuración'}
          </Button>
        </Grid>
      </Grid>
    </Box>
  )
}
```

## Step 2: EmitirFacturaModal (0.5일)

POS (`nueva-venta`) 판매 확정 직후 열리는 모달. 수신자 정보 입력 → CAE 취득 → PDF 팝업.

### `EmitirFacturaModal.tsx`

```tsx
// ventago-app/src/views/facturacion/EmitirFacturaModal.tsx
import { useState } from 'react'
import {
  Dialog, DialogTitle, DialogContent, DialogActions,
  TextField, Button, CircularProgress, Alert,
  FormControl, InputLabel, Select, MenuItem, Grid,
} from '@mui/material'
import { apiConnector } from 'src/services/api.service'

interface Props {
  open: boolean
  saleId: number
  totalAmount: number
  onClose: () => void
  onSuccess: (invoiceId: number) => void
}

interface EmitirFormState {
  customerName: string
  customerDocType: 'CUIT' | 'DNI' | 'CUIL' | 'SIN_IDENTIFICAR'
  customerDocNumber: string
  customerAddress: string
  customerCondicionIva: string
}

export default function EmitirFacturaModal({
  open,
  saleId,
  totalAmount,
  onClose,
  onSuccess,
}: Props) {
  const [form, setForm] = useState<EmitirFormState>({
    customerName: 'Consumidor Final',
    customerDocType: 'SIN_IDENTIFICAR',
    customerDocNumber: '0',
    customerAddress: '',
    customerCondicionIva: 'Consumidor Final',
  })
  const [loading, setLoading] = useState(false)
  const [error, setError] = useState<string | null>(null)

  const handleChange = (field: keyof EmitirFormState) => (e: any) => {
    setForm((prev) => ({ ...prev, [field]: e.target.value }))
  }

  const handleEmitir = async () => {
    setLoading(true)
    setError(null)

    try {
      const res: any = await apiConnector.post('/facturacion/emit', {
        saleId,
        ...form,
      })

      onSuccess(res.data.invoiceId)
    } catch (err: any) {
      const msg = err?.response?.data?.message ?? 'Error al emitir factura'
      setError(msg)
    } finally {
      setLoading(false)
    }
  }

  return (
    <Dialog open={open} onClose={onClose} maxWidth="sm" fullWidth>
      <DialogTitle>Emitir Factura</DialogTitle>
      <DialogContent>
        {error && (
          <Alert severity="error" sx={{ mb: 2 }}>
            {error}
          </Alert>
        )}

        <Grid container spacing={2} mt={0.5}>
          <Grid item xs={12}>
            <TextField
              label="Nombre / Razón Social"
              fullWidth
              value={form.customerName}
              onChange={handleChange('customerName')}
            />
          </Grid>
          <Grid item xs={6}>
            <FormControl fullWidth>
              <InputLabel>Tipo Doc.</InputLabel>
              <Select
                value={form.customerDocType}
                label="Tipo Doc."
                onChange={handleChange('customerDocType')}
              >
                <MenuItem value="SIN_IDENTIFICAR">Sin Identificar</MenuItem>
                <MenuItem value="DNI">DNI</MenuItem>
                <MenuItem value="CUIT">CUIT</MenuItem>
                <MenuItem value="CUIL">CUIL</MenuItem>
              </Select>
            </FormControl>
          </Grid>
          <Grid item xs={6}>
            <TextField
              label="Nro. Documento"
              fullWidth
              value={form.customerDocNumber}
              onChange={handleChange('customerDocNumber')}
              disabled={form.customerDocType === 'SIN_IDENTIFICAR'}
            />
          </Grid>
          <Grid item xs={12}>
            <TextField
              label="Condición IVA"
              fullWidth
              value={form.customerCondicionIva}
              onChange={handleChange('customerCondicionIva')}
              placeholder="Consumidor Final"
            />
          </Grid>
          <Grid item xs={12}>
            <TextField
              label="Domicilio (opcional)"
              fullWidth
              value={form.customerAddress}
              onChange={handleChange('customerAddress')}
            />
          </Grid>
        </Grid>
      </DialogContent>
      <DialogActions>
        <Button onClick={onClose} disabled={loading}>
          Cancelar
        </Button>
        <Button
          variant="contained"
          onClick={handleEmitir}
          disabled={loading}
          startIcon={loading ? <CircularProgress size={16} /> : null}
        >
          {loading ? 'Emitiendo...' : `Emitir ($ ${totalAmount.toFixed(2)})`}
        </Button>
      </DialogActions>
    </Dialog>
  )
}
```

CAE 취득 성공 후 PDF를 iframe으로 표시.

### `InvoicePdfViewer.tsx`

```tsx
// ventago-app/src/views/facturacion/InvoicePdfViewer.tsx
import {
  Dialog, DialogTitle, DialogContent, DialogActions,
  Button, Box,
} from '@mui/material'
import { API_HOST } from 'src/configs/appConfig'

interface Props {
  open: boolean
  invoiceId: number | null
  onClose: () => void
}

export default function InvoicePdfViewer({ open, invoiceId, onClose }: Props) {
  if (!invoiceId) {

    return null
  }

  const pdfUrl = `${API_HOST}/facturacion/invoices/${invoiceId}/pdf`

  return (
    <Dialog open={open} onClose={onClose} maxWidth="md" fullWidth>
      <DialogTitle>Factura emitida — CAE obtenido</DialogTitle>
      <DialogContent>
        <Box sx={{ height: 600 }}>
          <iframe
            src={pdfUrl}
            width="100%"
            height="100%"
            title="Factura PDF"
            style={{ border: 'none' }}
          />
        </Box>
      </DialogContent>
      <DialogActions>
        <Button href={pdfUrl} target="_blank" rel="noreferrer">
          Abrir en pestaña
        </Button>
        <Button onClick={onClose} variant="contained">
          Cerrar
        </Button>
      </DialogActions>
    </Dialog>
  )
}
```

### `FacturacionListView.tsx`

```tsx
// ventago-app/src/views/facturacion/FacturacionListView.tsx
import { useState, useEffect } from 'react'
import {
  Box, Typography, TextField, Button, Chip,
  Table, TableBody, TableCell, TableHead, TableRow,
  TableContainer, Paper, CircularProgress,
} from '@mui/material'
import { apiConnector } from 'src/services/api.service'

// Invoice 상태 색상 매핑
const STATUS_COLOR: Record<string, 'success' | 'error' | 'warning' | 'default'> = {
  ISSUED: 'success',
  CANCELLED: 'error',
  ERROR: 'warning',
}

export default function FacturacionListView() {
  const [rows, setRows] = useState<any[]>([])
  const [loading, setLoading] = useState(false)
  const [dateFrom, setDateFrom] = useState('')
  const [dateTo, setDateTo] = useState('')

  const load = async () => {
    setLoading(true)

    try {
      const params: any = {}
      if (dateFrom) params.dateFrom = dateFrom
      if (dateTo) params.dateTo = dateTo

      const res: any = await apiConnector.get('/facturacion/invoices', { params })
      setRows(res.data ?? [])
    } finally {
      setLoading(false)
    }
  }

  useEffect(() => {
    load()
  }, [])

  return (
    <Box p={3}>
      <Typography variant="h5" mb={2}>
        Comprobantes emitidos
      </Typography>

      {/* 필터 바 */}
      <Box display="flex" gap={2} mb={2} alignItems="center">
        <TextField
          label="Desde"
          type="date"
          size="small"
          value={dateFrom}
          onChange={(e) => setDateFrom(e.target.value)}
          InputLabelProps={{ shrink: true }}
        />
        <TextField
          label="Hasta"
          type="date"
          size="small"
          value={dateTo}
          onChange={(e) => setDateTo(e.target.value)}
          InputLabelProps={{ shrink: true }}
        />
        <Button variant="contained" size="small" onClick={load} disabled={loading}>
          Buscar
        </Button>
      </Box>

      {loading ? (
        <Box display="flex" justifyContent="center" py={4}>
          <CircularProgress />
        </Box>
      ) : (
        <TableContainer component={Paper}>
          <Table size="small">
            <TableHead>
              <TableRow>
                <TableCell>Fecha</TableCell>
                <TableCell>Comprobante</TableCell>
                <TableCell>Tipo</TableCell>
                <TableCell>Receptor</TableCell>
                <TableCell align="right">Neto</TableCell>
                <TableCell align="right">IVA</TableCell>
                <TableCell align="right">Total</TableCell>
                <TableCell>CAE</TableCell>
                <TableCell>Estado</TableCell>
                <TableCell>PDF</TableCell>
              </TableRow>
            </TableHead>
            <TableBody>
              {rows.map((row) => (
                <TableRow key={row.id} hover>
                  <TableCell>
                    {new Date(row.issueDate).toLocaleDateString('es-AR')}
                  </TableCell>
                  <TableCell>
                    {String(row.puntoVenta).padStart(4, '0')}-
                    {String(row.invoiceNumber).padStart(8, '0')}
                  </TableCell>
                  <TableCell>{row.invoiceType}</TableCell>
                  <TableCell>{row.customerName}</TableCell>
                  <TableCell align="right">
                    $ {Number(row.netAmount).toFixed(2)}
                  </TableCell>
                  <TableCell align="right">
                    $ {Number(row.ivaAmount).toFixed(2)}
                  </TableCell>
                  <TableCell align="right">
                    $ {Number(row.totalAmount).toFixed(2)}
                  </TableCell>
                  <TableCell sx={{ fontSize: '10px' }}>{row.cae}</TableCell>
                  <TableCell>
                    <Chip
                      label={row.status}
                      size="small"
                      color={STATUS_COLOR[row.status] ?? 'default'}
                    />
                  </TableCell>
                  <TableCell>
                    <Button
                      size="small"
                      href={`${process.env.NEXT_PUBLIC_API_HOST}/facturacion/invoices/${row.id}/pdf`}
                      target="_blank"
                      rel="noreferrer"
                    >
                      PDF
                    </Button>
                  </TableCell>
                </TableRow>
              ))}
            </TableBody>
          </Table>
        </TableContainer>
      )}
    </Box>
  )
}
```

```typescript
@Get('fiscal-config/:storeId')

async getFiscalConfig(@Param('storeId', ParseIntPipe) storeId: number) {
```
