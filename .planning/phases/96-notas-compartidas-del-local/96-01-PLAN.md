```sql
CREATE TABLE IF NOT EXISTS notas (
  id            SERIAL PRIMARY KEY,
  store_id      INTEGER      NOT NULL REFERENCES stores(id) ON DELETE CASCADE,
  sender_id     INTEGER      NOT NULL REFERENCES users(id),
  audience      VARCHAR(8)   NOT NULL CHECK (audience IN ('all','users')),
  title         VARCHAR(200) NOT NULL CHECK (length(btrim(title)) > 0),
  body          TEXT         NOT NULL DEFAULT '',
  importance    VARCHAR(12)  NOT NULL DEFAULT 'normal'
                CHECK (importance IN ('normal','importante','urgente')),
  pinned        BOOLEAN      NOT NULL DEFAULT false,
  expires_on    DATE,
  edited_at     TIMESTAMPTZ,
  archived_at   TIMESTAMPTZ,
  archived_by   INTEGER      REFERENCES users(id),
  last_reply_at TIMESTAMPTZ,
  last_reply_by INTEGER      REFERENCES users(id),
  created_at    TIMESTAMPTZ  NOT NULL DEFAULT now(),
  updated_at    TIMESTAMPTZ  NOT NULL DEFAULT now(),
  CONSTRAINT notas_pin_solo_todos CHECK (NOT pinned OR audience = 'all')
);
CREATE INDEX IF NOT EXISTS notas_store_created_idx ON notas (store_id, created_at DESC);
CREATE INDEX IF NOT EXISTS notas_store_sender_idx  ON notas (store_id, sender_id);

CREATE TABLE IF NOT EXISTS nota_recipients (
  id         SERIAL PRIMARY KEY,
  store_id   INTEGER     NOT NULL REFERENCES stores(id) ON DELETE CASCADE,
  nota_id    INTEGER     NOT NULL REFERENCES notas(id) ON DELETE CASCADE,
  user_id    INTEGER     NOT NULL REFERENCES users(id),
  pinned     BOOLEAN     NOT NULL DEFAULT false,
  created_at TIMESTAMPTZ NOT NULL DEFAULT now(),
  updated_at TIMESTAMPTZ NOT NULL DEFAULT now(),
  CONSTRAINT nota_recipients_nota_user_uq UNIQUE (nota_id, user_id)
);
CREATE INDEX IF NOT EXISTS nota_recipients_store_user_idx ON nota_recipients (store_id, user_id);

CREATE TABLE IF NOT EXISTS nota_reads (
  id              SERIAL PRIMARY KEY,
  store_id        INTEGER     NOT NULL REFERENCES stores(id) ON DELETE CASCADE,
  nota_id         INTEGER     NOT NULL REFERENCES notas(id) ON DELETE CASCADE,
  user_id         INTEGER     NOT NULL REFERENCES users(id),
  seen_at         TIMESTAMPTZ NOT NULL DEFAULT now(),
  ack_at          TIMESTAMPTZ,
  replies_seen_at TIMESTAMPTZ,
  created_at      TIMESTAMPTZ NOT NULL DEFAULT now(),
  updated_at      TIMESTAMPTZ NOT NULL DEFAULT now(),
  CONSTRAINT nota_reads_nota_user_uq UNIQUE (nota_id, user_id)
);
CREATE INDEX IF NOT EXISTS nota_reads_store_user_idx ON nota_reads (store_id, user_id);

CREATE TABLE IF NOT EXISTS nota_replies (
  id         SERIAL PRIMARY KEY,
  store_id   INTEGER     NOT NULL REFERENCES stores(id) ON DELETE CASCADE,
  nota_id    INTEGER     NOT NULL REFERENCES notas(id) ON DELETE CASCADE,
  user_id    INTEGER     NOT NULL REFERENCES users(id),
  body       TEXT        NOT NULL DEFAULT '',
  created_at TIMESTAMPTZ NOT NULL DEFAULT now(),
  updated_at TIMESTAMPTZ NOT NULL DEFAULT now()
);
CREATE INDEX IF NOT EXISTS nota_replies_nota_created_idx ON nota_replies (nota_id, created_at);

CREATE TABLE IF NOT EXISTS nota_reactions (
  id         SERIAL PRIMARY KEY,
  store_id   INTEGER     NOT NULL REFERENCES stores(id) ON DELETE CASCADE,
  nota_id    INTEGER     NOT NULL REFERENCES notas(id) ON DELETE CASCADE,
  reply_id   INTEGER     REFERENCES nota_replies(id) ON DELETE CASCADE,
  user_id    INTEGER     NOT NULL REFERENCES users(id),
  emoji      VARCHAR(16) NOT NULL CHECK (emoji IN ('thumbs_up','check','eyes','heart','joy')),
  created_at TIMESTAMPTZ NOT NULL DEFAULT now(),
  updated_at TIMESTAMPTZ NOT NULL DEFAULT now()
);
CREATE UNIQUE INDEX IF NOT EXISTS nota_reactions_nota_uq
  ON nota_reactions (nota_id, user_id, emoji) WHERE reply_id IS NULL;
CREATE UNIQUE INDEX IF NOT EXISTS nota_reactions_reply_uq
  ON nota_reactions (reply_id, user_id, emoji) WHERE reply_id IS NOT NULL;

CREATE TABLE IF NOT EXISTS nota_edit_history (
  id              SERIAL PRIMARY KEY,
  store_id        INTEGER      NOT NULL REFERENCES stores(id) ON DELETE CASCADE,
  nota_id         INTEGER      NOT NULL REFERENCES notas(id) ON DELETE CASCADE,
  edited_by       INTEGER      NOT NULL REFERENCES users(id),
  prev_title      VARCHAR(200) NOT NULL,
  prev_body       TEXT         NOT NULL,
  prev_importance VARCHAR(12)  NOT NULL,
  prev_expires_on DATE,
  created_at      TIMESTAMPTZ  NOT NULL DEFAULT now()
);
CREATE INDEX IF NOT EXISTS nota_edit_history_nota_idx ON nota_edit_history (nota_id, created_at DESC);

CREATE TABLE IF NOT EXISTS nota_attachments (
  id          SERIAL PRIMARY KEY,
  store_id    INTEGER      NOT NULL REFERENCES stores(id) ON DELETE CASCADE,
  nota_id     INTEGER      NOT NULL REFERENCES notas(id) ON DELETE CASCADE,
  reply_id    INTEGER      REFERENCES nota_replies(id) ON DELETE CASCADE,
  object_key  VARCHAR(300) NOT NULL UNIQUE CHECK (object_key LIKE 'notas/%'),
  file_name   VARCHAR(200) NOT NULL,
  mime_type   VARCHAR(100) NOT NULL
              CHECK (mime_type IN ('image/jpeg','image/png','image/webp','image/gif','application/pdf')),
  size_bytes  INTEGER      NOT NULL,
  uploaded_by INTEGER      NOT NULL REFERENCES users(id),
  created_at  TIMESTAMPTZ  NOT NULL DEFAULT now(),
  updated_at  TIMESTAMPTZ  NOT NULL DEFAULT now()
);
CREATE INDEX IF NOT EXISTS nota_attachments_nota_idx ON nota_attachments (nota_id);
```
