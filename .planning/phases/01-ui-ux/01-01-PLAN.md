   async updateUiMode(userId: number, uiMode: 'classic' | 'new'): Promise<{ uiMode: string }> {
     const user = await this.userModel.findByPk(userId);
     if (!user) {
       throw new NotFoundException('Usuario no encontrado');
     }
     await user.update({ uiMode });
     return { uiMode: user.uiMode };

   ) {
