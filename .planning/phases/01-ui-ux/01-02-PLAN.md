2. `ventago-app/src/context/UiModeContext.tsx` 신규 생성 (per D-06, Claude 재량으로 Context API 선택):
   ```typescript
   import { createContext, useContext, useState, useCallback, ReactNode, useMemo } from 'react'
   import { useAuth } from 'src/hooks/useAuth'
   import apiConnector from 'src/services/api.service'

   type UiModeContextType = {
     uiMode: 'classic' | 'new'
     isNewUi: boolean
     toggleUiMode: () => void
     canToggle: boolean
   }

   const UiModeContext = createContext<UiModeContextType>({
     uiMode: 'classic',
     isNewUi: false,
     toggleUiMode: () => {},
     canToggle: false,
   })

   export const UiModeProvider = ({ children }: { children: ReactNode }) => {
     const { user, setUser } = useAuth()

     // user 객체에서 uiMode 읽기 (서버에서 /me로 받은 값)
     const uiMode = user?.uiMode || 'classic'
     const isNewUi = uiMode === 'new'

     // admin/superadmin만 토글 가능 (per D-03)
     const canToggle = !!(user?.roles?.includes('admin') || user?.roles?.includes('superadmin'))

     const toggleUiMode = useCallback(async () => {
       if (!canToggle) return
       const newMode = uiMode === 'classic' ? 'new' : 'classic'
       try {
         await apiConnector.put('/users/ui-mode', { uiMode: newMode })

         // 로컬 상태 즉시 업데이트 (페이지 새로고침 없이 반영)
         if (user) {
           setUser({ ...user, uiMode: newMode })
         }
       } catch (error) {
         console.error('[UiMode] Error al cambiar modo:', error)
       }
     }, [canToggle, uiMode, user, setUser])

     const value = useMemo(() => ({
       uiMode, isNewUi, toggleUiMode, canToggle
     }), [uiMode, isNewUi, toggleUiMode, canToggle])

     return <UiModeContext.Provider value={value}>{children}</UiModeContext.Provider>
   }

   export const useUiMode = () => useContext(UiModeContext)
   ```
