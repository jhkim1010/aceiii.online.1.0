**6. dio_client.dart** — Dio 인스턴스 + JWT interceptor:
```dart
final dioClientProvider = Provider<DioClient>((ref) {
  final storage = ref.read(secureStorageProvider);
  return DioClient(ApiConfig.baseUrl, storage);
});

class DioClient {
  late Dio dio;
  
  DioClient(String baseUrl, SecureStorageService storage) {
    dio = Dio(BaseOptions(
      baseUrl: baseUrl,
      connectTimeout: ApiConfig.connectTimeout,
      receiveTimeout: ApiConfig.receiveTimeout,
    ));
    
    dio.interceptors.add(InterceptorsWrapper(
      onRequest: (options, handler) async {
        final token = await storage.read('vendor_token');
        if (token != null) {
          options.headers['Authorization'] = 'Bearer $token';
        }
        return handler.next(options);
      },
      onError: (error, handler) {
        // 401 처리는 auth provider에서 관리
        return handler.next(error);
      },
    ));
  }
}
```

**2. auth_repository.dart** — API 호출:

```dart
final authRepositoryProvider = Provider<AuthRepository>((ref) {
  final dio = ref.read(dioClientProvider);
  return AuthRepository(dio);

class AuthRepository {
  final DioClient _client;
  AuthRepository(this._client);
  
  Future<Map<String, dynamic>> login(String phone, String pin) async {
    try {
      final response = await _client.dio.post('/vendor-portal/auth/login', data: { 'phone': phone, 'pin': pin });
      return response.data;
    } on DioException catch (e) {
      throw ApiException(e.response?.data?['message'] ?? 'Error de conexión', statusCode: e.response?.statusCode);
    }
  }
  
  Future<Map<String, dynamic>> getMe() async {
    try {
      final response = await _client.dio.get('/vendor-portal/auth/me');
      return response.data;
    } on DioException catch (e) {
      throw ApiException(e.response?.data?['message'] ?? 'Error', statusCode: e.response?.statusCode);
    }
  }
}
```

**3. auth_provider.dart** — Riverpod AsyncNotifier (per CLAUDE.md Riverpod 선호):
```dart
final authNotifierProvider = AsyncNotifierProvider<AuthNotifier, AuthState?>(() => AuthNotifier());

// 현재 선택된 매장 인덱스
final selectedStoreIndexProvider = StateProvider<int>((ref) => 0);

// 현재 선택된 StoreInfo (computed)
final currentStoreProvider = Provider<StoreInfo?>((ref) {
  final auth = ref.watch(authNotifierProvider).value;
  final index = ref.watch(selectedStoreIndexProvider);
  if (auth == null || auth.stores.isEmpty) return null;
  return auth.stores[index.clamp(0, auth.stores.length - 1)];
});

class AuthNotifier extends AsyncNotifier<AuthState?> {
  @override
  Future<AuthState?> build() async {
    // 앱 시작 시 저장된 토큰 복구
    final storage = ref.read(secureStorageProvider);
    final token = await storage.read('vendor_token');
    final phone = await storage.read('vendor_phone');
    if (token == null || phone == null) return null;
    
    try {
      final data = await ref.read(authRepositoryProvider).getMe();
      return AuthState(token: token, phone: phone,
        stores: (data['stores'] as List).map((s) => StoreInfo.fromJson(s)).toList());
    } catch (_) {
      await storage.deleteAll();
      return null;
    }
  }
  
  Future<void> login(String phone, String pin) async {
    state = const AsyncLoading();
    state = await AsyncValue.guard(() async {
      final data = await ref.read(authRepositoryProvider).login(phone, pin);
      final authState = AuthState.fromLoginResponse(data, phone);
      final storage = ref.read(secureStorageProvider);
      await storage.write('vendor_token', authState.token);
      await storage.write('vendor_phone', phone);
      return authState;
    });
  }
  
  Future<void> logout() async {
    final storage = ref.read(secureStorageProvider);
    await storage.deleteAll();
    state = const AsyncData(null);
  }
}
```

```dart
final appRouterProvider = Provider<GoRouter>((ref) {
  final authState = ref.watch(authNotifierProvider);
  return GoRouter(
    initialLocation: '/login',
    redirect: (context, state) {
      final isLoggedIn = authState.value != null;
      final isLoginRoute = state.matchedLocation == '/login';
      if (!isLoggedIn && !isLoginRoute) return '/login';
      if (isLoggedIn && isLoginRoute) return '/home';
      return null;
    },
    routes: [
      GoRoute(path: '/login', builder: (_, __) => const LoginScreen()),
      GoRoute(path: '/home', builder: (_, __) => const HomeScreen()),
    ],
  );
});
```

```dart
class StoreTabBar extends ConsumerWidget {
  const StoreTabBar({super.key});
  @override
  Widget build(BuildContext context, WidgetRef ref) {
    final stores = ref.watch(authNotifierProvider).value?.stores ?? [];
    final selectedIndex = ref.watch(selectedStoreIndexProvider);
    if (stores.length <= 1) return const SizedBox.shrink(); // 1개 매장이면 탭 미표시
    return BottomNavigationBar(
      currentIndex: selectedIndex.clamp(0, stores.length - 1),
      onTap: (i) => ref.read(selectedStoreIndexProvider.notifier).state = i,
      items: stores.map((s) => BottomNavigationBarItem(
        icon: const Icon(Icons.store), label: s.aliasName ?? s.storeName,
      )).toList(),
    );
  }
}
```
