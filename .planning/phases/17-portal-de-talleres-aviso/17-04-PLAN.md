```dart
class EnvioRepository {
  final DioClient _client;
  EnvioRepository(this._client);
  
  Future<List<EnvioDto>> fetchEnvios(int storeId, {String? status, int page = 1, int pageSize = 20}) async {
    final params = { 'storeId': storeId, 'page': page, 'pageSize': pageSize };
    if (status != null) params['status'] = status;
    final response = await _client.dio.get('/vendor-portal/envios', queryParameters: params);
    final rows = response.data['rows'] as List? ?? response.data as List;
    return rows.map((e) => EnvioDto.fromJson(e)).toList();
  }
}
```

```dart
class StatusChip extends StatelessWidget {
  final String status;
  const StatusChip({super.key, required this.status});
  
  @override
  Widget build(BuildContext context) {
    final (color, label) = switch (status) {
      'PENDING' => (Colors.orange, 'Pendiente'),
      'PARTIAL' => (Colors.blue, 'Parcial'),
      'COMPLETED' => (Colors.green, 'Completado'),
      'CANCELLED' => (Colors.grey, 'Cancelado'),
      'OPEN' => (Colors.orange, 'Abierto'),
      'CLOSED' => (Colors.green, 'Cerrado'),
      _ => (Colors.grey, status),
    };
    return Chip(label: Text(label, style: TextStyle(color: Colors.white, fontSize: 12)),
      backgroundColor: color, padding: EdgeInsets.zero, materialTapTargetSize: MaterialTapTargetSize.shrinkWrap);
  }
}
```

**1. recepcion_repository.dart**:
```dart
final recepcionRepositoryProvider = Provider<RecepcionRepository>((ref) {
  return RecepcionRepository(ref.read(dioClientProvider));
});

class RecepcionRepository {
  final DioClient _client;
  RecepcionRepository(this._client);
  
  Future<Map<String, dynamic>> createRecepcion({
    required int envioId,
    required int receivedQuantity,
    int rejectedQuantity = 0,
    String? notes,
  }) async {
    final response = await _client.dio.post('/vendor-portal/recepciones', data: {
      'envioId': envioId,
      'receivedQuantity': receivedQuantity,
      'rejectedQuantity': rejectedQuantity,
      if (notes != null && notes.isNotEmpty) 'notes': notes,
    });
    return response.data;
  }
}
```
